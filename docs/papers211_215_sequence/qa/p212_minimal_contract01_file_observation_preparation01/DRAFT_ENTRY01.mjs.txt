// This is one fixed root-owned request input, not an extra host candidate.
// Its observed content/key sits OUTSIDE that request and authenticates no grant.
function loadBoundRequest(record) {
  let fd=null, succeeded=false;
  const chunks=[], digest=createHash('sha256');
  record.path=REQUEST_PATH; record.byte_limit=REQUEST_BYTE_LIMIT;
  record.bytes_read=0; record.eof=false; record.complete=false;
  record.close_succeeded=null; record.raw_hex=''; record.content=null;
  try {
    const lexical=fs.lstatSync(REQUEST_PATH,{bigint:true});
    record.lstat_before=fields(lexical);
    need(lexical.isFile()&&!lexical.isSymbolicLink(),'REQUEST_PHYSICAL_REGULAR_LEAF');
    need(lexical.size>=0n&&lexical.size<=BigInt(REQUEST_BYTE_LIMIT),'REQUEST_DECLARED_SIZE_BOUND');
    fd=fs.openSync(REQUEST_PATH,fs.constants.O_RDONLY|fs.constants.O_NOFOLLOW|fs.constants.O_NONBLOCK);
    const first=fs.fstatSync(fd,{bigint:true}); record.fstat_before=fields(first);
    need(first.isFile()&&same(record.fstat_before,record.lstat_before),'REQUEST_SAME_FD_BEFORE_BYTES');
    const buffer=Buffer.alloc(65536);
    for (;;) {
      const remaining=REQUEST_BYTE_LIMIT-record.bytes_read;
      need(remaining>=0,'REQUEST_REMAINING_BYTES');
      const count=fs.readSync(fd,buffer,0,Math.min(buffer.length,remaining+1),null);
      if (count===0) { record.eof=true; break; }
      const chunk=Buffer.from(buffer.subarray(0,count));
      record.bytes_read+=count; chunks.push(chunk); digest.update(chunk);
      need(count<=remaining,'REQUEST_OVERFLOW_SENTINEL_PRESERVED');
    }
    record.fstat_after=fields(fs.fstatSync(fd,{bigint:true}));
    record.lstat_after=fields(fs.lstatSync(REQUEST_PATH,{bigint:true}));
    need(same(record.fstat_before,record.fstat_after)&&same(record.lstat_before,record.lstat_after),'REQUEST_KEY_CHANGED');
    need(BigInt(record.bytes_read)===first.size,'REQUEST_WHOLE_EOF_SIZE');
    succeeded=true;
  } catch (error) {
    record.error=errorData(error); throw error;
  } finally {
    record.raw_hex=Buffer.concat(chunks).toString('hex');
    record.content={bytes:record.bytes_read,sha256:digest.digest('hex')};
    if (fd!==null) {
      try { fs.closeSync(fd); record.close_succeeded=true; }
      catch(error) { record.close_succeeded=false; record.close_error=errorData(error); throw error; }
    }
  }
  record.complete=succeeded&&record.close_succeeded===true;
  need(record.complete,'REQUEST_COMPLETE_CLOSED');
  const raw=Buffer.concat(chunks), text=raw.toString('utf8');
  need(Buffer.from(text,'utf8').equals(raw),'REQUEST_UTF8_REVERSIBLE');
  return JSON.parse(text);
}

// Canonical DATA encoding: sorted own object keys, original array order,
// safe integer numbers, ASCII JSON, every non-printing/non-ASCII UTF-16 code
// unit and quote/backslash escaped as lower-case \uXXXX, exactly one final LF.
// No received function, getter, toJSON hook or scientific/ELF parser is run.
function makeWriter(fd,initial,limit) {
  const writer={fd,initial,limit,written:0,nodes:0,ancestors:new Set()};
  writer.ascii = text => {
    need(typeof text==='string'&&/^[\x00-\x7f]*$/.test(text),'ENCODER_ASCII_ONLY');
    const buffer=Buffer.from(text,'ascii');
    need(writer.written+buffer.length<=writer.limit,'OUTPUT_BYTE_BOUND_PARTIAL_PRESERVED');
    let offset=0;
    while(offset<buffer.length) {
      const count=fs.writeSync(fd,buffer,offset,buffer.length-offset,null);
      need(Number.isInteger(count)&&count>0&&count<=buffer.length-offset,'OUTPUT_POSITIVE_WRITE_PROGRESS');
      offset+=count; writer.written+=count;
    }
  };
  writer.string = value => {
    writer.ascii('"');
    for(let at=0;at<value.length;at+=16384) {
      const piece=value.slice(at,at+16384).replace(/[^\x20-\x21\x23-\x5b\x5d-\x7e]/g,
        ch=>'\\u'+ch.charCodeAt(0).toString(16).padStart(4,'0'));
      writer.ascii(piece);
    }
    writer.ascii('"');
  };
  writer.value = (value,depth=0) => {
    need(++writer.nodes<=MAX_VALUE_NODES&&depth<=MAX_VALUE_DEPTH,'ENCODER_FINITE_STRUCTURE');
    if(value===null) { writer.ascii('null'); return; }
    if(typeof value==='string') { writer.string(value); return; }
    if(typeof value==='boolean') { writer.ascii(value?'true':'false'); return; }
    if(typeof value==='number') {
      need(Number.isSafeInteger(value)&&!Object.is(value,-0),'ENCODER_EXACT_INTEGER_ONLY');
      writer.ascii(String(value)); return;
    }
    need(typeof value==='object'&&!writer.ancestors.has(value),'ENCODER_DATA_ONLY_NO_CYCLE');
    writer.ancestors.add(value);
    try {
      if(Array.isArray(value)) {
        need(value.length<=MAX_VALUE_NODES&&Object.keys(value).length===value.length,'ENCODER_DENSE_ARRAY');
        writer.ascii('[');
        for(let i=0;i<value.length;i++) {
          const d=Object.getOwnPropertyDescriptor(value,String(i));
          need(d&&Object.hasOwn(d,'value'),'ENCODER_NO_ARRAY_ACCESSOR');
          if(i) writer.ascii(',');
          writer.value(d.value,depth+1);
        }
        writer.ascii(']');
      } else {
        const proto=Object.getPrototypeOf(value);
        need(proto===Object.prototype||proto===null,'ENCODER_PLAIN_DATA_OBJECT');
        const names=Object.keys(value).sort();
        need(names.length<=MAX_VALUE_NODES,'ENCODER_FINITE_MEMBERS');
        writer.ascii('{');
        for(let i=0;i<names.length;i++) {
          const name=names[i], d=Object.getOwnPropertyDescriptor(value,name);
          need(d&&Object.hasOwn(d,'value'),'ENCODER_NO_OBJECT_ACCESSOR');
          if(i) writer.ascii(',');
          writer.string(name); writer.ascii(':'); writer.value(d.value,depth+1);
        }
        writer.ascii('}');
      }
    } finally { writer.ancestors.delete(value); }
  };
  writer.document = value => { writer.value(value); writer.ascii('\n'); };
  writer.checkPoint = () => {
    const now=fields(fs.fstatSync(fd,{bigint:true}));
    need(IDENTITY.every(k=>now[k]===writer.initial[k]),'OUTPUT_SAME_DESCRIPTOR_IDENTITY');
    need(BigInt(now.size)===BigInt(writer.written),'OUTPUT_SIZE_EQUALS_ACTUAL_WRITES');
  };
  return writer;
}
