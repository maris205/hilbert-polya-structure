\begin{abstract}
Order a subset $A=\{a_0<\cdots<a_{k-1}\}$ of $\{0,\ldots,n-1\}$,
subtract the rank $j$ from $a_j$, discard multiplicity, and iterate the
resulting support map.  We prove that at time $t$ each original consecutive
gap $g$ contributes $g-t$ exactly when $g>t$, and otherwise disappears.  This
gives an exact pointwise clock and all basins; for $n\ge2$, there is a unique
state of depth $n-1$.  A nonempty set $B$ is in the time-$t$ image exactly when
$\max B+t(|B|-1)<n$.  Every such target has a coefficient-sum fibre formula
in which arbitrary short-gap words occupy the slots around forced long gaps;
at one step it reduces to $\binom{n-\max B}{|B|}$, and the first image has
$F_{n+2}$ states.  A bounded-gap generating function gives every depth
population.  Classical strict/weak sequence shifts, stars and bars, beta
sets, and Fibonacci summation receive no contribution credit.  The retained
conjunction is \textsc{owner\_amber}; circulation remains
\textsc{hold\_external}.
\end{abstract}

\section{The map and the subtraction boundary}

Let $[n]_0=\{0,\ldots,n-1\}$ and $\X_n=2^{[n]_0}$.  For
$A=\{a_0<\cdots<a_{k-1}\}$ define
\begin{equation}\label{eq:map}
 T_n(A)=\supp\{a_j-j:0\le j<k\},\qquad T_n(\varnothing)=\varnothing,
\end{equation}
where $\supp$ discards multiplicity.  The integers $a_j-j$ are weakly
increasing and lie in $[n]_0$, so this is a total finite self-map.

The strict-to-weak sequence shift behind $a_j-j$ and the resulting
stars-and-bars identities are classical \cite{Stanley2012}.  Beta sets are
also standard partition coordinates; see, for example, their use in core
partition theory \cite{Fayers2023}.  We assign those encodings, bounded
compositions, Fibonacci identities, and generic functional-graph
bookkeeping no contribution credit.  The scoped residual is only the
autonomous support iteration \eqref{eq:map}, its all-time gap law, and the
simultaneous pointwise/image/every-target inverse atlas below.  A bounded
search found no source stating that literal iterative conjunction.  This
non-hit is neither novelty nor priority evidence.

Internally, P163 acts on families by complemented shadows, P169 transfers
last occurrences between set-partition blocks, and P179 randomly isolates
labels through commuting idempotents.  None is \eqref{eq:map}; nevertheless
their shadow, partition, and inverse vocabulary is fully subtracted.  P100's
digit-erasure clock and P178's image-tower bookkeeping likewise receive no
separation credit.

For nonempty $A$, put $m(A)=a_0$ and define its positive gap word
\begin{equation}\label{eq:gaps}
 g(A)=(g_1,\ldots,g_{k-1}),\qquad g_j=a_j-a_{j-1}.
\end{equation}
The empty word is allowed.  Given a positive word $g$, let
$E_t(g)$ be the subsequence $(g_j-t:g_j>t)$, in the original order.

\section{Pointwise dynamics and the sharp clock}

\begin{theorem}[all-time gap normal form]\label{thm:gap}
For every $A\in\X_n$ and $t\ge0$, the empty set remains empty.  If $A$ is
nonempty, then $\min T_n^t(A)=m(A)$ and
\begin{equation}\label{eq:gap-iterate}
             g(T_n^t(A))=E_t(g(A)).
\end{equation}
Equivalently, start at $m(A)$ and cumulatively add, in order, the terms
$g_j-t$ for which $g_j>t$.

The recurrent states are precisely $\varnothing$ and the $n$ singletons.
Writing $\tau(A)$ for entrance time into this fixed set,
\begin{equation}\label{eq:clock}
 \tau(A)=
 \begin{cases}
 0,&|A|\le1,\\
 \max_{1\le j<|A|}g_j,&|A|\ge2.
 \end{cases}
\end{equation}
For $n\ge2$ the global height is $n-1$, attained uniquely by
$\{0,n-1\}$.  The basin of $\{m\}$ has size $2^{n-m-1}$, and the empty
basin has size one.
\end{theorem}

\begin{proof}
Put $b_j=a_j-j$.  Consecutive differences in the weakly increasing list
$(b_j)$ satisfy
\begin{equation}\label{eq:difference}
                 b_j-b_{j-1}=g_j-1.
\end{equation}
Taking support deletes exactly the zero differences.  Deleting repetitions
does not merge positive increments: it retains, in their old order, exactly
the positive values $g_j-1$.  The minimum remains $b_0=a_0$.  One application
therefore replaces the gap word by $E_1(g)$, and induction proves
\eqref{eq:gap-iterate}.

If $|A|\ge2$, all gaps have disappeared at time $\max g_j$, while a gap
equal to that maximum is still positive one epoch earlier.  This proves the
least-time formula \eqref{eq:clock} and excludes nonfixed recurrence.  A gap
is at most $n-1$, with equality only when two consecutive selected elements
are $0,n-1$; no third selected element can then occur.  This proves the
sharp and unique extremal statement.  Finally, a nonempty orbit preserves
its minimum and ends at that singleton.  The subsets with minimum $m$ are
$\{m\}\cup C$ with $C\subseteq\{m+1,\ldots,n-1\}$, giving the basin count.
\end{proof}

\section{All-time images and target-local fibres}

For $t\ge0$, write
\begin{equation}\label{eq:short-series}
 S_t(z)=z+\cdots+z^t,
\end{equation}
with $S_0(z)=0$.

\begin{theorem}[complete time-$t$ inverse atlas]\label{thm:fibre}
Fix $t\ge0$.  The empty target has the unique source $\varnothing$.  A
nonempty target $B=\{b_0<\cdots<b_r\}$ occurs at time $t$ if and only if
\begin{equation}\label{eq:image-condition}
                  b_r+tr<n.
\end{equation}
For every $B$ (with a negative upper limit interpreted as zero),
\begin{equation}\label{eq:fibre}
 |(T_n^t)^{-1}(B)|=
 \sum_{s=0}^{n-1-b_0-(b_r-b_0)-tr}
 [z^s]\bigl(1-S_t(z)\bigr)^{-(r+1)}.
\end{equation}
Consequently,
\begin{equation}\label{eq:image-size}
 |\im T_n^t|=1+\sum_{r\ge0}\binom{n-tr}{r+1},
\end{equation}
where impossible binomial coefficients vanish.  At one step,
\begin{align}
 |T_n^{-1}(B)|&=\binom{n-\max B}{|B|}\quad(B\ne\varnothing),
 \label{eq:first-fibre}\\
 |\im T_n|&=F_{n+2},                                      \label{eq:fibonacci}
\end{align}
for $F_0=0,F_1=1$.
\end{theorem}

\begin{proof}
Let $h_i=b_i-b_{i-1}$ for $1\le i\le r$.  By
Theorem~\ref{thm:gap}, a source mapping to $B$ at time $t$ has the unique gap
factorisation
\begin{equation}\label{eq:factorization}
 U_0,(h_1+t),U_1,(h_2+t),\ldots,(h_r+t),U_r,
\end{equation}
where each $U_i$ is an arbitrary ordered word with letters in
$\{1,\ldots,t\}$.  Conversely every such factorisation yields $B$: the
$U_i$ letters disappear and the distinguished gaps lose $t$.

The distinguished gaps consume total span
\begin{equation*}
 \sum_{i=1}^r(h_i+t)=b_r-b_0+tr.
\end{equation*}
The available span after the invariant minimum $b_0$ is $n-1-b_0$.
Each short-gap slot has ordinary generating function
$(1-S_t(z))^{-1}$, so summing its $(r+1)$-slot coefficients through the
remaining span proves \eqref{eq:fibre}.  With every $U_i$ empty, feasibility
is exactly \eqref{eq:image-condition}; this also proves sufficiency.

For fixed $r$, condition \eqref{eq:image-condition} chooses an $(r+1)$-set
from $\{0,\ldots,n-1-tr\}$, which proves \eqref{eq:image-size}.  If $t=1$,
then $S_1(z)=z$ and the hockey-stick identity reduces \eqref{eq:fibre} to
\eqref{eq:first-fibre}.  Finally,
$1+\sum_{r\ge0}\binom{n-r}{r+1}=F_{n+2}$, proving
\eqref{eq:fibonacci}.
\end{proof}

Formula \eqref{eq:fibre} is target-local and remains informative after the
first image: its $r+1$ factors distinguish the short-gap words before,
between, and after the long gaps that survive to $B$.  In particular, it is
not inferred by dividing the carrier size by the image size.

\section{Every depth population}

\begin{proposition}[bounded-gap depth census]\label{prop:depth}
Let $C_n(h)=|\{A\in\X_n:\tau(A)\le h\}|$ and put
$Q_h(z)=(1-S_h(z))^{-1}$.  For every $h\ge0$,
\begin{equation}\label{eq:depth-cdf}
 C_n(h)=1+\sum_{s=0}^{n-1}(n-s)[z^s]Q_h(z).
\end{equation}
Thus the exact depth-$h$ population is $C_n(h)-C_n(h-1)$ for $h\ge1$,
while $C_n(0)=n+1$.
\end{proposition}

\begin{proof}
Besides the empty set, choose the minimum $m$ and then an arbitrary ordered
gap word with letters at most $h$.  Such a word of total span $s$ is counted
by $[z^s]Q_h(z)$ and permits exactly $n-s$ minima
$m\in\{0,\ldots,n-1-s\}$.  The clock formula says precisely that these are
the states of depth at most $h$.  Summation proves \eqref{eq:depth-cdf}; at
$h=0$ only the empty gap word remains.
\end{proof}

For orientation, at $n=4$ the arrows include
\begin{equation*}
 \{0,3\}\longmapsto\{0,2\}\longmapsto\{0,1\}
 \longmapsto\{0\},
\end{equation*}
which witnesses the sharp height.  At $n=18$, exhaustive evaluation gives
6,765 first-image targets, 19 fixed states, maximum one-step fibre 2,002,
and the unique depth-17 state $\{0,17\}$.
