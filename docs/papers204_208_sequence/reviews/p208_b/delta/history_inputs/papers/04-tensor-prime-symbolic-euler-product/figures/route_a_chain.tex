\node[exact] (full) {%
  {\sffamily\scriptsize\bfseries [EXACT]}\quad
  \textbf{Finite full shifts + Cartesian tensor}\\[-1pt]
  $F_m\otimes F_n:=F_m\times F_n\cong F_{mn}$, with tensor unit $F_1$.

\node[exact, below=of full] (atoms) {%
  {\sffamily\scriptsize\bfseries [EXACT]}\quad
  \textbf{Tensor atoms + intrinsic entropy}\\[-1pt]
  $\operatorname{At}(\mathsf{FSh})=\{[F_p]:p\ \text{prime}\}$,
  $h_{\mathrm{top}}(F_p)=\log p$, and
  $F_n\cong\bigotimes_p F_p^{\otimes v_p(n)}$.

\node[construction, below=of atoms] (loops) {%
  {\sffamily\scriptsize\bfseries [CONSTRUCTION]}\quad

  The roof is inherited from entropy:
  $\ell(\gamma_p^r)=r\log p$.
};

\node[construction, below=of loops] (fredholm) {%
  {\sffamily\scriptsize\bfseries [HALF-PLANE]}\quad
  \textbf{Trace-class Fredholm ledger}\\[-1pt]
  $\mathcal L_s e_p=p^{-s}e_p$ is trace class for
  $\operatorname{Re}s>1$, and
  $D_\otimes(s):=\det(I-\mathcal L_s)=\prod_p(1-p^{-s})$.
};

\node[exact, below=of fredholm] (zeta) {%
  {\sffamily\scriptsize\bfseries [EXACT FOR $\operatorname{Re}s>1$]}\quad
  \textbf{Symbolic Euler product}\\[-1pt]
  $Z_\otimes(s):=D_\otimes(s)^{-1}
   =\prod_p(1-p^{-s})^{-1}
   =\sum_{n\geq1}n^{-s}=\zeta(s)$,\\[-1pt]
  $-Z_\otimes'(s)/Z_\otimes(s)=\sum_n\Lambda(n)n^{-s}$.
};
