\section{Proof details}
\label{app:proofs}

\subsection{Rank-one factorization and transport}

Since $E_x=e_xe_x^{\mathsf T}$,
\begin{equation}
 q_x=ZE_xM=(Ze_x)(e_x^{\mathsf T}M)=u_xv_x^{\mathsf T}.
\end{equation}
The definitions of $u_x,v_x$ follow from the entries of $Z$ and $M$.
Moreover,
\begin{equation}
 q_y^\sharp=W^{-1}v_yu_y^{\mathsf T}W,
\end{equation}
and hence
\begin{align}
 q_xq_y^\sharp
 &=u_x(v_x^{\mathsf T}W^{-1}v_y)u_y^{\mathsf T}W,\\
 \Tr(q_xq_y^\sharp)
 &=(v_x^{\mathsf T}W^{-1}v_y)(u_y^{\mathsf T}Wu_x).
\end{align}
This proves \cref{prop:general-gram}.  If $U_\phi$ is the permutation matrix
of a transported pointed isomorphism, then
\begin{equation}
 q'_{\phi x}=U_\phi q_xU_\phi^{-1},
 \qquad
 W'=U_\phi WU_\phi^*,
\end{equation}
so $G'_{\phi x,\phi y}=G_{xy}$.

\subsection{Divisibility Euler factors}

For a cover $p$, the lower-cone factor is $1+p^{2\eta}$.  The upper-tail
factor is
\begin{equation}
 \sum_{p\mid b}\frac{\mu(p,b)^2}{b^{2\eta}}
 =p^{-2\eta}\sum_{k\ge1}\frac{\mu(k)^2}{k^{2\eta}}
 =p^{-2\eta}C_\eta.
\end{equation}
Their product is \eqref{eq:Gdiag}.

For $p\ne q$, the only common lower bound is $1$.  A common upper-tail term
with nonzero M\"obius factors is $b=pqk$, where $k$ is squarefree and coprime
to $pq$.  The two M\"obius signs have product $+1$, so
\begin{align}
 G_{pq}
 &=(pq)^{-2\eta}
   \sum_{\substack{k\ge1\\(k,pq)=1}}
   \frac{\mu(k)^2}{k^{2\eta}}\\
 &=C_\eta\frac{(pq)^{-2\eta}}
 {(1+p^{-2\eta})(1+q^{-2\eta})}.
\end{align}
This proves \cref{thm:atom-gram} without a numeric prime list.

\subsection{Normal convergence}

Equation \eqref{eq:Gmixed} implies
\begin{equation}
 0<G_{pq}\le C_\eta(pq)^{-2\eta}.
\end{equation}
With $\alpha=2\eta+1/2>1$,
\begin{equation}
 \sum_{p<q}\frac{4G_{pq}}{\sqrt{pq}}
 \le2C_\eta\left(\sum_pp^{-\alpha}\right)^2<\infty.
\end{equation}
The Weierstrass test gives uniform convergence on the critical line.  If at
least one atom exceeds $X$, replace the corresponding prime tail by
$\sum_{n>X}n^{-\alpha}=O(X^{1-\alpha})$.  This proves
\eqref{eq:mixed-tail}.

For complex $s$, the two terms in \eqref{eq:mixed-holo} are dominated by
products with exponents $2\eta+\Re s$ and
$2\eta+1-\Re s$.  Both exceed one on compact subsets of
$1-2\eta<\Re s<2\eta$.  Normal convergence gives holomorphy, and exchanging
$p,q$ gives reflection.

\subsection{Kernel classification}

Polarization gives the unique Hermitian $K_{P,F}$ in
\eqref{eq:quadratic-kernel}.  Under an isomorphism, the transported vector is
$U_\phi a$.  Equality
\begin{equation}
 a^*K_{P,F}a=(U_\phi a)^*K_{P',\phi F}(U_\phi a)
\end{equation}
for every $a$ is equivalent, after polarization, to
\eqref{eq:kernel-equivariance}.  Equality on every vector supported in an
old set $F\subset F'$ is likewise equivalent to
\eqref{eq:kernel-prefix}.

Absolute convergence of the residual diagonal is
\begin{equation}
 \sum_p\frac{|G_{pp}-k_{pp}|}{p}<\infty.
\end{equation}
Because $G_{pp}-C_\eta=C_\eta p^{-2\eta}$ is already in
$\ell^1(p^{-1})$, this condition is equivalent to
\eqref{eq:diag-classification}.  The native mixed kernel belongs to the
weighted $\ell^1$ space by \cref{thm:mixed-summability}; triangle inequalities
then make residual mixed convergence equivalent to
\eqref{eq:mixed-classification}.  Absolute convergence removes dependence on
the enumeration of finite active sets.

\subsection{Sharp--Abel discrepancy}

The sharp finite part uses the constant $\Meta$ in
\eqref{eq:mertens}.  The Abel regulator uses the constant
$\Meta-\gamma$ in \eqref{eq:prime-zeta-expansion}.  The diagonal tail and
mixed series are absolutely summable, so dominated convergence carries them
unchanged between the two limits.  Multiplication by the coefficient
$2C_\eta$ proves \eqref{eq:abel-sharp}.

\subsection{Scheme-independent divisor}

If two finite parts differ by $h(s)$, cancellation of the common
$\detthree$ factor gives \eqref{eq:scheme-ratio}.  The complex exponential
never vanishes.  Therefore its product with a holomorphic function preserves
the zero divisor in $z$, including multiplicity.  This statement concerns
only the auxiliary determinant variable and supplies no external spectral
identification.
