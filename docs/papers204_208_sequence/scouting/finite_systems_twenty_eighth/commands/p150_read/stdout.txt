\title[Zero-totalized Lyness dynamics]{The All-Affine Functional Graph of the
Zero-Totalized Lyness Map over Odd Finite Fields}
\author{Anonymous}
\date{}
\subjclass[2020]{37P05, 37P25, 11T06}
\keywords{finite dynamics, finite field, Lyness map, functional graph, entry time, dynamical zeta function, fibre}

\begin{document}

\begin{abstract}
Let $\F_q$ be a finite field of odd order, set $\invz(0)=0$ and
$\invz(x)=x^{-1}$ for $x\ne0$, and consider the all-affine self-map
$\Lyn(x,y)=(y,(1+y)\invz(x))$.  We determine its complete functional graph.
The plane splits disjointly into a generic Lyness locus, the two coordinate
axes, and three exceptional tail layers.  The recurrent set has
$q^2-3q+5$ points, while the tail polynomial is
\[
 (q^2-3q+5)+(q-1)z+(q-2)z^2+(q-2)z^3;
\]
thus the maximum tail is exactly three for every odd $q$.  If $r_q$ counts
the roots of $a^2-a-1$, then the graph has $1+r_q$ fixed points, two
2-cycles, $(q-3)/2$ 4-cycles, and
$((q-2)(q-3)-r_q)/5$ 5-cycles.  We also prove a codomain-wide inverse law:
fibres have size $q$ at $(-1,0)$, size zero at $(-1,v)$ for $v\ne0$, and
size one elsewhere.  Classical Lyness five-periodicity, QRT and cluster
interpretations, and general finite-field birational dynamics receive zero
contribution credit.  The residual result is the exact zero-totalized affine
boundary completion.  External release remains on hold.
\end{abstract}

\section{Scope and complete statement}

The parameter-one Lyness recurrence and its five-cycle are classical.  The
same map is a basic QRT system and a type-$A_2$ cluster example
\citep{Lyness1942,HoneKouloukas2023,Hone2020}.  Integrable birational maps
over finite fields also admit general algebraic-geometric and
almost-good-reduction treatments~\citep{JogiaRobertsVivaldi2006,Kanki2013}.
Kanki resolves division-by-zero obstructions by extending/reducing the state
space, not by setting inverse zero to zero.  All of that material is
background here.
We study one boundary convention on the whole affine plane, including every
point at which the usual rational formula has a zero denominator.

Fix a finite field $\F_q$ of odd order.  Define
\begin{equation}\label{eq:invzero}
 \invz(x)=
 \begin{cases}
 0,&x=0,\\
 x^{-1},&x\ne0,
 \end{cases}
 \qquad
 \Lyn(x,y)=\bigl(y,(1+y)\invz(x)\bigr).
\end{equation}
Let $r_q$ be the number of roots in $\F_q$ of
\begin{equation}\label{eq:fixed-polynomial}
 a^2-a-1=0.
\end{equation}
For a point $P$, its tail $\tau(P)$ is the least $t\geq0$ such that
$\Lyn^t(P)$ is recurrent.

The five sets below will describe every point.  Put
\begin{align}
 \G&=\{(x,y):xy(1+x)(1+y)(1+x+y)\ne0\},\label{eq:generic}\\
 \A&=(\{0\}\times\F_q)\cup(\F_q\times\{0\}),\label{eq:axes}\\
 \E_1&=\{(a,-1):a\in\F_q\setminus\{0\}\},\label{eq:e1}\\
 \E_2&=\{(-1-a,a):a\in\F_q\setminus\{0,-1\}\},\label{eq:e2}\\
 \E_3&=\{(-1,-1-a):a\in\F_q\setminus\{0,-1\}\}.\label{eq:e3}
\end{align}

\begin{theorem}[Complete all-affine atlas]\label{thm:main}
For every odd prime power $q$, the following statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The five sets in \eqref{eq:generic}--\eqref{eq:e3} are pairwise
disjoint and their union is $\F_q^2$.  Their respective sizes are
\begin{equation}\label{eq:strata-sizes}
 (q-2)(q-3),\qquad 2q-1,\qquad q-1,\qquad q-2,\qquad q-2.
\end{equation}
The recurrent set is $\G\cup\A$, and $\tau$ equals $j$ on $\E_j$.
Consequently
\begin{equation}\label{eq:temporal}
 \sum_{P\in\F_q^2}z^{\tau(P)}
 =(q^2-3q+5)+(q-1)z+(q-2)z^2+(q-2)z^3.
\end{equation}
In particular, the maximum tail is exactly three.
\item The cycle census is
\begin{center}
\begin{tabular}{@{}cc@{}}
\toprule
cycle length & number of cycles\\
\midrule
$1$ & $1+r_q$\\
$2$ & $2$\\
$4$ & $(q-3)/2$\\
$5$ & $((q-2)(q-3)-r_q)/5$\\
\bottomrule
\end{tabular}
\end{center}
There are no other cycles.  Hence the dynamical zeta function is
\begin{equation}\label{eq:zeta}
 \zeta_{\Lyn}(z)=
 (1-z)^{-(1+r_q)}(1-z^2)^{-2}
 (1-z^4)^{-(q-3)/2}
 (1-z^5)^{-((q-2)(q-3)-r_q)/5}.
\end{equation}
\item For every target $(u,v)\in\F_q^2$,
\begin{equation}\label{eq:fibre-law}
 |\Lyn^{-1}(u,v)|=
 \begin{cases}
 q,&(u,v)=(-1,0),\\
 0,&u=-1\text{ and }v\ne0,\\
 1,&u\ne-1.
 \end{cases}
\end{equation}
Thus $|\im\Lyn|=q^2-q+1$, and the exceptional in-tree attached to the
two-cycle $(-1,0)\leftrightarrow(0,-1)$ is exactly the one displayed in
\cref{cor:tree} below.
\end{enumerate}
\end{theorem}

The next three sections prove \cref{thm:main}.  Enumeration is used only for
the independent controls reported in \cref{sec:controls}.
