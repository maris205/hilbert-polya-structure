\[

\[
\psi'(r)=\frac1{v(1+v)},\qquad
\psi''(r)=\frac{r(1+2v)}{v^3(1+v)^2}>0\quad(0<r<1).

\[
T_c(E)=\frac{2\sqrt2}{\sqrt c}

\[

\[

\[
