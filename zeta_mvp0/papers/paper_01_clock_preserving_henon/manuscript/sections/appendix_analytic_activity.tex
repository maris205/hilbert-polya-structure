\section{Analytic spectral activity of one H\'enon warp}
\label{app:analytic-activity}

This appendix proves the two one-step, nonmagnetic activity theorems in
\cref{sec:strict-ground-state,sec:relative-heat-activity}.  Throughout,
\begin{equation}
 \Psi_a(x,y)=(-2ar_ax-ax^2-y,x),
 \qquad
 r_a=\frac1{1+\sqrt{1+a}},
 \label{eq:app-activity-warp}
\end{equation}
\begin{equation}
 V_a(q)=2\pi e^{\pi|\Psi_a(q)|^2},
 \qquad
 \mathsf H_{a,\hbar}=-\frac{\hbar^2}{2}\Delta+V_a.
 \label{eq:app-activity-operator}
\end{equation}
The Friedrichs realization and compact resolvent were established in
\cref{app:compactness}.  The constants below may depend on fixed
\(a>-1\) and \(\hbar>0\), but not on \(t\).

\subsection{Strict ground-state ordering}
\label{app:strict-ground-state}

Let \(f^*\) denote the symmetric decreasing rearrangement of \(|f|\).
The area identity \(\det D\Psi_a=1\) implies, for every \(s\ge2\pi\),
\begin{equation}
 A_s:=\{V_a\le s\},
 \qquad
 A_s^*:=\{V_0\le s\},
 \qquad
 |A_s|=|A_s^*|.
 \label{eq:app-equimeasurable-sets}
\end{equation}
Thus \(V_0\) is the symmetric increasing rearrangement of \(V_a\).

\begin{lemma}[Rearranged Rayleigh inequality]
\label{lem:app-rayleigh-rearrangement}
For every \(f\) in the form domain of \(\mathsf H_{a,\hbar}\),
\begin{equation}
 Q_{0,\hbar}[f^*]\le Q_{a,\hbar}[f],
 \qquad \|f^*\|_2=\|f\|_2,
 \label{eq:app-rayleigh-rearrangement}
\end{equation}
where
\begin{equation}
 Q_{a,\hbar}[f]
 =\frac{\hbar^2}{2}\int|\nabla f|^2\dd q
 +\int V_a|f|^2\dd q.
\end{equation}
\end{lemma}

\begin{proof}
The diamagnetic inequality for the ordinary gradient followed by
P\'olya--Szeg\H{o} gives
\begin{equation}
 \int|\nabla f^*|^2
 \le\int|\nabla|f||^2
 \le\int|\nabla f|^2.
 \label{eq:app-polya-szego}
\end{equation}
For the unbounded increasing potential it is useful to keep the layer-cake
direction explicit.  The set form of Hardy--Littlewood and
\cref{eq:app-equimeasurable-sets} give
\begin{equation}
 \int_{A_s}|f|^2\dd q
 \le\int_{A_s^*}(f^*)^2\dd q.
\end{equation}
Subtracting from the common total \(L^2\) norm reverses this inequality on
the complements.  Since
\begin{equation}
 V_a=2\pi+\int_{2\pi}^{\infty}
 \mathbf1_{\{V_a>s\}}\dd s,
\end{equation}
Tonelli's theorem yields
\begin{equation}
 \int V_a|f|^2\dd q
 \ge\int V_0(f^*)^2\dd q.
 \label{eq:app-increasing-potential-rearrangement}
\end{equation}
The right-hand side of
\cref{eq:app-increasing-potential-rearrangement} is finite whenever \(f\)
lies in the form domain of \(\mathsf H_{a,\hbar}\); together with
\cref{eq:app-polya-szego}, this also proves that \(f^*\) lies in the form
domain of \(\mathsf H_{0,\hbar}\).  The assertion follows, with monotone
truncation justifying the layer-cake argument for the unbounded potential
\citep{LiebLoss2001}.
\end{proof}

\begin{proof}[Proof of \cref{thm:strict-ground-state}]
Compact resolvent gives a normalized ground state \(\phi_a\).  The scalar
heat semigroup is positivity improving on the connected space \(\R^2\);
hence the lowest eigenvalue is simple and \(\phi_a\) may be chosen strictly
positive \citep{ReedSimon1978IV}.  Smooth elliptic regularity applies, and
the real-analytic potential \(V_a\) makes \(\phi_a\) real analytic by
analytic elliptic regularity \citep{MorreyNirenberg1957}.  If equality is
assumed below, \(\phi_a^*\) is the ground state of the analytic radial
problem and is analytic as well.

The preceding lemma and the Rayleigh principle first give
\begin{equation}
 \lambda_1(\mathsf H_{0,\hbar})
 \le\lambda_1(\mathsf H_{a,\hbar}).
\end{equation}
If equality held, then
\begin{equation}
 \lambda_1(\mathsf H_{0,\hbar})
 \le Q_{0,\hbar}[\phi_a^*]
 \le Q_{a,\hbar}[\phi_a]
 =\lambda_1(\mathsf H_{0,\hbar}).
 \label{eq:app-ground-equality-chain}
\end{equation}
Consequently \(\phi_a^*\) attains the Rayleigh infimum for
\(\mathsf H_{0,\hbar}\) and is its radial ground state.  Because
\(\phi_a>0\), the difference between the right- and left-hand sides of
\cref{eq:app-rayleigh-rearrangement} is the sum of the two nonnegative
deficits
\begin{equation}
 \frac{\hbar^2}{2}\left(
 \int|\nabla\phi_a|^2-\int|\nabla\phi_a^*|^2\right)
 \label{eq:app-kinetic-deficit}
\end{equation}
and
\begin{equation}
 \int V_a\phi_a^2-\int V_0(\phi_a^*)^2.
 \label{eq:app-potential-deficit}
\end{equation}
Their sum is zero by \cref{eq:app-ground-equality-chain}; hence both vanish,
and in particular equality holds in P\'olya--Szeg\H{o}.

The positive superlevel sets of \(\phi_a^*\) have finite measure because
\(\phi_a^*\in L^2\).  Moreover, the Brothers--Ziemer exceptional set
\begin{equation}
 \left\{q:0<\phi_a^*(q)<\operatorname*{ess\,sup}\phi_a^*,\ 
 \nabla\phi_a^*(q)=0\right\}
 \label{eq:app-bz-exceptional-set}
\end{equation}
has measure zero.  Indeed, it is contained in the critical set of the
nonconstant real-analytic function \(\phi_a^*\); if that critical set had
positive measure, each analytic first derivative would vanish on a
positive-measure set and hence identically, forcing \(\phi_a^*\) to be
constant.  The Brothers--Ziemer equality classification therefore gives a
point \(q_0\in\R^2\) such that
\begin{equation}
 \phi_a(q)=\phi_a^*(q-q_0).
 \label{eq:app-ground-translate}
\end{equation}

The function in \cref{eq:app-ground-translate} solves both the warped
ground-state equation and the translate of the radial equation.  Subtracting
them and using \(\phi_a>0\) gives
\begin{equation}
 V_a(q)=V_0(q-q_0)\quad\text{for every }q.
 \label{eq:app-potential-radial-contradiction}
\end{equation}
This is impossible for \(a\ne0\).  Indeed,
\begin{equation}
 \frac1\pi\log\frac{V_a(x,y)}{2\pi}
 =x^2+(2ar_ax+ax^2+y)^2
 \label{eq:app-quartic-nonradiality}
\end{equation}
has degree-four homogeneous part \(a^2x^4\).  Translation does not change
the highest homogeneous part, whereas the degree-four part of a polynomial
radial about any point must be \(C(x^2+y^2)^2\).  Its missing \(y^4\)
coefficient would force \(C=0\), contradicting the nonzero \(x^4\)
coefficient.  Equality in \cref{eq:app-ground-equality-chain} is excluded,
which proves the strict inequality \citep{BrothersZiemer1988}.
\end{proof}

\subsection{Brownian-bridge proof of the uniform relative heat remainder}
\label{app:relative-heat-proof}

Set
\begin{equation}
 Q_a=\Psi_a^{-1},\qquad
 Q_a(u,v)=(v,-2ar_av-av^2-u),
 \qquad W(z)=2\pi e^{\pi|z|^2},
 \label{eq:app-heat-QW}
\end{equation}
and let \(\mathbf b=(\mathbf b_s)_{0\le s\le1}\) be a standard
two-dimensional Brownian bridge with
\begin{equation}
 \mathbb E[b_i(s)b_j(r)]
 =\delta_{ij}\bigl(\min(s,r)-sr\bigr).
 \label{eq:app-bridge-covariance}
\end{equation}
Write \(\varepsilon=\hbar\sqrt t\),
\(M=\sup_s|\mathbf b_s|\), and
\(E_1(\lambda)=\int_\lambda^\infty e^{-w}\dd w/w\).

\begin{lemma}[Common-coordinate representation]
\label{lem:app-common-heat-coordinate}
For every \(t>0\),
\begin{equation}
 \Theta_{a,\hbar}(t)
 =\frac1{2\pi\hbar^2t}\int_{\R^2}
 F_a(\varepsilon,z)\dd z,
 \label{eq:app-common-heat-coordinate}
\end{equation}
where
\begin{equation}
 F_a(\theta,z)=\mathbb E\exp\!\left[-t\int_0^1
 V_a(Q_a(z)+\theta\mathbf b_s)\dd s\right].
 \label{eq:app-Fa-definition}
\end{equation}
Moreover,
\begin{equation}
 F_a(0,z)=e^{-tW(z)},
 \label{eq:app-pointwise-classical-cancellation}
\end{equation}
independently of \(a\).
\end{lemma}

\begin{proof}
The diagonal Brownian-bridge Feynman--Kac formula gives
\begin{equation}
 K_{a,\hbar}(t;q,q)=\frac1{2\pi\hbar^2t}
 \mathbb E\exp\!\left[-t\int_0^1
 V_a(q+\hbar\sqrt t\,\mathbf b_s)\dd s\right]
\end{equation}
\citep{Simon2005FunctionalIntegration,BoldtGueneysu2023}.  The heat
semigroup is trace class; for example, Golden--Thompson and area preservation
give
\begin{equation}
 \Theta_{a,\hbar}(t)
 \le\frac1{2\pi\hbar^2t}\int e^{-tV_a(q)}\dd q
 =\frac{E_1(2\pi t)}{2\pi\hbar^2t}<\infty.
 \label{eq:app-GT-heat-bound}
\end{equation}
Integrate the diagonal and change variables \(q=Q_a(z)\).  Since
\(V_a(Q_a(z))=W(z)\), both assertions follow.
\end{proof}

The following explicit bounds provide the required global domination.  Set
\begin{equation}
 \Phi_a=\log\frac{V_a}{2\pi}=\pi|\Psi_a|^2,
 \qquad \sigma=\pi|z|^2.
 \label{eq:app-heat-Phi-sigma}
\end{equation}
For \(z=(u,v)\) and \(w=(\xi,\eta)\), direct substitution gives
\begin{equation}
 \Psi_a(Q_a(z)+w)
 =\bigl(u-2a(r_a+v)\xi-a\xi^2-\eta,\ v+\xi\bigr).
 \label{eq:app-exact-henon-displacement}
\end{equation}
Consequently,
\begin{equation}
 |\Phi_a(Q_a(z)+w)-\sigma|
 \le C_a(1+\sigma)(|w|+|w|^2).
 \label{eq:app-Phi-displacement-bound}
\end{equation}
Because \(\Phi_a\) is quartic, Fa\`a di Bruno gives, for \(1\le k\le4\),
\begin{equation}
 \|D^kV_a(Q_a(z)+w)\|
 \le C_{a,\delta,k}V_a(Q_a(z)+w)(1+\sigma)^k
 \label{eq:app-normalized-heat-derivatives}
\end{equation}
whenever \(|w|\le\delta/L\) and \(L\) is sufficiently large.  The same
calculation gives, for nonzero multi-indices \(\beta_j\) and
\(N=\sum_j|\beta_j|\),
\begin{equation}
 \int e^{-tV_a(q)}\prod_{j=1}^m
 |\partial^{\beta_j}V_a(q)|\dd q
 \le C_{a,\boldsymbol\beta}t^{-m}(1+L)^N.
 \label{eq:app-weighted-heat-derivatives}
\end{equation}
Indeed, radial integration in \(z\), followed by
\(w=tW(z)=2\pi t e^\sigma\), reduces the bound to
\begin{equation}
 \int_{2\pi t}^{\infty}e^{-w}w^{m-1}
 \left(1+\log\frac{w}{2\pi t}\right)^N\dd w
 =O_{m,N}((1+L)^N).
 \label{eq:app-gamma-derivative-bound}
\end{equation}

Freeze a sufficiently small \(\delta>0\) and define the symmetric good
event
\begin{equation}
 G_t=\left\{M\le\frac{\delta}{\hbar\sqrt t\,L}\right\}.
 \label{eq:app-good-bridge-event}
\end{equation}
The Gaussian bridge maximum satisfies
\begin{equation}
 \mathbb P(G_t^c)\le C_{\hbar}
 \exp\!\left(-\frac{c_{\hbar}}{tL^2}\right).
 \label{eq:app-bridge-tail}
\end{equation}
For every continuous deterministic path \(w_s\), convexity of
\(x\mapsto e^{-tx}\), followed by translation invariance, yields the
pathwise integral bound
\begin{equation}
 \int_{\R^2}e^{-t\int_0^1V_a(q+w_s)\dd s}\dd q
 \le\int_{\R^2}e^{-tV_a(q)}\dd q
 =E_1(2\pi t).
 \label{eq:app-pathwise-jensen}
\end{equation}
This order of integration prevents a small bridge probability from being
multiplied by the infinite volume of \(\R^2\).

\begin{lemma}[Integrated fourth-amplitude bound]
\label{lem:app-fourth-amplitude}
Let
\begin{equation}
 \mathcal A_a(\theta,z,\mathbf b)
 =t\int_0^1V_a(Q_a(z)+\theta\mathbf b_s)\dd s,
 \qquad f_a=e^{-\mathcal A_a}.
\end{equation}
Then
\begin{equation}
 \int_{\R^2}\sup_{|\theta|\le\varepsilon}
 \mathbb E\!\left[
 \mathbf1_{G_t}|\partial_\theta^4f_a(\theta,z,\mathbf b)|
 \right]\dd z
 \le C_aL^4.
 \label{eq:app-fourth-amplitude-bound}
\end{equation}
\end{lemma}

\begin{proof}
For \(1\le k\le4\),
\begin{equation}
 \mathcal A_a^{(k)}(\theta)
 =t\int_0^1D^kV_a(Q_a(z)+\theta\mathbf b_s)
 [\mathbf b_s,\ldots,\mathbf b_s]\dd s,
\end{equation}
and direct differentiation gives
\begin{equation}
 \partial_\theta^4e^{-\mathcal A}
 =e^{-\mathcal A}\left[(\mathcal A')^4
 -6(\mathcal A')^2\mathcal A''+3(\mathcal A'')^2
 +4\mathcal A'\mathcal A'''-\mathcal A^{(4)}\right].
 \label{eq:app-fourth-exponential-derivative}
\end{equation}
On \(G_t\), \(|\theta\mathbf b_s|\le\delta/L\).  Split at
\(\sigma_*=L+8\log L\).  On \(0\le\sigma\le\sigma_*\), set
\(Y=e^{\sigma-L}\).  Equations
\eqref{eq:app-Phi-displacement-bound}--
\eqref{eq:app-fourth-exponential-derivative} imply
\begin{equation}
 |\partial_\theta^4f_a|
 \le C_aM^4(1+\sigma)^4
 \sum_{m=1}^4Y^me^{-cY}.
 \label{eq:app-fourth-main-bound}
\end{equation}
Since \(\mathbb E[M^4]<\infty\) and \(\dd\sigma=\dd Y/Y\), its spatial
integral is \(O_a(L^4)\); no extra factor of \(L\) appears because every
summand contains at least one \(Y\).

For \(\sigma>\sigma_*\), put
\begin{equation}
 \alpha_L=C_a\left(\frac\delta L+\frac{\delta^2}{L^2}\right).
\end{equation}
The lower path potential is bounded by
\begin{equation}
 tV_a(Q_a(z)+\theta\mathbf b_s)
 \ge \exp\bigl[-L+(1-\alpha_L)\sigma-\alpha_L\bigr].
\end{equation}
After the corresponding change of variables, the lower endpoint is at least
\(L^7\), the upper potential is a power
\(p_L=(1+\alpha_L)/(1-\alpha_L)=1+O(L^{-1})\) of the lower one, and the
Jacobian factor \((1-\alpha_L)^{-1}\) is bounded.  The tail is therefore a
finite sum of integrals
\begin{equation}
 C_{a,\delta}\int_{L^7}^{\infty}
 (1+L+\log y)^4y^{mp_L-1}e^{-cy}\dd y,
 \qquad1\le m\le4,
\end{equation}
which is \(O_{a,N}(L^{-N})\) for every fixed \(N\).  Combining the two
regions proves the lemma.
\end{proof}

\begin{lemma}[Uniform amplitude expansion]
\label{lem:app-uniform-amplitude-expansion}
For fixed \(a\) and \(\hbar\),
\begin{equation}
 \int_{\R^2}\left|F_a(\varepsilon,z)-F_a(0,z)
 -\frac{\varepsilon^2}{2}\partial_\theta^2F_a(0,z)\right|\dd z
 \le C_{a,\hbar}t^2L^4.
 \label{eq:app-uniform-amplitude-expansion}
\end{equation}
\end{lemma}

\begin{proof}
On \(G_t\), invariance under \(\mathbf b\mapsto-\mathbf b\) makes the
restricted expectation even in \(\theta\).  Taylor's theorem,
\cref{lem:app-fourth-amplitude}, and \(\varepsilon=\hbar\sqrt t\) bound the
integrated remainder by \(C_{a,\hbar}t^2L^4\).

The bad event is exponentially smaller.  Equations
\eqref{eq:app-bridge-tail} and \eqref{eq:app-pathwise-jensen} bound its zeroth
term by
\begin{equation}
 C_{\hbar}E_1(2\pi t)e^{-c_{\hbar}/(tL^2)}.
 \label{eq:app-bad-event-zero}
\end{equation}
It remains to justify two derivatives through this expectation.  For fixed
\(z,t,\theta_0\), the exact polynomial displacement gives, uniformly for
\(|\theta|\le\theta_0\),
\begin{align}
 \frac{\|DV_a(Q_a(z)+\theta\mathbf b_s)\|}
 {V_a(Q_a(z)+\theta\mathbf b_s)}
 &\le C_{a,z,\theta_0}(1+M)^4,\\
 \frac{\|D^2V_a(Q_a(z)+\theta\mathbf b_s)\|}
 {V_a(Q_a(z)+\theta\mathbf b_s)}
 &\le C_{a,z,\theta_0}(1+M)^8.
\end{align}
Thus, with \(\mathcal A=\mathcal A_a(\theta,z,\mathbf b)\),
\begin{equation}
 |\mathcal A'|\le C(1+M)^5\mathcal A,
 \qquad
 |\mathcal A''|\le C(1+M)^{10}\mathcal A.
\end{equation}
Since \(xe^{-x}\) and \(x^2e^{-x}\) are bounded,
\begin{equation}
 \sup_{|\theta|\le\theta_0}|f_a''(\theta,z,\mathbf b)|
 \le C_{a,z,\theta_0}(1+M)^{10}.
 \label{eq:app-bad-event-dominator}
\end{equation}
The bridge maximum has Gaussian tails, so this is integrable and dominated
differentiation is valid.  At \(\theta=0\),
\begin{align}
 f_a''(0)=e^{-tV_a}\Bigg[&t^2
 \left(\nabla V_a\cdot\int_0^1\mathbf b_s\dd s\right)^2
 \notag\\[-1mm]
 &-t\int_0^1D^2V_a[\mathbf b_s,\mathbf b_s]\dd s\Bigg].
\end{align}
The weighted bound \eqref{eq:app-weighted-heat-derivatives} and a Gaussian
tail moment give an integrated bad-event contribution
\begin{equation}
 O_{a,\hbar}\!\left(L^2e^{-c_{\hbar}/(2tL^2)}\right).
\end{equation}
After multiplication by \(\varepsilon^2\), this and
\cref{eq:app-bad-event-zero} are smaller than \(t^2L^4\).  The lemma follows.
\end{proof}

\begin{lemma}[Second amplitude coefficient]
\label{lem:app-second-amplitude}
For every fixed \(a\),
\begin{equation}
 \frac1{2\pi\hbar^2t}\frac{\varepsilon^2}{2}
 \int_{\R^2}\partial_\theta^2F_a(0,z)\dd z
 =-\frac{t^2}{48\pi}I_a(t).
 \label{eq:app-second-amplitude-coefficient}
\end{equation}
\end{lemma}

\begin{proof}
The bridge covariances are
\begin{equation}
 \mathbb E\!\left[\left(\int_0^1b_i(s)\dd s\right)
 \left(\int_0^1b_j(s)\dd s\right)\right]=\frac{\delta_{ij}}{12},
 \qquad
 \int_0^1\mathbb E[b_i(s)b_j(s)]\dd s=\frac{\delta_{ij}}6.
\end{equation}
Hence
\begin{equation}
 \partial_\theta^2F_a(0,z)
 =e^{-tV_a(Q_a(z))}\left[
 \frac{t^2}{12}|\nabla V_a(Q_a(z))|^2
 -\frac t6\Delta V_a(Q_a(z))\right].
\end{equation}
Changing back to \(q\) and inserting the prefactor gives
\begin{equation}
 \frac{t^2}{48\pi}I_a(t)
 -\frac{t}{24\pi}\int e^{-tV_a}\Delta V_a\dd q.
\end{equation}
The weighted derivative bounds justify cutoff integration by parts, yielding
\begin{equation}
 \int e^{-tV_a}\Delta V_a\dd q
 =t\int e^{-tV_a}|\nabla V_a|^2\dd q=tI_a(t).
\end{equation}
Combining the two terms proves the claim.
\end{proof}

Apply \cref{lem:app-uniform-amplitude-expansion} to \(a\) and \(0\), insert
both expansions into \cref{eq:app-common-heat-coordinate}, and subtract.
The zeroth terms cancel by
\cref{eq:app-pointwise-classical-cancellation};
\cref{lem:app-second-amplitude} gives
\begin{equation}
 \Theta_{a,\hbar}(t)-\Theta_{0,\hbar}(t)
 =-\frac{t^2}{48\pi}\bigl(I_a(t)-I_0(t)\bigr)
 +O_{a,\hbar}(tL^4),
 \label{eq:app-relative-heat-remainder-final}
\end{equation}
because the common heat prefactor turns the integrated
\(O_{a,\hbar}(t^2L^4)\) amplitude remainder into
\(O_{a,\hbar}(tL^4)\).

It remains to compute the carrier.  In the coordinate
\(z=(u,v)=\Psi_a(q)\),
\begin{equation}
 I_a-I_0
 =\int e^{-tW(z)}(2\pi W(z))^2
 \left(|D\Psi_a(q)^Tz|^2-|z|^2\right)\dd z.
\end{equation}
At \(q=Q_a(z)\), direct multiplication gives
\begin{equation}
 |D\Psi_a(q)^Tz|^2-|z|^2
 =-4a(r_a+v)uv+4a^2(r_a+v)^2u^2.
\end{equation}
All terms odd in \(u\) or \(v\) vanish against the radial weight, leaving
\begin{equation}
 I_a-I_0=4a^2\int e^{-tW}(2\pi W)^2
 (r_a^2u^2+u^2v^2)\dd u\dd v>0
 \label{eq:app-positive-carrier-integral}
\end{equation}
for \(a\ne0\).  Polar coordinates, followed by
\(s=\pi|z|^2\) and \(w=2\pi te^s\), use
\begin{equation}
 \int_0^{2\pi}\cos^2\theta\dd\theta=\pi,
 \qquad
 \int_0^{2\pi}\cos^2\theta\sin^2\theta\dd\theta=\frac\pi4
\end{equation}
to give the exact identity
\begin{equation}
 I_a(t)-I_0(t)
 =\frac{2a^2}{t^2}
 \left[A_2(2\pi t)+4\pi r_a^2A_1(2\pi t)\right].
 \label{eq:app-exact-carrier-final}
\end{equation}

Finally, the Gamma moments
\begin{align}
 \int_0^\infty we^{-w}\log w\dd w&=1-\gamma,\\
 \int_0^\infty we^{-w}(\log w)^2\dd w
 &=\frac{\pi^2}{6}-2\gamma+\gamma^2
\end{align}
give the polynomial in \cref{eq:relative-heat-full-asymptotic}.  The
lower-tail correction can also be retained explicitly.  With
\(\lambda=2\pi t\),
\begin{align}
 A_1(\lambda)&=L+1-\gamma+\frac{\lambda^2}{4}+O(\lambda^3),\\
 A_2(\lambda)&=L^2+2(1-\gamma)L
 +\frac{\pi^2}{6}-2\gamma+\gamma^2
 -\frac{\lambda^2}{4}+O(\lambda^3).
\end{align}
Thus, if
\begin{equation}
 \mathcal B_a(t)=A_2(2\pi t)+4\pi r_a^2A_1(2\pi t),
 \qquad
 P_a(L)=L^2+\beta_aL+\kappa_a,
\end{equation}
then
\begin{equation}
 \frac{\mathcal B_a(t)-P_a(L)}{t^2}
 \longrightarrow\pi^2(4\pi r_a^2-1).
 \label{eq:app-carrier-scaled-limit}
\end{equation}
This is the independently derived dashed limit in
\cref{fig:relative-heat-carrier}.  Equations
\eqref{eq:app-relative-heat-remainder-final} and
\eqref{eq:app-exact-carrier-final} complete the proof of
\cref{thm:relative-heat-activity}.
