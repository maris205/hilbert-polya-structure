\section{Frozen source and main theorem}\label{sec:source}

Let
\[
 K=\QQ(\rho)=\QQ(\sqrt{-3}),\qquad \rho^2+\rho+1=0.
\]
For every rational prime \(p>3\) with \(p\equiv1\pmod3\), let
\[
 d_p=[\QQ(\zeta_p)^+:\QQ]=\frac{p-1}{2}.
\]
We retain the split-prime clock \(z_p=p^{-s}\) and the normalized
chronological moments
\[
 c_{p,n}=\frac{C_{p,n}}{d_p}.
\]
The local normalized logarithm is
\[
 \log G_p(z)=-\sum_{n\ge1}\frac{c_{p,n}}{n}z^n,
\]
where the branch is the source germ at \(z=0\).  The global source germ is
\[
 \cG(s)=\prod_{p\equiv1(3)}G_p(p^{-s}).                 \tag{2.1}
\]
The chronological phases, Galois trace, and normalization are frozen before
the present analysis.  In particular, neither an averaged transition matrix
nor a critical shift enters (2.1).

The second moment is governed by the smooth projective genus-four curve
\[
 C:\quad y^3=f(x),\qquad
 f(x)=-\frac{x(\rho^2x-1)}{\rho(x^3+1)}.                \tag{2.2}
\]
At a good split prime put
\[
 a_p=p+1-\#C(\Fp).
\]
The two reductions induced by \(\rho\) and \(\rho^{-1}\) are isomorphic and
have the same trace.  The inherited exact formula is
\[
 c_{p,2}=-\frac{28+4a_p}{p-1}.                         \tag{2.3}
\]

The new fourth-moment phase is
\[
 \Phi_{p,4}(x_0,\ldots,x_7)
 =2\sum_{i=0}^7x_i^3+\sum_{i=0}^6x_ix_{i+1}
   +\rho x_7x_0.                                       \tag{2.4}
\]
It is important that the closing edge remains \(\rho x_7x_0\).
Let \(Z_{p,4}=\#\Phi_{p,4}^{-1}(0)\) and
\[
 C_{p,4}=2p^{-3}Z_{p,4}-2p^4,\qquad
 c_{p,4}=\frac{2C_{p,4}}{p-1}.                         \tag{2.5}
\]

\begin{theorem}[Main theorem]\label{thm:main}
For the frozen H\'enon Euler germ \(\cG\), the following hold.
\begin{enumerate}
\item The automorphisms in \cref{prop:automorphisms} generate a
\(K\)-rational subgroup \(G\simeq C_2\times S_3\), and there are elliptic
curves \(E_\pm/K\) such that
\[
 \Jac(C)\sim_K E_+^2\times E_-^2.                      \tag{2.6}
\]
\item If
\[
 \ell_2(s)=\sum_{p\equiv1(3)}c_{p,2}p^{-2s},\qquad
 F_2(s)=\exp(-\ell_2(s)/2),
\]
then a holomorphic nonvanishing \(H_2\) on \(\RePart s>0\) satisfies
\[
 F_2(s)=\zeta_K(2s+1)^7L(H^1(C/K),2s+1)H_2(s)          \tag{2.7}
\]
initially on \(\RePart s>1/4\), and the right-hand side continues \(F_2\)
holomorphically to \(\RePart s>0\).  This continuation may vanish.
\item Outside a finite set of split primes,
\[
 C_{p,4}=-2-\frac{2A_p}{p^3}-\frac{2B_p}{p^2},          \tag{2.8}
\]
where
\[
 |A_p|\le86p^3,\qquad |B_p|\le168p^{5/2}.
\]
Consequently,
\[
 |c_{p,4}|
 \le\frac{348+672\sqrt p}{p-1}=O(p^{-1/2}).             \tag{2.9}
\]
\item The germ (2.1) has a canonical holomorphic continuation
\[
 \cG^{\mathrm{cont}}(s)\quad\text{to}\quad
 \RePart s>\frac15.                                    \tag{2.10}
\]
It is represented there by the normalized-semifinite graded determinant
in \cref{thm:det10}.  The continuation is not asserted to be nonvanishing.
\end{enumerate}
\end{theorem}

\begin{remark}[Scope]\label{rem:scope}
The theorem does not identify rational primes with primitive periodic orbits
of one real H\'enon map.  It gives no completion or functional equation for
\(\cG^{\mathrm{cont}}\), no Riemann--von Mangoldt law, no Riemann-zero
divisor, and no self-adjoint Hilbert--P\'olya operator.
\end{remark}
