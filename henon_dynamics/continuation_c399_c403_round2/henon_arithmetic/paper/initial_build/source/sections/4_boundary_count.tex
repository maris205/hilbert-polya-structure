\section{Affine reduction and the two boundary lengths}\label{sec:boundary}

\begin{lemma}[Reduced affine intersection]\label{lem:affine}
For every $n,r\geq1$, including $d^n=q^r$, the affine equalizer
$E_H(n,r)$ is finite and reduced. Every affine intersection point of
$\Gamma_g$ and $\Gamma_{\Phi_r}$ has multiplicity one.
\end{lemma}

\begin{proof}
The equalizer is cut out in $\A^2$ by
\begin{equation}\label{eq:affine_equations}
 g_1(x,y)-x^Q=0,\qquad g_2(x,y)-y^Q=0.
\end{equation}
Since $Q$ is a positive power of $p$, the two power terms have zero
differential. The Jacobian of \eqref{eq:affine_equations} is therefore
$Dg$. The Jacobian matrix of $H$ is
\[
 DH(x,y)=\begin{pmatrix}0&1\\-a&f'(y)\end{pmatrix},
\]
whose determinant is $a$. The chain rule gives $\det Dg=a^n\ne0$
at every point. The Jacobian criterion makes the equalizer smooth of
dimension zero at every solution. It is affine and of finite type over
$k$. Its coordinate ring is consequently a zero-dimensional reduced
finite-type $k$-algebra, hence an Artinian algebra and a finite product
of copies of $k$. This proves finiteness and reducedness, with the empty
scheme allowed at this stage. The local ring at each solution has length
one. Restricting the equations of one graph to the other gives precisely
\eqref{eq:affine_equations}, so the same length is the affine local
intersection multiplicity.
\end{proof}

\begin{lemma}[Location of the boundary]\label{lem:location}
The set-theoretic boundary of $\Gamma_g\cap\Gamma_{\Phi_r}$ consists
exactly of $(\Imn,\Imn)$ and $(\Ip,\Ip)$.
\end{lemma}

\begin{proof}
Every point of $\Gamma_{\Phi_r}$ is $(P,\Phi_r(P))$. Its two coordinates
are either both affine or both on $\Linfty$. Suppose $P\in\Linfty$.
If $P\ne\Ip$, Lemmas~\ref{lem:degrees} and~\ref{lem:graphcharts}
force its second coordinate on $\Gamma_g$ to be $\Imn$. In the formula
$[x:y:0]\mapsto[x^Q:y^Q:0]$, the only point above $\Imn$ is
$P=\Imn$. If $P=\Ip$, then $\Phi_r(P)=\Ip$. There are therefore
at most the two asserted boundary intersections.

Both occur. The regular fixed point $g(\Imn)=\Imn$ gives the first.
The regular fixed point $g^{-1}(\Ip)=\Ip$ and the inverse-graph chart
give the second. They also belong to the Frobenius graph, since both
points have coordinates in $\F_q$.
\end{proof}

\begin{lemma}[Length at the regular infinity point]\label{lem:minuslength}
If $D\ne Q$, the local intersection multiplicity at $(\Imn,\Imn)$ is
$Q\min(D,Q)$.
\end{lemma}

\begin{proof}
Use the chart $y\ne0$ in each factor, with local coordinates
$u=x/y$ and $v=z/y$ centered at $\Imn$. From \eqref{eq:homogeneous},
the local expression for $g$ is
\begin{equation}\label{eq:minuschart}
 (u,v)\longmapsto
 \left(\frac{A(u,v)}{C(u,v)},\frac{v^D}{C(u,v)}\right),
 \quad A(u,v)=G_1(u,1,v),\quad C(u,v)=G_2(u,1,v).
\end{equation}
The degree calculation gives
\begin{equation}\label{eq:unitdata}
 A(u,0)=0,\qquad C(u,0)=L_n\ne0.
\end{equation}
Thus $C$ is a unit in the completed local ring $S=k[[u,v]]$. The
Frobenius expression on this chart is $(u,v)\mapsto(u^Q,v^Q)$.
Since $\Gamma_g$ is the smooth graph parameterized by $(u,v)$ here,
the completed local intersection ring is
\begin{equation}\label{eq:localquotient}
 S/(B,v^D-v^Q C),\qquad B=A-u^QC.
\end{equation}
Multiplication by the denominator $C$ has not changed either defining
ideal, because $C$ is invertible.

Put $m=\min(D,Q)$. If $D>Q$, then
\[
 v^D-v^Q C=v^Q(v^{D-Q}-C),
\]
and the second factor has nonzero constant term $-L_n$. If $D<Q$, then
\[
 v^D-v^Q C=v^D(1-v^{Q-D}C),
\]
and the second factor has constant term $1$. Hence in both nonresonant
cases the ideal in \eqref{eq:localquotient} is exactly $(B,v^m)$.
By \eqref{eq:unitdata},
\begin{equation}\label{eq:Bmodv}
 B(u,0)=-L_nu^Q.
\end{equation}

Let $R=S/(B)$. We next verify that $v$ is a non-zero-divisor in $R$.
Suppose $vh=Bb$ in $S$. Reducing modulo $v$ gives
$0=-L_nu^Q b(u,0)$ in the domain $k[[u]]$. Thus $b(u,0)=0$, so
$b=vb_1$ for some $b_1\in S$. Cancelling $v$ in the domain $S$ gives
$h=Bb_1$. This proves the claimed injectivity in $R$.

For each $i\geq0$, multiplication by $v^i$ now induces an isomorphism
\begin{equation}\label{eq:filtrationiso}
 R/vR\longrightarrow v^iR/v^{i+1}R.
\end{equation}
Surjectivity follows from the definition of $v^iR$. For injectivity,
if $v^i h=v^{i+1}b$ in $R$, injectivity of multiplication by $v^i$
gives $h=vb$. Equation~\eqref{eq:Bmodv} identifies
\[
 R/vR\simeq k[[u]]/(u^Q),
\]
which has basis $1,u,\ldots,u^{Q-1}$ and length $Q$. The finite
filtration of $R/v^mR$ by the images of
$R,vR,\ldots,v^{m-1}R$ has $m$ successive quotients of this length.
It follows that
\begin{equation}\label{eq:minuslength}
 \length_k S/(B,v^m)=mQ=Q\min(D,Q).
\end{equation}
In particular the local intersection is isolated. Completion preserves
the length of a finite-length local module, so \eqref{eq:minuslength}
is also the length of the uncompleted local intersection ring.
\end{proof}

\begin{lemma}[Length at the inverse-regular infinity point]\label{lem:pluslength}
For every $n,r\geq1$, the local intersection at $(\Ip,\Ip)$ is
transverse and has length one.
\end{lemma}

\begin{proof}
Parameterize $\Gamma_g$ near $(\Ip,\Ip)$ by the second coordinate,
as $(g^{-1}(Y),Y)$, using Lemma~\ref{lem:graphcharts}. In this chart
the equation for intersection with $\Gamma_{\Phi_r}$ is
\begin{equation}\label{eq:plus_equation}
 Y=\Phi_r(g^{-1}(Y)).
\end{equation}
Choose affine coordinates $s,t$ on $x\ne0$ centered at $\Ip$.
The inverse is regular and fixes the origin in these coordinates.
The morphism $\Phi_r$ is $(s,t)\mapsto(s^Q,t^Q)$ and has zero
differential. Therefore the two entries of the left side of
\eqref{eq:plus_equation} minus the right side have linear terms $s$
and $t$, respectively.

In $k[[s,t]]$, let $J$ be the ideal generated by those two entries
and let $\mathfrak m=(s,t)$. Their linear terms imply
$\mathfrak m=J+\mathfrak m^2$. Applying Nakayama's lemma to the finite
module $\mathfrak m/J$ gives $\mathfrak m/J=0$. Hence
$J=\mathfrak m$ and $k[[s,t]]/J=k$, of length one. The independent
linear terms also prove transversality.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Lemma~\ref{lem:affine} proves affine finiteness and reducedness for all
positive clocks. Assume now $D\ne Q$. Lemma~\ref{lem:location}
exhausts the boundary support, and Lemmas~\ref{lem:minuslength}
and~\ref{lem:pluslength} show that both boundary intersections are
isolated. The two projective surfaces in the smooth fourfold $X$
therefore meet properly, in dimension zero.

The Frobenius graph is isomorphic to $\Pj^2$ and is smooth everywhere.
The graph $\Gamma_g$ is smooth at every intersection point: on the
affine graph this follows from the first projection, and at the two
boundary points it follows from Lemma~\ref{lem:graphcharts}. The local
rings of both surfaces are consequently Cohen--Macaulay. The classical
proper-intersection length formula applies at each point
\cite[Lemma~43.16.1, tag~0B01]{stacks2026intersection}. Thus the lengths
just computed are the local multiplicities in the Chow product.
% ref:stacks2026intersection; anchor:section:43.16-Lemma-1

Subtracting the two boundary contributions from \eqref{eq:total}, and
using that each affine point has length one, gives
\begin{align*}
 M_H(n,r)
 &=1+DQ+Q^2-Q\min(D,Q)-1\\
 &=DQ+Q^2-Q\min(D,Q)\\
 &=\max(DQ,Q^2).
\end{align*}
The same calculation establishes the boundary assertions in the theorem.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:allclocks}]
If $d^n=q^r=p^{er}$ for positive integers $n,r$, every prime divisor of
$d$ must be $p$. Hence $d$ is a power of $p$. Under the corollary's
hypothesis this is impossible, so every pair of clocks satisfies the
nonresonance condition in Theorem~\ref{thm:main}.
\end{proof}
