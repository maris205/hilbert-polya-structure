\begin{abstract}
We solve the Euclidean total-variation gradient flow on every finite
unweighted path.
\ifnum\CRevisionRound=0
Maximal-monotone well-posedness, one-homogeneous dissipation, and a path
Poincar\'e estimate give finite-time consensus at the initial mean.
\fi
\ifnum\CRevisionRound>0
An explicit interpolation of edge flux across each constant block gives a
piecewise-affine coalescence algorithm: blocks never split, simultaneous
collisions merge jointly, and at most $n-1$ event times occur.
\fi
\ifnum\CRevisionRound>1
The no-splitting invariant makes the time-averaged subgradient admissible at
the final state, proving that the flow equals the Rudin--Osher--Fatemi
minimizer for every time.  Exact two-engine evidence audits the finite event
implementation while a strict nonclaim separates this theorem from arithmetic
determinants and Hilbert--P\'olya.
\fi
\end{abstract}

\section{Frozen path flow}
Let $P_n$ be the path with vertices $1,\ldots,n$, where $n\geq1$.  On
$\R^n$ with its Euclidean inner product define
\begin{equation}\label{eq:J}
 (Dx)_i=x_{i+1}-x_i,\qquad
 J(x)=\lVert Dx\rVert_1=\sum_{i=1}^{n-1}|x_{i+1}-x_i|.
\end{equation}
We study the maximal-monotone clock
\begin{equation}\label{eq:flow}
 x'(t)\in-\partial J(x(t)),\qquad x(0)=x^0.
\end{equation}
This freezes unit edge weights and the ordinary vertex norm.  It is neither an
Euler time step nor a continuum limit.  Maximal-monotone semigroups provide
the general existence framework~\cite{Brezis1973}; graph total-variation flow
and its finite-time stabilization are treated in a broader setting by
Maz\'on, Solera, and Toledo~\cite{MST2020}.

For $a\in\R$, let $\Sgn(a)=\{\operatorname{sign}a\}$ if $a\ne0$ and
$\Sgn(0)=[-1,1]$.  The chain rule for~\eqref{eq:J} gives
\begin{equation}\label{eq:subgrad}
 \partial J(x)=\{D^{\mathsf T}z:z_i\in\Sgn((Dx)_i)\}.
\end{equation}
Since $J$ is finite, continuous, and convex, $\partial J$ is maximal monotone;
there is a unique global absolutely continuous solution of~\eqref{eq:flow}.

Write $\bar x=n^{-1}\sum_i x_i^0$ and
$\rho(t)=\lVert x(t)-\bar x\mathbf1\rVert_2$.  Every $D^{\mathsf T}z$ is
orthogonal to $\mathbf1$, so the mean is preserved.  One-homogeneity implies
$\langle g,x\rangle=J(x)$ for $g\in\partial J(x)$.  With $g=-x'$ and the
convex chain rule,
\begin{equation}\label{eq:dissipation}
 \frac{d}{dt}\frac{\rho^2}{2}=-J(x),\qquad
 \frac{d}{dt}J(x)=-\lVert x'\rVert_2^2
\end{equation}
almost everywhere.

\begin{lemma}[Finite consensus]\label{lem:consensus}
The solution reaches $\bar x\mathbf1$ by a time $T$ satisfying
\begin{equation}\label{eq:Tbound}
 T\leq\sqrt n\,\lVert x^0-\bar x\mathbf1\rVert_2,
\end{equation}
and remains there.
\end{lemma}
\begin{proof}
For each $i$, path telescoping and averaging give
\[
 |x_i-\bar x|\leq\frac1n\sum_j|x_i-x_j|\leq J(x).
\]
Hence $\rho\leq\sqrt n J(x)$.  While $\rho>0$, the first identity
in~\eqref{eq:dissipation} yields
$\rho'=-J(x)/\rho\leq-1/\sqrt n$.  Integration proves~\eqref{eq:Tbound}.
At a constant vector the minimal subgradient is zero, so uniqueness keeps the
state constant.
\end{proof}

\section{Exact block dynamics and joint collisions}
A maximal constant block $B=[\ell,r]$ has size $m=r-\ell+1$.  Define its
exterior signs
\[
 s_{\ell-1}=\begin{cases}\operatorname{sign}(x_\ell-x_{\ell-1}),&\ell>1,\\0,&\ell=1,
 \end{cases}\qquad
 s_r=\begin{cases}\operatorname{sign}(x_{r+1}-x_r),&r<n,\\0,&r=n.
 \end{cases}
\]

\begin{lemma}[Plateau flux]\label{lem:flux}
On the zero-jump edges internal to $B$, set
\begin{equation}\label{eq:flux}
 z_{\ell+k-1}=s_{\ell-1}+\frac{k}{m}(s_r-s_{\ell-1}),
 \qquad k=1,\ldots,m-1.
\end{equation}
Together with the exterior signs, this is an admissible subgradient flux and
every coordinate in $B$ has velocity
\begin{equation}\label{eq:velocity}
 v_B=\frac{s_r-s_{\ell-1}}{m}.
\end{equation}
\end{lemma}
\begin{proof}
The interpolation~\eqref{eq:flux} stays in $[-1,1]=\Sgn(0)$, so it is
admissible in~\eqref{eq:subgrad}.  Consecutive flux differences are all
$(s_{\ell-1}-s_r)/m$ on $B$.  Thus $D^{\mathsf T}z$ is constant there and
$-D^{\mathsf T}z$ equals~\eqref{eq:velocity}.  The same interpolation uniquely
minimizes the sum of squared consecutive flux differences with the two
boundary values.  Once the exterior nonzero-edge signs are fixed,
$\lVert D^{\mathsf T}z\rVert_2^2$ is the sum of these contributions over the
maximal plateaus, so the global minimization separates blockwise.  Thus the
local discrete Dirichlet minimizer is exactly the minimal-norm subgradient
selected by the right derivative of the flow.
\end{proof}

\begin{theorem}[Complete path coalescence atlas]\label{thm:atlas}
The solution of~\eqref{eq:flow} is piecewise affine.  Between events each
maximal block moves with~\eqref{eq:velocity}.  Blocks never split.  At an
event, every connected string of adjacent blocks whose affine heights agree is
replaced by its union; disjoint simultaneous collisions are handled at the
same time.  Every event lowers the block count, so there are at most $n-1$
distinct event times, after which consensus holds.
\end{theorem}
\begin{proof}
While adjacent heights are unequal, all exterior signs and block sizes are
fixed.  Lemma~\ref{lem:flux} therefore produces constant velocities and an
affine curve satisfying~\eqref{eq:flow}; uniqueness identifies it with the
semigroup solution.  A current plateau has one common velocity and cannot
split before the first adjacent equality.

At equality, union each maximal connected string of equal heights and apply
Lemma~\ref{lem:flux} to the union.  It again has one velocity; uniqueness
forbids a later split.  Joint union makes the result independent of any
artificial ordering among simultaneous collisions.  Each event removes at
least one block, proving the event bound.  If several blocks remained forever
without another equality, their affine solution would contradict finite
consensus from Lemma~\ref{lem:consensus}; hence the construction closes.
\end{proof}

If current blocks have heights $h_j$ and velocities $v_j$, the next event is
the least positive number among
\[
 \tau_j=\frac{h_{j+1}-h_j}{v_j-v_{j+1}}.
\]
All quantities are rational whenever $x^0$ is rational.  The formula also
covers endpoint blocks through the missing-flux value zero.  For $n=1$ or
constant data there are no events.

\section{The all-time ROF identity}
The denoising model of Rudin, Osher, and Fatemi~\cite{ROF1992} motivates the
finite path resolvent
\begin{equation}\label{eq:rof}
 R_t(x^0)=\underset{y\in\R^n}{\operatorname{argmin}}
 \left\{\frac12\lVert y-x^0\rVert_2^2+tJ(y)\right\}.
\end{equation}
The relationship between graph ROF and graph TV flow requires care: the full
one-dimensional equivalence does not persist on arbitrary graphs without
additional hypotheses~\cite{KSS2019}.  Space-discrete one-dimensional
flow--regularization equivalence was proved by Steidl et al.~\cite{Steidl2004},
and fused-lasso fusion paths have direct algorithmic precedent
\cite{Hoefling2010}.  We claim only the self-contained path certificate below,
not priority for equivalence or monotone fusion.

\begin{theorem}[Flow equals ROF on a path]\label{thm:rof}
For every $t\geq0$ and every $x^0\in\R^n$,
\[
 x(t)=R_t(x^0).
\]
This includes collision times and all post-consensus times.
\end{theorem}
\begin{proof}
Fix $t>0$ and write $-x'(s)=D^{\mathsf T}z(s)$ with
$z_i(s)\in\Sgn((Dx(s))_i)$ almost everywhere.  If $(Dx(t))_i\ne0$, edge $i$
has never entered a merged block: Theorem~\ref{thm:atlas} would otherwise keep
it zero.  Continuity prevents its sign from changing without first vanishing,
so $z_i(s)=\operatorname{sign}((Dx(t))_i)$ almost everywhere on $[0,t]$.
If $(Dx(t))_i=0$, every $z_i(s)$ merely lies in $[-1,1]$.

For $\bar z=t^{-1}\int_0^t z(s)\,ds$, the two cases imply
$\bar z_i\in\Sgn((Dx(t))_i)$.  Integrating~\eqref{eq:flow} gives
\[
 \frac{x^0-x(t)}{t}=D^{\mathsf T}\bar z\in\partial J(x(t)),
\]
or $0\in x(t)-x^0+t\partial J(x(t))$.  This is the KKT condition
for~\eqref{eq:rof}; strict convexity of the quadratic term makes its minimizer
unique.  The case $t=0$ is immediate.
\end{proof}
