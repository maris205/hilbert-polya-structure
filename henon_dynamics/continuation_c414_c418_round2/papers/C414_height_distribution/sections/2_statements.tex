\section{Heights, the exact distribution and the main theorems}\label{sec:statements}
Fix a prime power $q$, an integer $d\ge2$, a polynomial
$f\in\F_q[X]$ of degree exactly $d$, and $a\in\F_q^*$.
Let $R=\F_q[t]$ and
\begin{equation}\label{eq:map}
 H(x,y)=(y,f(y)-a x),\qquad
 H^{-1}(x,y)=(a^{-1}(f(x)-y),x).
\end{equation}
For $P=(x,y)\in R^2$ put $h(P)=\max(0,\deg x,\deg y)$,
with $\deg0=-\infty$. Our canonical heights are in units of
$\log q$:
\begin{equation}\label{eq:height}
 \hh^\pm(P)=\lim_{n\to\infty}d^{-n}h(H^{\pm n}P),\qquad
 \hh(P)=\hh^+(P)+\hh^-(P).
\end{equation}
Their existence follows from the degree patterns proved below.
Define the height Dirichlet series and its counting function by
\begin{equation}\label{eq:def-series}
 Z_H(s)=\sum_{P\in R^2}q^{-s\hh(P)},\qquad
 N_H(B)=\#\{P\in R^2:\hh(P)\le B\}.
\end{equation}
Here $q^z=\exp(z\log q)$ with the positive real logarithm of $q$.

For $M\ge1$, define the integer weight
\begin{equation}\label{eq:turn-weights}
 D_M=\begin{cases}
 (q-1)^2q^{(d+1)r},&M=dr,\quad r\ge1,\\
 (q-1)q^{(d+1)r+j+1},&M=dr+j,\quad r\ge0,\ 1\le j<d.
 \end{cases}
\end{equation}

\begin{theorem}[Coefficient-uniform distribution]\label{thm:distribution}
The limits \eqref{eq:height} exist for every polynomial point and
$h(P)\le\hh(P)$. The full distribution of $\hh$ depends only
on $q,d$, not on the leading coefficient of $f$, its lower
coefficients, or $a$. Its exact series is
\begin{align}\label{eq:positive-series}
 Z_H(s)=q^2
 &+(q-1)^2\sum_{\substack{m,n\ge1\\m<dn,\ n<dm}}
 q^{m+n}\sum_{k\in\Z}q^{-s(d^{-k}m+d^kn)}\notag\\
 &+\sum_{M\ge1}D_M\sum_{k\in\Z}
 q^{-sM(d^{-k}+d^{k-1})}.
\end{align}
This is first an identity of discrete nonnegative counting measures;
it is an identity of analytic series wherever absolutely convergent.
Height zero has multiplicity exactly $q^2$, and every bounded-height
set is finite.
\end{theorem}

\begin{theorem}[Complete poles and meromorphic boundary]\label{thm:poles}
The series $Z_H$ converges absolutely exactly when $\Re s>1$,
and it has a meromorphic continuation to $\Re s>0$. Put
\begin{equation}\label{eq:pole-lattice}
 L_k=d^k+d^{1-k},\qquad
 s_{k,\ell}=\frac{d+1}{L_k}+
 \frac{2\pi i\ell}{L_k\log q},\qquad k\ge1,\quad\ell\in\Z.
\end{equation}
These are exactly its poles in $\Re s>0$. The poles with $k=1$
and $(d+1)\mid\ell$ are double; all others are simple.
At every double pole, the coefficient of $(s-s_{1,\ell})^{-2}$ is
\begin{equation}\label{eq:double-coeff}
 \frac{(q-1)^2(d-1)}{(d+1)(\log q)^2}.
\end{equation}
An explicit nonzero residue at every simple pole is given in
\eqref{eq:residue}. Every point of the imaginary axis is an
accumulation point of poles. The axis is consequently a natural
boundary for meromorphic continuation.
\end{theorem}

\begin{theorem}[Real-height counting]\label{thm:counting}
As $B\to\infty$ through arbitrary real values,
\begin{equation}\label{eq:asymptotic}
 N_H(B)=\frac{d-1}{d+1}(q-1)q^{\lfloor B\rfloor+1}B
       +O_{q,d}(q^B).
\end{equation}
In particular, writing $\{B\}=B-\lfloor B\rfloor$,
\[
 \frac{N_H(B)}{Bq^B}
 =\frac{d-1}{d+1}(q-1)q^{1-\{B\}}+O_{q,d}(B^{-1}).
\]
The leading coefficient is not constant along real height bounds.
\end{theorem}

All three statements concern the entire polynomial-point domain, not
all of $\F_q(t)^2$. Constants form a finite invariant set; every
nonconstant polynomial orbit will be shown to escape in both
directions. Thus \eqref{eq:def-series} is not an ordinary
periodic-orbit zeta function.
