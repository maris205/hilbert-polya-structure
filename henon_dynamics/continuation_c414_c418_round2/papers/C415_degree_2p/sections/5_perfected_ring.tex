\section{A semilinear operator on a perfected polynomial ring}\label{sec:perfection}

We now assume the low-support form
\begin{equation}\label{eq:low-g}
 g(y)=by^{2p}+cy^{p+1}+R_0(y),\qquad \deg R_0\le p,
 \qquad b\ne0,
\end{equation}
with no condition on $c$. The proof will take roots of polynomial
expressions, but only in the following concrete algebraic domain:
\begin{equation}\label{eq:perfect-ring}
 \Rperf=\bigcup_{r\ge0}\F_q[x^{1/p^r},y^{1/p^r}].
\end{equation}
Every element is a finite sum of monomials with nonnegative
exponents in $\Z[1/p]$. No formal infinite series or completed
ring is involved. The monomial representation is unique, and
rational total degree satisfies
\[
 \deg(PQ)=\deg P+\deg Q,\qquad
 \deg(P^{1/p})=\frac1p\deg P
\]
for nonzero $P,Q$. The ordinary ring $R$ embeds injectively.

Let $L$ be absolute Frobenius on $\Rperf$, so $L(P)=P^p$.
This is a ring automorphism, whose inverse acts on coefficients by
\begin{equation}\label{eq:sigma}
 \sigma(d)=d^{1/p}=d^{p^{e-1}}\qquad(d\in\F_q).
\end{equation}
The maps $T,U$ extend uniquely from $R$ to $\Rperf$: evaluate an
ordinary polynomial and then take its unique prescribed root.
They commute with $L$ and $L^{-1}$. In addition,
\[
 \delta(P^p)=(TP)^p-(UP)^p=(\delta P)^p.
\]
Thus $\delta$ also commutes with $L$. Define the additive operator
\begin{equation}\label{eq:D-semi}
 \DD=L^{-1}\delta.
\end{equation}
The two identities that will control both degrees and coefficients are
\begin{equation}\label{eq:semi-identities}
 \delta^j=L^j\DD^j,\qquad
 \DD(dP)=\sigma(d)\DD(P)\quad(d\in\F_q).
\end{equation}
The second is semilinearity, not $\F_q$-linearity. It cannot be
discarded when $a,b,c$ lie outside the prime field. The first is
an identity of additive operators, not a conjugacy of point maps
and not a change of the time variable $n$.

Put
\begin{equation}\label{eq:low-notation}
 Q=q/p,\qquad B=\sigma(b)\ne0,\qquad C=\sigma(c),\qquad
 \varepsilon=(p-1)/p.
\end{equation}
The maps $V=L^{-1}T$ and $W=L^{-1}U$ are ring homomorphisms with
coefficient action $\sigma$, and $\DD=V-W$. Their coordinate
substitutions are
\begin{equation}\label{eq:VW}
 Vx=y^{1/p},\quad Wx=x^Q,\qquad
 Vy=y^Q+v_0,\quad Wy=y^Q,
\end{equation}
where
\begin{equation}\label{eq:v0}
 v_0=By^2+Cy^{1+1/p}+v_1,\qquad \deg v_1\le1.
\end{equation}
Indeed, every term of $R_0$ has degree at most $p$ before taking
a root, while $-ax$ becomes $-\sigma(a)x^{1/p}$. This also shows
explicitly why arbitrary low support and the determinant term
fit into the same tail bound.

\begin{lemma}[Uniform mixed-monomial bound]\label{lem:mixed}
For every $P\in\Rperf$,
\begin{equation}\label{eq:perfect-degree-bound}
 \deg\DD P\le Q\deg P.
\end{equation}
The bound is independent of the denominator level of $P$.
\end{lemma}

\begin{proof}
Take a mixed monomial $x^u y^v$ with $u,v\in\Z[1/p]_{
\ge0}$, and choose a common denominator $p^r$. Its image under
each substitution can be computed as the unique $p^r$th root
of the corresponding ordinary-power expression. Since $Q>2$,
\eqref{eq:VW}--\eqref{eq:v0} give
\[
 \deg V(x^u y^v)=u/p+Qv\le Q(u+v),\qquad
 \deg W(x^u y^v)=Q(u+v).
\]
For a finite sum, cancellations can only decrease degree. The
coefficient automorphism leaves a nonzero scalar of degree zero,
so the same bound holds for $V-W$ and every $P$.
\end{proof}

This estimate is deliberately uniform. The subsequent induction
will produce fractional exponents and mixed $x,y$ terms; it does
not assume that all omitted terms remain univariate or share a
fixed denominator.
