\section{Frozen convention and exact semigroup}
Use the ordered basis $|0\rangle,|1\rangle$, with $|0\rangle$ ground,
\[
 \sigma_z=|1\rangle\langle1|-|0\rangle\langle0|,
 \quad \sigma_-=|0\rangle\langle1|,
 \quad \sigma_+=|1\rangle\langle0|.
\]
For $\gamma_\downarrow,\gamma_\uparrow,\gamma_\phi\geq0$ and
$\omega\in\mathbb R$, consider
\begin{equation}\label{eq:gksl}
 \mathcal L(\rho)=-i[(\omega/2)\sigma_z,\rho]
 +\gamma_\downarrow\D[\sigma_-]\rho
 +\gamma_\uparrow\D[\sigma_+]\rho
 +\frac{\gamma_\phi}{2}(\sigma_z\rho\sigma_z-\rho),
\end{equation}
where $\D[L]\rho=L\rho L^\dagger-\tfrac12\{L^\dagger L,\rho\}$.
The factor $1/2$ in the last term is part of the model: pure dephasing alone
sends $\rho_{01}$ to $\ee^{(-\gamma_\phi+i\omega)t}\rho_{01}$.
Set
\[
 \Gamma_1=\gamma_\downarrow+\gamma_\uparrow,
 \qquad \Gamma_2=\frac{\Gamma_1}{2}+\gamma_\phi.
\]

\begin{theorem}[All-rate solution]\label{thm:flow}
If $\Gamma_1>0$, put
\[
 p=\frac{\gamma_\uparrow}{\Gamma_1},\qquad
 \eta=\ee^{-\Gamma_1t},\qquad
 c=\ee^{(-\Gamma_2+i\omega)t}.
\]
Then, for every density matrix and $t\geq0$,
\begin{equation}\label{eq:solution}
 \rho_{11}(t)=p+\eta(\rho_{11}(0)-p),\qquad
 \rho_{01}(t)=c\rho_{01}(0).
\end{equation}
The Liouvillian is diagonalizable on every parameter face and
\begin{equation}\label{eq:char}
 \det(\lambda I-\mathcal L)=
 \lambda(\lambda+\Gamma_1)\bigl((\lambda+\Gamma_2)^2+\omega^2\bigr).
\end{equation}
\end{theorem}
\begin{proof}
In the matrix-unit coordinate order $(E_{00},E_{01},E_{10},E_{11})$,
direct substitution in~\eqref{eq:gksl} gives
\[
 [\mathcal L]=\begin{pmatrix}
 -\gamma_\uparrow&0&0&\gamma_\downarrow\\
 0&-\Gamma_2+i\omega&0&0\\
 0&0&-\Gamma_2-i\omega&0\\
 \gamma_\uparrow&0&0&-\gamma_\downarrow
 \end{pmatrix}.
\]
The population block and two scalar coherence blocks yield
\eqref{eq:solution} and~\eqref{eq:char}.  They also supply independent
eigenoperators even at accidental eigenvalue collisions.  When $\Gamma_1=0$,
nonnegativity forces $\gamma_\downarrow=\gamma_\uparrow=0$, and the same
block decomposition applies directly.
\end{proof}

\section{Choi geometry and the exact EB time}
Let $|\Omega\rangle=(|00\rangle+|11\rangle)/\sqrt2$ and use the normalized
Choi state $J_t=(I\otimes\Phi_t)(|\Omega\rangle\langle\Omega|)$.  From
Theorem~\ref{thm:flow}, in the order $|00\rangle,|01\rangle,|10\rangle,
|11\rangle$,
\begin{equation}\label{eq:choi}
 J_t=\frac12\begin{pmatrix}
 a&0&0&c\\0&b&0&0\\0&0&d&0\\\bar c&0&0&e
 \end{pmatrix},
\end{equation}
where
\[
 a=1-p(1-\eta),\quad b=p(1-\eta),\quad
 d=(1-p)(1-\eta),\quad e=\eta+p(1-\eta).
\]
Define $q=2\Gamma_2/\Gamma_1\geq1$, so $|c|^2=\eta^q$.

\begin{theorem}[Exact entanglement-breaking threshold]\label{thm:eb}
For $\Gamma_1>0$, $\Phi_t$ is entanglement breaking if and only if
\begin{equation}\label{eq:eb}
 p(1-p)(1-\eta)^2\geq\eta^q.
\end{equation}
If $0<p<1$, there is a unique $\eta_*\in(0,1)$ solving equality, and
\[
 t_{\rm EB}=-\frac{\log\eta_*}{\Gamma_1},\qquad
 \Phi_t\text{ is EB}\Longleftrightarrow t\geq t_{\rm EB}.
\]
If $\gamma_\phi=0$, writing $r=p(1-p)$ gives
\begin{equation}\label{eq:radical}
 \eta_*=\frac{1+2r-\sqrt{1+4r}}{2r};
 \qquad p=\tfrac12:\ \eta_*=3-2\sqrt2.
\end{equation}
For $p=0$ or $p=1$, no finite $t$ is EB, but the limiting constant channel is.
\end{theorem}
\begin{proof}
First,~\eqref{eq:choi} is positive: all diagonal entries are nonnegative and
\[
 ae-|c|^2=p(1-p)(1-\eta)^2+\eta-\eta^q\geq0.
\]
Partial transpose moves $c$ into the middle $2\times2$ block.  Its determinant
is
\[
 \frac{bd-|c|^2}{4}
 =\frac{p(1-p)(1-\eta)^2-\eta^q}{4}.
\]
For a two-qubit state, positivity under partial transpose is equivalent to
separability; a channel is EB exactly when its Choi state is separable.
This proves~\eqref{eq:eb}.  For $r=p(1-p)>0$,
$F(\eta)=r(1-\eta)^2-\eta^q$ obeys
\[
 F(0)=r>0,\quad F(1)=-1,\quad
 F'(\eta)=-2r(1-\eta)-q\eta^{q-1}<0
\]
on $(0,1)$.  Hence the root is unique; $\eta$ decreases with time.  Setting
$q=1$ and taking the root in $(0,1)$ gives~\eqref{eq:radical}.  At $p=0,1$,
the left side of~\eqref{eq:eb} is zero while $\eta^q>0$ at finite time.
\end{proof}

\begin{proposition}[Sharp trace contraction]\label{prop:contract}
For $\Gamma_1>0$,
\[
 \sup_{\rho\ne\sigma}

 =\max\{\ee^{-\Gamma_1t},\ee^{-\Gamma_2t}\}.
