\section{Exact ancient solution and curvature}
We use the convention that a convex curve has positive curvature $\kappa$
with respect to its outward normal $n$, and curve shortening has outward
normal velocity $V_n=-\kappa$.  Put $r=e^t\in(0,1)$ and
\[
 F(x,y,t)=\cos x-r\cosh y,\qquad
 \Gamma_t=\{F=0,\ |x|<\pi/2\}.
\]
\begin{theorem}[Core implicit solution]\label{thm:core}
For every $t<0$, the central component $\Gamma_t$ is a smooth embedded strictly convex closed
curve.  It evolves by $V_n=-\kappa$, where
\begin{equation}\label{eq:kappa}
 \kappa(x,y,t)=\frac{\cos x}{\sqrt{1-e^{2t}}},\qquad
 \frac{e^t}{\sqrt{1-e^{2t}}}\leq\kappa\leq
 \frac1{\sqrt{1-e^{2t}}}.
\end{equation}
It is ancient on $(-\infty,0)$ and contracts to the origin as $t\uparrow0$.
\end{theorem}
\begin{proof}
The central component consists of the two graphs
\[
 y=\pm\arcosh(r^{-1}\cos x),\qquad |x|\leq\arccos r.
\]
On this level,
\begin{equation}\label{eq:cancellation}
 |\nabla F|^2=\sin^2x+r^2\sinh^2y=1-r^2>0,
 \qquad D^2F=-\cos x\,I.
\end{equation}
Thus the level is smooth and compact.  For
$n=-\nabla F/|\nabla F|$, the implicit curvature formula and
\eqref{eq:cancellation} give~\eqref{eq:kappa}, which is positive because
$r\leq\cos x\leq1$.  Differentiating $F=0$ along a moving point yields
$V_n=F_t/|\nabla F|=-\cos x/\sqrt{1-r^2}=-\kappa$.
The displayed graphs shrink to $(0,0)$ as $r\uparrow1$ and exist for every
$r>0$, proving the temporal statements and the sharp extrema.
The unrestricted zero set of $F$ is the disjoint union
$\bigsqcup_{k\in\mathbb Z}(\Gamma_t+(2\pi k,0))$; it is not one compact
curve.
\end{proof}

\section{Global geometry, arrival time, and two asymptotic regimes}
Write $K(k)=\int_0^{\pi/2}(1-k^2\sin^2\phi)^{-1/2}\,d\phi$; here $k$ is
the elliptic \emph{modulus}.
\begin{theorem}[Complete geometry and limits]\label{thm:global}
The horizontal width $W$, vertical height $H$, enclosed area $A$ and length
$L$ are
\begin{align}
 W(t)&=2\arccos(e^t),& H(t)&=2\arcosh(e^{-t}),\label{eq:spans}\\
 A(t)&=-2\pi t,&
 L(t)&=4\sqrt{1-e^{2t}}\,
 K\!\left(\sqrt{1-e^{2t}}\right).\label{eq:area-length}
\end{align}
The negative-time curves foliate
$(-\pi/2,\pi/2)\times\mathbb R\setminus\{(0,0)\}$ exactly once.  Their
arrival time
\begin{equation}\label{eq:arrival}
 T(x,y)=\log\cos x-\log\cosh y
\end{equation}
satisfies, away from the origin,
\begin{equation}\label{eq:arrival-pde}
 -|\nabla T|\,\diver\!\left(\frac{\nabla T}{|\nabla T|}\right)=1.
\end{equation}
Moreover,
\[
 (-2t)^{-1/2}\Gamma_t\longrightarrow S^1
 \quad\hbox{smoothly as }t\uparrow0.
\]
If $y_+(x,t)=\arcosh(e^{-t}\cos x)$ and
$h(t)=\arcosh(e^{-t})$, then
\begin{equation}\label{eq:grim}
 y_+(x,t)-h(t)\longrightarrow\log\cos x
\end{equation}
smoothly on compact subsets of $(-\pi/2,\pi/2)$ as $t\to-\infty$; the
lower tip has the reflected limit.
\end{theorem}
\begin{proof}
The extrema of the two graphs give~\eqref{eq:spans}.  On the upper graph,
\[
 y_x=-\frac{\sin x}{\sqrt{\cos^2x-r^2}},\qquad
 ds=\frac{\sqrt{1-r^2}}{\sqrt{\cos^2x-r^2}}\,dx.
\]
The substitution $\sin x=\sqrt{1-r^2}\sin\phi$, followed by the two graph
and two half-curve symmetries, proves the length formula.  Directly
differentiating the area integral (the endpoint value is zero) gives
\[
 A'(t)=-4\int_0^{\arccos r}
 \frac{\cos x}{\sqrt{\cos^2x-r^2}}\,dx=-2\pi.
\]
Since $A(t)\to0$ as $t\uparrow0$, the area formula follows.

On the open strip, $T\leq0$, with equality only at the origin, and $T=t$
is equivalent to the oval equation.  This proves the precise punctured-strip
foliation.  Setting $a=-\tan x$, $b=-\tanh y$ in the divergence of
$(a,b)/\sqrt{a^2+b^2}$ reduces the left side of~\eqref{eq:arrival-pde} to
\[
 \frac{\sec^2x\,\tanh^2y+\operatorname{sech}^2y\,\tan^2x}
 {\tan^2x+\tanh^2y}=1.
\]

In tangent-angle coordinates (up to an angle shift),
\begin{equation}\label{eq:pressure}
 \kappa^2(\theta,t)=\frac1{1-e^{2t}}-\sin^2\theta.
\end{equation}
After scaling, $(-2t)\kappa^2\to1$ with every angular derivative.
Symmetry fixes translation, so the fundamental theorem for strictly convex
plane curves gives the smooth circular limit.  Finally,
$\arcosh z=\log(2z)+O(z^{-2})$ with differentiated uniformity away from the
strip edges.  Apply it to $z=e^{-t}\cos x$ and to $z=e^{-t}$ to obtain
\eqref{eq:grim}; reflection gives the lower statement.
\end{proof}
