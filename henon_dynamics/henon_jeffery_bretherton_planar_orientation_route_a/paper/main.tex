\begin{abstract}
We classify a passive axisymmetric spheroid in every steady incompressible
planar linear flow.  The nonlinear head--tail director equation on $\RP^2$ is
the projectivization of one traceless matrix exponential.  A scalar
discriminant then yields the complete elliptic, hyperbolic, nilpotent, and
identity atlas.
\ifnum\CRevisionRound>0
We prove the full source--saddle--sink stratification, the nilpotent fixed
projective line, and the true head--tail versus oriented simple-shear periods.
\fi
\ifnum\CRevisionRound>1
Every stroboscopic fixed set and every sphere, rod, disk, and zero-flow boundary
is included.
\fi
The result is source-local: clean periodic continua supply no arithmetic
primitive-orbit owner or target determinant.
\end{abstract}

\section{Frozen director and its linear lift}
Let
\[
 L=\begin{pmatrix}a&b&0\\c&-a&0\\0&0&0\end{pmatrix},\qquad
 E=\tfrac12(L+L^{T}),\quad W=\tfrac12(L-L^{T}).
\]
For a spheroid of finite aspect ratio $r>0$, put
\begin{equation}
 \lambda=\frac{r^2-1}{r^2+1}\in(-1,1). \tag{1}
\end{equation}
The head--tail director is $[p]\in\RP^2$.  A unit representative obeys
\begin{equation}
 \dot p=Wp+\lambda\{Ep-(p^TEp)p\}. \tag{2}
\end{equation}
For $r\ne1$, $[p]$ is the intrinsic symmetry-axis director.  An unmarked
sphere ($r=1$) has no intrinsic shape director; whenever $r=1$ appears below,
$[p]$ denotes a marked material director carried by the sphere.
Jeffery~\cite{Jeffery1922} is cited only for the isolated-ellipsoid source
equation, and Bretherton~\cite{Bretherton1962} only for rigid-particle and
spheroidal-parameter lineage.  No classification or proof below is outsourced.

Set $B=W+\lambda E$.  Because $p^TWp=0$, (2) is
$\dot p=Bp-(p^TBp)p$.

\begin{lemma}[Exact projective lift]
For every nonzero $q_0\in\mathbb R^3$,
\begin{equation}
 [p(t)]=[e^{tB}q_0],\qquad
 p(t)=\frac{e^{tB}q_0}{\lVert e^{tB}q_0\rVert}. \tag{3}
\end{equation}
Conversely every solution of (2) has this form, so the flow is global on
$\RP^2$.
\end{lemma}
\begin{proof}
If $q'=Bq$ and $p=q/\lVert q\rVert$, differentiation gives
$p'=Bp-(p^TBp)p$.  Conversely, multiplying a unit solution by
$\exp\int_0^t p^TBp\,ds$ gives a solution of $q'=Bq$.  Matrix exponentials are
invertible, so normalization never vanishes.  Passing to $[p]$ removes the
representative sign exactly.
\end{proof}

Write $k=(b-c)/2$ and $s=(b+c)/2$.  The active block and its invariant are
\begin{equation}
 B_2=\begin{pmatrix}\lambda a&k+\lambda s\\-k+\lambda s&-\lambda a\end{pmatrix},
 \qquad
 \delta=\lambda^2(a^2+s^2)-k^2. \tag{4}
\end{equation}
Direct multiplication gives $B_2^2=\delta I$ and $\delta=-\det B_2$.

\section{The complete sign atlas}
\begin{theorem}[All projective orbit types]
The sign and rank in (4) give exactly four cases.

\begin{enumerate}
\item If $\delta<0$ and $\omega=\sqrt{-\delta}$, the vertical director is
the only fixed point.  Every equatorial director has least period $\pi/\omega$;
every director with nonzero horizontal and vertical parts has least period
$2\pi/\omega$.
\item If $\delta>0$ and $\rho=\sqrt\delta$, there are exactly three fixed
eigen-directors $[v_+]$, $[e_0]$, $[v_-]$ with exponents $\rho,0,-\rho$.
Writing $q=q_+v_++q_0e_0+q_-v_-$, its forward limit is $[v_+]$ if
$q_+\ne0$, $[e_0]$ if $q_+=0,q_0\ne0$, and $[v_-]$ otherwise; backward time
reverses $+$ and $-$.  Thus they are sink, saddle, and source.  Precisely,
\[
 W^s([e_0])=\mathbb P\Span(e_0,v_-)\setminus\{[v_-]\},\qquad
 W^u([e_0])=\mathbb P\Span(e_0,v_+)\setminus\{[v_+]\}.
\]
Their closures are the two invariant $\RP^1$ projective lines.
\ifnum\CRevisionRound>0
\item If $\delta=0$ but $B_2\ne0$, the fixed set is the entire projective line
$\mathbb P(\ker B)$.  Every other trajectory converges as $t\to\pm\infty$ to
the unique horizontal line $\mathbb P(\operatorname{im}B_2)
=\mathbb P(\ker B_2)$, at algebraic rate $|t|^{-1}$.
\item If $B_2=0$, the flow is the identity on all of $\RP^2$.
\fi
\end{enumerate}
\end{theorem}

\begin{proof}
Cayley--Hamilton gives, respectively,
\[
 e^{tB_2}=\cos(\omega t)I+\frac{\sin(\omega t)}{\omega}B_2,\qquad
 e^{tB_2}=\cosh(\rho t)I+\frac{\sinh(\rho t)}{\rho}B_2. \tag{5}
\]
In the first case there is no real horizontal eigenline.  At $\pi/\omega$ the
horizontal block is $-I$, which returns every equatorial head--tail line but
not a mixed line; at $2\pi/\omega$ the full matrix is $I$.  No earlier return
is possible because a non-scalar elliptic block has no real eigenline.

In the second case, exponentiating the three eigen-coordinates gives the stated
limits.  Within $\mathbb P\Span(e_0,v_-)$, every point except $[v_-]$ tends
forward to $[e_0]$; within $\mathbb P\Span(e_0,v_+)$, every point except
$[v_+]$ tends backward to $[e_0]$.  These are projective lines, and a point
outside them has both corresponding leading coordinates, proving the
stable/unstable completeness.  Coordinate ratios give gaps $\rho$ or $2\rho$.
\ifnum\CRevisionRound>0
At $\delta=0$, $e^{tB}=I+tB$.  A nonzero nilpotent $2$ by $2$ block has rank
one and image equal to kernel.  Hence $\mathbb P(\ker B)$ is fixed, whereas
$[q+tBq]\to[Bq]$ off that line.  Finally $B_2=0$ and the zero vertical block
give $B=0$.
\fi
\end{proof}

\section{Simple shear and singular shape faces}
For $L_2=\left(\begin{smallmatrix}0&\dot\gamma\\0&0\end{smallmatrix}\right)$
with $\dot\gamma\ne0$,
substitution of (1) in (4) yields
\begin{equation}
 \delta=-\frac{\dot\gamma^2r^2}{(r^2+1)^2},\qquad
 \omega=\frac{|\dot\gamma|r}{r^2+1}. \tag{6}
\end{equation}
Thus the physical head--tail equatorial period and the oriented-vector period
are
\begin{equation}
 T_{\RP^1}=\frac{\pi(r+r^{-1})}{|\dot\gamma|},\qquad
 T_{\rm oriented}=\frac{2\pi(r+r^{-1})}{|\dot\gamma|}. \tag{7}
\end{equation}
The second value is the least period of every nonvertical oriented vector and
also of a mixed $\RP^2$ director; the vertical oriented vector is fixed.  This
factor two is a state-space fact, not a convention change.

At $r=1$, $\lambda=0$ and strain drops out: under the marked-material-director
convention above, the marked direction follows only vorticity.  The unmarked
sphere itself has no observable orientation state.
As $r\to\infty$ or $r\to0$, $\lambda\to\pm1$, the simple-shear generator is
nilpotent and (7) diverges.  These rod/disk limits are not finite-aspect
periodic points.  At $\dot\gamma=0$, every director is fixed.

\section{Every elliptic strobe}
The first formula in (5) also resolves the return map without sampling.

\begin{theorem}[Stroboscopic fixed sets]
For $\delta<0$ and $\tau>0$,
\[
 \Fix_{\RP^2}(e^{\tau B})=
 \begin{cases}
 \{[e_0]\},&\omega\tau\notin\pi\mathbb Z,\\
 \RP^1_{\rm eq}\cup\{[e_0]\},&\omega\tau\in(2\mathbb Z+1)\pi,\\
 \RP^2,&\omega\tau\in2\pi\mathbb Z.
 \end{cases} \tag{8}
\]
\end{theorem}
\begin{proof}
Away from integer half-periods the horizontal block has no real eigenline, so
only the vertical eigenline fixes.  At odd half-period it is $-I$ while the
vertical multiplier is $+1$, fixing the two separate eigenspaces but no mixed
line.  At a full period every multiplier is one.
\end{proof}
