\section{The surface, its line field, and the main theorem}
\label{sec:setup-theorem}

\subsection{The frozen cubic and its two fields}

In homogeneous coordinates $(u_0,u_1,u_2,u_3)$, let
$Y=V(F)\subset\PP^3_\Q$, where
\begin{align}
F={}&75081586157u_0^3-28576620789u_0^2u_1
-122000922135u_0^2u_2-5364921951u_0^2u_3\notag\\
&+164150208636u_0u_1^2-415458334296u_0u_1u_2
+151070718312u_0u_1u_3\notag\\
&+1158143874300u_0u_2^2+114691988016u_0u_2u_3
+113572676646u_0u_3^2\notag\\
&+6898957820u_1^3+1132596902196u_1^2u_2
-30413540316u_1^2u_3\notag\\
&-2054867641020u_1u_2^2+151980984216u_1u_2u_3
+36794420832u_1u_3^2\notag\\
&+2646295985484u_2^3+560186573940u_2^2u_3
+706181383584u_2u_3^2+1884468968u_3^3.
\label{eq:frozen-cubic}
\end{align}
The coefficients have content one and the leading coefficient is positive,
so \cref{eq:frozen-cubic} fixes the scalar convention.  Exact Jacobian
elimination shows that $Y$ is smooth.  On the standard Grassmann chart
$U_{01}$, a line is the row span of
\[
 \begin{pmatrix}1&0&a&b\\0&1&c&d\end{pmatrix}.
\]
Restricting $F$ to this line gives four equations.  Their quotient is a field
of degree $27$, presented by a primitive irreducible eliminant
$g(d)\in\Z[d]$.  We write
\[
 E=\Q[d]/(g),\qquad K=\text{the normal closure of }E.
\]
The full coefficient vector of $g$, the rational back-substitutions for
$a,b,c$, and the oriented field identity used by the maximal-order
calculation are bound in the exact input record described in
\cref{app:local}.  Printing the several-page coefficient vector here would
add no local argument; \cref{eq:frozen-cubic} and the exact chart
construction specify the geometric object.

The line-incidence pairing embeds the Galois action into the Weyl group of
the $E_6$ root lattice.  An irreducibility certificate for $g$, a modular
order-five class, and an exact Picard-lattice enumeration give the frozen
input
\begin{equation}
 [E:\Q]=27,\qquad \Gal(K/\Q)=\WE,\qquad |\WE|=51840.
\label{eq:frozen-galois-input}
\end{equation}
This paper does not recompute the group from ramification; instead, it
determines the ramification inside the already identified group.  The
rational permutation module of the lines decomposes as
\begin{equation}
 \Q[27]=\one\oplus V_6\oplus V_{20}.
\label{eq:perm-decomposition}
\end{equation}
The $27$-point carrier is where the two nontrivial representations live.  A
second permutation carrier, the $36$ double-sixes on $Y_{\Qbar}$, separates
subgroup classes that have the same line-orbit data.

\subsection{Ramification notation}

Fix a finite prime $p$, an embedding
$K\hookrightarrow\overline{\Q}_p$, a decomposition group
$D_p\leq\WE$, and inertia $I_p\trianglelefteq D_p$.  We use the lower
numbering
\[
 I_0=I_p\supseteq I_1\supseteq I_2\supseteq\cdots.
\]
For a rational representation $V$ of $\WE$, put
\begin{align}
 \Sw_p(V)&=\sum_{i\geq1}\frac{|I_i|}{|I_0|}
       \codim V^{I_i},\label{eq:swan-definition}\\
 a_p(V)&=\codim V^{I_0}+\Sw_p(V).
\label{eq:artin-definition}
\end{align}
These formulas use inertia only.  In contrast, a bad Euler polynomial would
require decomposition data beyond the filtered groups determined below.

Set
\begin{equation}
\begin{split}
 q&=14932047182473291995860108491583652133938007263719,\\
 A&=181\cdot997\cdot2346241=423395612137,\\
 B&=283\cdot1801\cdot q.
\end{split}
\label{eq:qAB}
\end{equation}
Table-of-Marks labels always refer to the $350$ subgroup classes in the GAP
table of marks for \texttt{U4(2).2}.  They are reproducible identifiers
rather than intrinsic group names.

\subsection{Main theorem}

\begin{theorem}[Complete filtered inertia and global closure]
\label{thm:main}
For the cubic surface \cref{eq:frozen-cubic}, the degree-$27$ line field
$E$, and its normal closure $K$, the following statements hold.

\begin{enumerate}[label=\textup{(\alph*)},leftmargin=2.2em]
\item The nine-prime surface divided-discriminant envelope is
\[
 \cB_Y=\{2,3,5,181,283,997,1801,2346241,q\},
\]
whereas the finite ramified support of both $E$ and $K$ is exactly
\[
 \cR_{E,K}=\{3,5,181,283,997,1801,2346241,q\}.
\]
In the displayed order on $\cB_Y$,
\[
 \bigl(v_p(\Disc E)\bigr)_p=(0,46,36,18,6,18,6,18,6).
\]

\item At $p=3$, inertia is subgroup \Tom{140}, with
\[
 I_0\cong(C_3^2):C_2,\quad I_1=C_3^2,\quad
 I_2=\cdots=I_7=C_3,\quad I_8=1,
\]
and the deep $C_3$ is \Tom{7}.  The two and only two surviving ordered
pairs are
\[
 (D,I)=(140,140)\quad\text{or}\quad(206,140).
\]
Thus $|D_3|\in\{18,36\}$ is unresolved, and \Tom{206} is a decomposition
overgroup only.  At $p=5$,
\[
 D=I_0=\Tom{147}\cong C_5:C_4,\qquad
 I_1=I_2=I_3=C_5,\qquad I_4=1.
\]
At $181,997,2346241$, inertia is tame
$\Tom{6}\cong C_3$.  At $283,1801,q$, inertia is the tame root-reflection
subgroup $\Tom{2}\cong C_2$.

\item The local Swan and Artin pairs are
\[
\begin{array}{c|cc}
p&\Sw_p(V_6,V_{20})&a_p(V_6,V_{20})\\ \hline
3&(5,18)&(11,35)\\
5&(3,12)&(7,29)\\
181,997,2346241&(0,0)&(6,12)\\
283,1801,q&(0,0)&(1,5).
\end{array}
\]
Consequently
\[
 N(V_6)=3^{11}5^7A^6B,\qquad
 N(V_{20})=3^{35}5^{29}A^{12}B^5.
\]

\item The field discriminants are
\[
 \Disc E=3^{46}5^{36}A^{18}B^6,\qquad
 \Disc K=3^{106560}5^{80352}A^{34560}B^{25920},
\]
and $N(V_6)N(V_{20})=\Disc E$.

\item The signature of $E$ is $(r_1,r_2)=(3,12)$.  The subgroup generated by
complex conjugation is the Table-of-Marks subgroup \Tom{5}; separately,
complex conjugation belongs to character-table element-class index $17$ in
\texttt{CharacterTable("U4(2).2")} under CTblLib $1.3.1$.  The class has
size $540$ and centralizer order $96$, and
\[
 V_6:(d^+,d^-)=(3,3),\qquad
 V_{20}:(d^+,d^-)=(11,9).
\]
\end{enumerate}
\end{theorem}

The proof is local-to-global.  \Cref{prop:support,prop:theta-authority}
supply exact local arithmetic and the two action carriers.
\Cref{prop:p3-hits,prop:p3-deep,prop:p3-serre} prove the wild $3$-adic
classification, while
\cref{prop:p5-pair,prop:p5-filtration,prop:tame-c3,prop:reflection} treat
the remaining ramified primes.  The character calculation and both
discriminants appear in \cref{sec:conductors}; infinity is handled in
\cref{sec:infinity-scope}.

\begin{remark}[Decomposition is not inertia]
\label{rem:d3-boundary}
The notation $(D,I)$ is essential.  The alternatives are $(140,140)$ and
$(206,140)$, and neither alternative makes ToM $206$ an inertia group.
Every invariant in \cref{thm:main} depends on the common filtered inertia.
No argument below selects between the two decomposition groups.
\end{remark}

\begin{remark}[Exact scope firewall]
\label{rem:main-firewall}
\NoEuler: the theorem determines no decomposition Frobenius, bad Euler
polynomial or factor, local epsilon factor, local or global root number,
Artin holomorphy, automorphy, analytic continuation, or functional equation.
Even a later resolution of $D_3$ would not by itself establish any of these
independent assertions.
\end{remark}
