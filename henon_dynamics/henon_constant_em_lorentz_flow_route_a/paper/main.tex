\section{Convention and invariant factorization}
Use $\eta=\operatorname{diag}(1,-1,-1,-1)$ and $\kappa=q/m$.  We freeze
\begin{equation}
 A=\kappa\begin{pmatrix}
0&E_x&E_y&E_z\\ E_x&0&B_z&-B_y\\ E_y&-B_z&0&B_x\\ E_z&B_y&-B_x&0
\end{pmatrix}.                                                   \tag{1}
\end{equation}
Thus $A^T\eta+\eta A=0$, and for $u=(u^0,u^0\mathbf v)$ the spatial
part of $u'=Au$ is
$\mathbf u'=\kappa u^0(\mathbf E+\mathbf v\times\mathbf B)$.
This fixes both electric and magnetic signs.  The prime denotes proper time
$\tau$, and $x'=u$.  Exact constant-field Lorentz evolution has a long
lineage; see Chin~\cite{Chin}.  We make no priority claim.

\begin{theorem}[invariant scales and planes]
There are unique $a,b\ge0$ such that
\begin{align}
 \chi_A(z)&=(z^2-a^2)(z^2+b^2),                              \tag{2}\\
 a^2-b^2&=\kappa^2(|\mathbf E|^2-|\mathbf B|^2),\qquad
 a^2b^2=\kappa^4(\mathbf E\!\cdot\!\mathbf B)^2.            \tag{3}
\end{align}
If $D=a^2+b^2>0$, then
\begin{equation}
 P_h=\frac{A^2+b^2I}{D},\qquad P_r=\frac{-A^2+a^2I}{D}        \tag{4}
\end{equation}
are complementary projectors, with $A^2P_h=a^2P_h$ and
$A^2P_r=-b^2P_r$.
\end{theorem}
\begin{proof}
A direct determinant of (1) gives
$z^4+\kappa^2(|\mathbf B|^2-|\mathbf E|^2)z^2
-\kappa^4(\mathbf E\cdot\mathbf B)^2$; solving its quadratic in $z^2$
gives (2)--(3).  Cayley--Hamilton then verifies every identity in (4).
\end{proof}

\section{Exact velocity and position}
Write all quotients below by analytic continuation at $a=0$ or $b=0$.
\begin{theorem}[closed flow]
For $D>0$,
\begin{align}
 e^{\tau A}={}&P_h\left(\cosh(a\tau)I+\frac{\sinh(a\tau)}aA\right)
 +P_r\left(\cos(b\tau)I+\frac{\sin(b\tau)}bA\right),       \tag{5}\\
 \Phi(\tau)={}&P_h\left(\frac{\sinh(a\tau)}aI+
 \frac{\cosh(a\tau)-1}{a^2}A\right)\nonumber\\
 &+P_r\left(\frac{\sin(b\tau)}bI+
 \frac{1-\cos(b\tau)}{b^2}A\right).                       \tag{6}
\end{align}
The solution is $u(\tau)=e^{\tau A}u_0$ and
$x(\tau)=x_0+\Phi(\tau)u_0$.  Moreover
$\eta(u,u)=\eta(u_0,u_0)$ and $\det e^{\tau A}=1$.
\end{theorem}
\begin{proof}
Functional calculus on (4) gives (5), and direct integration gives (6).
Since $A^T\eta+\eta A=0$, differentiation of
$e^{\tau A^T}\eta e^{\tau A}$ gives zero.  Finally
$\det e^{\tau A}=e^{\tau\operatorname{tr}A}=1$.
\end{proof}

\section{Period, nonclosure, and boundary atlas}
On $P_h$ the flow has real exponents $\pm a$; on $P_r$ it rotates with
frequency $b$.  Hence a nonconstant $u(\tau)$ is periodic exactly when
\begin{equation}
 b>0,\quad P_ru_0\ne0,\quad\text{and}\quad(a=0\ \text{or}\ P_hu_0=0), \tag{7}
\end{equation}
and its least period is $2\pi/b$.  Notice the magnetic face: when $a=0<b$,
$P_hu_0$ is constant and does not obstruct periodicity.  When $a>0$, a
nonzero hyperbolic component does.  If $b=0$, no nonconstant velocity is

For a future-timelike physical initial velocity, the flow in (5) stays in the
identity component of the Lorentz group.  Thus $u^0(\tau)>0$, so
