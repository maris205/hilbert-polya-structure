\section{One normalization from matrix to chamber}

Let $\mathrm{Herm}_N$ carry the real trace inner product
$\langle A,B\rangle=\Tr(AB)$.  Standard Brownian motion $\mathcal B_t$ in
this Euclidean space drives
\begin{equation}
 \dd H_t=\dd\mathcal B_t-\frac12H_t\dd t.       \label{eq:matrix-ou}
\end{equation}
The trace-orthonormal basis consists of $E_{ii}$ and, for $i<j$,
\begin{equation}
 \frac{E_{ij}+E_{ji}}{\sqrt2},\qquad
 \frac{\mathrm i(E_{ij}-E_{ji})}{\sqrt2}.      \label{eq:trace-basis}
\end{equation}
Thus every diagonal matrix martingale in an instantaneous eigenbasis has
quadratic variation $\dd t$, while every off-diagonal complex entry has
expected squared modulus $\dd t$.  This sentence fixes the factor that is
often hidden in matrix-entry conventions.

Write
\begin{equation}
 W_N=\{x\in\RR^N:x_1<\cdots<x_N\},\qquad
 h(x)=\prod_{i<j}(x_j-x_i),\qquad d=\frac{N(N-1)}2.   \label{eq:chamber}
\end{equation}

\begin{theorem}[Radialization in the trace convention]
If $H_0$ has simple spectrum, then, up to their first collision, its
increasingly ordered eigenvalues solve
\begin{equation}
 \dd X_i=\dd B_i+\sum_{j\ne i}\frac{\dd t}{X_i-X_j}
             -\frac{X_i}{2}\dd t,                  \label{eq:dyson-sde}
\end{equation}
where the $B_i$ are independent standard real Brownian motions.
\ifnum\CRevisionRound>0
The collision time is infinite.
\fi
The generator on $W_N$ is
\begin{equation}
 \mathcal L=\frac12\sum_i\partial_i^2+
 \sum_i\left(\sum_{j\ne i}\frac1{x_i-x_j}-\frac{x_i}{2}\right)\partial_i.
                                                        \label{eq:generator}
\end{equation}
\end{theorem}

\begin{proof}
For a normalized eigenvector $u_i$, second-order Hermitian perturbation gives
\begin{equation}
 \dd X_i=\langle u_i,\dd H\,u_i\rangle+
 \sum_{j\ne i}\frac{|\langle u_j,\dd\mathcal B\,u_i\rangle|^2}{X_i-X_j}.
                                                        \label{eq:perturb}
\end{equation}
The covariance statement following \eqref{eq:trace-basis} turns the first
martingale term into $\dd B_i$, the second-order term into the displayed
reciprocal gaps, and the matrix drift into $-X_i\dd t/2$.  This proves the
formula until the first collision.
\ifnum\CRevisionRound>0
Section~\ref{sec:doob} proves that this time is infinite.
\fi
\end{proof}

\begin{theorem}[Ordered GUE reversible law]
The probability density
\begin{equation}
 \pi_N(x)=Z_N^{-1}\e^{-|x|^2/2}h(x)^2\,1_{W_N}(x),
 \qquad Z_N=(2\pi)^{N/2}\prod_{j=0}^{N-1}j!,          \label{eq:gue}
\end{equation}
is reversible for \eqref{eq:generator}.
\end{theorem}

\begin{proof}
The drift is one half of the logarithmic gradient:
\begin{equation}
 \frac12\partial_i\log(\e^{-|x|^2/2}h^2)
 =-\frac{x_i}{2}+\sum_{j\ne i}\frac1{x_i-x_j}.       \label{eq:log-gradient}
\end{equation}
Hence $\mathcal Lf=(2\pi_N)^{-1}\nabla\!\cdot(\pi_N\nabla f)$, where
$\pi_N$ denotes its density, proving symmetry.  For the normalizer, use
$h=\det[\operatorname{He}_{i-1}(x_j)]$ and scalar Hermite orthogonality in
Andr\'eief's identity.  The integral over $\RR^N$ is
$N!(2\pi)^{N/2}\prod_{j=0}^{N-1}j!$.  The integrand is symmetric, so the
ordered chamber contributes $1/N!$ of that value.  The invariant matrix
density is proportional to $\e^{-\Tr H^2/2}$; Weyl integration gives the same
ordered density.
\end{proof}

The $N=1$ face is ordinary scalar Ornstein--Uhlenbeck diffusion.  No
large-$N$ passage or entrance from a multiple eigenvalue is included.

\section{Killed determinant and conservative Doob transform}
\label{sec:doob}

The independent scalar generator is
\begin{equation}
 \mathcal L_0=\frac12\sum_i(\partial_i^2-x_i\partial_i).
                                                        \label{eq:l0}
\end{equation}
For $t>0$, put $r=\e^{-t/2}$ and $\sigma_t^2=1-r^2$.  Its Lebesgue density is
\begin{equation}
 p_t(a,b)=\frac1{\sqrt{2\pi\sigma_t^2}}
 \exp[-\frac{(b-ra)^2}{2\sigma_t^2}].                 \label{eq:mehler}
\end{equation}

\begin{lemma}[Vandermonde energy]
The polynomial $h$ is harmonic and homogeneous of degree $d$.  Consequently,
\begin{equation}
 \mathcal L_0h=-\frac d2h.                             \label{eq:h-energy}
\end{equation}
\end{lemma}

\begin{proof}
The permutation-invariant Laplacian sends the alternating polynomial $h$ to
an alternating polynomial of degree $d-2$.  Every nonzero alternating
polynomial is divisible by $h$, which already has degree $d$; hence
$\Delta h=0$.  Euler's homogeneous identity gives $x\cdot\nabla h=dh$.
Substitution in \eqref{eq:l0} proves \eqref{eq:h-energy}.
\end{proof}

\begin{theorem}[Exact chamber kernel and boundary]
The independent process killed at its first collision has density
\begin{equation}
 q_t(x,y)=\det[p_t(x_i,y_j)]_{i,j=1}^N.                \label{eq:km}
\end{equation}
Its Vandermonde mass is
\begin{equation}
 \int_{W_N}q_t(x,y)h(y)\dd y=r^dh(x).                 \label{eq:h-mass}
\end{equation}
Therefore
\begin{equation}
 k_t(x,y)=r^{-d}\frac{h(y)}{h(x)}q_t(x,y)              \label{eq:doob}
\end{equation}
is a conservative Markov density with generator \eqref{eq:generator}.
Starting from any $x\in W_N$, its continuous diffusion never reaches a
collision wall in finite time.
\end{theorem}

\begin{proof}
The reflection principle pairs every wall-hitting independent path with the
path reflected after its first hit.  The signed permutation sum that remains
is the Karlin--McGregor determinant \eqref{eq:km}.

Both $q_t(x,\cdot)$ and $h$ are alternating.  Their product is symmetric, so
the chamber integral in \eqref{eq:h-mass} is $1/N!$ of the full-space
integral.  Write $h(y)=\det[y_j^{i-1}]$.  Andr\'eief's identity gives
\begin{equation}
 \int_{\RR^N}q_t(x,y)h(y)\dd y
 =N!\det[\int_{\RR}p_t(x_i,y)y^{j-1}\dd y]_{i,j=1}^N.
                                                        \label{eq:andreief}
\end{equation}
The $(j-1)$st moment of $N(rx_i,1-r^2)$ is a degree-$(j-1)$ polynomial in
$x_i$ with leading coefficient $r^{j-1}$.  Lower-degree column operations
reduce the determinant in \eqref{eq:andreief} to
$r^{0+\cdots+(N-1)}h(x)=r^dh(x)$.  This proves \eqref{eq:h-mass}.

Equation \eqref{eq:h-mass} makes \eqref{eq:doob} conservative.  On smooth
compactly supported functions,
\begin{equation}
 h^{-1}\mathcal L_0(hf)+\frac d2f
 =\mathcal L_0f+\nabla\log h\cdot\nabla f=\mathcal Lf, \label{eq:conjugacy}
\end{equation}
because $\partial_i\log h=\sum_{j\ne i}(x_i-x_j)^{-1}$.  The minimal
transformed process has mass one at every finite time.  A boundary hit by a
continuous chamber path would be its killing time, so conservativity excludes
such a hit.  Local uniqueness before collision and the standard
electrostatic-repulsion theorem identify the transform with the unique global
strong solution of \eqref{eq:dyson-sde}.
\end{proof}

Scalar detailed balance in \eqref{eq:mehler} passes through the determinant;
the two Vandermonde factors in \eqref{eq:doob} then prove
$\pi_N(x)k_t(x,y)=\pi_N(y)k_t(y,x)$ directly.  This independently checks the
generator calculation.

\section{Complete symmetric-Hermite spectrum}

Let $\kappa=(\kappa_1\geq\cdots\geq\kappa_N\geq0)$ be a partition and define
\begin{equation}
 m_i=\kappa_{N+1-i}+i-1.                              \label{eq:m-index}
\end{equation}
Then $m_{i+1}-m_i=1+\kappa_{N-i}-\kappa_{N+1-i}\geq1$ and
$\sum_i m_i=d+|\kappa|$.  With monic probabilists' Hermites, set
\begin{equation}
 D_m(x)=\det[\operatorname{He}_{m_i}(x_j)]_{i,j=1}^N,
 \qquad \Phi_\kappa(x)=\frac{D_m(x)}{h(x)}.            \label{eq:phi}
\end{equation}
The quotient is a symmetric polynomial because every alternating polynomial
is divisible by $h$.

\begin{theorem}[Complete partition spectrum and exact norms]
The family $\{\Phi_\kappa\}$ is a complete orthogonal basis of
$L^2(W_N,\pi_N)$ and
\begin{align}
 \mathcal L\Phi_\kappa&=-\frac{|\kappa|}{2}\Phi_\kappa,
                                                        \label{eq:eigen}\\
 \|\Phi_\kappa\|_{L^2(\pi_N)}^2
 &=\frac{\prod_{i=1}^N m_i!}{\prod_{j=0}^{N-1}j!}.     \label{eq:norm}
\end{align}
Thus $-k/2$ has multiplicity $p_N(k)$, the number of partitions of $k$ with
at most $N$ parts.
\end{theorem}

\begin{proof}
The scalar equation
\begin{equation}
 \frac12(\operatorname{He}_n''-x\operatorname{He}_n')
 =-\frac n2\operatorname{He}_n                           \label{eq:hermite}
\end{equation}
and determinant expansion give
$\mathcal L_0D_m=-(\sum_i m_i)D_m/2$.  Apply
\eqref{eq:conjugacy} to $D_m=h\Phi_\kappa$ and subtract the filled-ground
energy $d/2$ to obtain \eqref{eq:eigen}.

Scalar Hermites form a complete Gaussian $L^2(\RR)$ basis.  Their
antisymmetrized tensor products $D_m$, indexed by strictly increasing $m$,
therefore form a complete basis of the antisymmetric Gaussian sector on
$\RR^N$.  Multiplication by $h$, followed by antisymmetric extension from
$W_N$, identifies this sector with $L^2(W_N,\pi_N)$ up to a fixed scalar.
Division by $h$ transfers the complete Slater basis to \eqref{eq:phi}; this is
the step that rules out missing nonpolynomial $L^2$ spectrum.

Finally, Andr\'eief and scalar Hermite orthogonality give
\begin{equation}
 \int_{\RR^N}D_mD_n\e^{-|x|^2/2}\dd x
 =\delta_{mn}N!(2\pi)^{N/2}\prod_i m_i!.              \label{eq:slater-norm}
\end{equation}
Divide by $N!$ for the chamber and by $Z_N$ to obtain \eqref{eq:norm}.
Equation \eqref{eq:m-index} bijects partitions with strict indices above
$(0,1,\ldots,N-1)$, proving the multiplicity.
\end{proof}

\begin{corollary}[Sharp gap, trace, and source determinant]
For $\int f\dd\pi_N=0$,
\begin{equation}
 \|P_tf\|_2\leq\e^{-t/2}\|f\|_2,
 \qquad
 \Var_{\pi_N}(f)\leq\int_{W_N}|\nabla f|^2\dd\pi_N.   \label{eq:poincare}
\end{equation}
Both constants are sharp.  For $t>0$,
\begin{align}
 \Tr P_t&=\prod_{j=1}^N(1-\e^{-jt/2})^{-1},            \label{eq:trace}\\
 \det(I-zP_t)&=\prod_{k=0}^{\infty}
 (1-z\e^{-kt/2})^{p_N(k)}.                            \label{eq:fredholm}
\end{align}
\end{corollary}

\begin{proof}
Theorem 4 has a simple zero level and first nonzero decay exponent $1/2$.
The function $x_1+\cdots+x_N$ lies in that first eigenspace, proving
sharpness.  The partition generating function
$\sum_\kappa q^{|\kappa|}=\prod_{j=1}^N(1-q^j)^{-1}$ with
$q=\e^{-t/2}$ proves \eqref{eq:trace}.  The finite value of the trace makes
$P_t$ trace class, and its eigenvalue product gives \eqref{eq:fredholm}.
This determinant belongs only to the source semigroup.
\end{proof}

$\psi_0=\e^{-|x|^2/4}h$.  Direct differentiation, using $\Delta h=0$ and
$x\cdot\nabla h=dh$, gives
\begin{equation}
 -\psi_0\mathcal L\psi_0^{-1}
 =-\frac12\Delta+\frac{|x|^2}{8}-\frac N4-\frac d2.    \label{eq:oscillator}
\end{equation}
This acts on the antisymmetric Dirichlet sector; the inverse-square Calogero
term cancels at beta two.  It explains the filled Slater sea but is only a
