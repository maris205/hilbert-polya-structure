\section{The source family and main theorem}
\label{sec:source-main}

Let
\[
K=\Q(\rho),\qquad \rho^2+\rho+1=0,\qquad
\tau(\rho)=\rho^2.
\]
For \(n\ge2\), set \(N=2n\) and
\begin{equation}
C_n=\sum_{i=0}^{N-1}x_i^3,\qquad
Q_{n,\rho}=\sum_{i=0}^{N-2}x_ix_{i+1}+\rho x_{N-1}x_0.
\label{eq:source-forms}
\end{equation}
The source variety is
\[
X_n=V(C_n,Q_{n,\rho})\subset\PP_K^{N-1}.
\]
The order of the variables and the coefficient on \(x_{N-1}x_0\) are
part of the definition.

Define \(\sigma(i)=-i\bmod N\), and put
\[
e_i=\begin{cases}
1,&i\ne0\text{ and }i\text{ is even},\\
0,&i=0\text{ or }i\text{ is odd}.
\end{cases}
\]
Let \(M_n\) be the monomial transformation
\begin{equation}
(M_nx)_i=\rho^{e_i}x_{\sigma(i)}.
\label{eq:M-definition}
\end{equation}
Finally set \(\theta=1+2\rho\), so that \(\theta^2=-3\) and
\(\tau(\theta)=-\theta\).

\begin{theorem}[Explicit family descent]
\label{thm:family-descent}
For every \(n\ge2\),
\[
C_n(M_nx)=C_n(x),\qquad
Q_{n,\rho}(M_nx)=\rho Q_{n,\rho^2}(x),\qquad
M_n\tau(M_n)=I.
\]
In rational coordinates
\[
(u_0,a_1,b_1,\ldots,a_{n-1},b_{n-1},c),
\]
define \(B_n\) by
\begin{align}
x_0&=u_0,\nonumber\\
x_i&=a_i+\theta b_i,\qquad
x_{N-i}=\rho^{e_i}(a_i-\theta b_i)
       &&(1\le i<n),\label{eq:B-pairs}\\
x_n&=\kappa_nc,\qquad
\kappa_n=\begin{cases}1,&n\text{ odd},\\1+\rho,&n\text{ even}.
\end{cases}\nonumber
\end{align}
Then
\begin{equation}
M_n\tau(B_n)=B_n,\qquad
\det B_n=(2\theta)^{n-1}\rho^{\lfloor(n-1)/2\rfloor}\kappa_n\ne0.
\label{eq:B-det}
\end{equation}
The forms
\begin{equation}
C_{n,0}=u_0^3+\sum_{i=1}^{n-1}(2a_i^3-18a_ib_i^2)
          +(-1)^{n+1}c^3
\label{eq:C0}
\end{equation}
and
\begin{align}
Q_{n,0}={}&u_0(a_1+3b_1)\nonumber\\
&+\sum_{i=1}^{n-2}
(a_ia_{i+1}+3a_ib_{i+1}+3b_ia_{i+1}-3b_ib_{i+1})
+R_n,\label{eq:Q0}\\
R_n={}&
\begin{cases}
(a_{n-1}+3b_{n-1})c,&n\text{ odd},\\
2a_{n-1}c,&n\text{ even}
\end{cases}\nonumber
\end{align}
satisfy
\[
C_n(B_nu)=C_{n,0}(u),\qquad
Q_{n,\rho}(B_nu)=(1+\rho)Q_{n,0}(u).
\]
Consequently
\[
X_{n,0}=V(C_{n,0},Q_{n,0})\subset\PP_\Q^{2n-1}
\]
base-changes to \(X_n\).
\end{theorem}

Theorem~\ref{thm:family-descent} is an identity theorem for equations.
The source work certifies smoothness only for \(n=2,3,4\); no later row is
declared smooth here.

\begin{theorem}[Rational packets and fourth-row repair]
\label{thm:packets-repair}
For \(n=2,3,4\), the normalized even and odd packets descend to a rational
packet
\[
\mathsf W_n=\mathsf E_n\oplus\mathsf O_n,\qquad
\rank\mathsf W_n=4^n-1.
\]
Their ranks are \(15,63,255\).  At every good split rational prime,
\[
\frac2n\Logz L_{K,p}(\mathsf W_{n,K},u)
=\frac4n\Logz L_{\Q,p}(\mathsf W_n,u).
\]
The reduced rational denominator is \(n/\gcd(n,4)\).  In particular,
\begin{equation}
\left(L_{K,p}(\mathsf W_{4,K},u)\right)^{1/2}_{\Logz}
=L_{\Q,p}(\mathsf W_4,u).
\label{eq:split-repair}
\end{equation}
The right side is one rank-\(255\), exponent-one rational local factor.
\end{theorem}

\begin{theorem}[Rational dihedral core]
\label{thm:core-main}
For \(n=4\), the order-\(24\) Reynolds middle projector descends to
\(\Q\).  It gives mutually orthogonal self-transpose Chow projectors
\(\pi_{\core,0}\) and \(\pi_{\lev,0}\) of ranks \(10\) and \(158\).
The full rational packet decomposes by ranks
\[
87+10+158=255.
\]
The raw rank-\(10\) summand
\(\mathsf M_0=(X_{4,0},\pi_{\core,0},0)\) has Hodge numbers
\[
h^{4,1}=h^{1,4}=1,\qquad h^{3,2}=h^{2,3}=4.
\]
\end{theorem}

\begin{theorem}[Good-prime polynomial]
\label{thm:compatible-main}
Outside a finite set, for every \(\ell\ne p\),
\[
P_p(T)=\det(1-\Frob_pT\mid H_\ell(\mathsf M_0))
=\sum_{k=0}^{10}a_kT^k
\]
is a degree-\(10\) polynomial in \(\Z[T]\), independent of \(\ell\), and
pure of weight \(5\).  It satisfies
\[
a_{10-k}=p^{25-5k}a_k
\quad(0\le k\le10).
\]
The twice-twisted system \(\mathsf M_0(2)\) has weight \(1\) and local
polynomial \(P_p(T/p^2)\in\Q[T]\).
\end{theorem}
