\section{Frozen memory owner}
Fix
\[
 a,b\geq0,\qquad r>0,\qquad
 K_r(s)=r^2s\ee^{-rs}\quad(s\geq0),
\]
and consider
\begin{equation}\label{eq:delay}
 \dot x(t)=-a x(t)-b\int_0^\infty K_r(s)x(t-s)\,ds .
\end{equation}
The kernel is normalized; it has mean $2/r$, variance $2/r^2$, and Laplace
transform $r^2/(\lambda+r)^2$.  A compatible fading-memory history is one
for which the following integrals exist and permit the standard
differentiation under the integral:
\[
 z_1(t)=\int_0^\infty r\ee^{-rs}x(t-s)\,ds,\qquad
 z_2(t)=\int_0^\infty K_r(s)x(t-s)\,ds.
\]

\begin{lemma}[Exact linear chain]\label{lem:chain}
For compatible initialization, equation~\eqref{eq:delay} is equivalent to
\begin{equation}\label{eq:chain}
 \dot x=-ax-bz_2,\qquad
 \dot z_1=r(x-z_1),\qquad
 \dot z_2=r(z_1-z_2).
\end{equation}
It is not asserted that arbitrary values of $(z_1,z_2)$ encode a prescribed
prehistory.
\end{lemma}

\begin{proof}
Integration by parts gives the two filter equations in~\eqref{eq:chain};
the first equation is~\eqref{eq:delay}.  Conversely, subtract the displayed
convolutions from a solution of the two stable filter equations.  The
differences solve homogeneous first-order equations and vanish initially,
so they vanish for all later times.  Thus compatible chain solutions recover
the delay equation exactly.
\end{proof}

Put $X=(x,z_1,z_2)^{\mathsf T}$.  Then $\dot X=\M X$, where
\[
 \M=\begin{pmatrix}-a&0&-b\\ r&-r&0\\0&r&-r\end{pmatrix},
\]
and direct expansion gives
\begin{align}
 p(\lambda)&=\det(\lambda I-\M)
 =(\lambda+a)(\lambda+r)^2+br^2 \label{eq:poly}\\
 &=\lambda^3+c_2\lambda^2+c_1\lambda+c_0,\nonumber\\
 c_2&=a+2r,\qquad c_1=r(r+2a),\qquad c_0=r^2(a+b).\nonumber
\end{align}

\begin{theorem}[Sharp stability and imaginary crossing]\label{thm:routh}
For $b>0$, the zero solution is exponentially stable if and only if
\begin{equation}\label{eq:bh}
 0<b<b_H,\qquad b_H=\frac{2(a+r)^2}{r}.
\end{equation}
At $b=b_H$,
\begin{equation}\label{eq:hopf-factor}
 p(\lambda)=(\lambda+a+2r)
 (\lambda^2+r(r+2a)),
\end{equation}
so the roots are $-a-2r$ and
$\lambda_\pm=\pm\ii\omega_H$, with
$\omega_H=\sqrt{r(r+2a)}$.  The conjugate pair is simple and crosses from
left to right:
\begin{equation}\label{eq:crossing}
 \operatorname{Re}\lambda_+'(b_H)
 =\frac{r^2}{2\{r(r+2a)+(a+2r)^2\}}>0.
\end{equation}
For $b>b_H$, exactly two roots lie in the open right half-plane.
\end{theorem}

\begin{proof}
For $b>0$, all $c_j$ are positive.  The remaining cubic Routh--Hurwitz
minor is
\[
 c_2c_1-c_0=r\{2(a+r)^2-br\}.
\]
This proves the equivalence in~\eqref{eq:bh}.  At equality, multiplication
verifies~\eqref{eq:hopf-factor}.  Implicit differentiation of
$p(\lambda,b)=0$ at $\lambda=\ii\omega_H$ yields~\eqref{eq:crossing}.
Above the threshold the first Routh column has signs $+,+,-,+$, hence two
sign changes and exactly two open-right-half-plane roots.
\end{proof}

round: this is an all-parameter analytic theorem, not stability inferred
from sampled roots.

\section{Complete degeneracy and propagator atlas}

\begin{theorem}[All repeated roots and Jordan blocks]\label{thm:jordan}
In the domain $a,b\geq0$, $r>0$, all repeated-root cases are precisely:
\begin{enumerate}
\item $b=0$.  The root $-r$ has one size-two Jordan block when $a\ne r$,
and at $a=r$ the root $-r$ has one size-three block.
\item $0\leq a<r$ and
\begin{equation}\label{eq:bd}
 b=b_D=\frac{4(r-a)^3}{27r^2}>0.
\end{equation}
Here
\[
 p(\lambda)=(\lambda-\mu)^2(\lambda-\nu),\qquad
 \mu=-\frac{r+2a}{3},\quad \nu=-\frac{4r-a}{3},
\]
and the Jordan sizes are two and one.
\end{enumerate}
The positive-$b$ defective face lies strictly inside the stable region.
\end{theorem}

\begin{proof}
Differentiating~\eqref{eq:poly} and taking the discriminant gives
\begin{equation}\label{eq:disc}
 p'(\lambda)=(\lambda+r)(3\lambda+r+2a),\qquad
 \operatorname{disc}p=br^2\{4(r-a)^3-27br^2\}.
\end{equation}
The first critical root belongs to $p$ exactly when $b=0$.  The second
belongs to $p$ exactly on~\eqref{eq:bd}; it has positive $b$ precisely when
$a<r$.  These observations exhaust~\eqref{eq:disc}, and substitution gives
the factorization.  Moreover,
\[
 \det[e_1,\M e_1,\M^2e_1]=r^3\ne0.
\]
Thus $\M$ is cyclic and its minimal polynomial equals $p$: each repeated
root has one Jordan block of its algebraic multiplicity.  Finally
\[
 \frac{b_D}{b_H}
 =\frac{2(r-a)^3}{27r(a+r)^2}<1,
\]
so the positive defective face is stable.
\end{proof}

For distinct roots $\lambda_1,\lambda_2,\lambda_3$, Lagrange interpolation
modulo $p$ gives
\begin{equation}\label{eq:simple-flow}
 \ee^{t\M}=\sum_{j=1}^3\ee^{t\lambda_j}
 \prod_{k\ne j}\frac{\M-\lambda_kI}{\lambda_j-\lambda_k}.
\end{equation}
On a double face, write $p=(\lambda-\mu)^2(\lambda-\nu)$ and set
\[
 P_\nu=\frac{(\M-\mu I)^2}{(\nu-\mu)^2},\qquad
 P_\mu=I-P_\nu,\qquad N_\mu=(\M-\mu I)P_\mu.
\]
Hermite interpolation yields
\begin{equation}\label{eq:double-flow}
 \ee^{t\M}=\ee^{\mu t}(P_\mu+tN_\mu)+\ee^{\nu t}P_\nu,
 \qquad N_\mu^2=0.
\end{equation}
At $a=r,b=0$, $N=\M+rI$ satisfies $N^3=0$ and
\begin{equation}\label{eq:triple-flow}
 \ee^{t\M}=\ee^{-rt}(I+tN+t^2N^2/2).
\end{equation}
Equations~\eqref{eq:simple-flow}--\eqref{eq:triple-flow} close the flow on
every spectral stratum.

revision: transversality, unstable-root count, discriminant exhaustion,

\section{Boundaries and claim discipline}
At $b=0$, $p=(\lambda+a)(\lambda+r)^2$.  Hence $a>0$ is exponentially
stable despite the defective $-r$ block.  At $a=b=0$, the simple zero root
is a constant $x$-mode and the memory transient contains $t\ee^{-rt}$.
For $a=0,b>0$, Theorem~\ref{thm:routh} specializes to stability for
$0<b<2r$, an imaginary pair at $b=2r$, and two right-half-plane roots
above it.  The point $a=r,b=0$ is the triple block in
\eqref{eq:triple-flow}.

The value $r=0$ is excluded: the displayed expression ceases to be a
normalized Erlang density.  For fixed bounded $\lambda$,
$r^2/(\lambda+r)^2\to1$ as $r\to\infty$, so the bounded slow root tends
to $-(a+b)$ while two memory modes are fast.  This limiting observation is
not substituted for any finite-$r$ theorem.

Theorem~\ref{thm:routh} proves a simple transverse imaginary
\emph{linear spectral crossing}.  A nonlinear vector field, its
nondegeneracy coefficient, and its center-manifold reduction have not been
specified.  Therefore neither existence nor stability of a nonlinear
periodic branch is claimed.
