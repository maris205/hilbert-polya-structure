\begin{abstract}
For every linear endomorphism of a finite-dimensional vector space over a
finite field, we derive its complete functional-graph census from its invariant
factors.  The result simultaneously covers nilpotent, nonsemisimple, and
inseparable cases.  It gives every fixed count, the periodic subspace and
maximal preperiod, exact periods, the Artin--Mazur zeta function, and the
characteristic polynomial of composition on the full function space.  Exact
enumeration, independent rank calculations, and symbolic controls verify eight
structurally distinct witnesses, including genuine $\F_4$ arithmetic.
\end{abstract}

\section{Invariant-factor fixed counts}
Let $A\in\operatorname{End}_{\F_q}(V)$ have invariant factors
$f_1\mid\cdots\mid f_r$, so that $V\simeq\bigoplus_i\F_q[X]/(f_i)$ and $A$
acts as multiplication by $X$.  Put
\[
 F_n=q^{\sum_i\deg\gcd(f_i,X^n-1)}.
\]

\begin{theorem}[complete finite dynamics]\label{thm:main}
For every $n\ge1$, $|\Fix(A^n)|=F_n$.  Write
$f_i=X^{e_i}g_i$ with $g_i(0)\ne0$.  Then $\Per(A)$ is a vector subspace of
dimension $\sum_i\deg g_i$, and the largest preperiod is $\max_i e_i$.
The numbers of least-period points and cycles are
\[
 P_n=\sum_{d\mid n}\mu(n/d)F_d,\qquad C_n=P_n/n,
\]
and
$\zeta_A(z)=\prod_{n\ge1}(1-z^n)^{-C_n}$.
\ifnum\CRevisionRound>1
For $U_A\phi=\phi\circ A$ on $\mathbb C^V$,
\[
 \chi_{U_A}(T)=T^{|V|-|\Per(A)|}
                 \prod_{n\ge1}(T^n-1)^{C_n}.
\]
\fi
\end{theorem}

\begin{proof}
On $\F_q[X]/(f_i)$, the kernel of multiplication by $X^n-1$ has dimension
$\deg\gcd(f_i,X^n-1)$.  Direct sums prove the fixed formula; no square-free
hypothesis on $X^n-1$ is used.  Primary decomposition separates the nilpotent
$X$-part, of height $e_i$, from the invertible $g_i$-part.  This proves the
periodic-subspace and tail assertions.  M\"obius inversion gives $P_n$, orbit
division gives $C_n$, and exponentiating
$\sum_{n\ge1}F_nz^n/n$ gives the zeta product.
\ifnum\CRevisionRound>1
Each functional-graph component consists of a directed cycle and incoming
trees.  Ordering the cycle before tree vertices makes the composition matrix
block triangular.  A $d$-cycle contributes $T^d-1$ and every noncycle vertex
contributes $T$, proving the last formula.
\fi
\end{proof}

\section{Transient and inseparable structure}
The theorem distinguishes two phenomena sometimes obscured by fixed-count
tables.  A factor $X^e$ contributes no nonzero periodic coordinate but fixes
the exact maximal tail $e$.  Conversely, repeated factors coprime to $X$ can
produce nonsemisimple periodic dynamics.  For example, $(X+1)^4$ over $\F_2$
is inseparable and has restriction order four, while $(X+\alpha)^2$ over
$\F_4$, with $\alpha^2+\alpha+1=0$, has order six.  The latter uses field
arithmetic, not arithmetic in $\mathbb Z/4\mathbb Z$.

\begin{lemma}
The theorem remains valid when the characteristic divides $n$ and
$X^n-1$ is inseparable.
\end{lemma}
\begin{proof}
The cyclic-module kernel calculation uses the ideal generated by
$\gcd(f_i,X^n-1)$ with its full multiplicity.  It never replaces the gcd by a
set of distinct roots.
\end{proof}
\fi
