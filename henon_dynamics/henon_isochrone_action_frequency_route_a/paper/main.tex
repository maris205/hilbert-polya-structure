\section{Model, conventions, and provenance}
Let \(\mu,b>0\).  In polar variables the unit-mass Hamiltonian is
\begin{equation}\label{eq:H}
 H(r,p_r,L)=\frac{p_r^2}{2}+\frac{L^2}{2r^2}
 -\frac{\mu}{b+\sqrt{b^2+r^2}}.
\end{equation}
Put \(\ell=\abs L\),
\begin{equation}\label{eq:AB}
 A=\sqrt{\ell^2+4\mu b},\qquad B=\ell+A,
\end{equation}
and use physical Hamiltonian time.  For \(\ell>0\), the radial action is
\begin{equation}\label{eq:actiondef}
 J_r=\frac1\pi\int_{r_p}^{r_a}
 \sqrt{2(E-U_\ell(r))}\,\D r,
 \quad
 U_\ell(r)=\frac{\ell^2}{2r^2}
 -\frac{\mu}{b+\sqrt{b^2+r^2}}.
\end{equation}
At \(\ell=0\) we use its continuous half-line limit and discuss the full
Cartesian return separately.

The potential and its orbit mechanics are classical, originating in
H\'enon's two 1959 papers~\cite{HenonI,HenonII}.  Ramond and Perez give a
modern isochrone-mechanics treatment~\cite{RamondPerez}; Fouvry and Prunet
record the frequency map in their official Appendix C~\cite{FouvryPrunet}.
We rederive the formulas to close a reproducible certificate and make no
claim of literature priority.

\section{Complete bound action theorem}
\begin{theorem}[Energy, action, and radial period]\label{thm:core}
For fixed \(\mu,b>0\) and \(\ell\geq0\), define
\begin{equation}\label{eq:circ}
 s_c=\frac{B^2}{4\mu},\qquad r_c^2=s_c^2-b^2,
 \qquad E_c=-\frac{\mu}{2s_c}=-\frac{2\mu^2}{B^2}.
\end{equation}
Bound motion exists exactly for \(E_c\leq E<0\).  On this domain,
\begin{align}
 J_r&=\frac{\mu}{\sqrt{-2E}}-\frac B2,\label{eq:J}\\
 H(J_r,\ell)&=-\frac{\mu^2}{2(J_r+B/2)^2},\label{eq:HJ}\\
 T_r&=\frac{2\pi\mu}{(-2E)^{3/2}},\qquad
 \Omega_r=\frac{(-2E)^{3/2}}{\mu}.\label{eq:Tr}
\end{align}
In particular, \(J_r\geq0\) is equivalent to \(E\geq E_c\) inside
\(E<0\), and \(T_r\) is independent of angular momentum.
\end{theorem}

\begin{proof}
Write \(s=\sqrt{b^2+r^2}\geq b\).  For \(\ell>0\), differentiation of
\[
 U_\ell(s)=-\frac{\mu}{b+s}
 +\frac{\ell^2}{2(s^2-b^2)}
\]
gives the circular equation \(\mu(s-b)^2=\ell^2s\).  Its unique solution
above \(b\) is \(s_c=B^2/(4\mu)\), and substitution gives~\eqref{eq:circ}.
The effective potential tends to \(+\infty\) at zero, to \(0^-\) at
infinity, and has only this critical point, hence this is its global
minimum.  When \(\ell=0\), the potential increases from
\(-\mu/(2b)\) to zero, and the same formulas give the central minimum.
This proves the asserted energy domain.

For the period, set
\begin{equation}\label{eq:xQ}
 x=b+\sqrt{b^2+r^2},\quad r^2=x(x-2b),\quad
 Q(x)=2Ex^2+(2\mu-4bE)x-(4\mu b+\ell^2).
\end{equation}
A direct substitution in the energy equation gives
\begin{equation}\label{eq:dt}
 \D t=\frac{x-b}{\sqrt{Q(x)}}\,\D x.
\end{equation}
For a noncircular bound orbit, put \(q=-2E>0\) and write
\(Q=q(x_a-x)(x-x_p)\).  Vieta's identities give
\begin{equation}\label{eq:vieta}
 x_p+x_a=2b+\frac{2\mu}{q},\qquad
 x_px_a=\frac{A^2}{q}.
\end{equation}
The arcsine substitution on \([x_p,x_a]\) therefore yields
\[
 \int_{x_p}^{x_a}\frac{x-b}{\sqrt{Q(x)}}\,\D x
 =\frac{\pi}{\sqrt q}\left(\frac{x_p+x_a}{2}-b\right)
 =\frac{\pi\mu}{q^{3/2}}.
\]
Doubling proves~\eqref{eq:Tr}; the circular value follows by continuity.

Finally, differentiation of the one-dimensional action integral gives
\(\partial_EJ_r=T_r/(2\pi)=\mu/(-2E)^{3/2}\).  Thus
\(J_r=\mu/\sqrt{-2E}+C(\ell)\).  The circular cycle collapses, so
\(J_r(E_c,\ell)=0\); since \(\mu/\sqrt{-2E_c}=B/2\), this fixes
\(C=-B/2\) and proves~\eqref{eq:J}.  Inversion gives~\eqref{eq:HJ},
and differentiating it in \(J_r\) gives \(\Omega_r\).
\end{proof}

Below the circular minimum no real radial interval exists.  At \(E=0\)
the motion reaches the marginal escape threshold, while \(E>0\) is
unbound.  Equations~\eqref{eq:J} and~\eqref{eq:Tr} show that both action and
period diverge as \(E\uparrow0\); the threshold is not a finite action point.

\section{Apsidal map and noncircular closure}
\begin{theorem}[Frequency map and closure]\label{thm:closure}
For \(L\geq0\) and \(\ell>0\),
\begin{equation}\label{eq:beta}
 \frac{\Omega_\phi}{\Omega_r}=\beta(\ell)
 =\frac12\left(1+\frac{\ell}{\sqrt{\ell^2+4\mu b}}\right),
 \qquad \frac12<\beta<1.
\end{equation}
The azimuthal advance in one radial period is \(2\pi\beta\).  A
noncircular bound phase-space orbit closes if and only if
\(\beta\in\Q\).  If \(\beta=p/q\) in lowest terms, its primitive period is
\(qT_r\).
\end{theorem}
\begin{proof}
Because \(\dot\phi=\ell/r^2\), equations~\eqref{eq:dt} and~\eqref{eq:xQ}
reduce the half-cycle angular integral by
\begin{equation}\label{eq:pf}
 \frac{x-b}{x(x-2b)}=\frac12\left(\frac1x+\frac1{x-2b}\right).
\end{equation}
Besides~\eqref{eq:vieta}, Vieta gives the shifted product
\begin{equation}\label{eq:shifted}
 (x_p-2b)(x_a-2b)=\frac{\ell^2}{q}.
\end{equation}
Using
\(
\int_{x_p}^{x_a}\D x/[x\sqrt{(x_a-x)(x-x_p)}]
=\pi/\sqrt{x_px_a}
\)
and its shifted counterpart in~\eqref{eq:pf}, the periapsis-to-apoapsis
angle is
\[
 \frac{\ell}{2\sqrt q}\left(
 \frac\pi{\sqrt{x_px_a}}+
 \frac\pi{\sqrt{(x_p-2b)(x_a-2b)}}\right)
 =\frac\pi2\left(1+\frac\ell A\right).
\]
Doubling proves the advance and~\eqref{eq:beta}.  The same ratio follows by
differentiating~\eqref{eq:HJ}: \(\Omega_\phi=\partial_\ell H\) and
\(\partial_\ell(B/2)=\beta\).

For a nonconstant radial phase, a phase-space return must occur after an
integer number \(q\) of radial cycles.  Angular return is then exactly
\(q\beta\in\mathbb Z\), equivalent to rational \(\beta\); lowest terms
give primitivity.
\end{proof}
