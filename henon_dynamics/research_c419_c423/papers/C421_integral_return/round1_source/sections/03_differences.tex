\section{A parameter-free difference identity}
\label{sec:differences}

Let $(x_i)_{i\in\Z}$ be a periodic integral sequence satisfying
\eqref{eq:scalar}. Define
\begin{equation}
 d_i=x_{i+2}-x_i,\qquad D=\max_i|d_i|.
 \label{eq:difference}
\end{equation}
Subtract the recurrence at $i-1$ from that at $i$. It gives
\begin{equation}
 d_{i-1}+d_{i+1}=(x_{i+1}+1)d_i,
 \qquad |(x_{i+1}+1)d_i|\le2D.
 \label{eq:key}
\end{equation}
The forcing cancels. Two adjacent zero differences force all
differences to be zero, by propagation of \eqref{eq:key} in
both directions.

Reversal of a scalar sequence preserves \eqref{eq:scalar}.
On triples it is the actual reversing involution
\begin{equation}
 R(x,y,z)=(z,y,x),\qquad RT_aR=T_a^{-1}.
 \label{eq:reversor}
\end{equation}
We use this symmetry to normalize an extremum, but restore both
time orientations at the end.

\begin{lemma}[Zero difference]
\label{lem:zero}
If $D=0$, the scalar sequence alternates two integers $u,v$ and
\begin{equation}
 a=u+v-uv=1-(u-1)(v-1).
 \label{eq:alternating}
\end{equation}
It is fixed when $u=v$ and otherwise has least period two.
\end{lemma}
\begin{proof}
The equation $d_i=0$ is $x_{i+2}=x_i$.
Substitution of $(u,v,u,v,\ldots)$ into the recurrence is
equivalent to \eqref{eq:alternating}. The assertion about the
least period follows directly.
\end{proof}

\begin{lemma}[The high-coordinate four-step channel]
\label{lem:height}
If $D>0$ and $|x_j+1|>3D$ at some phase, the orbit is a word
of the form $(-1,t,-1,a+1-t)$ repeated. Consequently, every
orbit not of this form satisfies
\begin{equation}
 |x_i+1|\le3D\quad\text{for all }i.
 \label{eq:heightbound}
\end{equation}
\end{lemma}
\begin{proof}
Rotate to $j=1$. Integrality in \eqref{eq:key} forces $d_0=0$.
Put $x_0=x_2=b$, $x_1=m$, and $e=d_1=-d_{-1}$.
We have $e\ne0$, since two adjacent zero differences would give
$D=0$, and $|e|\le D$. Moreover,
$x_{-1}=x_3=m+e$; in particular the sign in both of these
expressions is the same. Thus
$|x_{-1}+1|,|x_3+1|>2D$, forcing $d_{-2}=d_2=0$ by
\eqref{eq:key}. Its equation at $i=1$ is now $(b+1)e=0$,
so $b=-1$. Direct use of the scalar recurrence yields
$(-1,m,-1,a+1-m)$ repeated. Conversely this word satisfies
every recurrence equation. Its period is four unless
$2m=a+1$, when it is alternating (and fixed if also $m=-1$).
\end{proof}
