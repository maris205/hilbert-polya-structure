\section{Exhaustion when the maximum difference is large}
\label{sec:large}

\begin{proposition}[Uniform reduction]
\label{prop:large}
Every integral periodic sequence with $D>100$ is one of the
families $F_3,F_4,F_5,F_6,F_8,F_{12}$, allowing the
shorter-period degeneracies before choosing the ranges in
Table~\ref{tab:classification}.
\end{proposition}

Choose a phase and initially reverse if necessary so that
$d_0=D>0$. Write
\begin{equation}
 (x_0,x_1,x_2)=(u,s,v),\quad v-u=D,\quad
 p=d_{-1}=v-su+s-a,\quad q=d_1=sv+a-u-s.
 \label{eq:extremal}
\end{equation}
Equation~\eqref{eq:key} gives
\begin{equation}
 s\in\{-3,-2,-1,0,1\},\qquad
 p+q=(s+1)D,\qquad |p|,|q|\le D.
 \label{eq:fivecenters}
\end{equation}
All bounds below are consequences of the global maximum $D$;
a violation excludes a periodic sequence, not just a sampled
initial condition.

\subsection{The endpoint centers}
If $s=1$, then $p=q=D$, and \eqref{eq:key} gives
$d_{-2}=uD$, $d_2=vD$. Hence $u,v\in[-1,1]$, implying
$D\le2$, a contradiction. If $s=-3$, then $p=q=-D$ and
$d_{-2}=-(u+2)D$, $d_2=-(v+2)D$. Now
$u,v\in[-3,-1]$, giving the same contradiction.

\subsection{The center \texorpdfstring{$s=-1$}{s=-1}}
Set $E=u+v-a-1$, so $p=E$ and $q=-E$. We have
\[
 d_{-2}=(u+1)E-D,\qquad d_2=-(v+1)E-D.
\]
Their bounds imply $(u+1)E\ge0$ and $(v+1)E\le0$.
Since $u<v$, the case $E>0$ is impossible. If $E=0$,
the recurrence is the four-step word $F_4$ (up to rotation).
Otherwise write $E=-\kappa$, with $\kappa\in\Z_{>0}$.
Then
\[
 u\le-1\le v,\quad
 \kappa(-u-1)\le2D,\quad \kappa(v+1)\le2D.
\]
Adding the latter inequalities gives $\kappa\le4$.
Both $x_{-1}$ and $x_3$ equal $\kappa-1$, and one more
application of \eqref{eq:key} gives
\[
 d_{-3}+d_3=\kappa(\kappa-2)D.
\]
Its absolute value is at most $2D$, excluding $\kappa=3,4$.

For $\kappa=1$ we have $a=u+v$, $x_{-1}=x_3=0$,
$d_{-3}=-v$ and $d_3=u$. The centers of these differences
are both $a+1$, so
\[
 |a+2|\max(|u|,|v|)\le2D.
\]
Since the maximum is at least $D/2$, it follows that
$-6\le a\le2$. The forward sequence begins
\[
 u,-1,v,0,a+1,u,(a+1)u+a,(a+1)u^2+au-1.
\]
Consequently $d_5=(a+1)u^2+(a-1)u-1$.
If $a\ne-1$, put $U=|u|$. As $u=(a-D)/2<0$, we have
$U>49$ and $D\le2U+6$, while
\[
 |d_5|\ge U^2-7U-1>2U+6\ge D,
\]
a contradiction. If $a=-1$, the repeated word is
$(u,-1,-1-u,0,0)$, a rotation of $F_5$.

For $\kappa=2$ we have $a=u+v+1$, $d_{-3}=-2a$ and
$d_3=2a$, with centers $u+a+1$ and $v+a+1$.
Applying \eqref{eq:key} at these centers gives
\[
 |2a(u+a+2)|\le2D,\qquad |2a(v+a+2)|\le2D.
\]
The shifted coefficients differ by $D$, so one has absolute
value at least $D/2$. Hence $|a|\le2$, including $a=0$.
The forward sequence begins
\[
 u,-1,v,1,v+a+1,2a+1,(v+a+1)(2a+1)+a-1,
\]
and hence
\[
 d_4=2a(v+a+1)+a-1,\qquad D=2v-a+1,\qquad v>48.
\]
For $a=1$ we obtain $d_4=D+4$. For $a=2$ we obtain
$d_4=4v+13>D$, and for $a=-2$ we obtain
$|d_4|=4v-1>D=2v+3$. All are impossible.
For $a=-1$ direct iteration gives $F_8$; for $a=0$ it
gives $F_{12}$. In the latter case $u=-1-v$ and $m=v+1>0$;
the triple $(u,-1,v)$ occurs starting at position $9$ of
the displayed $F_{12}$ word, with zero-based indices.

\subsection{The signed normalization at \texorpdfstring{$s=0,-2$}{s=0,-2}}
For either remaining center, $|p+q|=D$, so at least one of
$|p|,|q|$ is at least $D/2$. Reverse if needed to arrange
$|p|\ge D/2$. This second reversal can change the extremum's
sign. We therefore write
\begin{equation}
 d_0=\varepsilon L,\quad L=D>100,\quad
 \varepsilon\in\{-1,1\},\quad v=u+\varepsilon L.
 \label{eq:signed}
\end{equation}
The bound at $p$ gives $|u+1|\le4$, so $-5\le u\le3$.
Since $|v+1|\ge L-6$, the bound at $q$ gives
$|q|\le2L/(L-6)<3$. Keeping $\varepsilon$ in the next
two cases is essential.

\subsection{The center \texorpdfstring{$s=0$}{s=0}}
Here $p+q=\varepsilon L$, so $p,q$ have sign
$\varepsilon$ or are zero. Write $q=\varepsilon r$,
$r\in\{0,1,2\}$. Then $a=u+\varepsilon r$,
$x_3=\varepsilon r$, and
\[
 d_2=(v+1)\varepsilon r-\varepsilon L.
\]
If $\varepsilon=-1$ and $r\ge1$, this is
$(r+1)L-r(u+1)>L$, a contradiction.
If $\varepsilon=1,r=2$, it is $L+2u+2\ge L-8$,
with center $x_3=2$. Equation~\eqref{eq:key} would require
$3(L-8)\le2L$, again impossible.

If $\varepsilon=1,r=1$, put $a=u+1$, so $-4\le a\le4$.
We obtain $d_2=a$, $d_3=2a-1$, with the latter center
$x_4=L+2a-1$. Thus
\[
 |(L+2a)(2a-1)|\le2L.
\]
For an integer $a\notin\{0,1\}$ the left side is at least
$3(L-8)>2L$. If $a=1$, the initial triple is $(0,0,L)$,
and $x_4=L+1$, $x_6=2L+2$, so $d_4=L+1>D$.
If $a=0$, the initial triple is $(-1,0,L-1)$, and
$x_7=1-L$, $x_9=1-(L-1)^2$, giving
$d_7=-L^2+3L-1$ of magnitude greater than $L$.

Only $r=0$ remains. Now $a=u$, $x_3=0$, $x_4=u$,
$x_5=-\varepsilon L$, and $d_2=d_3=-\varepsilon L$.
The bound at $d_3$ gives $-3\le u\le1$.
Also $d_4=-u\varepsilon L$, centered at
$x_5=-\varepsilon L$. Since $L>100$, the inequality
$|(-\varepsilon L+1)u\varepsilon L|\le2L$
forces $u=0$. Hence $a=0$ and the word is $F_6$.

\subsection{The center \texorpdfstring{$s=-2$}{s=-2}}
Here $p+q=-\varepsilon L$, so write
$q=-\varepsilon r$, $r\in\{0,1,2\}$. Then
\[
 d_2=-rL-\varepsilon\bigl(r(u+1)+L\bigr).
\]
For $\varepsilon=1,r\ge1$ this is at most
$-(r+1)L+4r<-L$. Consider $\varepsilon=-1$.
Then $q=r$, $v=u-L$, $a=2v+u+r-2$,
$x_3=r-2$, and $x_4=rv+u+r$.

If $r=2$, we have $x_4=2v+u+2$, $x_5=v+u$,
$a=2v+u$, and $d_4=x_4x_5+a-x_4$.
Using $|u|\le5$ gives the conservative estimate
\[
 |d_4|\ge(2L-17)(L-10)-4L-32>L
 \qquad(L>100),
\]
a contradiction. If $r=1$, the next entries are
\begin{align*}
 x_3&=-1,&x_4&=v+u+1,&x_5&=-2,\\
 x_6&=-u-2,&x_7&=v+2u+2,&
 x_8&=-uv-2u^2-5u-3.
\end{align*}
Thus $d_5=-L+3u+4$, centered at $-u-2$.
For $u\in[-5,3]\setminus[-3,1]$,
$|(u+1)d_5|\ge3(L-13)>2L$, impossible.
For $-3\le u\le1$, the difference
$d_6=uL-3u^2-4u-1$ has center
$x_7=-L+3u+2$. Its bound implies
\[
 |d_6|\le\frac{2L}{L-6}<3.
\]
If $u\ne0$, then $|d_6|\ge L-16>2$, contradicting
integrality and this strict bound. For $u=0$, direct
iteration instead gives $d_7=2L-7>L$.

Only $r=0$ remains, for either sign. Now
\begin{gather*}
 a=2v+u-2,\quad x_3=-2,\quad x_4=u,\quad
 x_5=\varepsilon L-2,\\
 d_2=-\varepsilon L,\quad d_3=\varepsilon L,
 \quad d_4=(u+2)\varepsilon L.
\end{gather*}
The bound for $d_4$ at center $x_5$ forces $u=-2$.
The resulting word is $(-2,-2,t)$ with $a=2t-4$, namely
$F_3$.

\subsection{Restoring orientation and concluding the reduction}
At the centered triple, reversal sends
$(d_0,p,q)$ to $(-d_0,-q,-p)$, exactly the rule used in
\eqref{eq:signed}. Reversing $F_3,F_4,F_6,F_8$ or
$F_{12}$ gives a rotation of the same word.
For the form $(u,-1,-1-u,0,0)$ of $F_5$, the reverse is
a rotation of $(-1-u,-1,u,0,0)$, which is also included
as the parameter ranges over all integers. Alternating words
are reversal-closed as well. Thus normalization has not lost
an oriented orbit. This completes the proof of
Proposition~\ref{prop:large}.

\begin{corollary}[The residual core]
\label{cor:core}
Every integral periodic orbit is either an alternating word,
one of $F_3,F_4,F_5,F_6,F_8,F_{12}$, or satisfies
\begin{equation}
 1\le D\le100,\qquad -3D-1\le x_i\le3D-1
 \quad\text{at every phase}.
 \label{eq:core}
\end{equation}
In particular the residual coordinates belong to $[-301,299]$.
\end{corollary}
\begin{proof}
Apply Lemma~\ref{lem:zero}, Proposition~\ref{prop:large},
and then Lemma~\ref{lem:height} outside $F_4$.
\end{proof}
