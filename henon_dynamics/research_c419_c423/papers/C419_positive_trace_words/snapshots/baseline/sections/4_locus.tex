\section{The 39 lines and their positive-word realization}\label{sec:locus}

\begin{proposition}\label{prop:cover}
Every periodic letter phase outside $\E$ belongs to $\D$.
\end{proposition}

\begin{proof}
By Corollary~\ref{cor:periodicsmall} the phase has a small
coordinate. If two coordinates are small, it is in $\D_0$.
If exactly one is small, it cannot be the third. Suppose
it is $x=c\in C$, so $|y|,|z|\ge2$. Repeated applications
of $A$ transform the last two coordinates by
\[
 M_c=\begin{pmatrix}0&1\\-1&c\end{pmatrix}.
\]
Direct multiplication gives
\begin{equation}\label{eq:rotations}
 M_0^4=I,\qquad M_1^3=-I,\qquad M_1^6=I,\qquad
 M_{-1}^3=I.
\end{equation}
The encountered absolute coordinate values lie in
\[
 \begin{cases}
 \{|y|,|z|\},&c=0,\\
 \{|y|,|z|,|y-z|\},&c=1,\\
 \{|y|,|z|,|y+z|\},&c=-1.
 \end{cases}
\]
These lists follow by applying the matrices for four, six
or three steps, respectively, after which they repeat.

If $c=0$, or if $c=\pm1$ and $|y-cz|\ge2$, both other
coordinates stay large through every $A$ step. The next
$B$ in the repeated admissible schedule removes the sole
small coordinate and, by Lemma~\ref{lem:exit}, enters the
cone. This is impossible for a periodic orbit. Therefore
$c=\pm1$ and $|y-cz|\le1$. Equivalently
$z=cy+b$ for $b\in C$, which is $\D_x$. The unique-small-$y$
case is its conjugate under $J$ and gives $\D_y$.
\end{proof}

\begin{proposition}\label{prop:realize}
Every line in $\D$ is fixed pointwise by an admissible positive
word. The set $\E$ is fixed pointwise by $B^2\circ A^2$.
\end{proposition}

\begin{proof}
Equations~\eqref{eq:rotations} show that $A^{12}$ is the
identity on every plane $x=c$, $c\in C$, and $B^{12}$
is the identity on every plane $y=c$, $c\in C$.
Thus $F=B^{12}\circ A^{12}$ fixes every point whose first
two coordinates lie in $C$.

Table~\ref{tab:realize}. The displayed images are obtained
by one or two direct substitutions; no interpolation in
the free integer parameter $t$ is used.

In each row the image under $G^r$ has its first two
coordinates in $C$, and the coordinate preserved by $G$
is a fixed member of $C$ on the original line. Therefore
\begin{equation}\label{eq:lineword}
 W_L=G^{12-r}\circ F\circ G^r
\end{equation}
fixes every point of that line. Indeed, $F$ fixes $G^r(P)$,
and the remaining composition is $G^{12}(P)=P$.
The word is positive and contains both letters, including
when $r=0$. Finally \eqref{eq:E} proves the assertion
on $\E$.
\end{proof}

Propositions~\ref{prop:cover} and \ref{prop:realize}, together
with Corollary~\ref{cor:periodicsmall}, prove the first two
parts of Theorem~\ref{thm:main}. Points on a listed line need
not be periodic for an arbitrary admissible word; their
realizing word is allowed to depend on the line.
