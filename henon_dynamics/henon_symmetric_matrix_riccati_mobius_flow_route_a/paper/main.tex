\section{Exact flow and maximal chart}
Let $X_0\in\Sym(n,\mathbb R)$ and consider
\begin{equation}\label{eq:ode}
 \dot X=I-X^2,\qquad X(0)=X_0.
\end{equation}
Put $C=\cosh t$, $S=\sinh t$.
\begin{theorem}[M\"obius, pole, and limit atlas]\label{thm:atlas}
On the connected component through zero of
$\{t:\det(CI+SX_0)\ne0\}$, the unique classical solution is
\begin{equation}\label{eq:flow}
 F_t(X_0)=(CX_0+SI)(CI+SX_0)^{-1}.
\end{equation}
If $\lambda_1,\ldots,\lambda_n$ are the eigenvalues of $X_0$, the only
finite chart singularities are
\begin{equation}\label{eq:poles}
 t_j=\operatorname{arctanh}(-1/\lambda_j),\qquad |\lambda_j|>1,
\end{equation}
with their spectral multiplicities.  Thus the flow is forward global iff
$\lambda_{\min}(X_0)\ge-1$.  In that case
\begin{equation}\label{eq:limit}
 F_t(X_0)\longrightarrow I-2P_{\ker(X_0+I)}.
\end{equation}
For each $\lambda>-1$ the exact scalar remainder is
\begin{equation}\label{eq:rate}
 f_t(\lambda)-1=
 \frac{2(\lambda-1)e^{-2t}}
 {(1+\lambda)+(1-\lambda)e^{-2t}}.
\end{equation}
Moreover $F_t\circ F_s=F_{t+s}$ whenever both expressions remain in one
classical chart.
\end{theorem}
\begin{proof}
Solve the symmetric block-linear system
\[
 \dot U=V,\qquad \dot V=U,\qquad U(0)=I,\quad V(0)=X_0.
\]
It gives $U=CI+SX_0$ and $V=SI+CX_0$.  While $U$ is invertible,
$X=VU^{-1}$ obeys~\eqref{eq:ode}, proving~\eqref{eq:flow}; local uniqueness
then gives maximality in that chart.  The spectral theorem reduces $F_t$ to
$f_t(\lambda)=(C\lambda+S)/(C+S\lambda)$.  Its denominator vanishes exactly
as in~\eqref{eq:poles}.  For $t>0$ this happens precisely when $\lambda<-1$.
The values $-1$ and $1$ are fixed, every $\lambda>-1$ tends to $1$, and
elementary exponential algebra yields~\eqref{eq:rate}.  Multiplication of
the two scalar M\"obius matrices gives the addition law, hence the semigroup
identity by functional calculus.
\end{proof}

\section{Gradient geometry and the complete equilibrium index}
Define
\begin{equation}\label{eq:phi}
 \Phi(X)=\tr(X^3/3-X).
\end{equation}
The Frobenius gradient is $X^2-I$, so along~\eqref{eq:ode}
\begin{equation}\label{eq:diss}
 \dot\Phi=-\|I-X^2\|_F^2.
\end{equation}
In particular, continuity plus strict decrease rules out every nonconstant
recurrent classical trajectory.

\begin{theorem}[Morse--Bott involution atlas]\label{thm:mb}
The equilibria are exactly the symmetric involutions.  The component with
$p$ eigenvalues $+1$ and $q=n-p$ eigenvalues $-1$ is
\[
 \mathcal E_{p,q}\simeq O(n)/(O(p)\times O(q)).
\]
At every point of this component, the stable, unstable, and center
dimensions of~\eqref{eq:ode} are respectively
\begin{equation}\label{eq:index}
 \frac{p(p+1)}2,\qquad \frac{q(q+1)}2,\qquad pq.
\end{equation}
The center space equals $T\mathcal E_{p,q}$, so the critical manifold of
$\Phi$ is Morse--Bott.
\end{theorem}
\begin{proof}
An equilibrium satisfies $S^2=I$ and is orthogonally conjugate to
$\operatorname{diag}(I_p,-I_q)$, which proves the homogeneous-space claim.
The linearization is $H\mapsto-(SH+HS)$.  Writing a symmetric perturbation
in plus/minus blocks gives eigenvalue $-2$ on the symmetric $p$ block,
$+2$ on the symmetric $q$ block, and zero on the $p\times q$ off-diagonal
block.  Their dimensions are~\eqref{eq:index}; infinitesimal orthogonal
conjugacy identifies the last block with the tangent space.
\end{proof}

\begin{theorem}[Full derivative of the time map]\label{thm:frechet}
In an eigenbasis of $X_0$, on every regular chart,
\begin{equation}\label{eq:loewner}
 (DF_t(X_0)H)_{ij}=
 \frac{H_{ij}}{(C+\lambda_iS)(C+\lambda_jS)}.
\end{equation}
\end{theorem}
\begin{proof}
For $i\ne j$, the divided difference of $f_t$ is
$(f_t(x)-f_t(y))/(x-y)=((C+xS)(C+yS))^{-1}$ because
$C^2-S^2=1$.  Its diagonal limit is $f_t'(x)=(C+xS)^{-2}$.
The standard spectral divided-difference formula, also valid across repeated
eigenvalues by this limit, gives~\eqref{eq:loewner}.
\end{proof}
