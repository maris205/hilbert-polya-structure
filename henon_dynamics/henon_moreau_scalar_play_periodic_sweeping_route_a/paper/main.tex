\section{Segment solution and return map}
Fix $r\geq0$.  Let $u\in W^{1,1}([0,T])$ satisfy
$u(0)=u(T)=m$.  For some $\tau$, assume that $u$ is nondecreasing from
$m$ to $M$ on $[0,\tau]$ and nonincreasing back to $m$ on $[\tau,T]$.
Plateaus are allowed.  Put $D=M-m$ and
\[
 C(t)=[u(t)-r,u(t)+r].
\]
We use $N_C(y)=\{v:v(z-y)\leq0\ \text{for all }z\in C\}$ and solve
\begin{equation}\label{eq:sweep}
 y(t)\in C(t),\qquad-\dot y(t)\in N_{C(t)}(y(t))
 \quad\text{a.e.}
\end{equation}

\begin{theorem}[Projection and Poincar\'e clamp]\label{thm:clamp}
For every $z\in[m-r,m+r]$, problem~\eqref{eq:sweep} with $y(0)=z$ has a
unique absolutely continuous solution.  On a monotone segment beginning at
$t_0$,
\begin{equation}\label{eq:projection}
 y(t)=\Proj_{[u(t)-r,u(t)+r]}y(t_0).
\end{equation}
The return map from one minimum to the next is
\begin{equation}\label{eq:P}
 P(z)=\min\{m+r,\max\{M-r,z\}\}.
\end{equation}
\end{theorem}

\begin{proof}
On a nondecreasing segment define $y(t)=\max\{y(t_0),u(t)-r\}$.
It is feasible.  Before the lower endpoint reaches the state, $\dot y=0$;
when the endpoint pushes it, $y=u-r$, $\dot y=\dot u\geq0$, and
$-\dot y$ lies in the lower-endpoint normal cone.  Feasibility and the
normal-cone sign force these alternatives for any solution, proving
uniqueness.  On a nonincreasing segment the corresponding formula is
$y(t)=\min\{y(t_0),u(t)+r\}$.  Both formulas are~\eqref{eq:projection}
and remain valid almost everywhere across plateaus and corners.

The rise ends at $z_\uparrow=\max\{z,M-r\}$.  The fall ends at
$\min\{z_\uparrow,m+r\}$, proving~\eqref{eq:P}.
\end{proof}

\section{Three chambers and finite-time entrainment}
\begin{theorem}[Complete periodic atlas]\label{thm:atlas}
The map $P$ is increasing, one-Lipschitz, and idempotent.  Every admissible
initial state therefore reaches a periodic response in at most one period.
Its fixed responses are as follows.
\begin{enumerate}
\item If $D<2r$, the fixed set is $[M-r,m+r]$, and each fixed response is
constant.
\item If $D=2r$, this interval collapses to the single constant response
$m+r=M-r$.
\item If $D>2r$, the unique fixed state is $m+r$ and it generates the unique
nonconstant periodic loop.
\end{enumerate}
For a common input, the solution maps preserve initial order and satisfy
$|y_1(t)-y_2(t)|\leq|y_1(0)-y_2(0)|$.
\end{theorem}

\begin{proof}
When $D<2r$, equation~\eqref{eq:P} is scalar projection onto
$[M-r,m+r]$; at equality the interval is a singleton.  When $D>2r$,
$M-r>m+r$, so~\eqref{eq:P} is the constant $m+r$.  This proves all fixed
sets and directly gives $P^2=P$.  A fixed point in the first two chambers
is never pushed through a positive distance.  In the third chamber, it
moves from $m+r$ to $M-r$ and back.  Projection on an interval is increasing
and one-Lipschitz; composing the two segment maps proves the last assertion.
\end{proof}

\section{Variation and dissipation}
Set $s=u-y\in[-r,r]$.  Almost everywhere, either $\dot y=0$ or the moving
endpoint pushes $y$ and $\dot s=0$.  Hence
\begin{equation}\label{eq:variation}
 \Var(u)=\Var(y)+\Var(s).
\end{equation}
On an active rising lower endpoint, $s=r$ and $\dot y\geq0$; on an active
falling upper endpoint, $s=-r$ and $\dot y\leq0$.  Thus
\begin{equation}\label{eq:dissipation}
 \int_0^T s\,dy=r\Var(y).
\end{equation}
For any periodic response,
\begin{equation}\label{eq:closed-var}
 \Var(y)=2\max\{D-2r,0\},\qquad
 \int_0^T s\,dy=2r\max\{D-2r,0\}.
\end{equation}
Indeed, the play is constant for $D\leq2r$; above threshold it travels the
distance $D-2r$ once in each direction.

At $r=0$, $C(t)=\{u(t)\}$, so $y=u$, $s=0$, and every identity survives.
At $D=0$, $C$ is fixed, $P$ is the identity on $[m-r,m+r]$, and every
feasible constant state is periodic.  Equality $D=2r$ has a unique constant
response and zero dissipation.  Plateau duration and $W^{1,1}$ corners do
not change any formula.
