\begin{abstract}
For the periodic binary box--ball system, we freeze every $L\ge2M$, every
soliton content and internal-symmetry sector, and every commuting carrier
evolution $T_l$.  The action--angle bijection and invariant-torus
decomposition are prior theorems of Kuniba--Takagi--Takenouchi and Takagi,
not novelty claims.  From their precise component matrix we derive a single
all-parameter theorem: augmented Smith normal form gives the exact order of
each $T_l$ translation; component formulas aggregate over symmetries,
contents, fixed mass, and the full positive-weight space; M\"obius inversion
gives primitive cycles; and the Artin--Mazur zeta is the reciprocal ordinary
determinant of the finite counting-measure Koopman permutation.  Vacuum,
half filling, and saturated carrier capacity are included.  An independent
periodic-carrier enumeration checks every binary state through $L=9$.
The exact cycles have no intrinsic rational-prime origin or logarithmic
arithmetic clock, so the candidate is mathematically retained and Route-A
rejected.
\end{abstract}
\noindent\textbf{Keywords:} periodic box--ball system; action--angle
variables; internal symmetry; Smith normal form; dynamical zeta; Koopman
permutation.

\section{Frozen source dynamics and action data}
Let $\mathcal P_{L,M}$ be the binary words on a ring of length $L$ with $M$
balls, where $0\le M\le\lfloor L/2\rfloor$.  The capacity-$l$ periodic
combinatorial-$R$ carrier defines $T_l$; one application is one unit of the
discrete physical clock.  KTT prove that the $T_l$ commute and preserve the
soliton content~\cite{KTT}
\[
m=(m_j)_{j\ge1},\qquad \sum_{j\ge1}j m_j=M.
\]
Put $H=\{j:m_j>0\}$ and
\begin{equation}\label{eq:vacancy}
p_j=L-2\sum_{k\in H}\min(j,k)m_k .
\end{equation}
Since $L\ge2M$, every $p_j\ge0$; for $N=\max H$,
$p_N=L-2M$.  Thus equality in the density bound is visible as $p_N=0$,
rather than excluded.

For every $j\in H$, choose
\[
\alpha_j\mid\gcd(m_j,p_j),\qquad \gcd(m,0)=m,
\]
and order $H$ increasingly.  Define
\begin{equation}\label{eq:Fh}
F^{(\alpha)}_{jk}
=\frac{\delta_{jk}p_k+2\min(j,k)m_k}{\alpha_k},
\qquad
h_l=(\min(j,l))_{j\in H}.
\end{equation}
The KTT/Takagi component theorem states that every connected component of
the symmetry-$\alpha$ level is equivariantly bijective to~\cite{Takagi}
\begin{equation}\label{eq:torus}
G_\alpha=\mathbb Z^H/F^{(\alpha)}\mathbb Z^H,
\qquad T_l[x]=[x+h_l].
\end{equation}
It has $D_\alpha=\det F^{(\alpha)}>0$ points.

\section{Augmented Smith order}
Fix one component, abbreviate $F=F^{(\alpha)}$, $h=h_l$, and
$D=\det F$, and let $B=[F\mid h]$.  Write the nonzero Smith factors of $B$
as $e_1,\ldots,e_s$.

\begin{theorem}[Exact component return order]\label{th:order}
The order of $T_l$ on the component is
\begin{equation}\label{eq:order}
q_{\alpha,l}=\min\{n\ge1:nh_l\in F^{(\alpha)}\mathbb Z^H\}
=\frac{D_\alpha}{e_1\cdots e_s}.
\end{equation}
If $UFV=\operatorname{diag}(d_1,\ldots,d_s)$, equivalently
\[
q_{\alpha,l}=\operatorname{lcm}_i
\frac{d_i}{\gcd(d_i,(Uh_l)_i)}.
\]
\end{theorem}

Indeed, quotienting $G_\alpha$ by the cyclic subgroup generated by $[h]$
gives
\[
G_\alpha/\langle[h]\rangle
\cong\mathbb Z^s/(F\mathbb Z^s+\mathbb Zh).
\]
The group on the right has size $e_1\cdots e_s$ by Smith normal form, while
$|G_\alpha|=D$.  Their ratio is the order of the cyclic subgroup, proving
\eqref{eq:order}.  The second formula is the coordinate divisibility
condition after applying $U$.  No determinant ratio is used without its
lattice-index justification.

Since $h_l=(j)_{j\in H}$ once $l\ge\max H$, all larger carriers induce the
same translation on this level.  If $H=\varnothing$, the empty determinant
and empty Smith product are one; the vacuum component is a singleton with
$q=1$.

\section{Fixed points, aggregation, and primitive cycles}
Translation by an element of order $q$ on a finite group has a fixed point
at iterate $n$ precisely when $q\mid n$, in which case every point is fixed.
Therefore one component satisfies
\begin{equation}\label{eq:component-fix}
\#\operatorname{Fix}(T_l^n|G_\alpha)
=D_\alpha\mathbf1_{q_{\alpha,l}\mid n}.
\end{equation}
After multiplying by the number of copies, the entire soliton level has
\begin{equation}\label{eq:level-fix}
F_{m,l}(n)=\sum_\alpha c(\alpha)D_\alpha
\mathbf1_{q_{\alpha,l}\mid n}.
\end{equation}

Because the action variable partitions the binary words,

F_{L,M,l}(n)=\sum_{m:\sum jm_j=M}F_{m,l}(n),
\qquad
F^+_{L,l}(n)=\sum_{M=0}^{\lfloor L/2\rfloor}F_{L,M,l}(n),
\end{equation}
with state totals $\binom LM$ and
$\sum_{M\le L/2}\binom LM$, respectively.  Thus ``all-state'' means an

Let $P_n$ be the number of points of exact period $n$ and $C_n=P_n/n$ the
number of primitive cycles.  M\"obius inversion gives
\begin{equation}\label{eq:primitive}
P_n=\sum_{d\mid n}\mu(n/d)F(d),\qquad C_n=P_n/n.
\end{equation}
Equivalently, one component contributes $D_\alpha/q_{\alpha,l}$ cycles of
length $q_{\alpha,l}$, multiplied by $c(\alpha)$.  This proves integrality
and completeness without a periodic-orbit cutoff.

\section{Zeta and the finite Koopman determinant}
For any one of the aggregations above, let $C_q$ denote its number of
length-$q$ cycles.  Equations \eqref{eq:component-fix}--\eqref{eq:primitive}
give
\begin{equation}\label{eq:zeta}
\zeta_{T_l}(z)
=\exp\!\left(\sum_{n\ge1}\frac{F(n)}n z^n\right)
=\prod_q(1-z^q)^{-C_q}.
\end{equation}
On $\ell^2(X)$ with counting measure, $U_lf=f\circ T_l$ is a canonical
finite unitary permutation.  A $q$-cycle has characteristic polynomial
$x^q-1$, hence
\begin{equation}\label{eq:koopman}
\det(I-zU_l)=\prod_q(1-z^q)^{C_q}=\zeta_{T_l}(z)^{-1}.
\end{equation}
This is the ordinary determinant of the same-clock finite operator.  It is
neither a regularized infinite-volume determinant nor a Hilbert--P\'olya
construction.
