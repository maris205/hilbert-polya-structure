\section{Dihedral geometry and genus}\label{sec:geometry}

The cubic discriminant is
\begin{equation}\label{eq:delta-eta}
 \Delta(\eta)=\Disc_X(f_\eta)
 =16\eta^4+88\eta^2+125.
\end{equation}
It is squarefree: its derivative is $8\eta(8\eta^2+22)$, and direct
substitution shows no common zero.  It is therefore nonsquare in
$\Qbar(\eta)$.  Lemma~\ref{lem:absolute-irreducible} and the cubic Galois
criterion give
\begin{equation}\label{eq:s3-geometric}
 \Gal(L_{\Qbar}/\Qbar(\eta))\simeq S_3.
\end{equation}
Thus $[L_{\Qbar}:\Qbar(\eta)]=6$.  Base extension cannot increase degree,
whereas the splitting field of a cubic over $\Q(\eta)$ has degree at most
six.  Therefore $[L:\Q(\eta)]=6$ as well, equality is preserved after base
change, and the constant field of $L$ is $\Q$.
Since $L$ is a cubic splitting field of degree six, this also gives
\begin{equation}\label{eq:s3-arithmetic}
 \Gal(L/\Q(\eta))\simeq S_3.
\end{equation}

\subsection{The twelve-state action}

Represent a directed state by $(\epsilon,(a,b,c))$, where
$\epsilon\in\{+1,-1\}$ records the sign of $\eta$ and $(a,b,c)$ is an
ordering of the three roots on that sheet.  The chronological shift and
reversal act by
\begin{equation}\label{eq:label-actions}
 \tau(\epsilon,(a,b,c))=(-\epsilon,(b,c,a)),
 \qquad
 \rho(\epsilon,(a,b,c))=(-\epsilon,(b,a,c)).
\end{equation}

\begin{theorem}[Dihedral ordered-cover group]\label{thm:dihedral}
The sheet involution $\iota$ defined by
\[
 \iota(\eta)=-\eta,\qquad
 \iota(\alpha_i)=m_\eta(\alpha_i)
\]
is a well-defined central involution of $L/\Q(A)$.  Moreover,
\begin{equation}\label{eq:d6-group}
 \Gal(L/\Q(A))\simeq S_3\times C_2\simeq\mathsf D_6,
 \qquad |\mathsf D_6|=12.
\end{equation}
The elements \eqref{eq:label-actions} satisfy
\begin{equation}\label{eq:d6-relations}
 \tau^6=\rho^2=1,
 \qquad
 \rho\tau\rho=\tau^{-1},
\end{equation}
and act freely and transitively on the twelve generic states.
\end{theorem}

\begin{proof}
Choose an ordering $\alpha_1,\alpha_2,\alpha_3$ of the roots.  The assignment
\[
 \eta\longmapsto-\eta,
 \qquad
 \alpha_i\longmapsto m_\eta(\alpha_i)
\]
respects the elementary symmetric relations for $f_{-\eta}$ by
Lemma~\ref{lem:matching}.  The universal property of a splitting field
therefore extends it to a $\Q(A)$-embedding $L\hookrightarrow L$.
Equation~\eqref{eq:inverse-identity} supplies the inverse assignment, so the
embedding is an involutive automorphism $\iota$.  Since the matching map is
applied coordinatewise, $\iota$ commutes with every permutation of the three
roots.  Consequently the extension over $\Q(A)$ contains the direct product
$S_3\times\langle\iota\rangle$.  Proposition~\ref{prop:recovery}
shows that the ordered-edge field contains $\eta$ and the full splitting
field, so its generic degree over $\Q(A)$ is $2\cdot6=12$; there is no larger
group.

If $c$ denotes a root three-cycle and $s$ a transposition, then
$\tau=\iota c$ and $\rho=\iota s$.  Hence
$\tau^2=c^2$ generates $A_3$, $\tau^3=\iota$, and
$s c s=c^{-1}$ gives \eqref{eq:d6-relations}.  Formula
\eqref{eq:label-actions} has one orbit of size twelve, proving freeness and
transitivity.  This also identifies the chronological action with the
ordered-edge shift in \eqref{eq:edge-shift}.
\end{proof}

\subsection{Branching over the radical line}

We compute genus through the certified degree-six $S_3$ cover
\begin{equation}\label{eq:eta-cover}
 E_6\longrightarrow\Pone_\eta.
\end{equation}
The four roots of \eqref{eq:delta-eta} are distinct.  Because the cubic is
monic, its splitting algebra is finite étale at every finite place where the
discriminant is a unit; hence there is no other finite branch value.  At each
root of the discriminant the cubic has one double and one simple root, so the
inertia is a transposition.  In
the regular degree-six cover, each branch value contributes
$6(1-1/2)=3$ to Riemann--Hurwitz.  It remains to exclude hidden inertia at
infinity.

\begin{lemma}[Unramified infinity]\label{lem:infinity}
The cover \eqref{eq:eta-cover} is unramified at $\eta=\infty$.
\end{lemma}

\begin{proof}
Put $t=1/\eta$ and $X=u/t$.  Equation \eqref{eq:F-plane} becomes
\begin{equation}\label{eq:infinity-chart}
 t^3F(1/t,u/t)
 =(u-1)^2(u+1)+t(1-u^2)+3t^2(1-u)+2t^3.
\end{equation}
The slope $u=-1$ is simple modulo $t$ and lifts to a branch over
$\Qbar[[t]]$.  At the double slope put $u=1+ct$ and divide the transformed
equation by $t^2$.  Its reduction at $t=0$ is
\[
 2c(c-1).
\]
The two roots $c=0,1$ are simple, so both lift over $\Qbar[[t]]$ as well.
All three $X$-roots lie in $t^{-1}\Qbar[[t]]$ as distinct Laurent series.
The splitting field therefore has trivial inertia at infinity.
\end{proof}

\begin{theorem}[Genus of the ordered cover]\label{thm:genus}
The smooth projective ordered-edge curve has genus
\begin{equation}\label{eq:genus-one}
 \boxed{g(E_6)=1.}
\end{equation}
\end{theorem}

\begin{proof}
The four finite transposition values contribute twelve in total, while
Lemma~\ref{lem:infinity} contributes nothing at infinity.
Riemann--Hurwitz for \eqref{eq:eta-cover} gives
\[
 2g(E_6)-2=6(-2)+4\cdot3=0.
\]
\end{proof}

\begin{remark}
The branch theorem just proved is over $\Pone_\eta$, which is sufficient for
the genus computation.  We do not promote the scalar discriminant
\eqref{eq:p6-disc} to a separately certified complete branch signature for
the degree-twelve cover over $\Pone_A$.
\end{remark}
