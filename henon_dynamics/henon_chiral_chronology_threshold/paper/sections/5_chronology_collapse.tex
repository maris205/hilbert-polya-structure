\section{The rotation quotient and cohomological collapse}\label{sec:collapse}

Let $\alpha,\beta,\gamma$ be the three roots of $f_\eta$ and put
\begin{equation}\label{eq:vandermonde}
 w=(\alpha-\beta)(\alpha-\gamma)(\beta-\gamma),
 \qquad
 w^2=\Delta(\eta).
\end{equation}
The $A_3$-fixed field in the cubic splitting field is $\Q(\eta,w)$.
The sheet involution does not fix $w$.

\begin{lemma}[Vandermonde sign]\label{lem:vandermonde-sign}
The central sheet involution acts by
\begin{equation}\label{eq:iota-vandermonde}
 \iota:(\eta,w)\longmapsto(-\eta,-w).
\end{equation}
\end{lemma}

\begin{proof}
The matching map sends a difference of two roots to
\[
 m_\eta(\alpha_i)-m_\eta(\alpha_j)
 =(\alpha_i-\alpha_j)(\alpha_i+\alpha_j).
\]
It therefore multiplies the Vandermonde by
\[
 \prod_{i<j}(\alpha_i+\alpha_j)=e_1e_2-e_3.
\]
For \eqref{eq:feta}, Vieta gives
$e_1=1+\eta$, $e_2=-A$, and $e_3=1-A(1+\eta)$, so
$e_1e_2-e_3=-1$.  The involution also sends $\eta$ to $-\eta$, proving
\eqref{eq:iota-vandermonde}.
\end{proof}

Recall from the proof of Theorem~\ref{thm:dihedral} that, if $c$ is a root
three-cycle,
\begin{equation}\label{eq:tau-subgroup}
 \tau=\iota c,\qquad
 \tau^2\text{ generates }A_3,\qquad
 \tau^3=\iota.
\end{equation}
Hence $\langle\tau\rangle=A_3\times\langle\iota\rangle$ inside
$S_3\times C_2$.

\begin{proposition}[Exact rotation fixed field]\label{prop:fixed-field}
Let $v=\eta w$.  Then
\begin{equation}\label{eq:fixed-field}
 \boxed{
 L^{\langle\tau\rangle}=\Q(A,v),
 \qquad
 v^2=(A-3)(16A^2-8A+5).}
\end{equation}
\end{proposition}

\begin{proof}
Equation \eqref{eq:iota-vandermonde} makes $v=\eta w$ invariant under
$\iota$, and $w$ is invariant under $A_3$.  Thus
$\Q(A,v)\subseteq L^{\langle\tau\rangle}$.  Moreover,
\[
 v^2=\eta^2\Delta(\eta)
 =(A-3)(16A^2-8A+5).
\]
The right-hand side has valuation one at $A=3$ (the quadratic factor equals
$125$ there), so it is not a square in $\Q(A)$ and the displayed field has
degree two over $\Q(A)$.  The subgroup $\langle\tau\rangle$ has order six
and index two in $\mathsf D_6$.  Its fixed field also has degree two.
Inclusion plus equality of degrees proves \eqref{eq:fixed-field}.
\end{proof}

Let $B_6$ be the smooth projective curve with function field
$\Q(A,v)$.

\begin{lemma}[Genus and unramified quotient]\label{lem:quotient-genus}
The curve $B_6$ has genus one, and the quotient map
\begin{equation}\label{eq:cyclic-quotient}
 E_6\longrightarrow B_6=E_6/\langle\tau\rangle
\end{equation}
is an unramified cyclic cover of degree six.
\end{lemma}

\begin{proof}
The cubic
\[
 (A-3)(16A^2-8A+5)
\]
has discriminant $-4{,}000{,}000$ and is therefore squarefree.  The smooth
projective model of its hyperelliptic equation has genus one.  Applying
Riemann--Hurwitz to the degree-six quotient and using
$g(E_6)=g(B_6)=1$ gives
\[
 0=6\cdot0+R.
\]
The total ramification $R$ is zero.
\end{proof}

\begin{theorem}[Chronology--cohomology collapse]\label{thm:collapse}
After base change to $\Qbar$ and a choice of origin on the genus-one curve
$E_6$, the chronological generator $\tau$ is translation by a point of
exact order six.  It acts trivially on geometric weight-one cohomology.  In
particular, over $\C$,
\begin{equation}\label{eq:tau-charpoly}
 \boxed{
 \chi_{\tau^*}(T)=(T-1)^2,
 \qquad
 \mu_{\tau^*}(T)=T-1.}
\end{equation}
For every prime $\ell$, the same identity holds on
$H^1_{\etale}(E_{6,\Qbar},\Q_\ell)$.  As a rational
$\mathsf D_6$-representation,
\begin{equation}\label{eq:reflection-sign}
 \boxed{
 H^1(E_6(\C),\Q)
 \simeq\varepsilon_{\mathrm{refl}}^{\oplus2},}
\end{equation}
where rotations act by $+1$ and reflections by $-1$.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:quotient-genus} says that the cyclic action has no fixed
point.  Choose a geometric origin on $E_6$.  Any automorphism of a genus-one
curve then has the form $t_P\circ\varphi$, where $t_P$ is translation and
$\varphi$ fixes the origin.  If $\varphi\ne1$, the nonzero endomorphism
$1-\varphi$ is an isogeny and is surjective over $\Qbar$; the equation
$(1-\varphi)Q=P$ would give a fixed point.  Thus $\varphi=1$ and $\tau=t_P$.
The exact order of $\tau$ in Theorem~\ref{thm:dihedral} makes $P$ an exact
six-torsion point.  Translation acts trivially on Betti and \(\ell\)-adic
cohomology, proving \eqref{eq:tau-charpoly}.

Since the rotation subgroup is trivial on the two-dimensional $H^1$, the
representation factors through
$\mathsf D_6/\langle\tau\rangle\simeq C_2$.  The invariants of the full
group have dimension $2g(E_6/\mathsf D_6)=2g(\Pone_A)=0$.  The reflection
therefore acts by $-1$ on both dimensions, giving
\eqref{eq:reflection-sign}.
\end{proof}

\begin{remark}
The theorem does not assert that $E_6$ has a chosen $\Q$-rational elliptic
origin, nor that the translating torsion point is rational over $\Q$.  The
translation statement is geometric.  The cohomological identity is
independent of the choice of origin.
\end{remark}
