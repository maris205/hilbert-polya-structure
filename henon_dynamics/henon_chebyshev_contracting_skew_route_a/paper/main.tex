\begin{abstract}
We freeze the triangular polynomial dynamics
$F(x,y)=(4x^3-3x,y/4+x)$ and prove a complete all-period theorem.
The base iterate is $T_{3^n}$; its fixed polynomial has exactly $3^n$
distinct real roots, and contraction gives one closing fiber coordinate above
each root.  Thus $\#\Fix(F^n)=3^n$, M\"obius inversion gives every primitive
orbit count, and the orbit-owned Artin--Mazur zeta is $(1-3z)^{-1}$.  We also
classify every fixed-point multiplier, stability determinant, orientation, and
primitive repetition.  Two exact controls show that unit fiber multiplier
destroys isolated closure and that the nearby-looking cubic $4x^3-2x$ creates
multiple roots and a neutral two-cycle.  This replaces another finite orbit
prefix by an all-period source theorem, but supplies no weighted target-facing
Fredholm determinant, analytic completion, or natural lift.
\end{abstract}

\section{Source lock and progress criterion}
Let
\[
 f(x)=T_3(x)=4x^3-3x,\qquad
 F(x,y)=\left(f(x),\frac14y+x\right),
\]
on $\mathbb R^2$, with one application of $F$ as one clock unit.  The
normalization is the unweighted isolated fixed-point count, and
\[
 \zeta_F(z)=\exp\!\left(\sum_{n\ge1}\#\Fix(F^n)\frac{z^n}{n}\right).
\]
No source parameter is fitted and no external target data are used.  The
progress criterion is all-period completeness: a larger finite word table, a
new low-period orbit, or a tangent matrix alone would not qualify.  The
period-twelve receipt later cited is a replay layer, not the proof horizon.

\section{Complete periodic atlas}
\begin{theorem}\label{thm:atlas}
For every $n\ge1$, $f^n=T_{3^n}$.  The polynomial
$T_{3^n}(x)-x$ has exactly $3^n$ distinct real roots, each with one unique
fiber lift fixed by $F^n$.  The lift preserves least period.  Consequently
$\#\Fix(F^n)=3^n$.
\end{theorem}
\begin{proof}
Put $m=3^n$.  The cosine definition gives
$T_a\circ T_b=T_{ab}$ and hence $f^n=T_m$.  Writing $x=\cos\theta$ with
$0\le\theta\le\pi$, the equation $\cos(m\theta)=\cos\theta$ yields
\[
 \left\{\cos\frac{2\pi k}{m-1}:0\le k\le\frac{m-1}{2}\right\},\qquad
 \left\{\cos\frac{2\pi k}{m+1}:0\le k\le\frac{m+1}{2}\right\}.
\]
Both lists are injective.  Since $\gcd(m-1,m+1)=2$, they intersect exactly at
$\{1,-1\}$ and contain
$(m+1)/2+(m+3)/2-2=m$ points in total.  At an interior root,
\[
 T_m'(x)=m\frac{\sin(m\theta)}{\sin\theta}=\pm m,
\]
while $T_m'(\pm1)=m^2$.  Therefore $(T_m-x)'$ never vanishes at a root,
so the degree-$m$ list is complete and simple.

Fiber iteration gives
\[
 y_n=4^{-n}y+\sum_{j=0}^{n-1}4^{-(n-1-j)}T_{3^j}(x).
\]
Because $1-4^{-n}\ne0$, every base root has the unique closing coordinate
\begin{equation}\label{eq:fiber}
 y_*(x,n)=\frac{\sum_{j=0}^{n-1}4^{-(n-1-j)}T_{3^j}(x)}{1-4^{-n}}.
\end{equation}
For a base point of least period $p\mid n$, the unique $F^p$ lift is already
fixed by $F^n$, so uniqueness identifies it with \eqref{eq:fiber}.  Projection
to the base prevents any period reduction.
\end{proof}

\section{Primitive counts and orbit-owned zeta}
Let $E_n$ count exact-period points and $P_n$ primitive orbits.  M\"obius
inversion of Theorem~\ref{thm:atlas} gives
\begin{equation}\label{eq:primitive}
 E_n=\sum_{d\mid n}\mu(d)3^{n/d},\qquad
 P_n=\frac1n\sum_{d\mid n}\mu(d)3^{n/d}.
\end{equation}

The trace series therefore closes exactly:
\begin{equation}\label{eq:zeta}
 \zeta_F(z)=\exp\left(\sum_{n\ge1}3^n\frac{z^n}{n}\right)
 =\frac1{1-3z}
 =\prod_{\gamma\ \mathrm{primitive}}(1-z^{p_\gamma})^{-1}.
\end{equation}
The logarithmic definition holds for $|z|<1/3$; the rational expression gives
its meromorphic continuation.  Equation~\eqref{eq:zeta} is an unweighted
source Artin--Mazur zeta, not a target-facing weighted Fredholm determinant.

\section{All-period stability, orientation, and repetition}
At a point fixed by $F^n$,
\begin{equation}\label{eq:jacobian}
 DF^n=\begin{pmatrix}T_m'(x)&0\\c_n(x)&4^{-n}\end{pmatrix},\qquad
 c_n(x)=\sum_{j=0}^{n-1}4^{-(n-1-j)}(T_{3^j})'(x).
\end{equation}
The two endpoints have unstable multiplier $m^2$; among the interior points,
$(m-3)/2$ have multiplier $+m$ and $(m-1)/2$ have multiplier $-m$.
Every fixed point is thus a saddle and
\begin{equation}\label{eq:stability}
 \det(I-DF^n)=(1-T_m'(x))(1-4^{-n})\ne0.
\end{equation}
The positive and negative unstable-orientation counts are $(m+1)/2$ and
$(m-1)/2$.  If $E_p^-$ counts exact-period points with negative primitive
orientation, then $(3^n-1)/2=\sum_{p\mid n,\,n/p\ \mathrm{odd}}E_p^-$;
odd-divisor M\"obius inversion gives
\[
 E_n^-=\frac12\sum_{\substack{d\mid n\\d\ \mathrm{odd}}}
 \mu(d)(3^{n/d}-1),\qquad P_n^-=E_n^-/n,
\]
which gives the final two rows of Table~\ref{tab:counts}.

For a primitive orbit $\gamma$ of period $p$ and unstable multiplier
$\alpha_\gamma$, its $r$-fold repetition has multipliers
$\alpha_\gamma^r$ and $4^{-pr}$.  Hence the full repetition law is
\begin{equation}\label{eq:repeat}
 \det(I-DF^{pr})=(1-\alpha_\gamma^r)(1-4^{-pr}),\qquad
 \operatorname{or}(\gamma^r)=\operatorname{sgn}(\alpha_\gamma)^r.
\end{equation}
For $p>1$, $\alpha_\gamma=\pm3^p$; at period one the endpoint multipliers are
$9$ and the central multiplier is $-3$.

\section{Exact controls}
The fiber contraction is not cosmetic.  For
$F_1(x,y)=(T_3(x),y+x)$, period-one closure requires $x=0$: the base point
$x=0$ carries an entire fixed line, while $x=\pm1$ admits no closing fiber.
The stable factor in \eqref{eq:stability} becomes $1-1=0$.

The Chebyshev coefficient is equally decisive.  For $g(x)=4x^3-2x$,
\begin{align*}
 g^2(x)-x&=x(2x-1)^3(2x+1)^3(4x^2-3),\\
 g^2(x)-T_9(x)&=x(192x^6-240x^4+80x^2-5).
\end{align*}
Thus the second fixed equation has only five distinct roots rather than nine;
$\pm1/2$ form a neutral period-two orbit.  Both controls destroy named theorem
clauses exactly, rather than degrading a fitted score.
