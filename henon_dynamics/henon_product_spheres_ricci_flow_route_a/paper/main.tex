\begin{abstract}
For arbitrary $m\geq1$, dimensions $d_i\geq1$, and scales $a_i>0$, we solve
the unnormalized Ricci flow of the product metric
$\bigoplus_i a_i g_{S^{d_i}}$ exactly.  Every tied first-collapse factor is
retained, yielding sharp scalar-curvature, full-curvature, volume, diameter,
and Type-I blowup laws.
\ifnum\CRevisionRound>0\space We derive the constant-volume conjugacy and
prove that partial collapse has finite normalized lifetime, whereas full
collapse is exactly the curved Einstein face and becomes a stationary
forward-eternal normalized solution.\space\fi
\ifnum\CRevisionRound>1 The sharp normalized-time asymptotics, pointed
product-shrinker limit, flat-circle and all-torus faces, executable
reconstruction, collision audit, and strict Route-A obstruction are closed.\fi
\space This is a nonlinear metric-evolution theorem, not a heat-trace,
spectral-zeta, or determinant construction.
\end{abstract}

\section{Frozen geometric owner}
Let
\[
 M=\prod_{i=1}^{m}S^{d_i},\qquad d_i\in\mathbb Z_{\geq1},
 \qquad g(0)=\bigoplus_{i=1}^{m}a_i g_i,\quad a_i>0,       \tag{1}
\]
where $g_i$ is the unit round metric.  Put $c_i=d_i-1$ and
$n=\sum_i d_i$.  We use the unnormalized equation
\begin{equation}
 \partial_tg=-2\Ric(g)                                    \tag{2}
\end{equation}
with its physical clock.  Hamilton's original program fixes (2) as a
geometric evolution equation \cite{Hamilton1982}; Chow--Knopf provide standard
special-geometry, scaling, and singularity conventions \cite{ChowKnopf2004}.
Those sources are cited only for equation lineage and vocabulary.  All
product, endpoint, and normalization claims below are derived here.

For $d_i\geq2$ define $T_i=a_i/(2c_i)$; for $d_i=1$ define
$T_i=\infty$.  This convention is essential: a circle is Ricci-flat, not a
factor with an undefined clock.

\begin{theorem}[exact flow and maximal interval]\label{thm:flow}
The solution of (2) is
\begin{equation}
 g(t)=\bigoplus_i a_i(t)g_i,\qquad a_i(t)=a_i-2c_it.       \tag{3}
\end{equation}
If some $c_i>0$, its maximal Riemannian interval is $(-\infty,T)$, where
$T=\min_iT_i$.  If every $d_i=1$, (3) is stationary for all $t\in\R$.
\end{theorem}
\begin{proof}
As a $(0,2)$ tensor, the Ricci tensor is unchanged by constant metric
rescaling.  The unit round factor satisfies $\Ric(g_i)=c_i g_i$, and the Ricci
tensor of a Riemannian product is the direct sum of factor tensors.  Hence
(2) is exactly $a_i'=-2c_i$, proving (3).  The coefficients remain positive
precisely on the stated interval.  The displayed solution solves the full
equation; compact Ricci-flow uniqueness identifies it with the ambient
solution while it exists.
\end{proof}

\section{Closed geometric observables}
Mixed sectional curvatures vanish, while the $i$th factor has sectional
curvature $a_i(t)^{-1}$.  Consequently
\begin{align}
 R(t)&=\sum_i\frac{d_ic_i}{a_i(t)},
 &|\Ric|^2(t)&=\sum_i\frac{d_ic_i^2}{a_i(t)^2},             \tag{4}\\
 |\Rm|^2(t)&=\sum_i\frac{2d_ic_i}{a_i(t)^2},
 &\frac{V(t)}{V(0)}&=\prod_i\left(\frac{a_i(t)}{a_i}\right)^{d_i/2}.
                                                                    \tag{5}
\end{align}
Here $V(t)=\Vol(M,g(t))$.  Differentiation gives
\begin{equation}
 \frac{d}{dt}\log V(t)=-R(t).                              \tag{6}
\end{equation}
The product distance also gives the exact diameter
\begin{equation}
 \diam(M,g(t))=\pi\sqrt{\sum_i a_i(t)}.                    \tag{7}
\end{equation}

The initial product is Einstein with positive constant $\lambda$ exactly when
every factor is curved and $c_i/a_i=\lambda$ for every $i$.  Equivalently,
all finite clocks coincide at $T=(2\lambda)^{-1}$, and then
\begin{equation}
 g(t)=(1-t/T)g(0).                                         \tag{8}
\end{equation}
If all factors are circles, the product is instead Ricci-flat for arbitrary
scales.  A mixture of flat and curved factors cannot be Einstein.

\section{Tied collapse and the full Type-I model}
Assume $T<\infty$ and freeze the complete first-collapse set
\[
 I=\{i:T_i=T\},\qquad D=\sum_{i\in I}d_i.                 \tag{9}
\]
No generic perturbation is used to split a tie.  For $i\in I$,
$a_i(t)=2c_i(T-t)$; for $j\notin I$, $a_j(T)>0$.

\begin{theorem}[sharp singularity atlas]\label{thm:sing}
As $t\uparrow T$,
\begin{align}
 (T-t)R(t)&\longrightarrow \frac D2,                       \tag{10}\\
 (T-t)^2|\Ric|^2(t)&\longrightarrow \frac D4,              \tag{10a}\\
 (T-t)^2|\Rm|^2(t)&\longrightarrow
       \sum_{i\in I}\frac{d_i}{2c_i},                    \tag{11}\\
 V(t)&\sim C_I(T-t)^{D/2},                                 \tag{12}
\end{align}
where
\[
 C_I=\prod_i\Vol(S^{d_i},g_i)
     \prod_{i\in I}(2c_i)^{d_i/2}
     \prod_{j\notin I}a_j(T)^{d_j/2}.
\]
The singularity is Type I and
\begin{equation}
 \diam(M,g(t))\longrightarrow
 \pi\sqrt{\sum_{j\notin I}a_j(T)}.                       \tag{13}
\end{equation}
For any $t_k\uparrow T$, let $Q_k=(T-t_k)^{-1}$ and
$g_k(s)=Q_kg(T+s/Q_k)$ for $s<0$.  At product basepoints,
\begin{equation}
 (M,g_k(s))\longrightarrow
 \prod_{i\in I}\bigl(S^{d_i},-2c_isg_i\bigr)
 \times(\R^{n-D},g_{\rm E})                              \tag{14}
\end{equation}
smoothly on compact pointed sets.
\end{theorem}
\begin{proof}
Substitution of $a_i(t)=2c_i(T-t)$ into (4)--(5) proves
(10), (10a)--(12); every noncollapsing summand stays bounded.  Formula (11) gives
both the Type-I upper bound and a nonzero Type-I residue, while (7) gives
(13).  In the rescaled flow, a collapsing coefficient is exactly
$-2c_is$.  A survivor has coefficient $Q_ka_j(T)-2c_js$, which diverges.
A round factor with scale tending to infinity converges pointed smoothly to
its Euclidean tangent space.  Taking products proves (14).
\end{proof}

\section{Constant-volume conjugacy and endpoint theorem}
Define
\begin{equation}
 C(t)=\left(\frac{V(0)}{V(t)}\right)^{2/n},\qquad
 \tau(t)=\int_0^t C(s)\,ds,\qquad
 \widehat g(\tau(t))=C(t)g(t).                            \tag{15}
\end{equation}

\begin{theorem}[partial versus full collapse]\label{thm:norm}
The metric $\widehat g$ has volume $V(0)$ and satisfies
\begin{equation}
 \partial_\tau\widehat g=-2\Ric(\widehat g)
       +\frac2n\widehat R\,\widehat g.                   \tag{16}
\end{equation}
It is ancient in normalized time.  At the forward endpoint:
\begin{enumerate}
\item If $D=n$, then every factor is curved, (1) is Einstein, (8) holds,
      $C(t)=(1-t/T)^{-1}$, and
      $\tau=-T\log(1-t/T)$.  Thus $\widehat g\equiv g(0)$ and the
      normalized lifetime is infinite.
\item If $D<n$, then $\tau_T=\lim_{t\uparrow T}\tau(t)<\infty$.
      Every normalized collapsing scale tends to zero, every survivor scale
      tends to infinity, and normalized curvature blows up.
\end{enumerate}
All-flat products form the separate stationary eternal case $C=1$, $\tau=t$.
\end{theorem}
\begin{proof}
Equation (15) makes $C^{n/2}V=V(0)$.  From (6),
$C'/C=2R/n$.  Constant rescaling leaves $\Ric$ fixed as a $(0,2)$ tensor and
gives $\widehat R=R/C$.  Dividing $d(Cg)/dt$ by $d\tau/dt=C$ yields (16).

At backward infinity, if $D_c$ is the total dimension of all curved factors,
then $C(t)$ is comparable to $|t|^{-D_c/n}$.  Its integral diverges for
$D_c<n$, and diverges logarithmically for $D_c=n$; hence normalized time tends
to $-\infty$.  At $T$, (12) gives
$C(t)\asymp(T-t)^{-D/n}$.  This is integrable exactly when $D<n$.  If
$D=n$, all clocks coincide and the Einstein calculation gives the explicit
stationary normalization.  If $D<n$, multiplication of collapsed affine
scales and surviving positive scales by $C(t)$ proves the asserted limits.
\end{proof}

\section{Sharp normalized-time asymptotics}
The preceding proof contains exact leading information.  Put
$\beta=D/n<1$.  There is $C_*>0$ such that, as $t\uparrow T$,
\begin{align}
 C(t)&\sim C_*(T-t)^{-\beta},                              \tag{17}\\
 \tau_T-\tau(t)&\sim\frac{C_*}{1-\beta}
                  (T-t)^{1-\beta}.                        \tag{18}
\end{align}
Thus, for $i\in I$,
\begin{equation}
 \widehat a_i(\tau)\sim
 2c_i(1-\beta)(\tau_T-\tau),                              \tag{19}
\end{equation}
while a survivor satisfies
$\widehat a_j\asymp(\tau_T-\tau)^{-\beta/(1-\beta)}$.
Moreover
\begin{equation}
 \widehat R(\tau)\sim
 \frac{D}{2(1-\beta)(\tau_T-\tau)}.                       \tag{20}
\end{equation}
Equations (17)--(20) show that finite normalized lifetime is a genuine
curvature singularity, not merely escape of the chosen coordinates.

\begin{corollary}[equivalence closure]
For nonflat data the following are equivalent: full first collapse; $D=n$;
positive Einstein initial metric; homothetic unnormalized shrinking;
stationary normalized flow; and infinite forward normalized lifetime.
Every other nonflat product has partial collapse and a finite normalized
singularity.  A flat factor therefore forces the partial branch whenever a
curved factor is present.
\end{corollary}
