\section{The chronological dyadic-solenoid system}
\label{sec:system}

Let \(R=\Z[1/2]\) and let \(\Xtwo=\widehat{R^2}\) be its Pontryagin dual.
The group \(R^2\) is torsion-free of rank two, so \(\Xtwo\) is a connected
compact abelian group of topological dimension two.  For
\(M\in\GL_2(R)\), define
\[
 \alpha_M=\widehat{M^{\mathsf T}}.
 \tag{3.1}\label{3.1}
\]
Pontryagin duality reverses composition; the transpose therefore gives
\(\alpha_M\circ\alpha_N=\alpha_{MN}\).

Let \(\SigmaTwo=\{a,b\}^{\Z}\) with the left shift \(\sigma\), and let
\(M_a=A\), \(M_b=B\), with \(A,B\) from \eqref{1.1}.  We study the
autonomous skew product
\[
 F(\omega,x)=\bigl(\sigma\omega,\alpha_{M_{\omega_0}}x\bigr).
 \tag{3.2}\label{3.2}
\]
Because \(\det A=\det B=8\) is a unit in \(R\), both fibre maps are
automorphisms.  With the fair \((1/2,1/2)\) Bernoulli measure on the base and
Haar measure on the
fibre, \(F\) is invertible and measure preserving.

If \(\omega\) is fixed by \(\sigma^n\), its based period block
\(w=w_0\cdots w_{n-1}\) has return matrix
\[
 M_w=M_{w_{n-1}}\cdots M_{w_0}.
 \tag{3.3}\label{3.3}
\]
A cyclic rotation conjugates \(M_w\) in \(\GL_2(R)\), and
\(M_{w^r}=M_w^r\).  Hence fixed data descend to base necklaces while their
repetitions retain the true monodromy.

\begin{proposition}[Uniform expansion and sign]
\label{prop:expansion}
For every word \(w\) of length \(n\), every eigenvalue of \(M_w\) has
modulus greater than one and
\[
 D_w:=\det(I-M_w)=8^n-\tr M_w+1>0.
 \tag{3.4}\label{3.4}
\]
If
\[
 \beta=\frac{7+\sqrt{17}}2,
 \qquad
 \gamma=\frac{7-\sqrt{17}}2,
\]
then
\[
 8^n-2\beta^n+1\le D_w\le8^n+1.
 \tag{3.5}\label{3.5}
\]
\end{proposition}

\begin{proof}
The matrices \(A\) and \(B\) are symmetric positive definite.  Their least
singular values are \(2\) and \(\gamma>1\), while their largest singular
values are \(4\) and \(\beta\).  Thus
\(s_{\min}(M_w)\ge\gamma^n\) and \(\lVert M_w\rVert_2\le\beta^n\).
The first inequality places every eigenvalue outside the unit disk.  All
entries of \(M_w\) are positive.  If its eigenvalues are real, their positive
product and positive sum make both positive; if they are nonreal, they are a
conjugate pair.  In either case \(\det(I-M_w)>0\).  Equation \eqref{3.4}
uses \(\det M_w=8^n\), and \eqref{3.5} follows from
\(0<\tr M_w\le2\lVert M_w\rVert_2\).
\end{proof}

\begin{theorem}[Intrinsic fixed-point index]
\label{thm:index}
For every nonempty word \(w\), the fibre fixed group is finite and
\[
 \#\Fix(\alpha_{M_w})
 =|D_w|_\infty |D_w|_2
 =\frac{D_w}{2^{\vTwo(D_w)}}
 =\oddpart(D_w).
 \tag{3.6}\label{3.6}
\]
\end{theorem}

\begin{proof}
Dualizing \(I-M_w^{\mathsf T}\) identifies the fixed group with the dual of
\(R^2/(I-M_w^{\mathsf T})R^2\).  Proposition~\ref{prop:expansion} makes the
determinant nonzero.  Smith normal form over the localized principal ideal
domain \(R\) shows that localization removes exactly the powers of two from
the finite integer cokernel.  Its remaining order is
\(D_w/2^{\vTwo(D_w)}\).  The local-product expression is the same identity in
normalized absolute-value notation.
\end{proof}

The full fixed-point count is therefore
\[
 \Nsol_n=\#\Fix(F^n)
 =\sum_{w\in\{a,b\}^n}\oddpart(D_w),
 \qquad
 \Zsol(z)=\exp\left(\sum_{n\ge1}\frac{\Nsol_n}{n}z^n\right).
 \tag{3.7}\label{3.7}
\]
As an exact control, omit the \(2\)-adic factor and count the corresponding
degree-eight torus endomorphisms.

\begin{proposition}[Archimedean comparison]
\label{prop:arch}
Let \(\Ninf_n=\sum_{|w|=n}D_w\).  Then
\[
 \Ninf_n=2^n+16^n-\tr\bigl((A+B)^n\bigr)
 \tag{3.8}\label{3.8}
\]
and
\[
 \Zinf(z)
 =\exp\left(\sum_{n\ge1}\frac{\Ninf_n}{n}z^n\right)
 =\frac{1-13z+33z^2}{(1-2z)(1-16z)}.
 \tag{3.9}\label{3.9}
\]
\end{proposition}

\begin{proof}
Distributivity gives the exact chronological identity
\(\sum_{|w|=n}M_w=(A+B)^n\).  This is a sum of all ordered products, not a
replacement of individual returns by an averaged orbit.  Summing
\eqref{3.4} yields \eqref{3.8}.  Finally,
\(\tr(A+B)=13\), \(\det(A+B)=33\), and
\(\log\det(I-z(A+B))=-\sum_{n\ge1}\tr((A+B)^n)z^n/n\), which gives
\eqref{3.9}.
\end{proof}
