\begin{abstract}
For the continuous-time branching chain with per-particle birth and death
rates $\lambda,\mu\geq0$, we prove one formula package valid for every initial
population $z\in\Nzero$.  The one-ancestor probability generating function
is a M\"obius semigroup.  Off criticality its clock is
$\delta=e^{-(\lambda-\mu)t}$; at criticality the nonsingular clock is
$\tau=\lambda t$.  Its zero-modified geometric expansion yields the exact
$z$-ancestor transition as a binomial mixture over surviving ancestral
lineages, with a conditional negative-binomial sum.  This prevents the
common, nonuniform replacement by one negative-binomial law.  Differentiation
gives the mean and variance, including all pure and zero-rate boundaries.
\ifnum\CRevisionRound>0
The same formulas prove the subcritical geometric quasi-stationary law, the
critical Yaglom exponential scaling, and the supercritical martingale limit
as an atom plus a binomial mixture of gamma laws.
\fi

Independent exact arithmetic, symbolic limits, byte replay and hostile

The source has no intrinsic prime arithmetic or target spectral object and

\fi
\end{abstract}

\section{Source lock and model}
Let $Z=(Z_t)_{t\geq0}$ be the minimal continuous-time Markov chain on
$\Nzero$ with generator
\begin{equation}\label{eq:generator}
 (Gf)(n)=\lambda n\{f(n+1)-f(n)\}
          +\mu n\{f(n-1)-f(n)\},
 \qquad \lambda,\mu\geq0.
\end{equation}
State zero is absorbing and $Z_0=z\in\Nzero$.  A Yule process with rate
$\lambda$ dominates the number of births, so the chain is nonexplosive at
every finite time.  Distinct initial particles found independent descendant
families.  Consequently, if
\[
 F_t(s)=\mathbb E_1[s^{Z_t}],\qquad 0\leq s\leq1,
\]
then $\mathbb E_z[s^{Z_t}]=F_t(s)^z$.  This precise source family and its
linear growth terminology are locked to the classical report of Karlin and
McGregor~\cite{KM58}; no priority claim is made here.

\section{The finite-time theorem}
Put $r=\lambda-\mu$.  For $r\ne0$ define
$\delta=e^{-rt}$; for $r=0$ put $\tau=\lambda t$.

\paragraph{Theorem 1 (M\"obius PGF semigroup).}
For all $\lambda,\mu\geq0$ and $t\geq0$,
\begin{align}
 r\ne0:\quad
 F_t(s)&=\frac{\mu(1-s)-\delta(\mu-\lambda s)}
 {\lambda(1-s)-\delta(\mu-\lambda s)}, \label{eq:offpgf}\\
 r=0:\quad
 F_t(s)&=\frac{\tau+(1-\tau)s}{1+\tau-\tau s}. \label{eq:critpgf}
\end{align}
The second line includes $\lambda=\mu=0$, when $\tau=0$ and $F_t(s)=s$.
Moreover $F_t\circ F_u=F_{t+u}$.

\paragraph{Proof.}
Conditioning over the first event of one particle gives the backward equation
\begin{equation}\label{eq:riccati}
 \partial_tF_t=\lambda(F_t^2-F_t)+\mu(1-F_t)
              =(F_t-1)(\lambda F_t-\mu),\qquad F_0(s)=s.
\end{equation}
Separating the two roots gives \eqref{eq:offpgf}; when the roots coalesce,
integrating $\partial_tF=\lambda(F-1)^2$ gives \eqref{eq:critpgf}.
The formulas also follow from the matrices, acting by
$\bigl(As+B\bigr)/\bigl(Cs+D\bigr)$,
\[
 M_\delta=\frac1r
 \begin{pmatrix}
 \delta\lambda-\mu&\mu(1-\delta)\\
 \lambda(\delta-1)&\lambda-\delta\mu
 \end{pmatrix},\qquad
 N_\tau=\begin{pmatrix}1-\tau&\tau\\-\tau&1+\tau\end{pmatrix}.
\]
Direct multiplication yields
$M_{\delta_1}M_{\delta_2}=M_{\delta_1\delta_2}$ and
$N_{\tau_1}N_{\tau_2}=N_{\tau_1+\tau_2}$; both matrices are the identity at
zero time.  Since $\delta(t+u)=\delta(t)\delta(u)$ and
$\tau(t+u)=\tau(t)+\tau(u)$, the semigroup claim follows. \hfill$\square$

\paragraph{Theorem 2 (one ancestor and every-$z$ transition).}
Define, for $r\ne0$,
\begin{equation}\label{eq:pb-off}
 p_0=\frac{\mu(1-\delta)}{\lambda-\mu\delta},\qquad
 \beta=\frac{\lambda(1-\delta)}{\lambda-\mu\delta},
\end{equation}
and, for $r=0$, define
\begin{equation}\label{eq:pb-critical}
 p_0=\beta=\frac{\tau}{1+\tau}.
\end{equation}
Then $0\leq p_0\leq1$, $0\leq\beta<1$, and
\begin{equation}\label{eq:one-law}
 \mathbb P_1(Z_t=0)=p_0,\qquad
 \mathbb P_1(Z_t=n)=(1-p_0)(1-\beta)\beta^{n-1},\quad n\geq1.
\end{equation}
For every $z\in\Nzero$,
\begin{align}
 \mathbb P_z(Z_t=0)&=p_0^z, \label{eq:z-zero}\\
 \mathbb P_z(Z_t=n)&=
 \sum_{k=1}^{\min(z,n)}\binom zk p_0^{z-k}(1-p_0)^k
 \binom{n-1}{k-1}(1-\beta)^k\beta^{n-k},\quad n\geq1. \label{eq:z-law}
\end{align}

\paragraph{Proof.}
Algebra rewrites either PGF in Theorem~1 as
\begin{equation}\label{eq:zmgeom}
 F_t(s)=p_0+(1-p_0)\frac{(1-\beta)s}{1-\beta s}.
\end{equation}
For $r>0$, $0<\delta\leq1$; for $r<0$, $\delta\geq1$.
These two sign cases in \eqref{eq:pb-off}, together with the pure-rate edges,
give the claimed parameter ranges.  Expanding the geometric denominator
proves \eqref{eq:one-law}.

For $z$ independent ancestral lines, let $K$ be the number nonextinct at
time $t$.  Then $K\sim\Bin(z,1-p_0)$.  Given $K=k$, the $k$ positive family
sizes are independent geometric variables on $\{1,2,\ldots\}$ with success
$1-\beta$; their sum has probability
$\binom{n-1}{k-1}(1-\beta)^k\beta^{n-k}$.  Mixing over $K$ gives
\eqref{eq:z-zero}--\eqref{eq:z-law}. \hfill$\square$

\paragraph{Why this is not uniformly one negative-binomial law.}
The mixing shape $K$ is random whenever several ancestral lines can
independently die.  Thus the all-parameter law must retain the binomial
survivor mixture; only after conditioning on $K=k$ is the positive sum one
negative-binomial law with fixed shape $k$.  At special parameter values the
mixture can collapse---for example, if $p_0+\beta=1$, then $F_t(s)^z$ itself
has an ordinary negative-binomial form.  This exception is exactly why the
claim is about the uniform formula rather than an impossible pointwise ban.

\paragraph{Corollary 3 (moments).}
For every $z\in\Nzero$,
\begin{align}
 \mathbb E_z Z_t&=ze^{rt}, \label{eq:mean}\\
 \operatorname{Var}_z Z_t&=
 \begin{cases}
 \displaystyle z\frac{\lambda+\mu}{r}e^{rt}(e^{rt}-1),&r\ne0,\\[4pt]
 2z\lambda t,&r=0.
 \end{cases} \label{eq:variance}
\end{align}

\paragraph{Proof.}
Differentiate $F_t(s)^z$ at $s=1$.  From \eqref{eq:zmgeom}, one ancestral
family has mean $(1-p_0)/(1-\beta)$ and second moment
$(1-p_0)(1+\beta)/(1-\beta)^2$.  Substitution of
\eqref{eq:pb-off} or \eqref{eq:pb-critical}, followed by addition across
independent ancestors, gives \eqref{eq:mean}--\eqref{eq:variance}. \hfill$\square$

\section{The three long-time regimes}
The finite-time exact ledger above uses rational sentinels for $\delta$ and
$\tau$.  The following limits are instead symbolic consequences of the
all-parameter formulas; they are not inferred from a finite cutoff.

\paragraph{Theorem 4 (subcritical quasi-stationarity).}
Suppose $0\leq\lambda<\mu$ and $z\geq1$.  With
$\rho=\lambda/\mu$,
\[
 \mathbb P_z(Z_t=n\mid Z_t>0)\longrightarrow
 (1-\rho)\rho^{n-1},\qquad n\geq1.
\]
This limiting law is genuinely quasi-stationary: its PGF
\begin{equation}\label{eq:qsd-pgf}
 g(s)=\frac{(1-\rho)s}{1-\rho s}
\end{equation}
satisfies, for every $t\geq0$,
\begin{equation}\label{eq:qsd-invariance}
 \frac{g(F_t(s))-g(F_t(0))}{1-g(F_t(0))}=g(s).
\end{equation}
For $\lambda=0$ this is the point mass at one, with $g(s)=s$.

\paragraph{Proof.}
Substitution of the off-critical formula \eqref{eq:offpgf} into
\eqref{eq:qsd-invariance} gives the identity after cancellation.  The
numerator in its left-hand side removes the absorption atom and the
denominator is the survival probability, so the identity is exactly
invariance under the transition semigroup conditional on nonabsorption.
As $t\to\infty$, $\delta\to\infty$, hence $p_0\to1$ and
$\beta\to\rho$.  Given survival of one line, \eqref{eq:one-law} is geometric
with success $1-\beta$.  For $z$ lines put $q_t=1-p_0\to0$.
Since $K\sim\Bin(z,q_t)$,
$\mathbb P(K\geq2)/\mathbb P(K\geq1)=O(q_t)$.  Conditional on population
survival, exactly one ancestral line therefore survives with limiting
probability one, proving the result. \hfill$\square$

\paragraph{Theorem 5 (critical Yaglom scaling).}
If $\lambda=\mu=c>0$ and $z\geq1$, then
\[
 \mathcal L\left(\frac{Z_t}{ct}\,\middle|\,Z_t>0\right)
 \Longrightarrow \Exp(1).
\]

\paragraph{Proof.}
Here $\tau=ct\to\infty$, $1-p_0=(1+\tau)^{-1}$, and again only one ancestral
line survives conditionally.  For that line, at $s=e^{-\theta/\tau}$,
\[
 \mathbb E[s^{Z_t}\mid Z_t>0]
 =\frac{s}{1+\tau-\tau s}\longrightarrow\frac1{1+\theta},
\]
the Laplace transform of a unit-rate exponential law.  The vanishing
multiple-lineage probability transfers the same limit to every fixed
$z\geq1$. \hfill$\square$

\paragraph{Theorem 6 (supercritical martingale and gamma mixture).}
Suppose $\lambda>\mu\geq0$, put $r=\lambda-\mu$ and
$W_t=e^{-rt}Z_t=\delta Z_t$.  Then $W_t\to W$ almost surely and in $L^2$.
For one ancestor,
\begin{equation}\label{eq:super-laplace}
 \mathbb E_1e^{-\theta W}
 =\frac\mu\lambda+
 \frac{r^2}{\lambda(r+\lambda\theta)}.
\end{equation}
For $z$ ancestors let $K\sim\Bin(z,r/\lambda)$.  The limit has atom
\[
 \mathbb P_z(W=0)=(\mu/\lambda)^z,
\]
and, conditional on $K=k\geq1$,
\[
 W\sim\GammaLaw\!\left(k,\text{rate }r/\lambda\right).
\]
Thus the positive law is a binomially weighted gamma mixture, with the
binomial weights conditioned on $K\geq1$ if one conditions on $W>0$.
Moreover $\mathbb E_zW=z$ and
$\operatorname{Var}_zW=z(\lambda+\mu)/r$.

\paragraph{Proof.}
The branching property and \eqref{eq:generator} make $W_t$ a nonnegative
martingale.  Equation \eqref{eq:variance} shows that its second moments are
uniformly bounded, giving almost-sure and $L^2$ convergence.  In
\eqref{eq:offpgf}, substitute $s=e^{-\theta\delta}$ and let
$\delta\downarrow0$ to obtain \eqref{eq:super-laplace}.  With
$q=r/\lambda$, its right side is
\[
 \frac\mu\lambda+q\frac{q}{q+\theta},
\]
the transform of an atom of mass $\mu/\lambda$ plus, with mass $q$, an
exponential variable of rate $q$.  The $z$-ancestor limit is the sum of $z$
independent copies.  Counting its nonzero summands gives the stated binomial
variable $K$, and a sum of $k$ rate-$q$ exponentials is gamma.  The limiting
moments follow either from this mixture or by scaling
\eqref{eq:mean}--\eqref{eq:variance}. \hfill$\square$

\paragraph{Asymptotic boundary ledger.}
The quasi-stationary and Yaglom statements require $z\geq1$ because
conditioning on survival is undefined at $z=0$.  The Yaglom statement
requires $c>0$; at $c=0$ the chain is the identity.  In the pure-birth edge
$\mu=0$, Theorem~6 has $K=z$ and a single $\GammaLaw(z,1)$ component (with
the usual point mass at zero when $z=0$).  These clauses close rather than
silently divide by every degenerate parameter.
