\begin{abstract}
For the normalized quadratic map $f(\zeta,t)=a(t)\zeta+b(t)\zeta^2$
we solve the constant-rate Polubarinova--Galin equation on its entire smooth
univalent branch.  The complex coefficient $a^2b$ is conserved, normalized
area is affine in physical time, injection is global and asymptotically
circular, while noncircular suction ends at an explicitly timed ordinary
semicubical cusp.  Circular collapse and nonsmooth or invalid initial faces
are separated.  This round is the \textbf{coefficient-reduction owner}.
\end{abstract}

\section{Normalized flow and exact reduction}
Let $a(t)>0$, $b(t)\in\mathbb C$, and
\begin{equation}\label{eq:model}
 f(\zeta,t)=a(t)\zeta+b(t)\zeta^2,
 \qquad
 \Real\!\left[f_t\,\overline{\zeta f_\zeta}\right]=q
 \quad (|\zeta|=1),
\end{equation}
where $q\in\mathbb R$ is constant.  The normalization $f(0,t)=0$ and
$f_\zeta(0,t)=a(t)>0$ fixes translation and rotation.  A \emph{smooth
univalent solution} means $a>2|b|$; then $f$ is conformal on a neighborhood
of $\overline{\D}$ and parametrizes a smooth Jordan boundary.

\begin{theorem}[Complete smooth coefficient branch]\label{thm:main}
For smooth univalent initial data, equation \eqref{eq:model} is equivalent to
\begin{equation}\label{eq:ode}
 \dot a=\frac{aq}{a^2-4|b|^2},\qquad
 \dot b=-\frac{2qb}{a^2-4|b|^2}.
\end{equation}
On its unique maximal smooth interval,
\begin{equation}\label{eq:invariants}
 \kappa:=a^2b\quad\hbox{is constant},\qquad
 M_0:=a^2+2|b|^2=\frac{\operatorname{Area}(f(\D))}{\pi},\qquad
 \dot M_0=2q.
\end{equation}
Put $u=a^2$ and $u_c=(4|\kappa|^2)^{1/3}$, taking $u_c=0$ when
$\kappa=0$.  Then
\begin{equation}\label{eq:F}
 M_0=F(u):=u+\frac{2|\kappa|^2}{u^2},\qquad
 F'(u)=1-\frac{4|\kappa|^2}{u^3},
\end{equation}
and smooth univalence is exactly $u>u_c$.

If $q>0$, the solution is smooth for every $t\ge0$, $u$ increases to
infinity, and $2|b|/a\to0$.  If $q=0$, the map is stationary.
\end{theorem}

\begin{proof}
On $|\zeta|=1$, expanding the left side of \eqref{eq:model} and equating its
constant and first Fourier modes gives
\begin{equation}\label{eq:fourier}
 a\dot a+2\Real(\dot b\,\overline b)=q,
 \qquad a\dot b+2\dot a b=0.
\end{equation}
The determinant is $a^2-4|b|^2$, so \eqref{eq:ode} follows.  The second
identity gives $(a^2b)^{\boldsymbol\cdot}=0$, and both identities give
$\dot M_0=2q$.  The area formula follows either from the coefficient area
theorem or from
$\tfrac12\int_0^{2\pi}\!\operatorname{Im}(\overline f f_\theta)d\theta$.
Substituting $b=\kappa/u$ proves \eqref{eq:F}.

Indeed,
\[
 f(z_1)-f(z_2)=(z_1-z_2)\{a+b(z_1+z_2)\},\qquad
 f'(\zeta)=a+2b\zeta.
\]
Thus $a>2|b|$ gives injectivity and nonvanishing derivative on
$\overline\D$; equality puts the sole critical point on its boundary, and
$a<2|b|$ puts it inside.  Since $F$ is strictly increasing on
$(u_c,\infty)$, $M_0(t)=M_0(0)+2qt$ uniquely determines that branch.
For $q>0$ it exists globally with $u\to\infty$ and
$2|b|/a=2|\kappa|/u^{3/2}\to0$.  For $q=0$, \eqref{eq:ode} vanishes.
\end{proof}

\section{Sharp suction endpoint and cusp normal form}
This revision is the \textbf{first-cusp owner}.

\begin{theorem}[First singularity under suction]\label{thm:cusp}
Assume $q<0$ and smooth univalent initial data.
If $\kappa\ne0$, the maximal smooth time is
\begin{equation}\label{eq:T}
 T=\frac{M_0(0)-M_c}{-2q},\qquad
 M_c=F(u_c)=\frac32u_c.
\end{equation}
As $t\uparrow T$, $u\downarrow u_c$, $a\downarrow\sqrt{u_c}$, and
$|b|\uparrow\sqrt{u_c}/2$.  No self-intersection on $\overline\D$ and no
critical point in $\overline\D$ occurs earlier.  At $T$ the limiting boundary has exactly one ordinary
semicubical cusp at
\begin{equation}\label{eq:location}
 \zeta_c=-\frac{a_c}{2b_c},\qquad
 z_c=f(\zeta_c,T)=-\overline{b_c}.
\end{equation}
\end{theorem}

\begin{proof}
On the upper branch $u>u_c$, equation \eqref{eq:F} is strictly increasing
and its infimum is $F(u_c)=3u_c/2$.  The affine area clock therefore reaches
that value at exactly \eqref{eq:T}; before then the strict quadratic
criterion in the preceding proof excludes both critical points and
self-intersections.  At the endpoint write $b_c=Be^{i\phi}$, so
$a_c=2B$, and set $\zeta=-e^{-i\phi}e^{is}$.  Then
\begin{align}\label{eq:cusp}
 e^{i\phi}\{f(\zeta,T)-z_c\}
 &=B\{-2e^{is}+e^{2is}+1\}\\
 &=-Bs^2-iBs^3+\frac7{12}Bs^4+O(s^5).\notag
\end{align}
With $X=-\Real(e^{i\phi}(z-z_c))$ and
$Y=-\operatorname{Im}(e^{i\phi}(z-z_c))$, one has
$X=Bs^2+O(s^4)$, $Y=Bs^3+O(s^5)$, and
\begin{equation}\label{eq:semi}
 \lim_{s\to0}\frac{Y^2}{X^3}=\frac1B.
\end{equation}
This is the ordinary semicubical normal form and proves uniqueness of the
boundary cusp.
\end{proof}

\section{Circular and excluded faces}
This revision is the \textbf{boundary-and-route owner}.

\begin{proposition}[Boundary atlas]\label{prop:boundary}
If $\kappa=0$, then $b=0$ throughout and
$a(t)^2=a(0)^2+2qt$.  Injection expands circles globally, $q=0$ fixes the
circle, and suction collapses it at
$T=a(0)^2/(-2q)$ without a pre-collapse cusp.

If $a(0)=2|b(0)|$ with $b(0)\ne0$, the datum already has the cusp
\eqref{eq:cusp} and its smooth lifespan is zero.  If
$0<a(0)<2|b(0)|$, then $f'$ vanishes inside $\D$ and no conformal
Laplacian-growth domain is asserted.  No surface-tension regularization,
weak post-cusp continuation, higher-degree classification, or multiply
connected extension is included.
\end{proposition}
