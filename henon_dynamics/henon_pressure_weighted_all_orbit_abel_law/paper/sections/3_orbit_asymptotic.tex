\section{The orbitwise cyclotomic asymptotic}

For $n>2$, reciprocal symmetry gives a polynomial $q_n\in\mathbb Z[T]$
such that
\begin{equation}
 q_n(X+X^{-1})=X^{-\varphi(n)/2}\Phi_n(X).
 \label{eq:q-poly}
\end{equation}
Thus $\beta_{\gamma,n}=q_n(t_\gamma)$.  The cyclotomic M\"obius product
implies
\begin{equation}
 \log|\sigma(\beta_{\gamma,n})|
 =-\frac{\varphi(n)}2\log|\rho_\sigma|
 +\sum_{d\mid n}\mu(n/d)\log|\rho_\sigma^d-1|.
 \label{eq:embedding-log}
\end{equation}

\subsection{Off-circle conjugates}

If $|\rho_\sigma|>1$, then
\[
 \log|\rho_\sigma^d-1|
 =d\log|\rho_\sigma|+\log|1-\rho_\sigma^{-d}|.
\]
Using $\sum_{d\mid n}\mu(n/d)d=\varphi(n)$ in
\eqref{eq:embedding-log} gives
\begin{equation}
 \log|\sigma(\beta_{\gamma,n})|
 =\frac{\varphi(n)}2\log|\rho_\sigma|
 +\epsilon_{\sigma,n}.
 \label{eq:off-circle}
\end{equation}
The correction is bounded uniformly in $n$, because
\[
 |\epsilon_{\sigma,n}|
 \le\sum_{d\ge1}-\log(1-|\rho_\sigma|^{-d})<\infty.
\]

\subsection{Unit-circle conjugates}

Suppose $|\rho_\sigma|=1$.  Yamada's two-logarithm estimate
\citep[Theorem~1.2 and consequence (15)]{Yamada2025} implies that for an
algebraic number of modulus one which is not a root of unity,
\[
 \log|\arg(\rho_\sigma^d)|
 \ge-C_\sigma(1+\log d)^2.
\]
Indeed, in Yamada's notation the integer nearest to the argument multiple is
$O(d)$, so $b'=O_\sigma(d)$ and the displayed parameter
$h=O_\sigma(1+\log d)$.  All finitely many small $d$ are absorbed into
$C_\sigma$.
With the principal argument $\theta\in[-\pi,\pi]$,
\[
 \frac2\pi|\theta|\le|1-e^{i\theta}|\le2.
\]
Consequently
\begin{equation}
 |\log|1-\rho_\sigma^d||
 \le C'_\sigma(1+\log d)^2.
 \label{eq:unit-bound}
\end{equation}
The elementary divisor bound $\tau(n)\le2\sqrt n$ then yields
\begin{equation}
 \left|
 \sum_{d\mid n}\mu(n/d)\log|1-\rho_\sigma^d|
 \right|
 \le2C'_\sigma\sqrt n(1+\log n)^2.
 \label{eq:unit-sum}
\end{equation}
There is no height main term because $\log|\rho_\sigma|=0$.

\begin{theorem}[Orbitwise Mahler-height law]
For every primitive orbit $\gamma$,
\begin{equation}
 b_{\gamma,n}
 =\frac{\varphi(n)}2\Hspec_\gamma+R_{\gamma,n},
 \qquad
 |R_{\gamma,n}|
 \le C_\gamma\sqrt n(1+\log n)^2.
 \label{eq:orbit-asymptotic}
\end{equation}
\end{theorem}

\begin{proof}
Sum \eqref{eq:off-circle} and \eqref{eq:unit-sum} over the finitely many
embeddings of $F_\gamma$.  The main terms sum to \eqref{eq:height}; the
embedding logarithms sum to \eqref{eq:mass}.
\end{proof}

The classical totient summatory estimate, or the elementary derivation
given in the appendix, gives
\begin{equation}
 \tau^2\sum_{n\ge1}\varphi(n)e^{-\tau n}
 \longrightarrow\frac6{\pi^2}.
 \label{eq:totient-laplace}
\end{equation}
Also
\[
 \tau^2\sum_{n\ge3}\sqrt n(1+\log n)^2e^{-\tau n}
 =O\!\left(\tau^{1/2}(1+|\log\tau|)^2\right).
\]
We therefore obtain the exact one-orbit coefficient.

\begin{corollary}[One-orbit Abel law]
\label{cor:one-orbit-abel}
For every primitive orbit $\gamma$,
\begin{equation}
 \tau^2\sum_{n\ge3}b_{\gamma,n}e^{-\tau n}
 \longrightarrow\frac3{\pi^2}\Hspec_\gamma.
 \label{eq:one-orbit-abel}
\end{equation}
\end{corollary}

The factor one half in \eqref{eq:orbit-asymptotic} is intrinsic to the
inversion-fixed packet.  Replacing the right side of
\eqref{eq:one-orbit-abel} by $6\Hspec_\gamma/\pi^2$ is false.
