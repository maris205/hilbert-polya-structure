\begin{abstract}
For the billiard inside a confocal ellipse, restricted to positively oriented
orbits with an elliptic caustic, we give the exact Jacobi covering and rigid
rotation number.
\ifnum\CRevisionRound>0 We prove strict monotonicity in both eccentricities,
both rotation endpoints, and that reduced rotation $p/q$ makes every orbit on
the caustic curve a least-period-$q$ Poncelet polygon.
\fi\space
\ifnum\CRevisionRound>1 The $q$th restricted return is the identity with unit
tangent derivative on a full periodic circle, obstructing the requested
ordinary isolated-orbit product.  A 613-cell receipt audits every convention.
\fi\space
The theorem excludes hyperbolic caustics and makes no ambient unipotent or
target spectral claim.
\end{abstract}

\section{The frozen confocal owner}
For $0<\varepsilon<1$ let
\[
 E(\varepsilon)=\left\{(x,y):\varepsilon^2x^2+
 \frac{\varepsilon^2}{1-\varepsilon^2}y^2=1\right\}.
\]
Its foci are $(\pm1,0)$.  Fix $0<f<e<1$: $E(f)$ is the outer
billiard boundary and $E(e)$ is an inner elliptic caustic.  We keep the
positive orientation and one-reflection clock.  The corresponding Poncelet
circle map is denoted $B_e^f:E(f)\to E(f)$.  This is precisely the
elliptic-caustic sector analyzed in the primary rotation-number
sources~\cite{LM,Kol,CF}; hyperbolic caustics are not included.

Our convention is
\[
 F(\phi,e)=\int_0^\phi\frac{d\tau}{\sqrt{1-e^2\sin^2\tau}},
 \qquad K(e)=F(\tfrac\pi2,e),
\]
so $e$ is the Jacobi modulus (software parameter $m=e^2$).  Define
\begin{align}
 \omega(e,f)&=\arcsin\sqrt{\frac{e^2-f^2}{e^2(1-f^2)}},\label{eq:omega}\\
 \rho(e,f)&=\frac{F(\omega(e,f),e)}{2K(e)}.\label{eq:rho}
\end{align}

\begin{theorem}[Jacobi covering and rigid rotation]
The period-one covering
\begin{equation}
 \pi_e^f(\theta)=\frac1f\left(
 -\sn(4K(e)\theta,e),
 \sqrt{1-f^2}\,\cn(4K(e)\theta,e)\right)                 \tag{3}
\end{equation}
satisfies
\[
 B_e^f\circ\pi_e^f=\pi_e^f\circ R_{\rho(e,f)},
 \qquad R_\rho(\theta)=\theta+\rho.
\]
In particular, \eqref{eq:rho} is the rotation number.
\end{theorem}
\begin{proof}
The identity $\sn^2+\cn^2=1$ puts (3) on $E(f)$, and the $4K(e)$
period gives the deck map $\theta\mapsto\theta+1$.  Set
$v=2F(\omega,e)$.  The Jacobi addition formulas, together with
\eqref{eq:omega}, show that the line through $\pi_e^f(\theta)$ and
$\pi_e^f(\theta+v/(4K(e)))$ obeys
\[
 C^2=A^2/e^2+(1-e^2)B^2/e^2
\]
when written as $Ax+By+C=0$.  This is the dual equation for tangency to
$E(e)$.  Positive orientation selects the next intersection, proving the
conjugacy; $v/(4K(e))$ is exactly \eqref{eq:rho}.  This also verifies the
hypotheses and conclusion of the confocal covering theorem in~\cite{LM}.
\end{proof}

\section{Strict parameter rigidity and endpoints}
The inverse form of \eqref{eq:rho} is especially effective:
\begin{equation}
 \rho(e,f)=\ell\quad\Longleftrightarrow\quad
 f=R(\ell,e):=e\,\cd(2K(e)\ell,e),
 \qquad 0<\ell<\tfrac12.                                \tag{4}
\end{equation}

\begin{theorem}[strict signs and the two endpoint values]
Throughout $0<f<e<1$,
\[
 \partial_e\rho(e,f)>0,\qquad \partial_f\rho(e,f)<0.
\]
Moreover,
\[
 \lim_{f\to e^-}\rho=\lim_{e\to f^+}\rho=0,
 \qquad
 \lim_{f\to0^+}\rho=\lim_{e\to1^-}\rho=\tfrac12,
\]
where the parameter not tending to its boundary is held fixed.
\end{theorem}
\begin{proof}
Let $u=2K(e)\ell$ and
$\mathcal E(u,e)=\int_0^u\dn^2(t,e)\,dt$.  Differentiating (4) gives
\[
 R_\ell=-\frac{2e(1-e^2)K(e)\sn(u,e)}{\dn^2(u,e)}<0
\]
and
\[
 R_e=\cd(u,e)+\frac{\sn(u,e)}{\dn^2(u,e)}
 [\mathcal E(u,e)-2\ell\mathcal E(K(e),e)].
\]
Since $\dn^2(t,e)$ strictly decreases on $(0,K(e))$, its average on
$[0,u]$ exceeds its average on $[0,K(e)]$; the bracket and hence $R_e$
are positive.  Implicit differentiation of $f=R(\rho,e)$ yields
$\rho_f=1/R_\ell<0$ and $\rho_e=-R_e/R_\ell>0$.

The two coalescing limits follow from $\omega\to0$; $f\to0$ gives
$\omega\to\pi/2$ and $F\to K$.  For fixed $f$ and $e\to1^-$, put
$e'=\sqrt{1-e^2}$.  Then $\cos\omega=O(e')$ and
\[
 0\le K(e)-F(\omega,e)=
 \int_\omega^{\pi/2}\frac{dt}{\sqrt{1-e^2\sin^2t}}=O(1),
\]
whereas $K(e)\sim\log(4/e')\to\infty$.  Thus $F/K\to1$ and the last
limit is $1/2$.
\end{proof}

\begin{theorem}[complete elliptic-caustic porism]
If $\rho(e,f)=p/q$ with $0<p/q<1/2$ and $\gcd(p,q)=1$, every orbit on
this caustic curve has minimal period $q$.
\end{theorem}
\begin{proof}
In the covering coordinate, the $q$th iterate sends
$\theta$ to $\theta+p$, a deck-equivalent point.  If $0<k<q$ returned,
then $kp/q$ would be integral; coprimality would force $q\mid k$, a
contradiction.  The arbitrary starting value of $\theta$ produces the
continuous Poncelet family, recovering the porism statement of~\cite{CF}.
\end{proof}
