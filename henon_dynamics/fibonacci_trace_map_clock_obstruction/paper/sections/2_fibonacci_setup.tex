\section{Chronological Fibonacci cocycle and its two clocks}

Let
\begin{equation}
w_{-1}=b,\qquad w_0=a,\qquad w_{k+1}=w_kw_{k-1}.
\label{eq:words}
\end{equation}
The word lengths satisfy
\begin{equation}
q_{-1}=q_0=1,\qquad q_{k+1}=q_k+q_{k-1},
\qquad q_k=F_{k+2},
\label{eq:lengths}
\end{equation}
where \(F_0=0,F_1=1\).

Fix \(V(a)=\lambda\), \(V(b)=0\), and define
\begin{equation}
A_v(E)=
\begin{pmatrix}E-v&-1\\1&0\end{pmatrix}\in SL_2(\C[E]).
\label{eq:site}
\end{equation}
For a chronological word \(w=c_1\cdots c_q\), later physical sites multiply
on the left:
\begin{equation}
M(w;E)=A_{V(c_q)}(E)\cdots A_{V(c_1)}(E).
\label{eq:chronological}
\end{equation}
Thus \(M(uv)=M(v)M(u)\) and
\(M_{k+1}=M_{k-1}M_k\).  We never replace this product by an averaged
letter-incidence matrix.

\begin{lemma}[Trace recursion]
Let \(d_k(E)=\Tr M(w_k;E)\), with the auxiliary seed \(d_{-2}=2\).  Then
\begin{equation}
d_{k+1}=d_kd_{k-1}-d_{k-2},\qquad
(d_{-2},d_{-1},d_0)=(2,E,E-\lambda).
\label{eq:trace-recurrence}
\end{equation}
\end{lemma}

\begin{proof}
The matrix recurrence follows from \eqref{eq:words} and
\eqref{eq:chronological}.  Applying the standard \(SL_2\) trace identity to
the two consecutive monodromy matrices yields
\eqref{eq:trace-recurrence}.  It may also be verified directly from the seed
matrices; the reproducibility package performs both checks symbolically.
\end{proof}

Set \(x_k=d_k/2\).  The recursion becomes
\(x_{k+1}=2x_kx_{k-1}-x_{k-2}\), generated by
\begin{equation}
T(x,y,z)=(2xy-z,x,y),\qquad
\ell_\lambda(E)=\left(\frac{E-\lambda}{2},\frac E2,1\right),
\label{eq:trace-map-line}
\end{equation}
with
\begin{equation}
T^k\ell_\lambda(E)=(x_k,x_{k-1},x_{k-2}).
\label{eq:iterate-line}
\end{equation}
The Fricke--Vogt invariant
\(I=x^2+y^2+z^2-2xyz-1\) obeys
\(I(\ell_\lambda(E))=\lambda^2/4\).

\begin{proposition}[Physical degree]
For every \(k\ge-1\), \(d_k(E)\) is monic and
\begin{equation}
\deg_Ed_k=q_k=F_{k+2}.
\label{eq:physical-degree}
\end{equation}
\end{proposition}

\begin{proof}
The assertion holds for \(d_{-1}=E\), \(d_0=E-\lambda\), and directly for
\(d_1=E(E-\lambda)-2\).  Starting with \(k=1\), suppose it holds at the
preceding levels.  In \eqref{eq:trace-recurrence}, the product
\(d_kd_{k-1}\) is monic of degree \(q_k+q_{k-1}=q_{k+1}\), strictly larger
than the degree \(q_{k-2}\) of the subtracted term.  Its leading term cannot
cancel.  Induction proves the claim.
\end{proof}

Equation \eqref{eq:physical-degree} exposes the two clocks used throughout:
\(k\) is trace-map renormalization time, while \(q_k\) is the number of
chronologically ordered lattice sites in the periodic approximant.
