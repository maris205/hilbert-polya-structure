\section{Projection dynamics and the exact quotient}
Let $A=A^T\in\R^{n\times n}$ and $1\le k\le n-1$.  We realize
$\Gr(k,n)$ as the rank-$k$ orthogonal projections
\[
 \mathcal P_{k,n}=\{P:P^T=P,\ P^2=P,\ \tr P=k\}
\]
and consider
\begin{equation}\label{eq:flow}
 \dot P=[P,[P,A]]=AP+PA-2PAP.
\end{equation}
The sign in~\eqref{eq:flow} selects the high-eigenvalue subspaces forward in
time.  The generator $A$ is fixed; the state is the subspace represented by
$P$.

\begin{theorem}[Global exact projector]\label{thm:global}
Let $Q_0\in\R^{n\times k}$ have full column rank and
$\Ran Q_0=\Ran P_0$.  For every $t\in\R$, set
\begin{equation}\label{eq:projector}
 Y(t)=e^{tA}Q_0,\qquad
 P(t)=Y(t)\bigl(Y(t)^TY(t)\bigr)^{-1}Y(t)^T.
\end{equation}
Then~\eqref{eq:projector} is the unique global solution of~\eqref{eq:flow};
in particular,
\[
 \Ran P(t)=e^{tA}\Ran P_0.
\]
If $Ae_i=\lambda_i e_i$ in an orthonormal eigenbasis, its projective
Pl\"ucker coordinates obey
\begin{equation}\label{eq:plucker}
 p_I(t)=e^{t\lambda_I}p_I(0)\quad\hbox{up to one common scale},
 \qquad \lambda_I=\sum_{i\in I}\lambda_i.
\end{equation}
\end{theorem}
\begin{proof}
The matrix $e^{tA}$ is invertible, so $Y(t)$ has full column rank and its Gram
matrix is positive definite for all real $t$.  Formula~\eqref{eq:projector}
is therefore the orthogonal projector onto $e^{tA}\Ran P_0$.  Since
$Y'=AY$ and $A=A^T$, differentiation of the Gram inverse gives
\[
 \dot P=AP+PA-2PAP=[P,[P,A]].
\]
Smooth finite-dimensional ODE uniqueness identifies it with the solution.
On $\bigwedge^k\R^n$, the induced map sends
$e_{i_1}\wedge\cdots\wedge e_{i_k}$ to
$e^{t(\lambda_{i_1}+\cdots+\lambda_{i_k})}$ times itself, proving
\eqref{eq:plucker}.
\end{proof}

This quotient identity is the subspace power-flow mechanism neighboring
Oja's principal-component dynamics and the continuous Grassmann flows of
Absil, Sepulchre, and Mahony.  We derive every statement needed below and make
no priority claim for that classical mechanism.

\section{Simple spectrum: every Schubert cell and its rate}
Assume in this section
\begin{equation}\label{eq:simple}
 \lambda_1<\lambda_2<\cdots<\lambda_n,
 \qquad F_j=\Span(e_1,\ldots,e_j).
\end{equation}
For a $k$-plane $V$, write
\[
 \mathcal B(V)=\{I\subset\{1,\ldots,n\}:|I|=k,\ p_I(V)\ne0\}.
\]
These are the bases of the representable matroid determined by the rows of
any frame for $V$.

\begin{lemma}[Matroid guard]\label{lem:greedy}
Distinct element weights $\lambda_1,\ldots,\lambda_n$ give a unique
maximum-weight basis $I_+(V)$ and a unique minimum-weight basis $I_-(V)$ in
$\mathcal B(V)$, even when two arbitrary $k$-subset sums coincide.
\end{lemma}
\begin{proof}
Run the matroid greedy algorithm in decreasing element weight.  To see
uniqueness directly, suppose two optimal bases differ and take the
largest-weight element in their symmetric difference.  Strong basis exchange
inserts that element into the basis missing it while removing a strictly
lower-weight element, producing a heavier basis.  This contradiction proves
the maximum statement; reverse the weights for the minimum.
\end{proof}

For $I=(i_1<\cdots<i_k)$, let $\Omega_I$ be the Schubert cell characterized by
\begin{equation}\label{eq:schubert}
 \dim(V\cap F_j)=\#\{a:i_a\le j\}\quad(1\le j\le n).
\end{equation}
Equivalently, $I$ is the lexicographically greatest nonzero Pl\"ucker index,
hence the greedy maximum in Lemma~\ref{lem:greedy}.

\begin{theorem}[Exact cell limits and actual-support rates]\label{thm:cells}
If $V_0\in\Omega_I$, then
\[
 \lim_{t\to+\infty}P(t)=P_I,
\]
the coordinate projection onto $E_I=\Span\{e_i:i\in I\}$.  The opposite
Schubert cell indexed by $I_-(V_0)$ gives the backward limit.

If $V_0\ne E_I$, define
\begin{equation}\label{eq:gap}
 \rho_2(V_0)=\max_{J\in\mathcal B(V_0),\,J\ne I}\lambda_J,
 \qquad \Delta_+(V_0)=\lambda_I-\rho_2(V_0)>0.
\end{equation}
Then
\begin{equation}\label{eq:rate}
 \lim_{t\to+\infty}-\frac1t
 \log\lVert P(t)-P_I\rVert_F=\Delta_+(V_0).
\end{equation}
The analogous backward rate uses the actual second-smallest nonzero
Pl\"ucker weight.  For an equilibrium the error is zero and its gap is
declared infinite.
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:greedy}, $I$ is the unique maximum supported weight despite
possible ambient subset-sum ties.  Factor $e^{t\lambda_I}$ from the Pl\"ucker
vector in~\eqref{eq:plucker}.  Every other supported coordinate decays, giving
the coordinate-plane limit.  The nonzero sum of all weight-$\rho_2$ coordinate
vectors is the first surviving transverse term, so projective distance is
$\Theta(e^{-t\Delta_+})$.  The Pl\"ucker embedding and the orthogonal-projector
chart are smooth local embeddings with equivalent norms; the logarithmic
rate is therefore~\eqref{eq:rate}.  Apply the argument to $-A$ for backward
time.
\end{proof}

\ifnum\CRevisionRound>0
\paragraph{Why the support qualifier matters.}
Simple eigenvalues do not imply distinct $k$-fold sums; for example
$0+4=1+3$.  Zero Pl\"ucker coordinates remove subsets from the orbit, and
ties may remain at subleading weights.  Lemma~\ref{lem:greedy} proves only the
needed top supported uniqueness, while~\eqref{eq:gap} measures the first term
that is actually present.

\section{Equilibria, modes, and Schubert dimensions}
Relative to $\Ran P\oplus\ker P$, write the off-diagonal block of $A$ as $B$.
Then $[P,[P,A]]$ has off-diagonal blocks $B$ and $B^T$.  Thus
\begin{equation}\label{eq:equilibria}
 \dot P=0\quad\Longleftrightarrow\quad[P,A]=0
 \quad\Longleftrightarrow\quad\Ran P\text{ is }A\text{-invariant}.
\end{equation}
For simple spectrum these are exactly the $\binom nk$ coordinate projections.

At $P_I$, a tangent graph component from selected $e_i$ to unselected $e_j$
has linearized equation
\begin{equation}\label{eq:mode}
 \dot x_{ji}=(\lambda_j-\lambda_i)x_{ji}.
\end{equation}
Consequently
\begin{align}
 d_s(I)&=\#\{(i,j):i\in I,j\notin I,\lambda_j<\lambda_i\},\\
 d_u(I)&=\#\{(i,j):i\in I,j\notin I,\lambda_j>\lambda_i\}.
\end{align}
If $I=(i_1<\cdots<i_k)$, then
$d_s(I)=\sum_{a=1}^k(i_a-a)=\dim\Omega_I$ and
$d_s(I)+d_u(I)=k(n-k)$.

\section{Repeated spectrum: associated grade and Morse--Bott strata}
Let the distinct eigenvalues be
$\mu_1<\cdots<\mu_s$, with eigenspaces $E_\alpha$ of dimensions $m_\alpha$,
and put $F_\alpha=\bigoplus_{\beta\le\alpha}E_\beta$.  For an initial plane
$V$, define
\begin{equation}\label{eq:graded}
 G_\alpha=\pi_\alpha(V\cap F_\alpha)\subset E_\alpha,
 \qquad
 k_\alpha=\dim(V\cap F_\alpha)-\dim(V\cap F_{\alpha-1}),
\end{equation}
where $\pi_\alpha$ is orthogonal projection to $E_\alpha$.

\begin{theorem}[Associated-graded limit and complete critical atlas]
Every initial plane has the basis-independent limit
\begin{equation}\label{eq:gradedlimit}
 \lim_{t\to+\infty}e^{tA}V=\bigoplus_{\alpha=1}^sG_\alpha.
\end{equation}
For every feasible occupancy vector
$0\le k_\alpha\le m_\alpha$, $\sum_\alpha k_\alpha=k$, the corresponding
equilibrium component is exactly
\begin{equation}\label{eq:critical}
 \mathcal C_{\boldsymbol k}=\prod_{\alpha=1}^s\Gr(k_\alpha,E_\alpha).
\end{equation}
It is Morse--Bott.  Its critical, stable-normal, and unstable-normal dimensions
are, respectively,
\begin{align}
 d_0&=\sum_\alpha k_\alpha(m_\alpha-k_\alpha),\label{eq:dims}\\
 d_s&=\sum_{\beta<\alpha}k_\alpha(m_\beta-k_\beta),\\
 d_u&=\sum_{\beta>\alpha}k_\alpha(m_\beta-k_\beta),
\end{align}
and $d_0+d_s+d_u=k(n-k)$.
\end{theorem}
\begin{proof}
Choose a basis of $V$ adapted to the filtration $V\cap F_\alpha$.  When a
basis vector first appears at level $\alpha$, its $E_\alpha$ component is
nonzero; these leading components are independent and span $G_\alpha$.
Under $e^{tA}$ every lower component decays relative to that leading one,
which proves~\eqref{eq:gradedlimit}.  Equivalently, the entire highest-weight
component of the decomposable Pl\"ucker vector survives.  We do not choose one
coordinate from a tied block.  The descending eigenflag gives the backward
limit.

Equation~\eqref{eq:equilibria} and self-adjoint spectral decomposition show
that every invariant plane splits as $\bigoplus V_\alpha$ with
$V_\alpha\in\Gr(k_\alpha,E_\alpha)$, proving completeness of
\eqref{eq:critical}.  Within-block graph modes have rate zero and are precisely
tangent to this product.  Every cross-block mode has the nonzero rate
$\mu_\beta-\mu_\alpha$; counting its sign gives~\eqref{eq:dims}.  Thus the
Hessian is nondegenerate normal to the critical product, which is the
Morse--Bott condition.  The dimension closure counts every selected-to-
unselected graph coordinate exactly once.
\end{proof}

\section{Strict Lyapunov law and recurrence}
Define $\Phi(P)=\tr(AP)$.  Cyclicity of trace and $P^2=P$ give
\begin{equation}\label{eq:lyapunov}
 \frac{\dd}{\dd t}\Phi(P(t))
 =\tr\bigl(A[P,[P,A]]\bigr)
 =\lVert[P,A]\rVert_F^2\ge0.
\end{equation}
Equality holds exactly at the invariant projections.  If a nonconstant orbit
returned arbitrarily close to its initial point, continuity of $\Phi$ would
force its values back toward $\Phi(P_0)$ after strict increase, contradicting
monotonicity.  Hence no nonconstant recurrent or periodic trajectory exists.
The explicit limits above are stronger than this obstruction.
