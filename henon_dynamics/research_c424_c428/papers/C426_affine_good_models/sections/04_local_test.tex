\section{A finite local test in every residue characteristic}\label{sec:localtest}

\begin{lemma}[Exact scalar condition]\label{lem:scalar}
For $S_{s,r}(x,y)=(sx+r,sy+r)$, goodness is equivalent to
$a\in O^\times$, $Q_{s,r}\in O[Y]$ and $bs^{d-1}\in O^\times$.
\end{lemma}
\begin{proof}
Direct substitution gives
\begin{align}
 S_{s,r}^{-1}FS_{s,r}(x,y)
 &=\left(y,\frac{f(sy+r)-(a+1)r}{s}-ax\right)\notag\\
 &=\bigl(y,Q_{s,r}(y)+ay-ax\bigr),\label{eq:scalarmodel}
\end{align}
whose inverse is
\begin{equation}\label{eq:scalarinverse}
 (x,y)\longmapsto
 \left(\frac{Q_{s,r}(x)+ax-y}{a},x\right).
\end{equation}
For unit $a$, integrality in both directions is equivalent to
$Q_{s,r}\in O[Y]$. Preservation of degree is equivalent to the
leading coefficient $bs^{d-1}$ being a unit. Under those conditions,
the top terms are $(0,bs^{d-1}y^d)$ and
$((bs^{d-1}/a)x^d,0)$. Their indeterminacy points are $[1:0:0]$ and
$[0:1:0]$, so regularity after reduction also holds.
\end{proof}

The leading-coefficient condition gives the unique possible exponent
$v(s)=-v(b)/(d-1)$. If that number is not an integer, no scale over
$E$ can work. Scales with the same valuation differ by a unit, so
one may fix any scale with the required valuation.

\begin{lemma}[Complete centre list]\label{lem:centres}
Suppose the Jacobian and scale conditions in
\eqref{eq:localconditions} hold. Every good centre is congruent modulo
$sO$ to a candidate in \eqref{eq:candidates}. The full coefficient
test for a candidate is
\begin{align}
 [Y^0]Q_{s,r}&=\frac{q(r)-r}{s},\label{eq:coefftest0}\\
 [Y^j]Q_{s,r}&=s^{j-1}\sum_{m=j}^d q_m\binom mj r^{m-j}
 \quad(1\le j\le d).\label{eq:coefftest}
\end{align}
\end{lemma}
\begin{proof}
The coefficient of $Y^{d-1}$ equals
\begin{equation}\label{eq:subleading}
 s^{d-2}(dbr+q_{d-1})=dbs^{d-2}(r-r_0).
\end{equation}
If $k=v(s)$ and $e=v(d)$, then $v(dbs^{d-2})=e-k$.
Integrality of \eqref{eq:subleading} therefore forces
$r\in r_0+(s/d)O$. When $d=2$, the coefficient here is
$q_1=f_1-a$, not $f_1$.

For $u\in O$, let $\tau_u(Y)=Y+u$. The identity
\begin{equation}\label{eq:centretranslation}
 Q_{s,r+su}=\tau_u^{-1}\circ Q_{s,r}\circ\tau_u
\end{equation}
shows that the integral, unit-leading-coefficient condition depends
only on $r$ modulo $sO$. The quotient of $(s/d)O$ by $sO$ is
isomorphic to $O/dO$, which has $|\kappa|^e$ elements. This proves
exhaustiveness and the stated count. Expanding $q(sY+r)$ proves
\eqref{eq:coefftest0}--\eqref{eq:coefftest}. Those formulas check all
coefficients, not merely the subleading one; they divide by no
factorial that might vanish in the residue field.
\end{proof}

\begin{lemma}[Unique good disc]\label{lem:unique}
All passing pairs $(s,r)$ determine the same disc $r+sO$.
In particular, for fixed $s$ there is at most one passing centre
class modulo $sO$.
\end{lemma}
\begin{proof}
If $Q\in O[Y]$ has degree $d\ge2$ and unit leading coefficient,
then $Q(O)\subset O$. If $|y|>1$, its top term strictly dominates
all lower terms, and
\begin{equation}\label{eq:escape}
 |Q(y)|=|y|^d,\qquad |Q^n(y)|=|y|^{d^n}\longrightarrow\infty.
\end{equation}
Thus the set of elements of $E$ with bounded forward $Q$-orbit is
exactly $O$. For $h(Y)=sY+r$ one has $Q_{s,r}=h^{-1}qh$, and
$q^n(h(y))=h(Q_{s,r}^n(y))$. A nonzero affine map preserves whether
a sequence is bounded. Hence the bounded-forward-orbit set of $q$
is exactly $r+sO$ whenever $(s,r)$ passes. That set is intrinsic to
$q$, proving the first assertion. Two discs with the same scale
are equal precisely when their centres differ by an element of $sO$.
\end{proof}

Lemmas~\ref{lem:rigidity}--\ref{lem:unique} prove the first two parts
of Theorem~\ref{thm:main}. Indeed, every good coordinate map reduces
to a passing scalar pair, all such pairs have the same image disc,
and two scalar maps with the same disc differ by an integral affine
map. Right composition gives exactly the coset
\eqref{eq:localcoset}.

\paragraph{Wild-centre check.}
The candidates cannot be collapsed to $r_0$ when $v(d)>0$.
For example, over $\QQ_2$ let $a=1$ and
$f(Y)=Y^2+Y+\tfrac14$. Then $q(Y)=Y^2+\tfrac14$, $s=1$ and
$r_0=0$. The two candidate classes are $0$ and $\tfrac12$ modulo
$\ZZ_2$. The first fails its constant coefficient, whereas
\[
 Q_{1,1/2}(Y)=(Y+\tfrac12)^2+\tfrac14-\tfrac12=Y^2+Y
\]
passes. This is a direct substitution illustrating the already proved
test, not a computational search or an additional classification.
