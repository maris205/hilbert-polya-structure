\begin{abstract}
We solve the inviscid nonadvective Constantin--Lax--Majda equation on the
circle for arbitrary conserved mean under one frozen periodic Hilbert
transform convention.  The periodic Tricomi identity turns the nonlocal PDE
into a pointwise complex Riccati equation, while an invariant Hardy
subspace proves that the Hilbert constraint is preserved.  This yields
exact M\"obius formulas in both the zero- and nonzero-mean strata.
\ifnum\CRevisionRound>0
For nonzero mean, forward smooth breakdown occurs if and only if the
initial vorticity has a zero; the first time is an explicit compact minimum
on the principal arccotangent branch.  Zero-free data are global and
periodic.  A separate signed-zero criterion treats zero mean, and the full
one-mode threshold diagram follows.
\fi
\ifnum\CRevisionRound>1
At every simple first pole we prove a local inverse-time self-similar
vorticity profile.  A global two-sided inverse-time bound is stated only
when all simultaneous first poles are simple; tangent and higher-order
zeros remain in the exact breakdown theorem but outside that rate claim.
The strict Route-A evaluation fails all five gates.
\fi
\end{abstract}

\section{Convention and exact Riccati closure}

Write $\TT=\RR/(2\pi\ZZ)$.  For a real smooth periodic function, define
\begin{equation}
 \HH(\e^{\ii kx})=-\ii\operatorname{sgn}(k)\e^{\ii kx},
 \qquad \HH1=0.                                  \label{eq:Hconvention}
\end{equation}
Thus $\HH\sin(kx)=-\cos(kx)$ and
$\HH\cos(kx)=\sin(kx)$ for $k>0$.  On mean-zero functions,
$\HH^2=-I$; on real $L^2$, $\HH$ is skew-adjoint.

Consider
\begin{equation}
 \partial_t\omega=\omega\HH\omega,
 \qquad \omega(0,x)=\omega_0(x).                \label{eq:clm}
\end{equation}
Let
\[
 \mu=\mean{\omega_0}=\frac1{2\pi}\int_{-\pi}^{\pi}\omega_0(x)\dd x,
 \quad f=\omega-\mu,\quad h=\HH f,\quad z=h+\ii f.
\]

\begin{lemma}[Periodic Tricomi identity with the fixed sign]\label{lem:tricomi}
For real mean-zero $f$,
\begin{equation}
 \HH(f\HH f)=\frac{(\HH f)^2-f^2}{2}.            \label{eq:tricomi}
\end{equation}
Moreover $\mean{f\HH f}=0$.
\end{lemma}

\begin{proof}
For a trigonometric polynomial, split its Fourier series into positive and
negative Hardy parts.  On those parts \eqref{eq:Hconvention} is multiplication
by $-\ii$ and $+\ii$, respectively.  Comparing the positive and negative
parts of the product gives \eqref{eq:tricomi}; the zero Fourier coefficient
vanishes by the skew-adjointness of $\HH$.  Density and boundedness of
$\HH$ on Sobolev spaces extend the identity to smooth $f$.
\end{proof}

\begin{theorem}[Arbitrary-mean pointwise flow]\label{thm:mobius}
The mean $\mu$ is conserved and
\begin{equation}
 z_t=\ii\mu z+\frac12z^2.                         \label{eq:riccati}
\end{equation}
Before the first zero of its denominator, the exact solution is
\begin{align}
 \mu=0:\quad &z(t,x)=\frac{2z_0(x)}{2-tz_0(x)},  \label{eq:zeroz}\\
 &\omega(t,x)=
 \frac{4\omega_0(x)}{[2-th_0(x)]^2+t^2\omega_0(x)^2}; \label{eq:zeroomega}\\
 \mu\ne0:\quad &z(t,x)=
 \frac{\e^{\ii\mu t}z_0(x)}
 {1-[\e^{\ii\mu t}-1]z_0(x)/(2\ii\mu)}.        \label{eq:nonzeroz}
\end{align}
If
\begin{equation}
 \Delta(t,x)=2\ii\mu-[\e^{\ii\mu t}-1]z_0(x),   \label{eq:delta}
\end{equation}
then the real vorticity formula for $\mu\ne0$ is
\begin{equation}
 \omega(t,x)=\frac{4\mu^2\omega_0(x)}{|\Delta(t,x)|^2}. \label{eq:nonzeroomega}
\end{equation}
\end{theorem}

\begin{proof}
Skew-adjointness gives
$\partial_t\mean{\omega}=\mean{\omega\HH\omega}=0$.  Hence
\[
 f_t=\mu h+fh,\qquad
 h_t=\mu\HH h+\HH(fh)=-\mu f+\tfrac12(h^2-f^2),
\]
where Lemma~\ref{lem:tricomi} and $\HH h=-f$ were used.  Combining the two
real equations proves \eqref{eq:riccati}.  Separation gives
\eqref{eq:zeroz}.  For $\mu\ne0$, differentiating $1/z$ gives a scalar
linear equation; solving it yields \eqref{eq:nonzeroz}.
Taking the imaginary part of \eqref{eq:zeroz} gives
\eqref{eq:zeroomega}.  Finally multiply \eqref{eq:nonzeroz} by
$2\ii\mu/\Delta$ and its complex conjugate.  Using
$z_0=h_0+\ii(\omega_0-\mu)$ gives exactly
\eqref{eq:nonzeroomega}.

It remains to check that the pointwise Riccati solution is still a solution
of the nonlocal PDE, rather than merely of a larger product system.  For
$s>1/2$, the negative-frequency Hardy subspace of $H^s(\TT)$ is a Banach
algebra.  Under \eqref{eq:Hconvention},
$z_0=\HH f_0+\ii f_0$ has only strictly negative Fourier modes.  The vector
field $z\mapsto\ii\mu z+z^2/2$ is locally Lipschitz and preserves that
subspace, so uniqueness for the Banach-space ODE shows that the explicit
$z(t)$ retains strictly negative modes.  Consequently its imaginary part
has mean zero and its real part is exactly the Hilbert transform of its
imaginary part: $h=\HH f$.

Equivalently, if $g=h-\HH f$ is the constraint defect in the product
system, Lemma~\ref{lem:tricomi} gives
\[
 g_t=(\HH f)g+\tfrac12g^2-\mu\HH g-\HH(fg).
\]
This equation is locally Lipschitz in the same Sobolev algebra, and
$g(0)=0$ forces $g\equiv0$.  On any compact time interval where the
displayed denominator has a positive lower bound, the explicit rational
formula remains smooth in $x$; the same local uniqueness argument continues
the PDE solution across the interval.  Thus a denominator zero, not a loss
of the Hilbert constraint, is the only obstruction used below.
\end{proof}

\noindent\textit{Round-zero certificate: round zero fixes the periodic
Hilbert sign, proves the Tricomi closure, conserves the arbitrary mean, and
derives both exact M\"obius solutions and real vorticity formulas.}

\section{Complete forward-time classification}

Use the principal branch
\begin{equation}
 \acot:\RR\longrightarrow(0,\pi),\qquad
 \frac{\dd}{\dd r}\acot r=-\frac1{1+r^2}.        \label{eq:acot}
\end{equation}

\begin{theorem}[Nonzero mean: zero set if and only if breakdown]
\label{thm:nonzeroclock}
Suppose $\mu\ne0$, put $a=|\mu|$, and let
$Z_0=\{x:\omega_0(x)=0\}$.  If $Z_0$ is empty, the solution is global,
smooth, and has time period $2\pi/a$ (not necessarily its least period).
If $Z_0$ is nonempty, the first positive denominator pole is
\begin{equation}
 T_+=\min_{x\in Z_0}\frac2a
       \acot( h_0(x)/a ),                         \label{eq:Tnonzero}
\end{equation}
and the classical smooth solution breaks down at $T_+$.  Thus forward
breakdown occurs if and only if the initial vorticity has a zero.
\end{theorem}

\begin{proof}
For $0<t<2\pi/a$, the equation $\Delta(t,x)=0$ is equivalent to
\[
 z_0(x)=\frac{2\ii\mu}{\e^{\ii\mu t}-1}
       =\mu\cot(\mu t/2)-\ii\mu.
\]
Its imaginary part is $\omega_0(x)-\mu=-\mu$, exactly the condition
$\omega_0(x)=0$.  Its real part is
\[
 h_0(x)=\mu\cot(\mu t/2)=a\cot(at/2).
\]
The last function decreases bijectively from $+\infty$ to $-\infty$ on
the displayed time interval.  Equation \eqref{eq:acot} therefore gives one
and only one pole time for each initial zero, and compactness of $Z_0$
gives the minimum \eqref{eq:Tnonzero}.  At a pole, $z$ has a Riccati pole,
so its real part $h=\HH(\omega-\mu)$ becomes unbounded and smooth evolution
fails.  If $Z_0$ is empty, $\Delta$ never vanishes.  It is periodic in time
and bounded away from zero on one compact time cell, proving global
smoothness and the stated period.
\end{proof}

\begin{theorem}[Zero mean: signed-zero clock]\label{thm:zeroclock}
Suppose $\mu=0$ and define
\[
 Z_+=\{x:\omega_0(x)=0,\ h_0(x)>0\}.
\]
If $Z_+$ is nonempty, the first positive denominator pole is
\begin{equation}
 T_+=\min_{x\in Z_+}\frac{2}{h_0(x)}.             \label{eq:Tzero}
\end{equation}
If $Z_+$ is empty, the exact solution is global forward in time.  In
particular, the identically zero datum is stationary.
\end{theorem}

\begin{proof}
The denominator in \eqref{eq:zeroomega} is a sum of two real squares.  It
vanishes precisely when $\omega_0(x)=0$ and $2-th_0(x)=0$.  A positive
solution requires $h_0(x)>0$ and is then unique.  Taking the first such time
proves \eqref{eq:Tzero}; if there is none, no positive-time denominator can
vanish.
\end{proof}

\section{Complete one-mode phase diagram}

\begin{corollary}[One Fourier mode]\label{cor:onemode}
Let $A\ne0$, $k\in\ZZ_{>0}$, and
\begin{equation}
 \omega_0(x)=\mu+A\sin(kx),\qquad h_0(x)=-A\cos(kx). \label{eq:onemode}
\end{equation}
For $\mu\ne0$:
\begin{align*}
 |\mu|>|A|&:\quad \text{global and periodic};\\
 |\mu|=|A|&:\quad T_+=\pi/|\mu|\quad\text{(tangent zero)};\\
 0<|\mu|<|A|&:\quad
 T_+=\frac{2}{|\mu|}\acot(
 \sqrt{A^2-\mu^2}/|\mu|).
\end{align*}
For $\mu=0$, $T_+=2/|A|$.  The clock is independent of $k$; the number of
spatial copies changes with $k$.
\end{corollary}

\begin{proof}
Zeros exist exactly when $|\mu|\le|A|$.  At a zero,
$\sin(kx)=-\mu/A$, so the two possible Hilbert values are
$h_0=\pm\sqrt{A^2-\mu^2}$.  Since \eqref{eq:acot} is decreasing, the
positive value gives the first time in the crossing regime.  At equality
the Hilbert value is zero.  For zero mean, the largest positive Hilbert
value at a zero is $|A|$, and Theorem~\ref{thm:zeroclock} applies.
\end{proof}

sufficient global/breakdown criteria for both mean strata, fixes the
principal arccot branch, proves the first forward clock, and closes the
entire one-mode phase diagram.}

\section{Transverse first-pole profiles}

The Riccati pole always detects loss of smoothness through $h$.  A
vorticity rate needs spatial transversality and is not asserted at a
tangent zero.

\begin{theorem}[Local simple-pole profiles]\label{thm:profile}
Let $x_*$ be a point attaining the first forward time $T=T_+$ and assume
the initial zero is simple: $\omega_0(x_*)=0$ and
$\omega_0'(x_*)\ne0$.  Put
\[
 z_*=z_0(x_*),\qquad z_*'=h_0'(x_*)+\ii\omega_0'(x_*),
 \qquad \tau=T-t.
\]
If $\mu\ne0$, set $E_*=\e^{\ii\mu T}$ and
\begin{equation}
 D_*(y)=\ii\mu E_*z_*-(E_*-1)z_*'y.              \label{eq:Dnonzero}
\end{equation}
Then, locally uniformly for real $y$,
\begin{equation}
 \lim_{\tau\downarrow0}\tau\,
 \omega(T-\tau,x_*+\tau y)
 =\frac{4\mu^2\omega_0'(x_*)y}{|D_*(y)|^2}.      \label{eq:Pnonzero}
\end{equation}
If $\mu=0$, then $z_*=h_0(x_*)>0$, $T=2/h_0(x_*)$, and
\begin{equation}
 \lim_{\tau\downarrow0}\tau\,
 \omega(T-\tau,x_*+\tau y)
 =\frac{4\omega_0'(x_*)y}
 {|h_0(x_*)-Tz_*'y|^2}.                          \label{eq:Pzero}
\end{equation}
Both denominators are nonzero for every real $y$, and both profiles are
nontrivial.
\end{theorem}

\begin{proof}
For $\mu\ne0$, Taylor expansion of \eqref{eq:delta} at $(T,x_*)$ gives
\[
 \Delta(T-\tau,x_*+\tau y)=\tau D_*(y)+O(\tau^2),
 \qquad
 \omega_0(x_*+\tau y)=\tau\omega_0'(x_*)y+O(\tau^2).
\]
Insert these expressions into \eqref{eq:nonzeroomega}.  To see that
$D_*(y)$ cannot vanish for real $y$, write
$E_*=\e^{\ii\theta}$ and use the pole relation
$z_*=2\ii\mu/(E_*-1)$.  The ratio of the constant and linear coefficients
in \eqref{eq:Dnonzero} is
\[
 \frac{\ii\mu E_*z_*}{(E_*-1)z_*'}
 =\frac{\mu^2}{2\sin^2(\theta/2)}\frac1{z_*'}.
\]
It is not real because $\operatorname{Im}z_*'=\omega_0'(x_*)\ne0$.

For $\mu=0$, expand $2-tz_0(x)$ at $(T,x_*)$:
\[
 2-(T-\tau)z_0(x_*+\tau y)
 =\tau[h_0(x_*)-Tz_*'y]+O(\tau^2).
\]
Equation \eqref{eq:zeroomega} gives \eqref{eq:Pzero}.  A real zero of its
linear denominator would force $\omega_0'(x_*)y=0$, then also
$h_0(x_*)=0$, a contradiction.  Choosing any sufficiently small nonzero
$y$ proves that each profile is nontrivial.
\end{proof}

\begin{corollary}[Conditional global inverse-time rate]
If every point attaining $T_+$ is a simple zero, then there are constants
$0<c<C<\infty$ such that, for $t$ sufficiently close to $T_+$,
\begin{equation}
 \frac{c}{T_+-t}\le\|\omega(t)\|_{L^\infty(\TT)}
 \le\frac{C}{T_+-t}.                             \label{eq:rate}
\end{equation}
\end{corollary}

\begin{proof}
Simple zeros on a compact circle are finite.  At each first pole, the real
two-variable map from $(T-t,x-x_*)$ to the complex denominator has
invertible derivative by the nonvanishing argument in
Theorem~\ref{thm:profile}.  Hence locally its modulus is comparable to
$[(T-t)^2+(x-x_*)^2]^{1/2}$, while the numerator in either real formula is
$O(|x-x_*|)$.  The elementary bound
$r/(\tau^2+r^2)\le1/(2\tau)$ gives the upper estimate.  Away from these
finitely many neighborhoods the denominator stays bounded below.  A
nonzero value of either limiting profile supplies the lower estimate.
\end{proof}
