\section{The period-four packet and its totient main term}

The source orbit has positive unstable multiplier
\begin{equation}
L=289+24\sqrt{145}>1,
\qquad L+L^{-1}=578,
\qquad N_{\mathbb Q(\sqrt{145})/\mathbb Q}(L)=1.
\label{eq:L}
\end{equation}
For $n\ge3$ define the inversion-fixed packet
\begin{equation}
\beta_n=L^{-\varphi(n)/2}\Phi_n(L).
\label{eq:beta}
\end{equation}
Cyclotomic reciprocity expresses \eqref{eq:beta} as a polynomial in
$T=L+L^{-1}=578$.  In particular the orbit trace field is
$\mathbb Q(T)=\mathbb Q$, so $\beta_n$ is an ordinary rational integer and
the divisor below is genuinely a divisor over $\mathbb Q$.  Pairing conjugate roots of
$\Phi_n$ makes it positive.  We use the effective source-tagged divisor
\begin{equation}
D_n=\Div_{\mathbb Q}(\beta_n)
=\sum_p v_p(\beta_n)[\gamma_4,n,p].
\label{eq:Dn}
\end{equation}
The weighted tagged space inherited from HCS-P51 assigns weight $\log p$ to
the displayed basis atom.  Hence
\begin{equation}
b_n:=\|D_n\|_{\rm tag}=\log\beta_n.
\label{eq:mass}
\end{equation}
The index $n$ is part of the tag, so different packets have disjoint support
even when the same rational prime divides several packets.

\begin{theorem}[Uniform totient packet formula]
For every $n\ge3$,
\begin{equation}
b_n=\frac{\varphi(n)}2\log L+\varepsilon_n,
\qquad
\varepsilon_n=\sum_{d\mid n}\mu(n/d)\log(1-L^{-d}).
\label{eq:totientformula}
\end{equation}
Moreover
\begin{equation}
|\varepsilon_n|\le C_L:=\sum_{d\ge1}-\log(1-L^{-d})<0.001735.
\label{eq:CL}
\end{equation}
\end{theorem}

\begin{proof}
Cyclotomic M\"obius inversion gives
\[
\Phi_n(L)=\prod_{d\mid n}(L^d-1)^{\mu(n/d)}.
\]
Writing $L^d-1=L^d(1-L^{-d})$ and using
$\sum_{d\mid n}\mu(n/d)d=\varphi(n)$ proves
\eqref{eq:totientformula}.  Since $0<L^{-d}<1$,
\[
|\varepsilon_n|
\le\sum_{d\mid n}|\log(1-L^{-d})|
\le C_L.
\]
The inequality $-\log(1-x)\le x/(1-x)$ bounds the tail geometrically.
Evaluating the first 32 terms and this explicit tail gives
\eqref{eq:CL}; the calculation is independently reproduced by the project
checker.
\end{proof}

In particular $b_n>0$ for all $n\ge3$.  The correction is not merely lower
order on average: it is uniformly bounded while the main term records the
exact cyclotomic degree.
