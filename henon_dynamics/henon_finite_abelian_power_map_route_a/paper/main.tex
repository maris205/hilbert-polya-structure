\begin{abstract}
For every finite abelian group $G$ and integer $d\geq1$, we classify the
entire functional graph of $g\mapsto g^d$.  A canonical prime-support split
separates a periodic automorphism from a nilpotent transient factor.  It gives
all fixed and primitive counts and the complete finite dynamical zeta product.
\ifnum\CRevisionRound>0
Every periodic vertex receives the same explicitly specified rooted in-tree,
including every tail layer and its saturation height; the singular constant
map $d=0$ is closed separately.
\fi
\ifnum\CRevisionRound>1
For composition on the full function space we further determine every power
rank, every zero Jordan-block size, and the characteristic polynomial.
\fi
The theorem is parameter-uniform; exact finite enumeration is used only as a
replayable regression certificate.
\end{abstract}

\section{Prime-support splitting}
Write the group additively as
\[
 G=\bigoplus_{i=1}^r C_{n_i},\qquad T_d(x)=dx.
\]
For $d\geq1$, factor $n_i=a_i b_i$ so that $(a_i,d)=1$ and every prime
divisor of $b_i$ divides $d$.  Thus $(a_i,b_i)=1$.  Put
\[
 A=\bigoplus_i C_{a_i},\qquad B=\bigoplus_i C_{b_i}.
\]
The Chinese remainder theorem identifies $G=A\times B$.  Multiplication by
$d$ is an automorphism on $A$, whereas every element of $B$ reaches zero.
Functional graphs of power maps over finite groups provide the established
lineage for this decomposition~\cite{QR}; the theorem below is proved locally,
and no literature-priority claim is made.

\begin{theorem}[complete cycle atlas]
The periodic set of $T_d$ is exactly $A\times\{0\}$.  For every $n\geq1$,
\begin{equation}
 F_n:=|\Fix(T_d^n)|=\prod_i\gcd(d^n-1,a_i).                 \tag{1}
\end{equation}
The numbers of exact-period points and cycles are respectively
\begin{equation}
 P_m=\sum_{e\mid m}\mu(m/e)F_e,\qquad C_m=P_m/m.           \tag{2}
\end{equation}
In particular, only finitely many $C_m$ are nonzero and
\begin{equation}
 \zeta_{T_d}(t):=\exp\!\left(\sum_{n\geq1}F_n\frac{t^n}{n}\right)
 =\prod_{m\geq1}(1-t^m)^{-C_m}.                            \tag{3}
\end{equation}
\end{theorem}
\begin{proof}
If $(a,b)$ is periodic, then $d^k b=b$ for some $k>0$.  But a power of $d$
annihilates $B$, so $b=0$; conversely the finite automorphism on $A$ makes
every $(a,0)$ periodic.  On $C_{a_i}$ the equation
$(d^n-1)x=0$ has $\gcd(d^n-1,a_i)$ solutions, proving (1).  Partitioning
$F_m$ by exact period and applying M\"obius inversion gives (2).  Expanding
$-\log(1-t^m)$ and collecting the coefficient of $t^n/n$ proves (3).
\end{proof}

\section{Uniform transient trees and boundary faces}
Define
\[
 K_j=\left|\ker(T_d^j|_B)\right|=\prod_i\gcd(d^j,b_i),\qquad K_0=1.
\]
If $b\in B$ first reaches zero after $j$ steps, then its exact tail is $j$.
For each periodic vertex, invertibility on $A$ supplies a unique $A$-coordinate
at every backward step.  Hence its attached rooted tree is canonically the
tree of $b\mapsto db$ on $B$, and its depth-$j$ population is
\begin{equation}
 L_j=K_j-K_{j-1}\quad(j\geq1),\qquad L_0=1.                \tag{4}
\end{equation}
This specifies the complete graph: cycles are given by (2), and every cycle
vertex is decorated by the same explicitly defined $B$-tree.  The maximal
tail height is
\begin{equation}
 h=\max_{p\mid d}\left\lceil
 \frac{\max_i v_p(b_i)}{v_p(d)}\right\rceil,              \tag{5}
\end{equation}
with an empty maximum interpreted as zero.

The identity face $d=1$ lies inside the theorem: $A=G$, $B=0$, every point is
fixed, and there is no transient sector.  The exponent $d=0$ is kept outside
the prime-support notation.  It is the constant map to the identity, with one
fixed point and $|G|-1$ vertices of exact tail one; its source zeta is
$(1-t)^{-1}$.

\section{Full-function Koopman Jordan atlas}
Let $U:\mathbb C^G\to\mathbb C^G$ be the composition operator
$(Uf)(x)=f(T_d x)$.  Distinct rows of its zero--one matrix are indexed by the
distinct values of $T_d$, so
\begin{equation}
 \rank(U^j)=R_j:=|\operatorname{im}T_d^j|
 =|A|\prod_i\frac{b_i}{\gcd(d^j,b_i)}.                   \tag{6}
\end{equation}
The ranks stabilize at $|A|$.  For a nilpotent Jordan block of size $s$, the
rank loss from power $j-1$ to $j$ is one exactly when $s\geq j$.  Taking a
second difference therefore gives the number of zero blocks of exact size
$j$:
\begin{equation}
 Z_j=R_{j-1}-2R_j+R_{j+1}.                                \tag{7}
\end{equation}
The recurrent quotient is a disjoint union of the cycles counted in (2), so
its permutation characteristic factors are $\lambda^m-1$.  Consequently the
entire operator, including its transient algebraic multiplicity, satisfies
\begin{equation}
 \det(\lambda I-U)=\lambda^{|G|-|A|}
 \prod_{m\geq1}(\lambda^m-1)^{C_m}.                      \tag{8}
\end{equation}
For $d=0$, $U^2=U$; it is diagonalizable with eigenvalues $1$ once and zero
$|G|-1$ times.  In general $U$ need not be normal or unitary; only the
bijective face $B=0$ is a permutation operator.
