\section{The source and the main theorem}

Fix
\[
 K=\Q(\rho),\qquad \rho^2+\rho+1=0,
\]
and define
\begin{equation}
\label{eq:source}
 C(x)=\sum_{i=0}^{7}x_i^3,\qquad
 Q(x)=\sum_{i=0}^{6}x_ix_{i+1}+\rho x_7x_0.
\end{equation}
Let \(X=\{C=Q=0\}\subset\PP^7_K\).  Characteristic-zero smoothness is
inherited from the preceding fourth-moment construction; the release
machine source lock records the exact C50 certificate path and SHA-256.
The variety
has dimension \(5\), degree \(6\), and middle Hodge numbers
\[
 h^{4,1}=h^{1,4}=1,\qquad
 h^{3,2}=h^{2,3}=83.
\]

A \emph{projective monomial source symmetry} is a projective
transformation represented by
\[
 x_i\longmapsto \rho^{e_i}x_{\sigma(i)}
\]
that preserves \(C\) and \(Q\) up to their individual nonzero scalars.
Let \(\Gm\) be the group of all such transformations.

Put \(h=c_1(\mathcal O_X(1))\).  Define
\begin{equation}
\label{eq:ambient-projectors}
 \pi_{2i}=\frac16h^{5-i}\times h^i\quad(0\le i\le5),
 \qquad
 \pi_5=\Delta_X-\sum_{i=0}^{5}\pi_{2i},
\end{equation}
and
\begin{equation}
\label{eq:middle-projectors}
 e_G=\frac1{24}\sum_{g\in\Gm}[\Gamma_g],\qquad
 \pi_{\core}=\pi_5e_G,\qquad
 \pi_{\lev}=\pi_5-\pi_5e_G.
\end{equation}

\begin{theorem}[Dihedral middle splitting]
\label{thm:main}
The projective monomial source stabilizer is
\[
 \Gm\cong\Dih(C_{12})=C_{12}\rtimes C_2,
 \qquad |\Gm|=24.
\]
The Reynolds correspondence \(e_G\) is a self-transpose idempotent.
The two middle correspondences \(\pi_{\core}\) and \(\pi_{\lev}\) are
mutually orthogonal self-transpose idempotents in
\(\CH^5(X\times_KX)_{\Q}\).  Their middle realizations have Hodge ledgers
\[
\begin{aligned}
 \pi_{\core}H^5(X)&:
 (4,1)^1+(3,2)^4+(2,3)^4+(1,4)^1,\\
 \pi_{\lev}H^5(X)&:
 (3,2)^{79}+(2,3)^{79}.
\end{aligned}
\]
In particular, their ranks are \(10\) and \(158\).  After one Tate
twist the first ledger is
\[
 (3,0)^1+(2,1)^4+(1,2)^4+(0,3)^1.
\]
The same Chow projectors split every standard Weil realization; each
\(\ell\)-adic projector is \(G_K\)-equivariant.
\end{theorem}

\begin{theorem}[Graph-algebra optimum]
\label{thm:optimum}
Let \(q\in\Q[\Gm]\) be an idempotent whose graph correspondence acts as
the identity on \(H^{4,1}(X)\).  Then
\[
 \rank(qH^5(X))\ge10.
\]
The bound is attained by \(e_G\).  In particular, the rational graph
algebra cannot isolate only the rank-two extreme Hodge pair.
\end{theorem}

Theorem~\ref{thm:optimum} is confined to graph correspondences in
\(\Q[\Gm]\).  It does not exclude a \(K\)-rational algebraic
correspondence outside that algebra, a projector after extending
coefficients, or a different categorical construction.
