\begin{abstract}
We study the hybrid process \(\dot X=a>0\) with jump intensity \(\rho X\) and
multiplicative decrease \(X\mapsto\beta X\), \(0<\beta<1\).  For the
pre-jump value \(Y_n\), completing the integrated hazard gives the exact
square-affine chain \(Y_{n+1}^2=\beta^2Y_n^2+2aE_{n+1}/\rho\).  This yields a
unique stationary affine perpetuity, a source-local Laplace q-product, all
squared moments, the generator moment recursion, and a stationary
Markov-renewal/Palm occupation formula.  The receipt covers 27 rational
parameter tuples and exact reward skeletons, with independent checking,
symbolic identities, replay, and hostile mutations.  For positive \(\beta\)
the jump sequence is Markov rather than iid regenerative; \(\beta=0\) is the
separate reset face.  No arithmetic target determinant is claimed, so the
strict Route-A verdict is \texttt{ROUTE\_A\_REJECTED}.
\end{abstract}

\section{Process and the pre-jump chain}
Between jumps let \(X_t=\beta Y_n+at\), where \(Y_n\) is the value immediately
before the preceding jump.  The next waiting time \(T_{n+1}\) is determined by
the integrated hazard
\begin{equation}
 \rho\left(\beta Y_nT_{n+1}+\frac{aT_{n+1}^2}{2}\right)=E_{n+1},
 \qquad E_{n+1}\sim\operatorname{Exp}(1),
 \tag{1}
\end{equation}
and \(Y_{n+1}=\beta Y_n+aT_{n+1}\).

\begin{theorem}[Square-affine recurrence]
For every \(a,\rho>0\) and \(0<\beta<1\),
\begin{equation}
 Y_{n+1}^2=\beta^2Y_n^2+\frac{2a}{\rho}E_{n+1}. \tag{2}
\end{equation}
\end{theorem}
\begin{proof}
Expand \((\beta Y_n+aT)^2-\beta^2Y_n^2=2a(\beta Y_nT+aT^2/2)\) and use (1).
The factor \(a\) cannot be dropped.  \end{proof}

\begin{theorem}[Perpetuity and uniqueness]
With \(Z_n=Y_n^2\), synchronous coupling under common exponentials satisfies
\(|Z_n-Z_n'|=\beta^{2n}|Z_0-Z_0'|\).  Consequently
\[
 Z_\infty=\frac{2a}{\rho}\sum_{j\ge0}\beta^{2j}E_{-j}
\]
converges almost surely and is the unique stationary jump-chain law; every
finite initial state converges to it.
\end{theorem}
\begin{proof}
The geometric coefficients have finite sum
\(\mathbb E Z_\infty=(2a/\rho)/(1-\beta^2)<\infty\), so the nonnegative
perpetuity series is finite almost surely.  The coupling identity contracts every
initial pair, so two stationary versions coincide and the finite-start law
converges weakly (indeed in Wasserstein metrics with a finite first moment).
\end{proof}

\section{Laplace product and moments}
Independence of the exponential innovations gives the source-local transform
\begin{equation}
 \psi(s)=\mathbb E[e^{-sZ_\infty}]
 =\prod_{j\ge0}\left(1+\frac{2a}{\rho}\beta^{2j}s\right)^{-1}. \tag{3}
\end{equation}
The release stores a 12-factor rational prefix and three rational spot values
for every frozen parameter tuple.  Equation (3) is probability bookkeeping,
not an arithmetic Euler product or a target determinant.

For the continuous-time stationary law let
\(\varphi(s)=\mathbb E[e^{-sX}]\).  Applying the generator (6) to \(e^{-sx}\) gives the
source-local Laplace-generator identity
\begin{equation}
 \varphi'(s)-\varphi'(\beta s)=\frac{a}{\rho}s\,\varphi(s). \tag{4}
\end{equation}
It is not a target continuation or functional equation.

Writing \(m_k=\mathbb E[Z^k]\), expansion of (2) yields
\begin{equation}
 (1-\beta^{2k})m_k=
 \sum_{j=0}^{k-1}\binom{k}{j}\beta^{2j}
 \left(\frac{2a}{\rho}\right)^{k-j}(k-j)!m_j,\qquad m_0=1. \tag{5}
\end{equation}
All entries in the finite receipt are exact rational numbers.

\section{Stationary occupation formula}
The continuous-time generator is
\begin{equation}
 (Lf)(x)=a f'(x)+\rho x\,[f(\beta x)-f(x)]. \tag{6}
\end{equation}
For stationary moments \(M_m=\mathbb E[X^m]\), applying (6) to \(x^m\) gives
\begin{equation}
 \rho(1-\beta^m)M_{m+1}=amM_{m-1},\qquad m\ge1. \tag{7}
\end{equation}
Thus \(M_0=1\), \(M_1=\mu\), and all higher moments are exact affine expressions
in the single Palm-determined mean \(\mu\).

Let \(\pi\) denote the stationary pre-jump law.  A cycle from \(\beta Y_n\) to
\(Y_{n+1}\) has duration \(T=(Y_{n+1}-\beta Y_n)/a\) and monomial reward
\begin{equation}
 \int_0^T(\beta Y_n+at)^m\,dt
 =\frac{Y_{n+1}^{m+1}-(\beta Y_n)^{m+1}}{a(m+1)}. \tag{8}
\end{equation}
The stationary Markov-renewal/Palm ratio is therefore
\begin{equation}
 M_m=\frac{1-\beta^{m+1}}{(m+1)(1-\beta)}
 \frac{\mathbb E_\pi[Y^{m+1}]}{\mathbb E_\pi[Y]},
 \qquad
 \mu=\frac{a}{\rho(1-\beta)\mathbb E_\pi[Y]}. \tag{9}
\end{equation}
For \(\beta>0\), \(\pi\) is a stationary Markov jump-cycle law: (8) is not an
iid regeneration assertion.

\section{Boundary faces and Route-A scope}

is the genuine reset/regeneration face.  Its pre-jump density is
\(f_Y(y)=(\rho/a)y e^{-\rho y^2/(2a)}\), and its continuous-time occupation
