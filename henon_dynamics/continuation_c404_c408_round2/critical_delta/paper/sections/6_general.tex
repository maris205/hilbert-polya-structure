\section{Asymptotic couplings and the hard and soft sides}
\label{sec:general}

The exact critical sequence determines the full asymptotic
class by min--max comparison. The order of the limiting
operations is important: the spectral frequency tends to
infinity before the comparison couplings approach their
limiting values.

\subsection{Rate-free stability at a finite positive ratio}

\begin{proof}[Completion of Theorem~\ref{thm:critical}]
The form and spectral assertions follow from
Proposition~\ref{prop:compact}, and
Proposition~\ref{prop:coefficient-range} proves the assertions
about $C$. It remains to pass from exact linear couplings
to \eqref{eq:critical-assumption}.

Fix $0<\eta<\kappa$. Outside a finite set $F_\eta$ of
vertices, the couplings satisfy
\begin{equation}
 (\kappa-\eta)n\le b_n\le(\kappa+\eta)n.
 \label{eq:coupling-sandwich}
\end{equation}
Eventual two-sided comparability implies equality of the
three form domains: weighted trace summability is equivalent
on the tail, and each of the finitely many initial point
penalties is finite for an $H^1$ function. On their common
subdomain with $f(x_n)=0$ for $n\in F_\eta$, the forms
are ordered as in \eqref{eq:coupling-sandwich}.

Put $r_\eta=\abs{F_\eta}$. Combining this form ordering
with Lemma~\ref{lem:cuts} yields, for every $k$,
\begin{equation}
 N_{((\kappa+\eta)n)}(k^2)-r_\eta
       \le N_b(k^2)
       \le N_{((\kappa-\eta)n)}(k^2)+r_\eta.
 \label{eq:count-sandwich}
\end{equation}
Subtract $k\log k$, divide by $k$, and let $k\to\infty$.
Proposition~\ref{prop:exact-law} gives
\begin{align*}
 C(\kappa+\eta)
 &\le\liminf_{k\to\infty}\frac{N_b(k^2)-k\log k}{k}\\
 &\le\limsup_{k\to\infty}\frac{N_b(k^2)-k\log k}{k}
 \le C(\kappa-\eta).
\end{align*}
Continuity of $C$ permits $\eta\downarrow0$ and proves
\eqref{eq:main-law}. The integer $r_\eta$ is fixed as
$k\to\infty$; no bound on its growth as $\eta\downarrow0$
is required. This explains why the theorem imposes no rate
or monotonicity in \eqref{eq:critical-assumption}.
\end{proof}

\subsection{The fully Dirichlet comparison}

Define
\[
 \D_\infty=\{f\in\V:f(x_n)=0\text{ for every }n\ge1\},
 \qquad q_\infty[f]=\int_0^\infty\abs{f'}^2\dd x.
\]
This is the orthogonal sum of the Dirichlet forms on the
intervals $I_n$. Its domain is contained in every $\D_b$,
and the forms agree there. Thus its eigenvalues lie above
those of $H_b$ in min--max order, and
\begin{equation}
 N_D(k^2)\le N_b(k^2).
 \label{eq:dirichlet-lower-count}
\end{equation}
This comparison uses an infinite domain restriction but
does not assign it a finite-dimensional count error.

The cell eigenvalues are $(jn)^2$, $j,n\ge1$, so the
Dirichlet count is
\begin{equation}
 N_D(k^2)=\sum_{n\le k}\left\lfloor\frac{k}{n}\right\rfloor
            =k\log k+(2\gamma-1)k+O(\sqrt{k}).
 \label{eq:dirichlet-count}
\end{equation}
This is the classical divisor law underlying the model
of~\cite{egger2011trace}. To verify the stated error and
its real-$k$ convention directly, put
$h=\lfloor\sqrt{k}\rfloor$. Counting lattice points
under the hyperbola in the two coordinate strips gives
\[
 \sum_{n\le k}\left\lfloor\frac{k}{n}\right\rfloor
      =2\sum_{n\le h}\left\lfloor\frac{k}{n}\right\rfloor-h^2
      =2k\harm_h-h^2+O(h).
\]
The relation $\harm_h=\log h+\gamma+O(h^{-1})$, together
with $h^2=k+O(\sqrt{k})$, gives
\eqref{eq:dirichlet-count} for every real $k\to\infty$.

\subsection{Hard couplings}

\begin{proof}[Proof of \eqref{eq:hard-law}]
Suppose $b_n/n\to\infty$, and fix $K>0$.
Outside a finite set $F_K$, $b_n\ge Kn$.
Here the domains need not be equal: one has
\begin{equation}
 \D_b\subset\D_{(Kn)}.
 \label{eq:hard-domain}
\end{equation}
Indeed, tail summability with the larger weights implies
summability with the smaller ones, and finitely many
initial traces are harmless. On imposing zero traces at
$F_K$, the form $q_b$ dominates $q_{(Kn)}$ on the smaller
domain \eqref{eq:hard-domain}. Min--max and the finite-cut
comparison therefore give
\begin{equation}
 N_b(k^2)\le N_{(Kn)}(k^2)+\abs{F_K}.
 \label{eq:hard-count-upper}
\end{equation}
Use \eqref{eq:dirichlet-lower-count} for the lower bound
and Proposition~\ref{prop:exact-law} for the upper bound.
After subtracting $k\log k$, dividing by $k$, and taking
$k\to\infty$, one obtains
\[
 2\gamma-1\le\liminf\frac{N_b(k^2)-k\log k}{k}
     \le\limsup\frac{N_b(k^2)-k\log k}{k}\le C(K).
\]
Finally $K\to\infty$ and
Proposition~\ref{prop:coefficient-range} prove
\eqref{eq:hard-law}.
\end{proof}

\subsection{Soft couplings}

\begin{proof}[Proof of \eqref{eq:soft-law}]
Suppose $b_n/n\to0$, retaining
\eqref{eq:compact-assumption}. For any fixed $K>0$,
outside a finite set $F_K$ one has $b_n\le Kn$.
The domain inclusion is now
\begin{equation}
 \D_{(Kn)}\subset\D_b.
 \label{eq:soft-domain}
\end{equation}
On the domains with zero traces at $F_K$, the $b$-form
is bounded above by the $Kn$-form on the smaller domain.
Thus the variational comparison has the reverse direction
from \eqref{eq:hard-count-upper}:
\begin{equation}
 N_b(k^2)\ge N_{(Kn)}(k^2)-\abs{F_K}.
 \label{eq:soft-count-lower}
\end{equation}
It follows that
\[
 \liminf_{k\to\infty}\frac{N_b(k^2)-k\log k}{k}
                         \ge C(K)
\]
for every $K>0$. Since $C(K)\to\infty$ as
$K\downarrow0$, this is precisely convergence to positive
infinity in \eqref{eq:soft-law}. Compactness continues to
depend on the separate condition $n b_n\to\infty$.
In particular this argument gives neither a discrete
spectrum for the zero-coupling half-line nor a universal
soft-regime leading counting law.
\end{proof}
