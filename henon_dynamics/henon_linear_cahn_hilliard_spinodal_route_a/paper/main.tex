classify the linear Cahn--Hilliard flow

lattice-shell spectrum, and the complete stable/critical/spinodal atlas.

\section{One theorem for the entire parameter space}
Let $\T^d=(\mathbb R/2\pi\mathbb Z)^d$, $H=L^2_0(\T^d)$, and
\[
 A=-\kappa\Delta^2-\alpha\Delta,\qquad
 D(A)=H^4(\T^d)\cap H,
 \quad \kappa>0,\quad\alpha\in\mathbb R.
\]
Write $r_d(n)=\#\{k\in\mathbb Z^d:|k|^2=n\}$.

\begin{theorem}[Full linear spinodal atlas]\label{thm:atlas}
For every finite $d\ge1$ the operator $A$ is self-adjoint, bounded above,
and has compact resolvent. It generates a self-adjoint analytic semigroup
$S(t)$, trace class for every $t>0$. On shell $n>0$,
\begin{equation}\label{eq:rate}
 Ae_k=\sigma_ne_k,\qquad \sigma_n=\alpha n-\kappa n^2,
\end{equation}
with multiplicity $r_d(n)$. The mean-zero equilibrium is exponentially
stable iff $\alpha<\kappa$, critical iff $\alpha=\kappa$, and spinodally
unstable iff $\alpha>\kappa$. Its unstable Morse index and kernel dimension
are
\begin{equation}\label{eq:index}
 M=\sum_{n<\alpha/\kappa}r_d(n),\qquad
 K=\sum_{n=\alpha/\kappa}r_d(n),
\end{equation}
where absent shells contribute zero. At criticality $K=r_d(1)=2d$.

The spectral bound is attained precisely on
\begin{equation}\label{eq:fast}
 \mathcal N_*=\operatorname*{argmax}_{n\ge1,\ r_d(n)>0}
 (\alpha n-\kappa n^2),
\end{equation}
including every tie. For $u_0\ne0$, let
$\lambda_*=\max\{\sigma_n:P_nu_0\ne0\}$. Then
\begin{equation}\label{eq:limit}
 e^{-\lambda_*t}S(t)u_0\longrightarrow
 \sum_{\sigma_n=\lambda_*}P_nu_0\quad\hbox{in }L^2.
\end{equation}
Every recurrent state is stationary; hence there is no nonstationary
periodic solution.
\end{theorem}

\begin{proof}
For $e_k=(2\pi)^{-d/2}e^{ik\cdot x}$, $\Delta e_k=-|k|^2e_k$, so
\eqref{eq:rate} follows. The real diagonal values tend to $-\infty$ as
$|k|\to\infty$, proving self-adjointness, compact resolvent, and an upper
bound. The spectral theorem yields the analytic semigroup, while
\[
 \sum_{k\ne0}e^{t(\alpha|k|^2-\kappa|k|^4)}<\infty\qquad(t>0)
\]
by quartic domination, proving trace class. The sign factorization
$\sigma_n=n(\alpha-\kappa n)$ proves the chamber and index formulas.

The maximization in \eqref{eq:fast} is over the whole represented spectrum,
not a numerical cutoff. Indeed
\begin{equation}\label{eq:square}
 \sigma_n=\frac{\alpha^2}{4\kappa}
 -\kappa\left(n-\frac{\alpha}{2\kappa}\right)^2.
\end{equation}
If $\alpha/\kappa\le1$, then for $n>1$,
$\sigma_n-\sigma_1=(n-1)[\alpha-\kappa(n+1)]<0$. If
$\alpha/\kappa>1$, shell one has positive rate whereas every
$n\ge\alpha/\kappa$ has nonpositive rate. Thus an explicit finite set of
represented integers contains every maximizer, and all ties are retained.
For a fixed datum the same decay at infinity gives a largest rate in its
actual support. Factoring $e^{\lambda_*t}$ and dominated convergence prove
\eqref{eq:limit}. Finally, recurrence along $t_j\to\infty$ forces
$e^{t_j\sigma_n}\widehat u_0(k)\to\widehat u_0(k)$ coefficientwise; every
nonzero coefficient must have $\sigma_n=0$, so the state is stationary.
\end{proof}

\section{Gradient-flow law and singular faces}
Set
\[
 \mathcal F(u)=\frac12\int_{\T^d}
 (\kappa|\nabla u|^2-\alpha|u|^2)\dd x,
 \qquad \mu=-\kappa\Delta u-\alpha u.
\]
The equation is $u_t=\Delta\mu$. Periodic integration by parts gives the
exact dissipation identity
\begin{equation}\label{eq:energy}
 \frac{\dd}{\dd t}\mathcal F(u(t))
 =\langle\mu,\Delta\mu\rangle=-\|\nabla\mu\|_2^2.
\end{equation}
This explains the energy signature without turning linear instability into
a nonlinear saturation or coarsening theorem.

\begin{proposition}[Boundary completion]\label{prop:boundary}
On full $L^2$, the constant mode is stationary and equals the conserved
mean. On the mean-zero space $H$, the singular face $\kappa=0$ is
interpreted with its natural generator domain: for $\alpha\ne0$, the
operator $-\alpha\Delta$ has domain $H^2(\T^d)\cap H$, while for
$\alpha=0$ the zero generator has domain all of $H$. It gives forward heat
when $\alpha<0$, the identity when $\alpha=0$, and no bounded $L^2$ $C_0$
semigroup when $\alpha>0$.
\end{proposition}
\begin{proof}
The zero shell has rate zero. On the stated natural domains, the
$\kappa=0$ positive shells have rates $\alpha n$. Negative $\alpha$ gives
heat, zero gives identity, and positive $\alpha$ makes
$\sup_ne^{t\alpha n}=\infty$ for every $t>0$.
\end{proof}
