\section{Frozen class and profile equation}
Let $m>0$, $M>0$, and consider
\begin{equation}\label{eq:pde}
 u_t=(u^m)_{xx},\qquad x\in\R,\quad t>0.
\end{equation}
We classify only centered, nonnegative, integrable first-kind profiles of mass
$M$ such that $F^m\in W^{1,1}_{\mathrm{loc}}(\R)$ and the integrated
zero-flux law below holds almost everywhere.  Uniqueness is up to
almost-everywhere equality (equivalently, for the continuous representative
determined by $F^m$).  Translates, signed profiles, higher dimensions, and
arbitrary Cauchy solutions are outside the claim.  Source-type ownership and
the surrounding theory are classical
\cite{Barenblatt,Vazquez}; no formula priority is claimed.

Mass-preserving scaling has
\[
 u(x,t)=t^{-\alpha}F(\xi),\qquad \xi=xt^{-\alpha},\qquad
 \alpha=\frac1{m+1}.
\]
Indeed exponent matching gives $\alpha+1=m\alpha+2\alpha$.
The profile equation (in distributions) and its zero-flux integral (almost
everywhere) are
\begin{equation}\label{eq:profile}
 -\alpha(F+\xi F')=(F^m)'',\qquad
 (F^m)'+\alpha\xi F=0.
\end{equation}

\section{All positive exponents and exact mass}
\begin{theorem}\label{thm:atlas}
In the frozen class there is exactly one mass-$M$ profile.  For $m>1$, set
\[
 p=\frac1{m-1},\qquad k=\frac{m-1}{2m(m+1)}.
\]
Then
\begin{equation}\label{eq:porous}
 F(\xi)=(C-k\xi^2)_+^p,\qquad
 M=C^{p+1/2}k^{-1/2}\B\!\left(\frac12,p+1\right).
\end{equation}
For $m=1$,
\begin{equation}\label{eq:heat}
 F(\xi)=\frac{M}{2\sqrt\pi}\e^{-\xi^2/4}.
\end{equation}
For $0<m<1$, set
\[
 q=\frac1{1-m},\qquad b=\frac{1-m}{2m(m+1)}.
\]
Then
\begin{equation}\label{eq:fast}
 F(\xi)=(C+b\xi^2)^{-q},\qquad
 M=C^{1/2-q}b^{-1/2}\B\!\left(\frac12,q-\frac12\right).
\end{equation}
In each nonlinear regime the displayed equation uniquely fixes $C>0$.
\end{theorem}
\begin{proof}
Where $F>0$, equation~\eqref{eq:profile} gives
\[
 (F^{m-1})'=-\frac{\alpha(m-1)}m\xi \quad(m\ne1),
 \qquad (\log F)'=-\alpha\xi \quad(m=1).
\]
Integration on each positivity component gives
\eqref{eq:porous}--\eqref{eq:fast}.  Local absolute continuity makes $F$
continuous.  In the porous case a finite endpoint forces
$F^{m-1}=C-k\xi^2=0$; the two endpoints therefore have equal modulus, so the
only nonempty component is $(-\sqrt{C/k},\sqrt{C/k})$.  In the fast and heat
cases the displayed finite expressions cannot approach zero at a finite
component endpoint, while integrability excludes a missing infinite end;
their positivity set is all of $\R$.  Thus no extra positivity component or
zero interval is possible.  Substituting $z=k\xi^2/C$ or
$z=b\xi^2/C$ evaluates the two mass integrals by Euler's Beta integral;
Gaussian integration gives~\eqref{eq:heat}.  The porous mass is strictly
increasing in $C$, while the fast mass is strictly decreasing because
$q>1$.  Thus mass fixes $C$.  Conversely the formulas satisfy
\eqref{eq:profile}, have mass $M$, and are centered.  This proves existence
and uniqueness inside the frozen profile class.
\end{proof}

\section{All absolute moments and the sharp tail boundary}
\begin{proposition}\label{prop:moments}
For every $r>-1$, the porous and Gaussian absolute moments are respectively
\begin{equation}\label{eq:pmoment}
 \int_\R |\xi|^rF_{\mathrm{por}}(\xi)\,d\xi
 =C^{p+(r+1)/2}k^{-(r+1)/2}
 \B\!\left(\frac{r+1}{2},p+1\right).
\end{equation}
For the Gaussian,
\begin{equation}\label{eq:gmoment}
 \int_\R |\xi|^rF_{\mathrm{G}}(\xi)\,d\xi
 =M\,2^r\frac{\Gamma((r+1)/2)}{\sqrt\pi}.
\end{equation}
Both are finite.  In fast diffusion the moment is finite exactly when
\begin{equation}\label{eq:threshold}
 r<2q-1=\frac{1+m}{1-m},
\end{equation}
and then equals
\begin{equation}\label{eq:fmoment}
 \int_\R |\xi|^rF_{\mathrm{fast}}(\xi)\,d\xi
 =C^{-q+(r+1)/2}b^{-(r+1)/2}
 \B\!\left(\frac{r+1}{2},q-\frac{r+1}{2}\right).
\end{equation}
At equality divergence is logarithmic; above it divergence is a power.
Hence the fast second moment is finite exactly for $m>1/3$, logarithmically
divergent at $m=1/3$.
\end{proposition}
\begin{proof}
Evenness and the substitutions used for mass give
\eqref{eq:pmoment}--\eqref{eq:fmoment}.  Alternatively,
$F(\xi)\asymp |\xi|^{-2q}$ in the fast regime, so the tail integral has
power $r-2q$ and converges exactly when $r-2q<-1$.  Equality yields
$\int^R d\xi/\xi$; the second-moment statement follows by solving
$2<(1+m)/(1-m)$.
\end{proof}

\section{Pressure, free boundary, and rescaled dissipation}
For $m>1$, the pressure
\[
 P=\frac{m}{m-1}u^{m-1}
\]
is quadratic on the positivity set.  If $R_M=\sqrt{C/k}$, then
$\operatorname{supp}u(\cdot,t)=[-R_Mt^\alpha,R_Mt^\alpha]$ and the two
interfaces $X_\pm(t)=\pm R_Mt^\alpha$ obey the exact one-sided pressure law
\[
 X_\pm'(t)=\frac{\alpha X_\pm(t)}t
 =-\lim_{x\to X_\pm(t),\,u(x,t)>0}P_x(x,t).
\]

Put $\tau=\log t$, $\xi=xt^{-\alpha}$, and $v=t^\alpha u$.  Then
\begin{equation}\label{eq:rescaled}
 v_\tau=(v^m)_{\xi\xi}+\alpha(\xi v)_\xi
 =\partial_\xi(v\partial_\xi\mu),
\end{equation}
where
\[
 \mu=\frac{m}{m-1}v^{m-1}+\frac\alpha2\xi^2\ (m\ne1),
 \qquad \mu=\log v+\frac\alpha2\xi^2\ (m=1).
\]
This is the first variation of the explicitly branched free energy
\begin{equation}\label{eq:energy}
 \mathcal F_m[v]=
 \begin{cases}
 \displaystyle\int_\R\left(\frac{v^m}{m-1}
       +\frac\alpha2\xi^2v\right)d\xi,&m\ne1,\\[5pt]
 \displaystyle\int_\R\left(v\log v-v
       +\frac14\xi^2v\right)d\xi,&m=1.
 \end{cases}
\end{equation}
Every profile in Theorem~\ref{thm:atlas} is stationary.  For sufficiently
regular positive rescaled solutions for which the displayed terms in
\eqref{eq:energy} are finite (so no $\infty-\infty$ is used) and whose decay
justifies integration by parts,
\begin{equation}\label{eq:dissipation}
 \frac{d\mathcal F_m}{d\tau}
 =\int_\R\mu\,\partial_\xi(v\partial_\xi\mu)\,d\xi
 =-\int_\R v|\partial_\xi\mu|^2\,d\xi.
\end{equation}
This identity is not asserted outside that class.  For the fast Barenblatt
profiles, the finite-second-moment/free-energy setting used here starts at
$m>1/3$.  At $m\le1/3$ that functional is not finite on the Barenblatt
profile, so equation~\eqref{eq:dissipation} is not applied to it.
