\section{Polynomial convention and theta/q-series lock}
Let a prime mean $\dd/\dd\tau$ and fix
\begin{equation}\label{eq:DH}
 x_1'=x_2x_3-x_1(x_2+x_3),\quad
 x_2'=x_3x_1-x_2(x_3+x_1),\quad
 x_3'=x_1x_2-x_3(x_1+x_2).
\end{equation}
For $\tau\in\HH$, set $Q=e^{\pi\ii\tau}$ and use theta constants with this
nome.  Define
\begin{equation}\label{eq:seed}
 (x_1,x_2,x_3)=-2\frac{\dd}{\dd\tau}
 (\log\vartheta_2,\log\vartheta_3,\log\vartheta_4).
\end{equation}

\begin{theorem}[Theta seed, exact series, and arithmetic bridge]\label{thm:theta}
The triple~\eqref{eq:seed} solves~\eqref{eq:DH}.  Put
$D=Q\,\dd/\dd Q$ and $X_j=x_j/(\pi\ii)$.  Then
$DX_1=X_2X_3-X_1(X_2+X_3)$, cyclically, and
\begin{align}
\vartheta_2&=2Q^{1/4}\sum_{m\geq0}Q^{m(m+1)},\nonumber\\
\vartheta_3&=1+2\sum_{m\geq1}Q^{m^2},\qquad
\vartheta_4=1+2\sum_{m\geq1}(-1)^mQ^{m^2}.
\label{eq:theta-series}
\end{align}
In particular $X(0)=(-1/2,0,0)$.  Moreover
\begin{equation}\label{eq:E2}
 X_1+X_2+X_3=-\frac12E_2(\tau)
 =-\frac12+12\sum_{n\geq1}\sigma_1(n)Q^{2n}.
\end{equation}
\end{theorem}
\begin{proof}
Jacobi's triple products give~\eqref{eq:theta-series}.  Their logarithmic
derivatives converge normally on compact subsets of $|Q|<1$.  The standard
theta derivative identities are
\begin{align*}
 \frac1{\pi\ii}(\log\vartheta_2)'&=\frac1{12}
 (E_2+\vartheta_3^4+\vartheta_4^4),\\
 \frac1{\pi\ii}(\log\vartheta_3)'&=\frac1{12}
 (E_2+\vartheta_2^4-\vartheta_4^4),\\
 \frac1{\pi\ii}(\log\vartheta_4)'&=\frac1{12}
 (E_2-\vartheta_2^4-\vartheta_3^4).
\end{align*}
Substitution, using $\vartheta_3^4=\vartheta_2^4+\vartheta_4^4$,
\[
 E_4=\tfrac12(\vartheta_2^8+\vartheta_3^8+\vartheta_4^8),\qquad
 DE_2=\tfrac16(E_2^2-E_4)
\]
for the present square-root nome $Q$, reduces each residual in~\eqref{eq:DH} to
zero.  Since $\dd/\dd\tau=\pi\ii D$, the scaled polynomial law follows
without an omitted factor.  Finally
$\vartheta_2\vartheta_3\vartheta_4=2\eta^3$ and
$(\log\eta)'=\pi\ii E_2/12$ give~\eqref{eq:E2}; its last equality is the
standard $E_2$ series in $Q^2$.
\end{proof}

Equation~\eqref{eq:E2} is an intrinsic quasimodular divisor-sum identity.
It is neither a target $\Lambda$-function nor a prime-orbit correspondence.

\section{PSL2 covariance, Chazy, and discriminant}
Let $F_i(x)$ denote the right side of~\eqref{eq:DH}.  A direct expansion
gives, for a common scalar $r$,
\begin{equation}\label{eq:shift}
 F_i(w_1+r,w_2+r,w_3+r)=F_i(w)-2rw_i-r^2.
\end{equation}

\begin{theorem}[Affine weight-two covariance]\label{thm:covariance}
If $\gamma\in\mathrm{SL}_2(\CC)$ has entries $a,b,c,d$ with $ad-bc=1$
and $x$ solves~\eqref{eq:DH}, then, wherever defined,
\begin{equation}\label{eq:covariance}
 x_i^\gamma(\tau)=(c\tau+d)^{-2}x_i(\gamma\tau)
 +\frac c{c\tau+d}
\end{equation}
is another solution.  For the theta seed, the modular generators act by
\begin{align}
 T:\quad &(x_1,x_2,x_3)(\tau+1)=(x_1,x_3,x_2)(\tau),\label{eq:T}\\
 S:\quad &\tau^{-2}(x_1,x_2,x_3)(-1/\tau)+\tau^{-1}(1,1,1)
 =(x_3,x_2,x_1)(\tau).\label{eq:S}
\end{align}
\end{theorem}
\begin{proof}
Write $u=c\tau+d$.  Since $(\gamma\tau)'=u^{-2}$, differentiating
\eqref{eq:covariance} gives
$u^{-4}F_i(x)-2cu^{-3}x_i-c^2u^{-2}$; equation~\eqref{eq:shift} gives the
same expression on its right side.  This proves covariance.  The standard
theta transformations under $\tau\mapsto\tau+1$ and
$\tau\mapsto-1/\tau$, followed by logarithmic differentiation, give
\eqref{eq:T}--\eqref{eq:S}; constant eighth-root multipliers disappear.
\end{proof}

\begin{theorem}[Chazy and relative discriminant laws]\label{thm:chazy}
For
\[
 S=x_1+x_2+x_3,\qquad
 \Delta=(x_1-x_2)(x_2-x_3)(x_3-x_1),
\]
every solution satisfies
\begin{equation}\label{eq:chazy}
 S'''=-4SS''+6(S')^2,
 \qquad \Delta'=-2S\Delta.
\end{equation}
Consequently $\Delta=0$ is an invariant union of collision strata.
\end{theorem}
\begin{proof}
Summing~\eqref{eq:DH} first gives
$S'=-(x_1x_2+x_2x_3+x_3x_1)$.  Differentiate twice and eliminate every
$x_i'$ with~\eqref{eq:DH}; collecting symmetric monomials yields the first
identity.  Subtracting two cyclic equations gives, for example,
$(x_1-x_2)'=-2x_3(x_1-x_2)$.  Multiplication of the three corresponding
relations proves the second identity and invariance.
\end{proof}

\begin{theorem}[Complete pair-collision branches]\label{thm:collision}
On $x_1=x_2=a$, $x_3=b$, the reduced equations are
\begin{equation}\label{eq:reduced}
 a'=-a^2,\qquad b'=a^2-2ab.
\end{equation}
For $c,C\in\CC$, every local solution with $a$ not identically zero is,
on a domain where $\tau\ne c$,
\begin{equation}\label{eq:collision}
 a=\frac1{\tau-c},\qquad
 b=\frac1{\tau-c}+\frac C{(\tau-c)^2}.
\end{equation}
The layer $C=0$ is the nonzero fully diagonal Riccati solution.  If
$a\equiv0$, then $b\equiv B$ is arbitrary for $B\in\CC$, giving the equilibrium axis
$(0,0,B)$.  Cyclic permutation gives all collision strata and all three
coordinate axes; the axes meet at the origin.
\end{theorem}
\begin{proof}
Substitution gives~\eqref{eq:reduced}.  If $a$ vanishes once, uniqueness for
the polynomial ODE forces $a\equiv0$.  Otherwise a Riccati solution is the
first formula in~\eqref{eq:collision}; its second equation is linear and
integration gives the stated $C$.  If $a=0$, the second equation says
$b'=0$.  Cyclic symmetry completes the atlas.
\end{proof}
