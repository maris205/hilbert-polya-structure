\begin{abstract}
A canonical Euler product for odd reflection packets can assign one aggregate
mean to every period, but this loses the distribution of individual orbit
weights.  We compute that loss exactly for the minimal cross-axis observable
$\chi_{\rm R}(s)=\mathbf1\{s_{-1}=s_1\}$ in the full H\'enon horseshoe.
Odd cyclic decimation conjugates its orbit sum to nearest-neighbor equality
energy on a reflected binary chain.  For $n=2m+1$, the complete packet
polynomial is
\[
 F_n(q)=2q(1+q^2)^m,
\]
and the primitive polynomial satisfies the dilation M\"obius law
$E_n(q)=\sum_{k\mid n}\mu(k)F_{n/k}(q^k)$.  Consequently the
orbit-resolved primitive pressure is
\[
 P_{\rm orb}(s)=\tfrac12\log(1+e^{-2s}).
\]
The aggregate-mean pressure is $\frac12\log2-\frac s2$, so their exact gap is
$\frac12\log\cosh s$, strictly positive for every $s\ne0$.  This proves a
global cumulant obstruction to mean-field replacement while supplying exact
coefficient data for a full orbit-resolved Euler product.  It gives no
arithmetic trace or operator.
\end{abstract}

\section{The missing distribution}

For the area-preserving map $H(x,y)=(6-x^2-y,x)$, the real horseshoe is
conjugate to the full two-shift and the reversor acts as sequence reversal
\cite{DevaneyNitecki1979,Arai2007}; see \citet{LindMarcus1995} for the
symbolic framework.  At odd period, write $A_n$ for the primitive marked
reflection packet.  The canonical packet-mean Euler product assigns to all
elements of $A_n$ the common weight generated by their average.  That
construction is intrinsic under coboundaries, but it does not retain
within-period fluctuations.

We use the minimal cross-axis separator
\begin{equation}\label{eq:chi}
 \chiR(s)=\mathbf1\{s_{-1}=s_1\},\qquad
 S_n\chiR(s)=\sum_{j=0}^{n-1}\chiR(\sigma^js).
\end{equation}
The complete and primitive moment polynomials are
\begin{align}
 F_n(q)&=\sum_{\substack{s:\,\sigma^ns=s\\s_{-j}=s_j}}
 q^{S_n\chiR(s)},\label{eq:Fdef}\\
 E_n(q)&=\sum_{s\in A_n}q^{S_n\chiR(s)}.\label{eq:Edef}
\end{align}

\section{A reflected rank-two transfer matrix}

\begin{proposition}[Odd decimation]\label{prop:decimation}
Let $n=2m+1$.  A reflection-fixed word is determined by
$a_j=s_j$, $0\le j\le m$, and
\begin{equation}\label{eq:energy}
 S_n\chiR=1+2\sum_{j=0}^{m-1}\mathbf1\{a_j=a_{j+1}\}.
\end{equation}
Consequently
\begin{equation}\label{eq:Fclosed}
 F_{2m+1}(q)=2q(1+q^2)^m.
\end{equation}
\end{proposition}

\begin{proof}
Multiplication by $2$ permutes $\mathbb Z/n\mathbb Z$, commutes with
$j\mapsto-j$, and sends the distance-two comparisons in \eqref{eq:chi} to
nearest-neighbor comparisons.  Under reflection, each path edge occurs twice
and the edge joining the two terminal reflected sites is always equal.  This
gives \eqref{eq:energy}.  Equivalently, with
\[
 T(q)=\begin{pmatrix}q^2&1\\1&q^2\end{pmatrix},
\]
one has $F_{2m+1}(q)=q(1,1)T(q)^m(1,1)^{\mathsf T}$.  The vector $(1,1)$
has eigenvalue $1+q^2$, proving \eqref{eq:Fclosed}.
\end{proof}

\begin{remark}[Exact energy distribution]
There are $2\binom mk$ reflected words of energy $1+2k$.  Thus the equality
increments form an exact binomial ledger; no asymptotic independence is
assumed.
\end{remark}

\section{Primitive subtraction with energy dilation}

Ordinary M\"obius subtraction would be wrong because repetition multiplies
both length and energy.

\begin{theorem}[Primitive moment polynomial]\label{thm:primitive}
For odd $n$,
\begin{align}
 F_n(q)&=\sum_{d\mid n}E_d(q^{n/d}),\label{eq:dilation}\\
 E_n(q)&=\sum_{k\mid n}\mu(k)F_{n/k}(q^k).\label{eq:mobius}
\end{align}
\end{theorem}

\begin{proof}
A word of least period $d\mid n$ repeats $n/d$ times, so its $n$-step
energy is $(n/d)S_d\chiR$.  Partitioning \eqref{eq:Fdef} by least period
gives \eqref{eq:dilation}; M\"obius inversion in the semigroup of the
dilations $q\mapsto q^k$ gives \eqref{eq:mobius}.
\end{proof}

At $q=1$, \eqref{eq:mobius} recovers the primitive count
$D_n=\sum_{d\mid n}\mu(n/d)2^{(d+1)/2}$.  Coefficientwise nonnegativity is
also immediate from the defining sum \eqref{eq:Edef}.

\section{Nonlinear pressure and the exact gap}

\begin{theorem}[Orbit-resolved pressure]\label{thm:pressure}
For each fixed real $s$,
\begin{equation}\label{eq:pressure}
 P_{\rm orb}(s):=
 \lim_{\substack{n\to\infty\\n\ {\rm odd}}}
 \frac1n\log E_n(e^{-s})
 =\frac12\log(1+e^{-2s}).
\end{equation}
\end{theorem}

\begin{proof}
The $k=1$ term in \eqref{eq:mobius}, using \eqref{eq:Fclosed}, has the
displayed rate.  A proper divisor has odd $k\ge3$ and rate
\[
 \frac1{2k}\log(1+e^{-2ks})
 <\frac12\log(1+e^{-2s}).
\]
For $s\ge0$ this follows from monotonicity.  For $s<0$, put
$x=e^{-2s}>1$ and use $1+x^k<(1+x)^k$.  Proper-divisor terms are therefore
exponentially smaller; the divisor count is subexponential.  Since
$E_n(e^{-s})$ is a positive defining sum, the leading asymptotic follows.
\end{proof}

The fair Bernoulli expectation of $\chiR$ is $1/2$.  The aggregate-mean
packet pressure is consequently
\begin{equation}\label{eq:mean}
 P_{\rm mf}(s)=\frac12\log2-\frac s2.
\end{equation}

\begin{corollary}[Exact cumulant obstruction]\label{cor:gap}
For every real $s$,
\begin{equation}\label{eq:gap}
 P_{\rm orb}(s)-P_{\rm mf}(s)=\frac12\log\cosh s.
\end{equation}
The gap is zero only at $s=0$, and $P_{\rm orb}''(0)=1/2$.
\end{corollary}

\begin{proof}
Use $1+e^{-2s}=2e^{-s}\cosh s$ in \eqref{eq:pressure}.  Strictness and the
second derivative are immediate.
\end{proof}

This is not merely a limiting Jensen inequality.  For every odd $n\ge5$,
the primitive reflected word with one centered $1$ has energy $n-2$, while
the centered block $111$ has energy $n-4$.  Thus finite primitive variance is
already positive.

\section{Audit and route boundary}

The certificate constructs exact integer polynomials through $n=31$.
An independent implementation enumerates every reflected word through
$n=17$ and builds the binomial rows by Pascal recurrence.  Eight tests pass
in normal and optimized modes, six dependencies are hash locked, and 25
claim mutations are rejected.

The result upgrades the packet pressure from a first moment to a full
cumulant generating function for one natural potential.  It does not attach
rational primes, von Mangoldt amplitudes, a Fredholm operator, or a
self-adjoint spectrum.  Route A remains exploratory and Route B is not
authorized.  The exact polynomials \eqref{eq:mobius} are now the natural
input for a genuinely orbit-resolved Euler product and its analytic boundary.
