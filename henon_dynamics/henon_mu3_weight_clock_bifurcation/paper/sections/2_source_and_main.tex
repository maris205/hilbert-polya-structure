\section{The source and the main theorem}\label{sec:source}

Let
\[
 K=\QQ(\rho),\qquad \rho^2+\rho+1=0.
\]
For \(p>3\), \(p\equiv1\pmod3\), choose one reduction of \(\rho\) in
\(\Fp\).  For \(n\in\{2,3,4\}\), the source is the ordered \(2n\)-variable
phase
\begin{equation}\label{eq:phase}
 \Phi_{p,n}(x_0,\ldots,x_{2n-1})
 =2\sum_{i=0}^{2n-1}x_i^3+
   \sum_{i=0}^{2n-2}x_ix_{i+1}+\rho x_{2n-1}x_0.
\end{equation}
The last term is the chronological closing edge.  We do not average it or
replace it by a cyclic transition matrix.

Set
\[
 Z_{p,n}=\#\Phi_{p,n}^{-1}(0),\qquad
 C_{p,n}=2p^{-(n-1)}Z_{p,n}-2p^n,
\]
\begin{equation}\label{eq:normalization}
 d_p=\frac{p-1}{2},\qquad c_{p,n}=\frac{C_{p,n}}{d_p}.
\end{equation}
In \(\PP^{2n-1}\), write
\[
 \cC_n=\sum_{i=0}^{2n-1}x_i^3,\qquad
 \cQ_{n,\rho}=\sum_{i=0}^{2n-2}x_ix_{i+1}
               +\rho x_{2n-1}x_0,
\]
\[
 S_n=V(\cC_n),\qquad Q_n=V(\cQ_{n,\rho}),\qquad
 X_n=S_n\cap Q_n.
\]
The predecessor source calculations prove the required smoothness for
\(n=2,3,4\) outside finite bad sets.  Whenever we state a family formula
for arbitrary \(n\), smoothness of the displayed member is an explicit
hypothesis.

We use geometric Frobenius and the convention that it acts by \(p\) on
\(\Ql(-1)\).  Define
\begin{equation}\label{eq:packets}
 E_n=\Ql(0)\oplus
 H^{2n-2}_{\mathrm{prim}}(S_{n,\overline K},\Ql)(n-1),
 \qquad
 O_n=H^{2n-3}(X_{n,\overline K},\Ql)(n-2).
\end{equation}
Fix \(S\) once and for all as the finite set of rational primes that
ramify in \(K\) or occur in the inherited source-defined bad-reduction
sets for \(n=2,3,4\).

\begin{theorem}[Weight--clock bifurcation]\label{thm:main}
At every good split prime and for \(n=2,3,4\), the following statements
hold.
\begin{enumerate}
\item The packets \(E_n\) and \(O_n\) are pure of weights \(0\) and \(1\),
respectively, and
\begin{equation}\label{eq:trace-main}
 C_{p,n}=-2\{\Tr(F_p\mid E_n)+\Tr(F_p\mid O_n)\}.
\end{equation}
\item For any smooth member of the displayed family,
\[
 \rank E_n=\frac{4^n+5}{3},\qquad
 \rank O_n=\frac{2(4^n-4)}{3},\qquad
 \rank(E_n\oplus O_n)=4^n-1.
\]
\item Put
\[
 \ell_n(s)=\sum_{p\equiv1(3)}c_{p,n}p^{-ns},\qquad
 F_n(s)=\exp(-\ell_n(s)/n).
\]
There is a canonical holomorphic nonzero \(H_{n,S}\) on \(\RePart s>0\)
such that, initially for \(\RePart s>1/(2n)\),
\begin{equation}\label{eq:main-extraction}
 F_n(s)=
 \exp\!\left(\frac2n\Logzero
 L_K^{(S)}(E_n\oplus O_n,ns+1)\right)H_{n,S}(s).
\end{equation}
\item The exact denominator tower is \(u_{n,j}=ns+j\), \(j\ge1\).
A pure weight-\(w\) standard reflection maps to the center
\[
 s_{n,j}(w)=\frac{(w+1)/2-j}{n}.
\]
Consequently, the leading odd centers all equal \(0\); the leading even
centers and the higher odd centers are, respectively,
\[
 \begin{gathered}
  \bigl(s_{2,1}(0),s_{3,1}(0),s_{4,1}(0)\bigr)
    =\bigl(-\tfrac14,-\tfrac16,-\tfrac18\bigr),\\
  s_{n,j}(1)=-\frac{j-1}{n}\qquad (j\ge2).
 \end{gathered}
\]
\item These centers are invariant under every consistent Tate relabeling.
No semisimple finite-rank \(K\)-system in the direct source category can
realize the factorwise fractional leading roots at \(n=3,4\) while both
retaining the \(E_n/O_n\) weight decomposition and preserving the
split-prime trace.
\end{enumerate}
\end{theorem}

The final clause is categorical only within its stated source class.
Restriction of scalars and Galois-orbit counterpackets change that class
and are not excluded.  The inherited normalized-semifinite determinant is
also not excluded.

For \(n=2\), the exact predecessor factorization is
\begin{equation}\label{eq:n2-factor}
 F_2(s)=\zeta_K(2s+1)^7
 L(H^1(C/K),2s+1)H_2(s).
\end{equation}
The elliptic factors of \(H^1(C)\) are modular over
\(K=\QQ(\sqrt{-3})\) by \cite[Theorem 1.1]{CaraianiNewton2025};
their standard automorphic analytic package follows from
\cite[Global Theory, pp.~136--184]{GodementJacquet1972}.
These are the complete \(n=2\) factors.  On the initial Euler domain,
passing between them and the incomplete factors in
\eqref{eq:main-extraction} changes the residual only by the fixed standard
local units, which are holomorphic and nonzero there.  No corresponding
theorem is invoked for \(O_3\) or \(O_4\).

\begin{remark}[Fermat refinement at \(n=3\)]
The invariant packet is
\[
 E_3=\Ql(0)\oplus H^4_{\mathrm{prim}}(S_3)(2).
\]
Over \(K\), the primitive term has twenty normalized Tate lines and one
rank-two non-Tate Jacobi packet.  The extra trivial line gives twenty-one
Tate lines in \(E_3\).  This is the classical Fermat/Jacobi decomposition
\cite{Weil1952,Brunjes2003}; it is not a claim that the entire primitive
cohomology is Tate.
\end{remark}
