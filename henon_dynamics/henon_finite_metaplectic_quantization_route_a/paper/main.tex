\begin{abstract}
For the toral H\'enon map generated by
$A=\left(\begin{smallmatrix}3&-1\\1&0\end{smallmatrix}\right)$, we construct
an explicit unitary on $\mathbb C^7$ and prove exact Weyl Egorov covariance,
clock preservation, and antiunitary time reversal.  The trace is a finite
quadratic action sum whose stationary equation is the H\'enon recurrence.
Exact arithmetic in $\mathbb Q(\zeta_7,\sqrt7)$ then gives $U^8=I$ and
$\det(I-zU)=(1-z^8)/(1+iz)$.  Thus the natural finite-level lift is genuine,
but its periodic traces cannot carry the exponentially growing real-torus
fixed-point counts at all times.  This simultaneously closes a natural
quantization subgate and proves an aliasing obstruction.
\end{abstract}

\section{Classical source and reversal}

On $\mathbb T^2$, let
\[
 H(x,y)=(3x-y,x),\qquad
 A=\begin{pmatrix}3&-1\\1&0\end{pmatrix},\qquad
 R=\begin{pmatrix}0&1\\1&0\end{pmatrix}.
\]
Then $\det A=1$ and $RAR=A^{-1}$.  Writing $a_n=\Tr A^n$, one has
$a_0=2$, $a_1=3$, and $a_n=3a_{n-1}-a_{n-2}$.  Hence
\begin{equation}\label{eq:fix}
 \#\operatorname{Fix}(H^n)=|\det(A^n-I)|=a_n-2.
\end{equation}
The primitive-cycle counts follow by M\"obius inversion.  This real-torus
clock is kept separate from the finite quantum clock below.

\section{Frozen Weyl convention}

Let $N=7$, $\omega=e^{2\pi i/7}$, and interpret $1/2$ as $4$ in $\F_7$.
On the basis $|x\rangle$, define
\[
 Q|x\rangle=\omega^x|x\rangle,\qquad
 P|x\rangle=|x+1\rangle,
\]
\[
 W(q,p)=\omega^{-qp/2}Q^qP^p,\qquad
 \mathcal F_{xy}=7^{-1/2}\omega^{xy},
\]
\[
 C_{xx}=\omega^{(3/2)x^2},\qquad U=C\mathcal F^{-1}.
\]
Freezing these signs is essential; changing a Fourier sign changes the
classical matrix implemented by conjugation.

The half-phase convention also fixes an odd-level boundary.  At the control
modulus $N=8$, the congruence $2h=1\pmod 8$ has no solution, so the literal
formula $\omega^{-qp/2}$ cannot be copied unchanged to even level.  This does
not exclude doubled-phase or other separately defined even-level
Weil/metaplectic conventions.

\begin{theorem}[exact natural lift]\label{thm:egorov}
The matrix $U$ is unitary and, for every $(q,p)\in\F_7^2$,
\begin{equation}\label{eq:egorov}
 U W(q,p)U^{-1}=W(3q-p,q).
\end{equation}
If $K$ denotes coefficientwise complex conjugation and
$\Theta=\mathcal F K$, then
\[
 \Theta^2=I,\qquad \Theta U\Theta^{-1}=U^{-1}.
\]
\end{theorem}

\begin{proof}
The finite Fourier orthogonality sum proves unitarity.  Direct use of
$PQ=\omega^{-1}QP$, including the half-phase in $W$, gives
\eqref{eq:egorov} first on $Q$ and $P$ and then on all Weyl operators.  The
same orthogonality sum gives $\mathcal F\overline{\mathcal F}=I$ and the two
antiunitary identities.  The release certificate verifies every matrix entry
in $\mathbb Q(\zeta_7,\sqrt7)$, not by a floating tolerance.
\end{proof}

Iteration of \eqref{eq:egorov} implements $A^n$ after exactly $n$ powers of
$U$; no time rescaling occurs.  This is the clock-preservation statement used
here.

\section{Action sum and stationary recurrence}

Since
\[
 U_{xy}=7^{-1/2}\omega^{(3/2)x^2-xy},
\]
closing the matrix indices gives, for every $n\ge1$,
\begin{equation}\label{eq:action}
 \Tr U^n=7^{-n/2}
 \sum_{x_0,\ldots,x_{n-1}\in\F_7}
 \omega^{\sum_j((3/2)x_j^2-x_jx_{j-1})}.
\end{equation}
The stationary equations are
\[
 3x_j-x_{j-1}-x_{j+1}=0\pmod7,
\]
which are exactly the coordinate recurrence of $H$.  Equation
\eqref{eq:action} is an exact finite action sum, not an asserted
semiclassical correspondence with the real periodic points in
\eqref{eq:fix}.

\section{Exact aliasing obstruction}

\begin{theorem}[finite-level obstruction]\label{thm:alias}
For the frozen convention above,
\[
 U^8=I,
\]
and the seven eigenvalues are all eighth roots of unity except $-i$.
Consequently
\begin{align}
 \det(I-zU)
 &=\frac{1-z^8}{1+iz}\label{eq:det}\\
 &=1-iz-z^2+iz^3+z^4-iz^5-z^6+iz^7.
\end{align}
Moreover $\Tr U^n=7$ if $8\mid n$ and
$\Tr U^n=-(-i)^n$ otherwise.
\end{theorem}

\begin{proof}
Cyclotomic matrix multiplication gives $U^8=I$ and the first eight traces
$i,1,-i,-1,i,1,-i,7$.  Newton identities reconstruct the characteristic
polynomial $(t^8-1)/(t+i)$, proving the spectral and determinant statements.
\end{proof}

The classical count in \eqref{eq:fix} grows like the expanding eigenvalue of
$A$, whereas the fixed-level quantum trace is 8-periodic.  Therefore this
single finite level cannot carry the classical periodic growth for all $n$.
The mismatch is an exact obstruction, not a failed numerical fit.
