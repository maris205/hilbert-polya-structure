\begin{abstract}
For arbitrary ordered real positions and strictly positive velocities, we
prove that the Calogero goldfish polynomial pencil has simple real roots at
every real time.  A decreasing partial-fraction map gives exact signed-time
interlacing, global collision avoidance, and strict coordinate monotonicity.

\section{Polynomial linearization and global interlacing}
Fix $N\geq2$, positions $x_1<\cdots<x_N$, and velocities $v_i>0$.  Define
\begin{equation}\label{eq:pencil}
 P(z)=\prod_{i=1}^N(z-x_i),\qquad
 Q(z)=\sum_{i=1}^Nv_i\prod_{j\ne i}(z-x_j),\qquad
 p(z,t)=P(z)-tQ(z).
\end{equation}
Let $y_1<\cdots<y_{N-1}$ denote the roots of $Q$.

\begin{theorem}[Complete positive-cone flow]\label{thm:global}
For every $t\in\mathbb R$, $p(\cdot,t)$ has $N$ simple real roots
$z_1(t)<\cdots<z_N(t)$.  They obey
\begin{align}
 t>0:&\quad x_i<z_i(t)<y_i\ (i<N),\qquad x_N<z_N(t),\label{eq:pos}\\
 t<0:&\quad z_1(t)<x_1,\qquad y_i<z_{i+1}(t)<x_{i+1}\ (i<N).
 \label{eq:neg}
\end{align}
Every $\dot z_i$ is positive, $\sum_i\dot z_i=V:=\sum_i v_i$, and
\begin{equation}\label{eq:goldfish}
 \ddot z_i=2\sum_{j\ne i}\frac{\dot z_i\dot z_j}{z_i-z_j}.
\end{equation}
Thus~\eqref{eq:pencil} is a complete collision-free goldfish solution with
$z_i(0)=x_i$ and $\dot z_i(0)=v_i$.
\end{theorem}

\begin{proof}
Off the poles $x_i$, partial fractions give
\begin{equation}\label{eq:R}
 R(z):=\frac{Q(z)}{P(z)}=\sum_{i=1}^N\frac{v_i}{z-x_i},
 \qquad R'(z)=-\sum_{i=1}^N\frac{v_i}{(z-x_i)^2}<0.
\end{equation}
Hence $R=0$ has exactly one root $y_i$ in each interval
$(x_i,x_{i+1})$.  When $t\ne0$, $p(z,t)=0$ is equivalent to $R(z)=1/t$.
Strict decrease and the one-sided pole limits give exactly the intervals
in~\eqref{eq:pos}--\eqref{eq:neg}, including the sign-selected exterior
interval.  All roots are simple.  At $t=0$ they are the simple roots of
$P$, so this construction covers all real time.

For $t\ne0$, differentiation of $R(z_i(t))=1/t$ gives
\begin{equation}\label{eq:velocity}
 \dot z_i=-\frac1{t^2R'(z_i)}>0.
\end{equation}
At zero, implicit differentiation instead gives
$\dot z_i(0)=Q(x_i)/P'(x_i)=v_i$.  The coefficient of $z^{N-1}$ in $p$
gives the exact law
\begin{equation}\label{eq:vieta}
 \sum_i z_i(t)=\sum_i x_i+Vt,
\end{equation}
and therefore $\sum_i\dot z_i=V$.  Differentiating
$p(z,t)=\prod_i(z-z_i(t))$ in $t$ shows that the time-independent $Q$ is
\begin{equation}\label{eq:invariant}
 Q(z)=\sum_i\dot z_i(t)\prod_{j\ne i}(z-z_j(t)).
\end{equation}
At a root $z_i$, elementary logarithmic differentiation gives
\begin{align}
 \frac{p_{zz}}{p_z}&=2\sum_{j\ne i}\frac1{z_i-z_j},\label{eq:pzz}\\
 \frac{Q'}{p_z}&=\sum_{j\ne i}\frac{\dot z_i+\dot z_j}{z_i-z_j}.
 \label{eq:qprime}
\end{align}
Twice differentiating $p(z_i(t),t)=0$ and inserting
\eqref{eq:pzz}--\eqref{eq:qprime} yields~\eqref{eq:goldfish}.  Finally,
$p(z,t+s)=p(z,t)-sQ(z)$ proves the exact flow group law.
\end{proof}

\section{One-carrier scattering and boundary sharpness}
Put
\begin{equation}\label{eq:data}
 M=\sum_i v_ix_i,\qquad c=\frac MV,\qquad
 \beta_i=-\frac{P(y_i)}{Q'(y_i)},\qquad B=\sum_{i=1}^{N-1}\beta_i.
\end{equation}
\begin{theorem}[Two-ended rank-transfer scattering]\label{thm:scatter}
Every $\beta_i$ is positive.  As $t\to-\infty$,
\begin{align}
 z_1(t)&=Vt+c+\frac Bt+O(t^{-2}),\label{eq:ballminus}\\
 z_{i+1}(t)&=y_i-\frac{\beta_i}{t}+O(t^{-2})\quad(i<N),
 \label{eq:finminus}
\end{align}
whereas, as $t\to+\infty$,
\begin{align}
 z_i(t)&=y_i-\frac{\beta_i}{t}+O(t^{-2})\quad(i<N),
 \label{eq:finplus}\\
 z_N(t)&=Vt+c+\frac Bt+O(t^{-2}).\label{eq:ballplus}
\end{align}
All expansions differentiate termwise.  Thus the incoming velocity vector
tends to $(V,0,\ldots,0)$ and the outgoing vector to
$(0,\ldots,0,V)$ in rank order.
\end{theorem}

\begin{proof}
At a simple zero $y_i$ of $Q$, the implicit-function theorem with
$s=1/t$ applied to $sP(z)-Q(z)=0$ gives
\begin{equation}\label{eq:finite-expansion}
 z=y_i+\frac{P(y_i)}{tQ'(y_i)}+O(t^{-2})
   =y_i-\frac{\beta_i}{t}+O(t^{-2}).
\end{equation}
Since $y_i\in(x_i,x_{i+1})$, the signs of $P(y_i)$ and $Q'(y_i)$ are
opposite; hence $\beta_i>0$.  Comparing the two leading coefficients in
$Q=V\prod_i(z-y_i)$ yields
\begin{equation}\label{eq:anchor-sum}
 \sum_{i=1}^{N-1}y_i=\sum_{i=1}^Nx_i-c.
\end{equation}
Subtracting all finite expansions~\eqref{eq:finite-expansion} from the
exact sum~\eqref{eq:vieta}, and using~\eqref{eq:anchor-sum}, gives the
ballistic expansions including the $B/t$ term.  Interlacing
\eqref{eq:pos}--\eqref{eq:neg} fixes their ordered ranks.  Analyticity in
$s$ justifies differentiated remainders and completes the velocity claim.
\end{proof}

The strict positive cone is sharp.  For $x=(0,1)$ and $v=(1,0)$,
$p=(z-1)(z-t)$ collides at $t=1$.  For $v=(1,-1)$,
$p=z^2-z+t$ has a double root at $t=1/4$ and nonreal roots afterwards.
The $N=1$ face is a free particle; coincident initial positions are singular;
all-negative velocities follow by time reversal but are not merged into
Theorems~\ref{thm:global}--\ref{thm:scatter}.
