\section{Frozen Hamiltonian and physical sheet}
Let
\[
 \HH=\ell^2(\mathbb Z)\oplus\mathbb C|d\rangle,
 \qquad J>0,\quad \varepsilon\in\mathbb R,\quad g\in\mathbb R,
\]
and define
\begin{align}
 (Hu)_n&=J(u_{n+1}+u_{n-1})+g u_d\delta_{n0},\label{eq:chain}\\
 (Hu)_d&=\varepsilon u_d+g u_0.\label{eq:impurity}
\end{align}
This is a bounded self-adjoint operator on all of $\HH$.  Write $H_0$ for
the free chain.  Fourier transform makes $H_0$ multiplication by
$2J\cos k$, so
\[
 \sigma(H_0)=\sigma_{\ac}(H_0)=[-2J,2J]
\]
with multiplicity two in the open band.

Fix the analytic branch
\[
 S(z)=\sqrt{z^2-4J^2},\qquad S(z)/z\longrightarrow1
 \quad(z\longrightarrow\infty).
\]
Thus $S(E)>0$ for $E>2J$ and $S(E)<0$ for $E<-2J$.  A contour integral,
with the normalization checked at infinity, gives
\begin{equation}\label{eq:m}
 m(z)=\langle0,(z-H_0)^{-1}0\rangle=\frac1{S(z)}.
\end{equation}
Block inversion at $|d\rangle$ now yields
\begin{equation}\label{eq:schur}
 G_{dd}(z)=\langle d,(z-H)^{-1}d\rangle
 =\frac1{D(z)},\qquad D(z)=z-\varepsilon-\frac{g^2}{S(z)}.
\end{equation}

\begin{theorem}[Two physical-sheet bound states]\label{thm:poles}
If $g\ne0$, $H$ has exactly two exterior eigenvalues.  They are simple and
are the unique solutions
\begin{align}
 E_+-\varepsilon-\frac{g^2}{\sqrt{E_+^2-4J^2}}&=0,
 &E_+&>\max(2J,\varepsilon),\label{eq:upper}\\
 E_--\varepsilon+\frac{g^2}{\sqrt{E_-^2-4J^2}}&=0,
 &E_-&<\min(-2J,\varepsilon).\label{eq:lower}
\end{align}
Squaring produces
\begin{equation}\label{eq:quartic}
 (E-\varepsilon)^2(E^2-4J^2)-g^4=0,
\end{equation}
but a real root of~\eqref{eq:quartic} is an eigenvalue only when it also
satisfies the corresponding inequality in
\eqref{eq:upper} or~\eqref{eq:lower}.
\end{theorem}

\begin{proof}
For $E>2J$ put
\[
 D_+(E)=E-\varepsilon-\frac{g^2}{\sqrt{E^2-4J^2}}.
\]
It tends to $-\infty$ at $2J+$ and to $+\infty$ at infinity, while
\[
 D_+'(E)=1+\frac{g^2E}{(E^2-4J^2)^{3/2}}>0.
\]
Hence it has exactly one simple zero.  The equation itself forces
$E_+>\varepsilon$.  Below the band use
\[
 D_-(E)=E-\varepsilon+\frac{g^2}{\sqrt{E^2-4J^2}}.
\]
It tends to $-\infty$ at minus infinity and to $+\infty$ at $-2J-$, and
\[
 D_-'(E)=1-\frac{g^2E}{(E^2-4J^2)^{3/2}}>0.
\]
The unique zero is simple and satisfies $E_-<\varepsilon$.

At an exterior zero, equations~\eqref{eq:chain}--\eqref{eq:impurity}
give the square-summable vector
$u_{\mathbb Z}=g u_d(E-H_0)^{-1}|0\rangle$.  Conversely, an exterior
eigenvector cannot have $u_d=0$, because $H_0$ has no exterior eigenvalue;
its impurity equation therefore gives $D(E)=0$.  This establishes the
pole/eigenvalue equivalence.  Squaring loses the sign of $S(E)$ and the sign
of $E-\varepsilon$, which proves the stated filter.
\end{proof}

\section{Full spectral measure}
Let reflection send $u_n$ to $u_{-n}$.  The odd and even subspaces reduce
$H$.  On the odd subspace, $u_0=u_d=0$ and $H$ is the free Dirichlet
half-line Jacobi matrix.  It has simple purely absolutely continuous spectrum
on $[-2J,2J]$.

On the even-plus-impurity subspace, use the ordered basis
\[
 |d\rangle,\ |0\rangle,\qquad
 |e_n\rangle=\frac{|n\rangle+|-n\rangle}{\sqrt2}\quad(n\ge1).
\]
The Jacobi diagonal is $\varepsilon,0,0,\ldots$ and its successive
off-diagonals are $g,\sqrt2J,J,J,\ldots$.  If $g\ne0$, $|d\rangle$ is
cyclic.  Its scalar resolvent is~\eqref{eq:schur}, so that scalar measure
therefore determines the entire even spectral type.

Let $\mu_d$ be that scalar spectral measure.  Our convention is
\[
 G_{dd}(z)=\int_{\mathbb R}\frac{d\mu_d(x)}{z-x}.
\]
Thus $G_{dd}$ is an anti-Herglotz Cauchy transform on the upper half-plane;
$M_{dd}=-G_{dd}=\langle d,(H-z)^{-1}d\rangle$ is the standard Herglotz
function.  In particular the density has the minus sign below.

For $|E|<2J$, set $s(E)=\sqrt{4J^2-E^2}$.  The upper boundary value is
\[
 G_{dd}(E+\ii0)=\frac1{E-\varepsilon+\ii g^2/s(E)}.
\]
Stone--Stieltjes inversion gives the positive density
\begin{equation}\label{eq:density}
 \rho_d(E)=-\frac1\pi\operatorname{Im}G_{dd}(E+\ii0)
 =\frac{g^2s(E)}
 {\pi\{(E-\varepsilon)^2s(E)^2+g^4\}}.
\end{equation}
Here is the measure-theoretic closure of this formula.  Fix
$K\Subset(-2J,2J)$.  The chosen square root has a continuous upper boundary
value on $K$, and the denominator in the preceding display has nonzero
imaginary part $g^2/s(E)$.  Hence
$G_{dd}(E+\ii\eta)\to G_{dd}(E+\ii0)$ uniformly on $K$.  For every
$\phi\in C_c(K)$, Stone's formula, followed by this uniform limit, gives
\begin{equation}\label{eq:stone}
 \int\phi\,d\mu_d
 =\lim_{\eta\downarrow0}\int\phi(E)
   [-\frac1\pi\operatorname{Im}G_{dd}(E+\ii\eta)]\,dE
 =\int\phi(E)\rho_d(E)\,dE.
\end{equation}
Compact exhaustion therefore identifies the full open-band measure, not only
its absolutely continuous part.

On each real exterior, $G_{dd}$ is real analytic except at the unique simple
physical pole proved in Theorem~\ref{thm:poles}.  Formula~\eqref{eq:stone}
on pole-free exterior intervals gives zero measure there, and the two poles
give precisely two atoms.  Finally, for $E_0=\pm2J$,
\[
 G_{dd}(z)=\frac{S(z)}{(z-\varepsilon)S(z)-g^2}\longrightarrow0,
 \qquad
 \mu_d(\{E_0\})=\lim_{\eta\downarrow0}\ii\eta
 G_{dd}(E_0+\ii\eta)=0.
\]
Any measure supported on the two leftover edge points would be atomic.
Thus the density, the two exterior atoms, and the zero edge masses exhaust
$\mu_d$; no singular continuous component or embedded atom remains.

At an exterior pole $E_*$, differentiation of~\eqref{eq:schur} gives
\begin{equation}\label{eq:weight}
 w_* = \underset{z=E_*}{\operatorname{Res}}G_{dd}(z)
 =\frac1{1-g^2m'(E_*)}.
\end{equation}
On both physical exteriors $m'(E_*)<0$, hence $0<w_*<1$.  Moreover
\begin{equation}\label{eq:moments}
 G_{dd}(z)=\frac1z+\frac{\varepsilon}{z^2}
 +\frac{\varepsilon^2+g^2}{z^3}
 +\frac{\varepsilon^3+2\varepsilon g^2}{z^4}+O(z^{-5}).
\end{equation}
The coefficient of $z^{-1}$ in the Cauchy transform is the total mass, so
\begin{equation}\label{eq:mass}
 \int_{-2J}^{2J}\rho_d(E)\,dE+w_-+w_+=1.
\end{equation}

\begin{theorem}[Complete spectral type]\label{thm:spectrum}
For $J>0$ and $g\ne0$,
\[
 \sigma(H)=[-2J,2J]\cup\{E_-,E_+\},\qquad
 \sigma_{\scs}(H)=\varnothing.
\]
The band is purely absolutely continuous of multiplicity two almost
everywhere; $E_\pm$ are simple atoms with weights~\eqref{eq:weight} in the
impurity measure.  Neither band edge is an eigenvalue.
\end{theorem}

\begin{proof}
The odd component contributes one simple absolutely continuous copy of the
band.  Cyclicity, the explicit scalar measure above, and
Theorem~\ref{thm:poles} give one further simple absolutely continuous copy
and exactly the two simple atoms in the even component.  At $E=\pm2J$, the
free recurrence has affine rather than decaying half-line solutions.  An
$\ell^2$ solution must vanish on both tails and then
equations~\eqref{eq:chain}--\eqref{eq:impurity} force all remaining
coordinates to vanish.  Thus there is no edge eigenvector.
\end{proof}

\section{Stationary scattering and the Fano zero}
Let $E=2J\cos k$ with $0<k<\pi$.  Away from $E=\varepsilon$, eliminating
$u_d$ gives the on-shell origin potential
\[
 V(E)=\frac{g^2}{E-\varepsilon}.
\]
Matching an incident wave, its reflection, and its transmission at the origin
gives, in a consistent phase convention,
\begin{equation}\label{eq:amplitudes}
 t(E)=\frac{2\ii J\sin k}{2\ii J\sin k+V(E)},\qquad
 r(E)=\frac{-V(E)}{2\ii J\sin k+V(E)}.
\end{equation}
Therefore
\begin{align}
 T(E)&=\frac{4J^2\sin^2k\,(E-\varepsilon)^2}
 {g^4+4J^2\sin^2k\,(E-\varepsilon)^2},\label{eq:T}\\
 R(E)&=\frac{g^4}
 {g^4+4J^2\sin^2k\,(E-\varepsilon)^2},
 &T(E)+R(E)&=1.\label{eq:R}
\end{align}
At $E=\varepsilon$, elimination is invalid, but the original impurity
equation gives $g u_0=0$.  Thus $u_0=0$ and $T=0$ directly.  Consequently a
continuum Fano zero occurs at $E=\varepsilon$ if and only if
$|\varepsilon|<2J$; equality is a closed channel edge, not an interior
antiresonance.

\section{Degenerate faces}
If $g=0$, the operator is the direct sum of the free chain and the isolated
level $\varepsilon$.  The latter is an embedded eigenvalue for
$|\varepsilon|<2J$, an eigenvalue at a band edge for equality, and an
isolated eigenvalue outside the band.  This face cannot be obtained by merely
substituting $g=0$ into formulas whose derivation divided by $g$.

Changing $g$ to $-g$ is conjugation by the unitary that multiplies
$|d\rangle$ by $-1$.  Thus the sign of the coupling changes phases in a
chosen coordinate convention but not the spectrum, density, $T$, or $R$.

At $J=0$, the origin and impurity form the two-dimensional block $B_0$
with entries $(B_0)_{00}=0$, $(B_0)_{0d}=(B_0)_{d0}=g$, and
$(B_0)_{dd}=\varepsilon$.  Its eigenvalues are
\[
 E_{\pm}^{(0)}=\frac{\varepsilon\pm\sqrt{\varepsilon^2+4g^2}}2,
\]
while every other lattice coordinate belongs to an infinite-multiplicity
zero eigenspace.  This pure-point degeneration is not assigned the band
multiplicity of Theorem~\ref{thm:spectrum}.

The weak-coupling pole limit is nonuniform.  If $|\varepsilon|<2J$, the two
poles approach the two band edges.  If $\varepsilon>2J$, the upper pole
approaches $\varepsilon$ and the lower pole approaches $-2J$; for
$\varepsilon<-2J$ the reflected statement holds.  At
$\varepsilon=\pm2J$, both descriptions meet at a square-root threshold.
These limits do not replace the exact direct-sum classification at $g=0$.
