\section{The unstable roof and a self-consistent cone bound}
\label{sec:roof}

The source cone proof uses normalized tangent coordinates with half-widths
\[
 r_x=\frac7{48},\qquad r_y=\frac{41}{256},
 \qquad r=\frac{r_y}{r_x}=\frac{123}{112},
 \qquad a_0=r^{-1}.
\]
Write the normalized unstable line as \((1,\mu(z))\).  Its projective
recurrence is
\[
 \mu(\Had z)=\frac{a_0}{-12q(z)-r\mu(z)}.
 \tag{4.1}\label{eq:slope-recurrence}
\]

\begin{lemma}[Self-consistent cone sharpening]
\label{lem:cone}
Let
\[
 \rzero=\frac{\sqrt{17}-\sqrt{13}}2.
\]
Then
\[
 \sup_{\Lam}|\mu|
 \le a_0\rzero,
 \qquad
 \Jad(z)=|-12q(z)-r\mu(z)|
 \ge \rzero^{-1}
 =\frac{\sqrt{17}+\sqrt{13}}2.
 \tag{4.2}\label{eq:cone-sharpening}
\]
Moreover,
\[
 \Jad(z)\le 3\sqrt6+\rzero.
 \tag{4.3}\label{eq:j-upper}
\]
\end{lemma}

\begin{proof}
Compactness makes \(M=\max_{\Lam}|\mu|\) well defined.  The inherited cone
gives \(M\le1/2\).  Equations~\eqref{eq:q-range} and
\eqref{eq:slope-recurrence} imply
\[
 M\le \frac{a_0}{\sqrt{17}-rM},
 \quad\text{hence}\quad
 rM^2-\sqrt{17}M+a_0\ge0.
\]
The two roots are \(a_0\rzero\) and \(a_0/\rzero\).  The second is larger
than \(1/2\), so the inherited branch selection forces
\(M\le a_0\rzero\).  Since \(ra_0=1\) and
\(\rzero+\rzero^{-1}=\sqrt{17}\), the lower bound follows.  The upper bound
uses \(12|q|\le3\sqrt6\) and \(rM\le\rzero\).
\end{proof}

Define the adapted roof
\[
 \tad(z)=\log\Jad(z).
\]
It is positive and H\"older.  To compare it with the Euclidean geometric
potential, use the physical frame
\[
 e^u_{\mathrm{ad}}(z)=(r_x,r_y\mu(z)).
\]
Equation~\eqref{eq:slope-recurrence} is equivalent to
\[
 D\Had(z)e^u_{\mathrm{ad}}(z)
 =\lambda(z)e^u_{\mathrm{ad}}(\Had z),
 \qquad \lambda(z)=-12q(z)-r\mu(z).
\]

\begin{proposition}[Explicit norm-gauge coboundary]
\label{prop:coboundary}
Let \(b(z)=\log\|e^u_{\mathrm{ad}}(z)\|_2\).  Then
\[
 \teu(z)=\tad(z)+b(\Had z)-b(z).
 \tag{4.4}\label{eq:roof-coboundary}
\]
Consequently the two roofs have the same pressure and identical sums on every
periodic orbit.
\end{proposition}

\begin{proof}
Normalize \(e^u_{\mathrm{ad}}(z)\) to Euclidean length one in the displayed
invariant-frame identity.  Taking logarithms gives
\eqref{eq:roof-coboundary}.  The unstable
line is H\"older on a compact uniformly hyperbolic set, and the frame never
vanishes, so \(b\) is bounded H\"older.  Pressure is invariant under a
coboundary, and a periodic sum telescopes.
\end{proof}

As an immediate quantitative consequence, the absolute periodic majorant for
the instability-weighted Euler series is valid for
\[
 |z|<\frac{\rzero^{-1}}{\varphi}
 =\frac{\sqrt{17}+\sqrt{13}}{1+\sqrt5}
 =2.38828632613\ldots .
 \tag{4.5}\label{eq:euler-radius}
\]
This is an instability-weight radius, not a scalar BPS flat-trace radius.
