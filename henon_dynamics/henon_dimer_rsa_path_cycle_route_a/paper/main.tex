\begin{abstract}
We give a finite-size solution of dimer random sequential adsorption on a
path and a simple cycle, viewed as a uniformly random greedy edge order.
\ifnum\CRevisionRound=0
Conditioning on the first edge closes a convolution for the probability
generating polynomials and a Riccati ordinary generating function.  It gives
an exact mean and the cycle identity $G_n(z)=zF_{n-2}(z)$.
\fi

\section{Model, ownership, and claim boundary}
Let $P_n$ have vertices $1,\ldots,n$ and edges
$e_i=\{i,i+1\}$, with $P_0$ empty.  Assign iid continuous priorities to the
edges and inspect them in increasing order.  An edge is accepted exactly when
both endpoints are currently unmatched.  Equivalently, scan a uniformly
random permutation of the labeled edges.  Let $M_n$ be the number accepted and
write
\[
 F_n(z)=\E z^{M_n}.
\]
For the simple labeled cycle $C_n$, $n\ge3$, write the corresponding count and
PGF as $K_n$ and $G_n(z)$.

The terminal edge set is a jammed \emph{maximal} matching: no further edge can
be added.  It need not be a \emph{maximum} matching.  This distinction is part
of every statement below.  Continuous priorities merely generate a uniform
order; a discrete-priority model needs a separate tie-breaking convention.

One-dimensional blocking goes back at least to Flory's polymer calculation
\cite{Flory1939}; Evans reviews random and cooperative sequential adsorption
\cite{Evans1993}.  Penrose gives rigorous general jamming-limit context
\cite{Penrose2001}, while Dyer and Frieze place randomized greedy matching in
its algorithmic lineage \cite{DyerFrieze1991}.  We reconstruct an exact finite
theorem and integrity chain.  We do not claim invention or literature
priority for the model, the limiting density, or random greedy matching.

\section{The finite theorem}

\begin{theorem}[Path/cycle PGFs, moments, support, and boundary effect]
\label{thm:main}
Set $F_0=F_1=1$.  For every $n\ge2$,
\begin{equation}\label{eq:convolution}
 (n-1)F_n(z)=z\sum_{a+b=n-2}F_a(z)F_b(z).
\end{equation}
Thus the formal OGF $F(x,z)=\sum_{n\ge0}F_n(z)x^n$ is the unique formal
solution with $[x^0]F=[x^1]F=1$ of
\begin{equation}\label{eq:riccati}
 F_x=\frac{F-1}{x}+zxF^2,
 \qquad xF_x-F+1=zx^2F^2.
\end{equation}

Put $H_r(x)=\left.\partial_z^rF(x,z)\right|_{z=1}$.  Then
$H_0=(1-x)^{-1}$, and for every $r\ge1$, $H_r(0)=0$ and
\begin{align}
 H_r'-\left(\frac1x+\frac{2x}{1-x}\right)H_r
  ={}&x\sum_{j=1}^{r-1}\binom{r}{j}H_jH_{r-j} \notag\\
   &+rx\sum_{j=0}^{r-1}\binom{r-1}{j}H_jH_{r-1-j}.
\label{eq:triangle}
\end{align}
This triangular system determines every factorial moment.  In particular,
for $n\ge2$,
\begin{equation}\label{eq:meanexact}
 \E M_n=\sum_{j=1}^{n-1}
 (n-j)\frac{(-1)^{j+1}2^{j-1}}{j!}.
\end{equation}
If $a=\e^{-2}$ and $\alpha=(1-a)/2$, then
\begin{equation}\label{eq:pathasym}
 \E M_n=\alpha n-a+o(1),\qquad
 \Var(M_n)=a^2n+2a^2+o(1).
\end{equation}

The support is $\{0\}$ for $n=0,1$.  For $n\ge2$,
\begin{equation}\label{eq:pathsupport}
 \supp(M_n)=
 \left\{\left\lceil\frac{n-1}{3}\right\rceil,
 \ldots,\left\lfloor\frac n2\right\rfloor\right\}.
\end{equation}
For every simple cycle $n\ge3$,
\begin{equation}\label{eq:cycleidentity}
 G_n(z)=zF_{n-2}(z),
 \qquad
 \supp(K_n)=
 \left\{\left\lceil\frac n3\right\rceil,
 \ldots,\left\lfloor\frac n2\right\rfloor\right\}.
\end{equation}
Consequently
\begin{align}
 \E K_n&=1+\E M_{n-2}=\alpha n+o(1),\label{eq:cyclemean}\\
 \E K_n-\E M_n&=\e^{-2}+o(1),\label{eq:boundarymean}\\
 \Var(K_n)&=\Var(M_{n-2})=\e^{-4}n+o(1).
 \label{eq:cyclevariance}
\end{align}
Both expected occupied-vertex fractions tend to $1-\e^{-2}$.
\end{theorem}

\section{First-edge decomposition and the Riccati OGF}
The first edge in any order is accepted.  Conditional on $e_i$ being first,
its endpoints $i,i+1$ are blocked and the eligible residual vertices split
into $P_{i-1}$ and $P_{n-i-1}$.  Given this event, the relative orders induced
on those two disjoint edge sets are independent and uniform.  Their
interleaving has no effect because the vertex sets are disjoint.  Therefore
\[
 M_n\ \stackrel{d}{=}\ 1+M_{i-1}^{(L)}+M_{n-i-1}^{(R)}
 \quad\hbox{given }e_i\hbox{ first},
\]
where the two variables on the right are independent.  Averaging over the
$n-1$ equally likely choices proves~\eqref{eq:convolution}.  Multiplication by
$x^{n-1}$ and summation over $n\ge2$ gives
\[
 F_x-\frac{F-1}{x}=zxF^2,
\]
which is~\eqref{eq:riccati}.  This is a formal-power-series derivation, so it
does not presume analytic convergence in two variables.

At $z=1$, normalization gives $F_n(1)=1$, hence
$F(x,1)=(1-x)^{-1}$.  Differentiate $zF^2$ exactly $r$ times:
\begin{align*}
 \left.\partial_z^r(zF^2)\right|_{z=1}
 ={}&\sum_{j=0}^r\binom{r}{j}H_jH_{r-j}
 +r\sum_{j=0}^{r-1}\binom{r-1}{j}H_jH_{r-1-j}.
\end{align*}
The $j=0,r$ terms in the first sum are $2H_0H_r$.  Moving them to the
left proves the all-$r$ equation~\eqref{eq:triangle}; all other indices are
strictly below $r$.

For $r=1$, the equation is
\[
 H_1'-\left(\frac1x+\frac{2x}{1-x}\right)H_1
 =\frac{x}{(1-x)^2},
\]
and its formal solution regular at zero is
\begin{equation}\label{eq:H1}
 H_1(x)=\frac{x(1-\e^{-2x})}{2(1-x)^2}.
\end{equation}
Expanding the exponential and extracting $[x^n]$ proves
\eqref{eq:meanexact}.

\section{Second factorial moment and variance}
For $r=2$, the triangular equation becomes
\begin{equation}\label{eq:H2ode}
 H_2'-\left(\frac1x+\frac{2x}{1-x}\right)H_2
 =4xH_0H_1+2xH_1^2.
\end{equation}
The unique formal solution regular at zero is
\begin{equation}\label{eq:H2}
 H_2(x)=
 \frac{x\e^{-4x}
 (4x^2\e^{2x}-3x\e^{4x}-x+\e^{4x}-1)}{4(x-1)^3}.
\end{equation}
Direct differentiation verifies~\eqref{eq:H2ode}.  The formula also follows
by multiplying by the integrating factor
$\e^{2x}(1-x)^2/x$ and integrating from zero.

To extract the variance without fitting finite data, set $s=1-x$,
$a=\e^{-2}$, and $\alpha=(1-a)/2$.  Equations~\eqref{eq:H1}
and~\eqref{eq:H2} have pole parts
\begin{align}
 H_1&=\frac{\alpha}{s^2}-\frac{1+a}{2s}+R_1(x),\label{eq:H1pole}\\
 H_2&=\frac{c_3}{s^3}+\frac{c_2}{s^2}+\frac{c_1}{s}+R_2(x),\label{eq:H2pole}
\end{align}
where
\begin{equation}\label{eq:cs}
 c_3=\frac{(1-a)^2}{2},\quad
 c_2=-\frac54+a+\frac54a^2,\quad
 c_1=\frac34+a+\frac54a^2.
\end{equation}
After these pole parts are removed, $R_1,R_2$ are entire combinations of
exponentials and polynomials divided by removable powers of $1-x$.  Their
Taylor coefficients are therefore $o(1)$ (indeed factorially small up to a
polynomial factor).  It follows that
\begin{align*}
 \E M_n&=\alpha(n+1)-\frac{1+a}{2}+o(1)
       =\alpha n-a+o(1),\\
 \E(M_n)_2&=c_3\binom{n+2}{2}+c_2(n+1)+c_1+o(1).
\end{align*}
Since $c_3/2=\alpha^2$, substituting these expressions into
$\Var(M_n)=\E(M_n)_2+\E M_n-(\E M_n)^2$ cancels the quadratic term.  The
linear coefficient simplifies to $a^2$ and the constant to $2a^2$, proving
\eqref{eq:pathasym}.  Thus the requested leading law is
$\Var(M_n)=\e^{-4}n+O(1)$, with its boundary constant also identified.

\section{Exact support: binary gap words}
\begin{lemma}[Path support]\label{lem:pathsupport}
A cardinality $k$ occurs as a maximal matching of $P_n$ exactly when
$2k\le n\le3k+1$, with the separate empty cases $n=0,1$.
\end{lemma}
\begin{proof}
List the $k$ selected dimers from left to right.  There are $k+1$ gaps of
unmatched vertices: two end gaps and $k-1$ internal gaps.  Maximality is
equivalent to every gap having size at most one, since two adjacent unmatched
vertices would support another edge.  Hence
\[
 n=2k+g_0+\cdots+g_k,\qquad g_j\in\{0,1\},
\]
which is possible exactly for $0\le n-2k\le k+1$.  Conversely, choosing any
$n-2k$ of these gaps to have size one explicitly constructs a maximal
matching.  This proves the equivalence.
\end{proof}

Every such maximal matching occurs with positive probability under RSA:
place all of its mutually disjoint edges first in the scan.  They are accepted,
and every remaining edge meets one of them by maximality, so it is rejected.
Lemma~\ref{lem:pathsupport} therefore proves~\eqref{eq:pathsupport}, not merely
a bound on possible sizes.

For a cycle with $k$ selected dimers there are $k$ cyclic gaps, each again zero
or one.  Thus $n=2k+g_1+\cdots+g_k$, so $2k\le n\le3k$, and the same ordering
argument realizes every constructed matching.  This proves the support in
\eqref{eq:cycleidentity}.  Notice that smaller support values are maximal but
not maximum; for example, the middle edge alone is a maximal matching of
$P_4$, while a maximum matching has size two.

\section{Cycle identity and the boundary correction}
The first edge of $C_n$ is accepted.  Removing its two endpoints from further
competition leaves exactly a path on $n-2$ vertices; conditional on the first
edge, the residual relative order is uniform.  The identity
$K_n\stackrel d=1+M_{n-2}$ follows for every $n\ge3$, including $C_3$ and
$C_4$, and proves~\eqref{eq:cycleidentity}.  Taking moments gives
\eqref{eq:cyclemean} and
$\Var(K_n)=\Var(M_{n-2})$.  Finally,
\[
 1+\alpha(n-2)-a=\alpha n,
 \qquad 1-2\alpha-a=0,
\]
which explains the exact cancellation of the path mean's constant boundary
term.  Comparison with the same-size path leaves the positive correction
$a=\e^{-2}$ in~\eqref{eq:boundarymean}.  The cycle variance has no constant
term at this precision because
$a^2(n-2)+2a^2=a^2n$.
