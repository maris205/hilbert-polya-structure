\begin{abstract}
For every prime $q>5$ with $q\equiv1\pmod4$, six Hamilton quaternions of
norm five define a six-regular Lubotzky--Phillips--Sarnak Cayley graph.
We close its arithmetic and dynamical specialization in one normalization.
The Legendre symbol $(5/q)$ selects a connected $\PSL_2(\Fq)$ or
$\PGL_2(\Fq)$ chamber and exactly controls bipartiteness.
\ifnum\CRevisionRound>0
The oriented-edge Hashimoto operator has a complete Bass characteristic
factorization, all-iterate trace formula, M\"obius primitive-cycle inversion,
and finite-graph Ihara product.
\fi
\ifnum\CRevisionRound>1
The LPS Ramanujan bound then places every nontrivial quadratic Hashimoto
root on $|\mu|=\sqrt5$; the two chambers each have conditional prime density
$1/2$. Five full finite groups provide exact regression evidence, not a
proof by sampling. This is a source-arithmetic Route-A exploration, with no
target Euler product, zero correspondence, or Hilbert--P\'olya claim.
\fi
\end{abstract}

\section{Arithmetic quaternion chambers}
Let
\[
 \mathcal S_5=\{(1,\pm2,0,0),(1,0,\pm2,0),(1,0,0,\pm2)\}.
\]
Choose $\iota\in\Fq$ with $\iota^2=-1$ and put
\begin{equation}\label{eq:matrix}
 M_\iota(a)=
 \begin{pmatrix}
 a_0+\iota a_1&a_2+\iota a_3\\
 -a_2+\iota a_3&a_0-\iota a_1
 \end{pmatrix}.
\end{equation}
We use its projective class. Write $\chi_q=(5/q)$ and
\[
 G_q=\begin{cases}
 \PSL_2(\Fq),&\chi_q=1,\\
 \PGL_2(\Fq),&\chi_q=-1.
 \end{cases}
\]

\begin{lemma}[Norm, inverses, and gauge]\label{lem:gauge}
The six classes $[M_\iota(a)]$ are distinct and inverse closed. Replacing
$\iota$ by $-\iota$ conjugates the generating set and gives an isomorphic
labeled Cayley dynamics.
\end{lemma}

\begin{proof}
Expansion gives
\[
 \det M_\iota(a)=a_0^2+a_1^2+a_2^2+a_3^2=5.
\]
Quaternion conjugation changes the sign of the unique nonzero tail
coordinate and satisfies $M(a)M(\bar a)=5I$. The paired projective classes
are inverses. For $q>5$, equality of two classes, or equality with the
identity class, would force one of $2,4,5$ to vanish modulo $q$. Thus the
six classes are distinct and nonidentity. For
$J=\left(\begin{smallmatrix}0&1\\-1&0\end{smallmatrix}\right)$, direct
multiplication gives $M_{-\iota}(a)=JM_\iota(a)J^{-1}$.
\end{proof}

\begin{theorem}[LPS chamber closure]\label{thm:chamber}
The Cayley graph $X^{5,q}=\operatorname{Cay}(G_q,\mathcal S_5)$ is connected,
simple, six-regular, and Ramanujan. More precisely,
\[
 |G_q|=\begin{cases}
 q(q^2-1)/2,&\chi_q=1,\\
 q(q^2-1),&\chi_q=-1.
 \end{cases}
\]
The first chamber is nonbipartite. The second is bipartite, with the square
and nonsquare projective determinant classes as its two equal parts.
\end{theorem}

\begin{proof}
Connectedness, the group identification, and the Ramanujan bound are the
LPS theorem specialized to quaternion prime five~\cite{LPS}. Lemma
\ref{lem:gauge} gives simplicity and degree six, while the displayed orders
are the standard projective-group orders. Projective determinant square
class is well defined because scalar rescaling changes determinant by a
square. If $\chi_q=-1$, every edge flips this character; connectedness and
one generator give a bijection between the two parts. If $\chi_q=1$, a
connected bipartite Cayley graph would define a nontrivial word-parity map
$\PSL_2(\Fq)\to\{\pm1\}$. Its kernel would have index two, contradicting
the simplicity of $\PSL_2(\Fq)$ for $q\ge13$.
\end{proof}

\section{Exact primitive nonbacktracking dynamics}
Let $n_q=|G_q|$ and $m_q=3n_q$. On the $2m_q=6n_q$ oriented edges, define
$H_q(e,e')=1$ when the terminus of $e$ is the origin of $e'$ and
$e'\ne\bar e$. Let $A_q$ be adjacency on vertices.

\begin{theorem}[Complete determinant and orbit ledger]\label{thm:bass}
One has the polynomial identities
\begin{align}
 \det(tI-H_q)&=(t^2-1)^{2n_q}
 \det(t^2I-tA_q+5I),\label{eq:char}\\
 \det(I-uH_q)&=(1-u^2)^{2n_q}
 \det(I-uA_q+5u^2I).\label{eq:bass}
\end{align}
If $N_q(r)=\Tr(H_q^r)$ and $\Pi_q(r)$ counts prime oriented cycles of
length $r$, modulo cyclic shift but not reversal, then
\begin{equation}\label{eq:mobius}
 \Pi_q(r)=\frac1r\sum_{d\mid r}\mu(d)N_q(r/d).
\end{equation}
Moreover, in $\mathbb Q[[u]]$ and as a rational function,
\begin{equation}\label{eq:zeta}
 \prod_{[C]}(1-u^{\ell(C)})^{-1}
 =\exp\!\left(\sum_{r\ge1}\frac{N_q(r)}r u^r\right)
 =\det(I-uH_q)^{-1}.
\end{equation}
\end{theorem}

\begin{proof}
The Bass--Hashimoto formula for a graph with degree matrix $D$ is
\[
 \det(I-uH)=(1-u^2)^{m-n}\det(I-uA+u^2(D-I)).
\]
Here $D=6I$ and $m-n=2n$, which proves \eqref{eq:bass}; substitution
$u=t^{-1}$ proves \eqref{eq:char}~\cite{Hashimoto,Bass,KotaniSunada}.
By construction, a diagonal entry of $H_q^r$ selects a closed oriented-edge
walk with no inverse pair, including across the terminal seam. Thus its
trace counts cyclically nonbacktracking walks. Every such walk is a unique
repeat of a prime oriented cycle, so
$N_q(r)=\sum_{d\mid r}d\Pi_q(d)$. M\"obius inversion proves
\eqref{eq:mobius}; grouping all repeats, or taking the formal logarithm of
the finite determinant, proves \eqref{eq:zeta}.
\end{proof}

\ifnum\CRevisionRound>1
\section{Ramanujan circle and route boundary}
\begin{theorem}[Complete Hashimoto spectral placement]\label{thm:circle}
For every $\lambda\in\Spec(A_q)$, the two roots assigned by
\eqref{eq:char} solve $\mu^2-\lambda\mu+5=0$.
The factor $(t^2-1)^{2n_q}$ adds $+1$ and $-1$, each with multiplicity
$2n_q$. The trivial adjacency root $6$ gives $5,1$. The root $-6$ gives
$-5,-1$ and occurs exactly in the bipartite $\PGL_2$ chamber. Every other
quadratic root obeys $|\mu|=\sqrt5$.
\end{theorem}

\begin{proof}
Equation \eqref{eq:char} is a complete factorization of degree $6n_q$.
The LPS Ramanujan theorem gives $|\lambda|\le2\sqrt5$ after removing $6$
and the possible bipartite root $-6$. In the strict case the quadratic
roots are conjugate with product five; at equality they coincide at
$\pm\sqrt5$. Hence both have modulus $\sqrt5$. The two trivial
factorizations are $(\mu-1)(\mu-5)$ and $(\mu+1)(\mu+5)$. A connected
regular graph has eigenvalue minus its degree exactly when it is bipartite,
and Theorem~\ref{thm:chamber} identifies that chamber.
\end{proof}

\begin{proposition}[Prime chamber density]\label{prop:density}
Among primes $q>5$ with $q\equiv1\pmod4$, the $\PSL_2$ and $\PGL_2$
chambers each have conditional natural density $1/2$. They are respectively
$q\equiv1,9\pmod{20}$ and $q\equiv13,17\pmod{20}$.
\end{proposition}

\begin{proof}
Quadratic reciprocity gives $(5/q)=(q/5)$. The four displayed reduced
classes modulo 20 each have natural density $1/8$ among all primes by the
prime number theorem for arithmetic progressions~\cite{Davenport}.
Conditioning on their union gives weight $1/4$ per class and $1/2$ per pair.
\end{proof}
