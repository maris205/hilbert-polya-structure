\begin{abstract}
We solve the Caputo time-fractional heat equation on $(0,\pi)$ in its exact
Dirichlet sine basis and prove inverse-stable subordination, positivity, and
contraction for every $0<\beta\leq1$.
\ifnum\CRevisionRound>0 For $0<\beta<1$ the solution family is not a semigroup;
at every positive time it gains exactly two spatial derivatives and belongs
to $\mathcal S_p$ exactly when $p>1/2$.
\fi\space
\ifnum\CRevisionRound>1 Its scaled long-time limit is the Dirichlet resolvent
in operator norm, whereas $\beta=1$ recovers all-order heat smoothing and
exponential decay; 378 executable cells audit every endpoint.
\fi\space
No source operator ideal is promoted to target arithmetic data.
\end{abstract}

\section{The frozen memory owner}
Let $X=L^2(0,\pi)$ and
\[
 A=-\frac{d^2}{dx^2},\qquad
 D(A)=H^2(0,\pi)\cap H_0^1(0,\pi).
\]
The normalized eigenvectors $e_n=(2/\pi)^{1/2}\sin(nx)$ satisfy
$Ae_n=n^2e_n$.  We freeze
\begin{equation}
 {}^C D_t^\beta u(t)+Au(t)=0,\qquad u(0)=u_0\in X,
 \qquad 0<\beta\leq1.                                      \tag{1}
\end{equation}
For $0<\beta<1$,
\[
 {}^C D_t^\beta f(t)=\frac1{\Gamma(1-\beta)}
 \int_0^t(t-s)^{-\beta}f'(s)\,ds,
\]
and $\beta=1$ means the ordinary derivative.  Bounded-domain fractional
diffusion-wave problems admit eigenfunction constructions of this
type~\cite{SY}; here the interval permits sharp operator thresholds.

Writing
\[
 E_\beta(z)=\sum_{k=0}^\infty\frac{z^k}{\Gamma(\beta k+1)},
\]
the coefficient identity
\[
 {}^C D_t^\beta\frac{t^{\beta k}}{\Gamma(\beta k+1)}
 =\frac{t^{\beta(k-1)}}{\Gamma(\beta(k-1)+1)},\quad k\geq1,
\]
gives the unique solution family
\begin{equation}
 S_\beta(t)u_0=\sum_{n\geq1}
 E_\beta(-n^2t^\beta)\langle u_0,e_n\rangle e_n.          \tag{2}
\end{equation}

Pollard proved that $E_\beta(-x)$ is completely monotone on $x\geq0$ for
$0<\beta<1$~\cite{Pollard}.  More precisely, if $\eta_\beta(t,s)$ is the
inverse-stable density characterized by
\[
 \int_0^\infty e^{-zt}\eta_\beta(t,s)\,dt
 =z^{\beta-1}e^{-sz^\beta},
\]
then scalar Laplace inversion in (2) yields
\begin{equation}
 S_\beta(t)=\int_0^\infty e^{-sA}\eta_\beta(t,s)\,ds.     \tag{3}
\end{equation}
Thus $S_\beta(t)$ is self-adjoint and positive, preserves nonnegative
functions, and is contractive.  Dominated convergence in (2) gives strong
continuity at zero.

\section{The sharp memory theorem}
We use the spectral Sobolev scale $X^s=D(A^{s/2})$.  In particular,
$X^2=D(A)=H^2\cap H_0^1$; ``two derivatives'' includes this endpoint.

\begin{theorem}[category, smoothing, and Schatten boundaries]
Fix $0<\beta<1$ and $t>0$.
\begin{enumerate}
\item The strongly continuous family $S_\beta$ is not a semigroup.
\item Within the declared domain $\theta\geq0$,
$A^\theta S_\beta(t)$ is bounded on $X$ if and only if $\theta\leq1$.
Equivalently, within $s\geq0$, $S_\beta(t):X\to X^s$ is bounded if and only
if $s\leq2$.
\item For every $p>0$, $S_\beta(t)\in\mathcal S_p$ exactly when $p>1/2$.
The endpoint $p=1/2$ fails.
\end{enumerate}
\end{theorem}
\begin{proof}
If $S_\beta$ were a semigroup, the continuous first-mode multiplier
$E_\beta(-t^\beta)$ would be an exponential.  But
\[
 E_\beta(-t^\beta)=1-\frac{t^\beta}{\Gamma(1+\beta)}
 +O(t^{2\beta}),
\]
which is not exponential when $\beta<1$.

On the positive ray, the standard Mittag--Leffler expansion gives
\begin{equation}
 E_\beta(-x)=\frac1{x\Gamma(1-\beta)}+O(x^{-2})
 \quad(x\to\infty).                                      \tag{4}
\end{equation}
If a later reciprocal-Gamma coefficient vanishes, (4) only improves.  Hence
\[
 n^{2\theta}E_\beta(-n^2t^\beta)
 \sim\frac{t^{-\beta}}{\Gamma(1-\beta)}n^{2\theta-2}.
\]
Continuity on compact $x$-intervals supplies the upper bound at
$0\leq\theta\leq1$, and the positive asymptotic proves failure at every
$\theta>1$.  If $\theta<0$, then $A\geq I$ makes $A^\theta$ bounded, so
$A^\theta S_\beta(t)$ is also bounded; these negative powers lie outside the
declared $\theta\geq0$ smoothing domain.  The singular values themselves are
asymptotic to a positive
constant times $n^{-2}$, so their $p$th powers sum exactly for $2p>1$.
\end{proof}

The endpoint is substantive: memory retains only one bounded power of the
second-order generator.  This differs from merely asserting some unspecified
regularization, and it prevents importing the heat flow's analytic smoothing
into the fractional clock.

\ifnum\CRevisionRound>1
\section{Long time and the classical face}
\begin{theorem}[operator-norm resolvent limit]
For $0<\beta<1$,
\begin{equation}
 t^\beta S_\beta(t)\longrightarrow
 \frac{A^{-1}}{\Gamma(1-\beta)}\quad\hbox{in }\mathcal B(X),          \tag{5}
\end{equation}
and
$t^\beta\lVert S_\beta(t)\rVert\to1/\Gamma(1-\beta)$.
For $\beta=1$, $S_1(t)=e^{-tA}$, $\lVert S_1(t)\rVert=e^{-t}$,
$A^\theta S_1(t)$ is bounded for every $\theta\geq0$, and
$S_1(t)\in\mathcal S_p$ for every $p>0$.
\end{theorem}
\begin{proof}
Set $g(x)=xE_\beta(-x)-1/\Gamma(1-\beta)$.  Equation (4) gives
$g(x)\to0$.  The $n$th diagonal entry of the difference in (5) is
\[
 \frac{g(n^2t^\beta)}{n^2},
\]
whose supremum is bounded by $\sup_{x\geq t^\beta}|g(x)|\to0$.
Complete monotonicity makes the first multiplier the norm, proving the scalar
limit.  At $\beta=1$, (2) becomes $e^{-n^2t}$; exponential mode decay proves
all remaining claims.
\end{proof}

The heat face is therefore a singular structural boundary, not a uniform
extension of (4): algebraic memory tails disappear, the solution family
becomes a semigroup, and finite smoothing saturates to all orders.
