\section{The physical pressure pole}

Logarithmic differentiation of \eqref{eq:zeta} in its convergence half-plane
gives
\begin{equation}
-\frac{\zeta'_{\widehat\tau}}{\zeta_{\widehat\tau}}(s)
=\sum_\gamma\sum_{k\ge1}
\widehat\ell_\gamma e^{-sk\widehat\ell_\gamma}.
\label{eq:logder}
\end{equation}
Because the suspension is weak mixing and
\(P(-\widehat\tau)=0\), \citet[Theorem~6.3 and
Corollary~6.3.1]{ParryPollicott1990} give
\begin{equation}
-\frac{\zeta'_{\widehat\tau}}{\zeta_{\widehat\tau}}(s)
=\frac1{s-1}+G_0(s)
\label{eq:zeta-principal}
\end{equation}
as a meromorphic germ, with \(G_0\) holomorphic near \(s=1\).

\begin{lemma}[Repetition tail]
The series
\begin{equation}
R(s)=\sum_\gamma\sum_{k\ge2}
\widehat\ell_\gamma e^{-sk\widehat\ell_\gamma}
\label{eq:tail}
\end{equation}
converges normally for \(\Re s>1/2\).
\end{lemma}

\begin{proof}
The prime orbit theorem \eqref{eq:pot} implies convergence of
\(\sum_\gamma\widehat\ell_\gamma e^{-u\widehat\ell_\gamma}\) for every
real \(u>1\), by Stieltjes integration.  Fix a compact set on which
\(\Re s\ge\sigma>1/2\), and let \(L_{\min}>0\) be the minimum primitive
suspension length.  Then
\[
 \sum_{k\ge2}\widehat\ell_\gamma
 |e^{-sk\widehat\ell_\gamma}|
 \le
 \frac{\widehat\ell_\gamma e^{-2\sigma\widehat\ell_\gamma}}
 {1-e^{-\sigma L_{\min}}}.
\]
The sum of the right-hand side over \(\gamma\) converges because
\(2\sigma>1\).  The Weierstrass test proves normal convergence.
\end{proof}

\begin{theorem}[Physical Mahler pressure pole]
The primitive physical amplitude has a meromorphic germ at \(s=1\) and
\begin{equation}
\boxed{
\Aamp_{\rm phys}(s)
=\frac{3}{\pi^2h_*}\frac1{s-1}+G_{\rm phys}(s)},
\label{eq:physical-pole}
\end{equation}
where \(G_{\rm phys}\) is holomorphic near \(1\).  Moreover,
\begin{equation}
1.093445200412297389\ldots
<\operatorname*{Res}_{s=1}\Aamp_{\rm phys}
<1.093472735186032499\ldots.
\label{eq:residue-interval}
\end{equation}
\end{theorem}

\begin{proof}
Subtract \eqref{eq:tail} from \eqref{eq:logder} and use
\eqref{eq:zeta-principal}.  The remaining first-repetition series has
principal part \((s-1)^{-1}\).  Since
\(\ell_\gamma=\widehat\ell_\gamma/h_*\), equation
\eqref{eq:physical} gives \eqref{eq:physical-pole}.  The endpoint values in
\eqref{eq:residue-interval} follow from \eqref{eq:pressure-root} and the
monotonicity of \(3/(\pi^2h_*)\).
\end{proof}

The theorem is all-period and source-backed.  No finite orbit section is used
to infer the pole.
