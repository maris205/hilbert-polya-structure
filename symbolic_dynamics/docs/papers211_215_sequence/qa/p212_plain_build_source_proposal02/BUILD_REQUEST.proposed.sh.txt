#!/bin/bash
# PROPOSED SOURCE ONLY. Never run before root receives the strict pair,
# documentary delta, exact request and finite runtime/trust binding.
# Not a grant, observer, scientific verifier, reviewer or terminal build.
set -euo pipefail
umask 077
if [[ "${1-}" != "--execute-under-separate-root-grant" || "$#" != 1 ]]; then
  printf '%s\n' 'P212_BUILD_NOT_AUTHORIZED_BY_THIS_SOURCE' >&2
  exit 78
fi
P212_PREP='/root/autodl-tmp/symbolic_dynamics/docs/papers211_215_sequence/qa/p212_plain_build_source_proposal02'
P212_SRC='/root/autodl-tmp/symbolic_dynamics/papers/212-closed-pointer-orbits'
P212_BIND='/root/autodl-tmp/symbolic_dynamics/docs/papers211_215_sequence/qa/p212_plain_build_binding01'
P212_OUT='/root/autodl-tmp/symbolic_dynamics/papers/212-closed-pointer-orbits/qa_initial/plain_build01'
P212_COLD="$P212_OUT/source_only"
P212_PINFILE="$P212_PREP/PROPOSED_SOURCE_ONLY.sha256"
# These paths are proposals, not observed or created by preparation.
# Root pins the whole exact manifest and source/request in its later grant.
[[ -f "$P212_BIND/RUNTIME_INPUTS.sha256" && ! -L "$P212_BIND/RUNTIME_INPUTS.sha256" ]]
[[ -f "$P212_PINFILE" && ! -L "$P212_PINFILE" ]]
# Exclusive new tree only. Existing/failed tree is never cleaned or resumed.
mkdir -- "$P212_OUT"
mkdir -- "$P212_OUT/raw" "$P212_OUT/pass_artifacts" "$P212_OUT/pages" "$P212_COLD"
mkdir -- "$P212_COLD/sections"
exec 3>&1 4>&2
set -C
exec > "$P212_OUT/controller.stdout.raw" 2> "$P212_OUT/controller.stderr.raw"
p212_finish() {
  local p212_rc=$?
  trap - EXIT
  printf '%s\n' "$p212_rc" > "$P212_OUT/controller.exit"
  printf 'P212_INITIAL_BUILD_SUPERVISOR_EXIT=%s\n' "$p212_rc" >&3
  exit "$p212_rc"
}
trap p212_finish EXIT
P212_SOURCES=(main.tex math_commands.tex references.bib sections/01_setup.tex sections/02_returns.tex sections/03_period_set.tex sections/04_census.tex sections/05_scope.tex)
P212_PRODUCTS=(main.aux main.bbl main.blg main.log main.fls main.out main.toc main.pdf)
cp --no-clobber -- "$P212_PINFILE" "$P212_OUT/SOURCE_EXPECTED.sha256"
cp --no-clobber -- "$P212_BIND/RUNTIME_INPUTS.sha256" "$P212_OUT/RUNTIME_EXPECTED.sha256"
sha256sum -- "$P212_PREP/BUILD_REQUEST.proposed.sh.txt" "$P212_PREP/NATIVE_REQUEST.proposed.json" "$P212_BIND/RUNTIME_INPUTS.sha256" > "$P212_OUT/REQUEST_AND_BINDING.sha256"
# Fixed selected runtime paths come from root's separately received manifest.
# No which/help/version/proc query, recursive discovery, package installation,
# dynamic FLS-path following or P211 observer is hidden in this script.
sha256sum -c --strict "$P212_OUT/RUNTIME_EXPECTED.sha256" > "$P212_OUT/runtime.before.stdout" 2> "$P212_OUT/runtime.before.stderr"
(cd "$P212_SRC" && sha256sum -c --strict "$P212_OUT/SOURCE_EXPECTED.sha256") > "$P212_OUT/live_source.before.stdout" 2> "$P212_OUT/live_source.before.stderr"
for p212_name in "${P212_SOURCES[@]}"; do
  [[ -f "$P212_SRC/$p212_name" && ! -L "$P212_SRC/$p212_name" ]]
  cp --no-clobber -- "$P212_SRC/$p212_name" "$P212_COLD/$p212_name"
  cmp -- "$P212_SRC/$p212_name" "$P212_COLD/$p212_name"
done
cd "$P212_COLD"
sha256sum -c --strict "$P212_OUT/SOURCE_EXPECTED.sha256" > "$P212_OUT/cold_source.initial.stdout" 2> "$P212_OUT/cold_source.initial.stderr"
for p212_name in "${P212_PRODUCTS[@]}"; do
  [[ ! -e "$p212_name" && ! -L "$p212_name" ]]
done
p212_snapshot() {
  local p212_dest="$P212_OUT/pass_artifacts/$1"
  mkdir -- "$p212_dest"
  for p212_name in "${P212_PRODUCTS[@]}"; do
    if [[ -f "$p212_name" && ! -L "$p212_name" ]]; then
      cp --no-clobber -- "$p212_name" "$p212_dest/$p212_name"
      sha256sum -- "$p212_name" >> "$p212_dest/PRESENT.sha256"
    elif [[ -e "$p212_name" || -L "$p212_name" ]]; then
      printf 'UNEXPECTED_PRODUCT_TYPE %s\n' "$p212_name" >> "$p212_dest/TYPE_FAILURE.txt"
      return 93
    else
      printf '%s\n' "$p212_name" >> "$p212_dest/ABSENT.txt"
    fi
  done
}
P212_STEP_STATUS=0
p212_run() {
  local p212_label=$1 p212_seconds=$2
  shift 2
  printf 'cwd=%s\n' "$P212_COLD" > "$P212_OUT/raw/$p212_label.request.txt"
  printf 'supervisor_argv=' >> "$P212_OUT/raw/$p212_label.request.txt"
  printf '%q ' /usr/bin/timeout --signal=TERM --kill-after=10s "${p212_seconds}s" "$@" >> "$P212_OUT/raw/$p212_label.request.txt"
  printf '\n' >> "$P212_OUT/raw/$p212_label.request.txt"
  set +e
  /usr/bin/timeout --signal=TERM --kill-after=10s "${p212_seconds}s" "$@" < /dev/null > "$P212_OUT/raw/$p212_label.stdout.raw" 2> "$P212_OUT/raw/$p212_label.stderr.raw"
  P212_STEP_STATUS=$?
  set -e
  printf '%s\n' "$P212_STEP_STATUS" > "$P212_OUT/raw/$p212_label.supervisor_exit"
  # A timeout/launch failure is not assigned an invented native TeX exit.
}
p212_build_pass() {
  local p212_label=$1
  shift
  p212_snapshot "$p212_label.before"
  p212_run "$p212_label" 600 "$@"
  p212_snapshot "$p212_label.after"
  sha256sum -c --strict "$P212_OUT/SOURCE_EXPECTED.sha256" > "$P212_OUT/raw/$p212_label.sources.stdout" 2> "$P212_OUT/raw/$p212_label.sources.stderr"
  if [[ "$P212_STEP_STATUS" != 0 ]]; then exit "$P212_STEP_STATUS"; fi
}
P212_TEX=(/usr/bin/pdflatex -no-shell-escape -interaction=nonstopmode -halt-on-error -file-line-error -recorder main.tex)
p212_build_pass pass1 "${P212_TEX[@]}"
p212_build_pass bibtex /usr/bin/bibtex main
p212_build_pass pass2 "${P212_TEX[@]}"
p212_build_pass pass3 "${P212_TEX[@]}"
[[ -s main.pdf ]]
p212_run pdfinfo 180 /usr/bin/pdfinfo main.pdf
[[ "$P212_STEP_STATUS" == 0 ]] || exit "$P212_STEP_STATUS"
p212_run pdffonts 180 /usr/bin/pdffonts main.pdf
[[ "$P212_STEP_STATUS" == 0 ]] || exit "$P212_STEP_STATUS"
p212_run pdftotext 180 /usr/bin/pdftotext -layout main.pdf -
[[ "$P212_STEP_STATUS" == 0 ]] || exit "$P212_STEP_STATUS"
p212_run final_diagnostics 180 /usr/bin/awk '/Warning|undefined|Overfull|Underfull|Missing character|Rerun|Label.s. may have changed|Error|error|^!/ {print FILENAME ":" FNR ":" $0}' main.log main.blg
[[ "$P212_STEP_STATUS" == 0 ]] || exit "$P212_STEP_STATUS"
P212_PAGES=$(/usr/bin/awk '/^Pages:[[:space:]]/ {print $2}' "$P212_OUT/raw/pdfinfo.stdout.raw")
[[ "$P212_PAGES" =~ ^[1-9][0-9]*$ ]]
printf '%s\n' "$P212_PAGES" > "$P212_OUT/PAGE_COUNT.txt"
for ((p212_page=1; p212_page<=P212_PAGES; p212_page++)); do
  printf -v p212_label 'page-%04d' "$p212_page"
  p212_run "$p212_label" 180 /usr/bin/pdftoppm -f "$p212_page" -l "$p212_page" -singlefile -png -r 150 main.pdf "$P212_OUT/pages/$p212_label"
  [[ "$P212_STEP_STATUS" == 0 ]] || exit "$P212_STEP_STATUS"
  [[ -s "$P212_OUT/pages/$p212_label.png" ]]
  sha256sum -- "$P212_OUT/pages/$p212_label.png" >> "$P212_OUT/PAGES.sha256"
done
sha256sum -c --strict "$P212_OUT/RUNTIME_EXPECTED.sha256" > "$P212_OUT/runtime.after.stdout" 2> "$P212_OUT/runtime.after.stderr"
(cd "$P212_SRC" && sha256sum -c --strict "$P212_OUT/SOURCE_EXPECTED.sha256") > "$P212_OUT/live_source.after.stdout" 2> "$P212_OUT/live_source.after.stderr"
sha256sum -c --strict "$P212_OUT/SOURCE_EXPECTED.sha256" > "$P212_OUT/cold_source.final.stdout" 2> "$P212_OUT/cold_source.final.stderr"
sha256sum -- main.pdf main.log main.fls main.aux main.bbl main.blg > "$P212_OUT/FINAL_PRODUCTS.sha256"
printf '%s\n' 'CAPTURED_PENDING_FULL_LOG_FLS_CONFIG_PDF_AND_ACTUAL_ALL_PAGE_RECEPTION' > "$P212_OUT/STATUS.txt"
# No PDF adoption, Round0, visual PASS or package seal is produced here.
