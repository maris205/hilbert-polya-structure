\begin{abstract}
For $d\ge3$ and a finite field $\F_q$, we study the self-map of the full
matrix space
\[
 \Psi(A)=\adj(\adj A),\qquad A\in M_d(\F_q).
\]
Jacobi's classical identity gives $\Psi(A)=\det(A)^{d-2}A$; we assign that
identity zero novelty credit and determine its finite temporal consequences.
Every singular matrix enters zero in one step.  With
$\alpha=(d-1)^2$ and $E_k=(\alpha^k-1)/d$, every invertible matrix satisfies
\[
 \Psi^k(A)=\det(A)^{E_k}A.
\]
Consequently
\[
 |\Fix(\Psi^k)|=1+|\SL_d(q)|\gcd(E_k,q-1),
 \qquad
 |\im(\Psi^k)|=1+\frac{|\GL_d(q)|}{\gcd(\alpha^k,q-1)}.
\]
The second formula yields the recurrent-core size and a sharp
prime-valuation stabilization time.  The first gives every cycle count and
the Artin--Mazur zeta function by M\"obius inversion, including alternating
and third-period fixed-count anomalies.  A second route computes two
adjugates literally from minors and exhausts small matrix spaces.  The note
is owner-subtracted against Jacobi identities, projective adjugate/Cremona
maps, and scalar power-map functional graphs; external release remains on
hold.
\end{abstract}

\section{Introduction}

The adjugate is simultaneously a polynomial matrix operation and, on
invertible matrices, determinant times inversion.  Its projectivization is
a classical Cremona transformation \cite{Dolgachev2012}.  Jacobi's complementary-minor identity
and its hyperadjugate extensions are established material
\cite{Lawrence1964}; standard finite-field facts used below can be found in
\cite{LidlNiederreiter1997}.  Functional graphs of power maps on finite
groups likewise have a direct modern literature \cite{QureshiReis2023}.
None of those ingredients is claimed here.

We instead ask what happens when the adjugate is applied twice at every
time step on the \emph{entire} finite matrix space.  Two regimes coexist.
All singular ranks collapse immediately to zero, whereas every invertible
projective scalar line is invariant and carries an affine power map on its
nonzero scalars.  This separation makes both the fixed sequence and the
shrinking image sequence exact.  Their arithmetic is different: fixed
counts see $(\alpha^k-1)/d$, while image sizes see $\alpha^k$ itself.

The residual contribution is the conjunction of the iterate normal form,
singular basin, fixed and image formulas, valuation stabilization, and full
cycle/zeta census over finite fields.  A bounded source search did not find
this temporal package, but it is not an absolute novelty certificate.
External submission, public release, and priority language remain
\textbf{HOLD}.

\section{Normal form and singular collapse}

Fix a prime power $q$ and an integer $d\ge3$.  Put
\[
 X_{q,d}=M_d(\F_q),\qquad \Psi(A)=\adj(\adj A).
\]
The restriction $d\ge3$ is structural: for $d=2$, adjugation is already a
linear involution and its square is the identity.

\begin{lemma}[Double-adjugate identity]\label{lem:jacobi}
For every $A\in M_d(\F_q)$,
\begin{equation}\label{eq:double-adj}
 \Psi(A)=\det(A)^{d-2}A.
\end{equation}
In particular, every singular matrix maps to zero, and zero is fixed.
\end{lemma}

\begin{proof}
For invertible $A$, the identities
$\adj(A)=\det(A)A^{-1}$ and
$\det(\adj A)=\det(A)^{d-1}$ give \eqref{eq:double-adj} immediately.
Both sides of \eqref{eq:double-adj} are polynomial in the matrix entries,
and Jacobi's identity extends the equality to all matrices
\cite{Lawrence1964}.  Alternatively, the rank-$d-1$ case has
$\operatorname{rank}(\adj A)=1$, whose adjugate vanishes for $d\ge3$;
lower ranks already have $\adj A=0$.  If $\det A=0$, the right side of
\eqref{eq:double-adj} is zero.
\end{proof}

Define
\begin{equation}\label{eq:alpha-E}
 \alpha=(d-1)^2,\qquad E_k=\frac{\alpha^k-1}{d}\quad(k\ge1).
\end{equation}
The quotient is integral because $\alpha\equiv1\pmod d$.

\begin{theorem}[Iterate normal form]\label{thm:iterate}
If $A\in\GL_d(q)$ and $\delta=\det A$, then for every $k\ge1$,
\begin{equation}\label{eq:iterate}
 \boxed{\Psi^k(A)=\delta^{E_k}A},
 \qquad
 \det(\Psi^k(A))=\delta^{\alpha^k}.
\end{equation}
Every nonzero singular matrix has transient depth one and then stays at
zero.
\end{theorem}

\begin{proof}
By \cref{lem:jacobi}, if $B$ is invertible with determinant $\eta$, then
$\Psi(B)=\eta^{d-2}B$ and
\[
 \det\Psi(B)=\eta^{d(d-2)+1}=\eta^{(d-1)^2}=\eta^\alpha.
\]
Thus the scalar exponent in the $k$th iterate obeys
$E_{k+1}=E_k+(d-2)\alpha^k$, with $E_1=d-2$.  Since
$\alpha-1=d(d-2)$, summing the geometric progression gives
$E_k=(\alpha^k-1)/d$.  The singular assertion follows from
\cref{lem:jacobi}.
\end{proof}

\section{Fixed sequence and cycle census}

Write
\begin{equation}\label{eq:group-sizes}
 |\GL_d(q)|=\prod_{j=0}^{d-1}(q^d-q^j),
 \qquad |\SL_d(q)|=\frac{|\GL_d(q)|}{q-1}.
\end{equation}

\begin{theorem}[All iterate-fixed counts]\label{thm:fixed}
For every $k\ge1$,
\begin{equation}\label{eq:fixed}
 \boxed{\displaystyle
 |\Fix(\Psi^k)|
 =1+|\SL_d(q)|\gcd(E_k,q-1).}
\end{equation}
Consequently, the number $C_r$ of cycles of exact length $r$ is
\begin{equation}\label{eq:cycles}
 C_r=\frac1r\sum_{e\mid r}\mu(r/e)
 \left(1+|\SL_d(q)|\gcd(E_e,q-1)\right),
\end{equation}
and
\begin{equation}\label{eq:zeta}
 \zeta_\Psi(z)=\prod_{r\ge1}(1-z^r)^{-C_r}
 =\exp\left(\sum_{k\ge1}
 \frac{1+|\SL_d(q)|\gcd(E_k,q-1)}{k}z^k\right).
\end{equation}
\end{theorem}

\begin{proof}
A singular matrix fixed by a positive iterate must equal zero.  For
$A\in\GL_d(q)$, \cref{thm:iterate} gives
$\Psi^k(A)=A$ exactly when $(\det A)^{E_k}=1$.  The cyclic group
$\F_q^\times$ has $\gcd(E_k,q-1)$ such determinants, and every determinant
