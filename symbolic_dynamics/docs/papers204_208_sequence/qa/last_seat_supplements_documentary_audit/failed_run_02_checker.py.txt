never a hash search and never a replacement for original recorded pins.

def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def raw(path):
    p = pathlib.Path(path)
    assert p.is_absolute() and '..' not in p.parts, str(p)
    assert p.is_file() and not p.is_symlink() and p.resolve() == p, str(p)
    if p not in CACHE:
        CACHE[p] = p.read_bytes()
    return CACHE[p]


def data(path):
    return json.loads(raw(path))


def digest(path):
    return sha(raw(path))


def meta(path):
    return dict(sha256=digest(path), bytes=len(raw(path)), resolved=str(pathlib.Path(path)), symlink=None)


def scoped(path):
    p = pathlib.Path(path)
    assert p in EXTERNAL or any(p.is_relative_to(BASE / row[0]) for row in PACKAGES), str(p)
    return p


def pin(path, expected):
    p = scoped(path)
    wanted = expected if isinstance(expected, str) else expected['sha256']
    effective = ALIASES.get((str(p), wanted), p)
    assert digest(effective) == wanted, (str(p), str(effective), wanted, digest(effective))
    if isinstance(expected, dict):
        assert expected['bytes'] == len(raw(effective))
        assert expected['resolved'] == str(p) and expected['symlink'] is None


def pinmaps(folder, before_name, after_name, relative=False):
    before, after = data(folder / before_name), data(folder / after_name)
    assert isinstance(before, dict) and before == after, str(folder)
    for name, expected in before.items():
        pin(LNR / name if relative else name, expected)
    return before


def census(folder):
    paths = []
    for p in folder.rglob('*'):
        assert not p.is_symlink() and '__pycache__' not in p.parts and p.suffix != '.pyc', str(p)
        assert p.is_dir() or p.is_file(), str(p)
        if p.is_file():
            paths.append(p)
    return set(paths)


def sealed_packages():
    rows = []
    for name, seal, count, expected in PACKAGES:
        folder = BASE / name
        assert digest(folder / seal) == expected
        entries = {}
        for line in raw(folder / seal).decode().splitlines():
            h, rel = line.split('  ', 1)
            assert re.fullmatch('[0-9a-f]{64}', h)
            p = folder / rel
            assert not pathlib.Path(rel).is_absolute() and '..' not in pathlib.Path(rel).parts
            assert p not in entries and p != folder / seal
            entries[p] = h
        physical = census(folder)
        assert len(entries) == count and physical == set(entries) | {folder / seal}, name
        for p, h in entries.items():
            assert digest(p) == h, str(p)
            if p.suffix == '.json':
                data(p)
        CENSUS[folder] = physical
        rows.append(dict(package=name, payloads=count, physical_files=len(physical), seal_sha256=expected))
    return rows

def lower_native(folder, role=None, count_key='input_count', pathset=False):
    receipt = data(folder / 'receipt.json')
    before = pinmaps(folder, 'inputs_before.json', 'inputs_after.json')
    assert receipt['exit'] == 0 and receipt['unchanged'] is True
    assert receipt[count_key] == len(before)
    if role is not None:
        assert receipt['role'] == role
    if pathset:
        names = data(folder / 'pathset.json')
        assert len(names) == len(set(names)) == len(before) and set(names) == set(before)
    for stream in ('stdout', 'stderr'):
        assert digest(folder / (stream + '.raw')) == receipt[stream + '_sha256']
    assert not raw(folder / 'stderr.raw')
    if 'finished_epoch' in receipt:
        assert receipt['started_epoch'] <= receipt['finished_epoch']
    if 'timed_out' in receipt:
        assert receipt['timed_out'] is False and receipt['elapsed_seconds'] >= 0
    argv = receipt['argv']
    assert isinstance(argv, list) and argv and all(isinstance(a, str) for a in argv)
    assert pathlib.Path(receipt['cwd']).is_absolute()
    stdout = raw(folder / 'stdout.raw')
    read_type = 'raw_stream_hash_and_declared_argv'
    if argv[0] == 'sed':
        assert argv[1] == '-n' and re.fullmatch(r'\d+,\d+p', argv[2])
        start, stop = map(int, argv[2][:-1].split(','))
        bodies = [raw(scoped(p)) for p in argv[3:]]
        joined = b''.join(bodies)
        # sed uses one cumulative line number across multiple input files.
        expected = b''.join(joined.splitlines(keepends=True)[start - 1:stop])
        assert stdout == expected, str(folder)
        if start == 1 and stop == 4000:
            assert len(joined.splitlines()) <= 4000 and stdout == joined
            read_type = 'full_read_stdout_byte_equality'
        else:
            read_type = 'exact_declared_excerpt_stdout_byte_equality'
    elif argv[0] == 'cp':
        assert stdout == b'' and argv[1] in ('-p', '--parents')
    elif argv[0] == 'rg':
        if argv[1] == '--files':
            assert argv[:3] == ['rg', '--files', '-g'] and len(argv) == 4
            lines = stdout.decode().splitlines()
            assert len(lines) == len(set(lines)) and all(pathlib.Path(s).name == s and s.endswith('.md') for s in lines)
            read_type = 'filename_only_original_raw_output_not_current_discovery_replay'
        else:
            assert argv[1] == '-n' and argv[3] == '--'
            matches = {}
            for p in argv[4:]:
                matches[p] = []
                for n, line in enumerate(raw(scoped(p)).decode().splitlines(), 1):
                    if re.search(argv[2], line):
                        matches[p].append((p + ':' if len(argv[4:]) > 1 else '') + str(n) + ':' + line + '\n')
            if len(argv[4:]) > 1:
                # rg's unsorted parallel file groups need not follow argv order.
                order = list(dict.fromkeys(line.split(':', 1)[0] for line in stdout.decode().splitlines()))
                assert set(order) == {p for p, lines in matches.items() if lines}
            else:
                order = argv[4:]
            assert stdout == ''.join(line for p in order for line in matches[p]).encode(), str(folder)
            read_type = 'declared_simple_regex_per_file_byte_equality_native_group_order'
    elif argv[0] == 'sha256sum':
        assert stdout == ''.join(digest(scoped(p)) + '  ' + p + '\n' for p in argv[1:]).encode()
    elif argv[0] == 'python3':
        assert argv[1:3] == ['-I', '-B'] and len(argv) == 4
        assert scoped(argv[3]).name in ('check.py', 'audit.py')
        result = data(folder / 'stdout.raw')
        assert isinstance(result, dict)
    else:
        assert argv[0] == 'curl' and folder.is_relative_to(NED / 'commands')
    COMMANDS.append(dict(path=str(folder.relative_to(BASE)), schema='lowercase_native', input_count=len(before), exit=0,
                         argv=argv, stdout_bytes=len(stdout), stdout_check=read_type))
    return receipt


def upper_native(folder, relative=False, expected_exit=0):
    receipt, attempt = data(folder / 'RECEIPT.json'), data(folder / 'ATTEMPT.json')
    before = pinmaps(folder, 'INPUTS_BEFORE.json', 'INPUTS_AFTER.json', relative)
    assert attempt['status'] == 'ATTEMPTED' and attempt['exit_code'] is None
    assert receipt['status'] == 'COMPLETED' and receipt['exit_code'] == expected_exit
    assert receipt['inputs_unchanged'] is True and receipt['started_utc'] <= receipt['ended_utc']
    for k in ('argv', 'cwd', 'environment', 'started_utc'):
        assert attempt[k] == receipt[k]
    assert receipt['environment'] == ENV and receipt['cwd'] == str(ROOT)
    for stream in ('stdout', 'stderr'):
        assert receipt[stream] == meta(folder / stream)
    if 'result' in receipt:
        assert receipt['result'] == meta(LNR / 'AUDIT_RESULT.json')
    COMMANDS.append(dict(path=str(folder.relative_to(BASE)), schema='uppercase_native', input_count=len(before),
                         exit=expected_exit, argv=receipt['argv'], stdout_bytes=len(raw(folder / 'stdout')),
                         stdout_check='full_raw_stream_metadata_equality'))
    return receipt
