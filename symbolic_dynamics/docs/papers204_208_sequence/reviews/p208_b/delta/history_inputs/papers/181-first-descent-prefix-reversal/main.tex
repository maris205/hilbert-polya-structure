\begin{abstract}
Given a nonidentity permutation, locate its first descent and reverse the
prefix ending at the descent's smaller follower.  We determine the complete
functional graph of this autonomous map.  Its image is the half of
$\mathfrak S_n$ beginning with an ascent.  For $n\ge3$, the recurrent set is
the identity together with the $n!/3$ permutations having a peak at position
two; the latter form $n!/6$ two-cycles.  Every tail has length at most two,
with exact populations $n!/3+1,n!/2,n!/6-1$.  The full predecessor set of a
target is indexed by the decreasing run starting at its second position.
Consequently, for $n\ge4$ the maximum fibre is $n-1$, attained at exactly
$n-1$ targets.  The exceptional $n=1,2,3$ atlases are explicit.  Arbitrary
prefix reversal, pancake sorting, descent counts, and finite-map bookkeeping
are treated as background.  The residual conjunction is
\textsc{owner\_amber}, and external circulation is
\textsc{hold\_external}.
\end{abstract}

\section{The autonomous rule and theorem package}

Fix $n\ge1$.  For $1\le k\le n$, let $\rho_k$ reverse the first $k$ entries of a
permutation:
\begin{equation}\label{eq:prefix-reversal}
 \rho_k(\pi_1\cdots\pi_k\pi_{k+1}\cdots\pi_n)
 =\pi_k\cdots\pi_1\pi_{k+1}\cdots\pi_n.
\end{equation}
Write $\iota=12\cdots n$.  If $\pi\ne\iota$, let
\begin{equation}\label{eq:first-descent}
 d(\pi)=\min\{i: \pi_i>\pi_{i+1}\}
\end{equation}
and define the first-descent prefix-reversal map
\begin{equation}\label{eq:map}
 F_n(\iota)=\iota,\qquad
 F_n(\pi)=\rho_{d(\pi)+1}(\pi)\quad(\pi\ne\iota).
\end{equation}
The extra position after the first descent is part of the rule.

Prefix reversal and its pancake-sorting distance are classical
\cite{GatesPapadimitriou1979}; prefix-reversal sorting on fixed-alphabet
strings is also established \cite{HurkensEtAl2007}.  Pudwell and Smith study
four shuffle maps selected by a cut after the longest increasing prefix
\cite{PudwellSmith2024}, but those maps do not reverse the full prefix in
\eqref{eq:map}.  We assign arbitrary prefix reversal, pancake graphs and
distances, longest-increasing-prefix selection, elementary descent and peak
counts, and generic finite-map bookkeeping no contribution credit.

There is also a sharp negative control.  The rule called ``First Sort'' in
Project Euler Problems 523--524 moves the follower $\pi_{d+1}$ to the front;
it does not reverse the selected prefix.\footnote{Official statements:
\url{https://projecteuler.net/problem=523} and
\url{https://projecteuler.net/problem=524}.}
For example, it sends $1324$ to $2134$, whereas \eqref{eq:map} sends $1324$
to $2314$.  We therefore do not use ``First Sort'' as a name for $F_n$.

For $n\ge3$, set
\begin{equation}\label{eq:sets}
 \cI_n=\{\tau\in\Sn:\tau_1<\tau_2\},\qquad
 \cP_n=\{\tau\in\Sn:\tau_1<\tau_2>\tau_3\},\qquad
 \cR_n=\{\iota\}\cup\cP_n.
\end{equation}
Let $\mu(\pi)$ be the distance from $\pi$ to the recurrent set of $F_n$.
For $\tau\in\cI_n$, let
\begin{equation}\label{eq:run}
 r(\tau)=\max\{r:1\le r\le n-1,
                  \ \tau_2>\tau_3>\cdots>\tau_{r+1}\},
\end{equation}
where the condition is empty when $r=1$.

\begin{theorem}[complete functional-graph atlas]\label{thm:main}
For every $n\ge3$, the following hold.
\begin{enumerate}[(i)]
\item The image is exactly $\cI_n$, and $|\cI_n|=n!/2$.

\item The recurrent set is exactly $\cR_n$.  The identity is fixed, while
the $n!/3$ states in $\cP_n$ form $n!/6$ disjoint two-cycles under $\rho_3$.

\item Every tail has length at most two, and
\begin{equation}\label{eq:tail-census}
 \#\{\mu=0\}=\frac{n!}{3}+1,\qquad
 \#\{\mu=1\}=\frac{n!}{2},\qquad
 \#\{\mu=2\}=\frac{n!}{6}-1.
\end{equation}

\item A target outside $\cI_n$ has no predecessor.  For $\tau\in\cI_n$,
the complete set of distinct predecessors is
\begin{equation}\label{eq:fibre-set}
 F_n^{-1}(\tau)=
 \{\rho_k(\tau):2\le k\le r(\tau)+1\}
 \mathbin{\cup}
 \begin{cases}\{\iota\},&\tau=\iota,\\ \varnothing,&\tau\ne\iota.
 \end{cases}
\end{equation}
Consequently,
\begin{equation}\label{eq:fibre-size}
             |F_n^{-1}(\tau)|=r(\tau)+\mathbf1_{\{\tau=\iota\}}.
\end{equation}

\item For $n\ge4$, the maximum fibre size is $n-1$.  Its complete maximizing
set is
\begin{equation}\label{eq:maximizers}
 \cM_n=\{\tau\in\Sn:\tau_2=n,
              \ \tau_2>\tau_3>\cdots>\tau_n\},
 \qquad |\cM_n|=n-1.
\end{equation}
For $n=3$, the maximum remains $2=n-1$, but the identity is one additional
maximizer beyond the two states in \eqref{eq:maximizers}.
\end{enumerate}
\end{theorem}

The five conclusions can be read at a glance as follows.

\section{Half-image and recurrent core}

\begin{proof}[Proof of Theorem~\ref{thm:main}(i)]
If $d=d(\pi)$, the first two entries of $F_n(\pi)$ are
$\pi_{d+1}<\pi_d$.  Thus every nonfixed output begins with an ascent; so
does $\iota$.  Hence $\operatorname{im}F_n\subseteq\cI_n$.

Conversely, take $\tau\in\cI_n$ and put $\pi=\rho_2(\tau)$.  The first
comparison of $\pi$ is the descent $\tau_2>\tau_1$, so
$F_n(\pi)=\rho_2(\pi)=\tau$.  This proves equality.  Swapping the first two
entries is a fixed-point-free involution pairing permutations in $\cI_n$
with those beginning with a descent, and therefore $|\cI_n|=n!/2$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}(ii)]
For $\tau\in\cP_n$, the first descent is at position two.  Hence
$F_n(\tau)=\rho_3(\tau)$.  The new first three entries are
$\tau_3<\tau_2>\tau_1$, so the image is again in $\cP_n$; applying
$\rho_3$ once more returns to $\tau$.  The two states are distinct because
their first and third entries cannot agree.

Now let $\tau\in\cI_n\setminus\cR_n$.  Since it is neither a peak at two nor
the identity, its first three entries increase and its first descent occurs
at some $d\ge3$.  The first three entries of $F_n(\tau)$ are
\begin{equation}\label{eq:new-peak}
              \tau_{d+1}<\tau_d>\tau_{d-1},
\end{equation}
so $F_n(\tau)\in\cP_n$.  Every state first enters $\cI_n$ by part (i), and
every image state then lies in or enters $\cR_n$.  This proves that no other
cycle exists.

Among the six relative orders of the first three entries, exactly two put
their maximum in the middle.  Each order occurs for $n!/6$ permutations, so
$|\cP_n|=n!/3$.  Pairing by $\rho_3$ gives $n!/6$ two-cycles.
\end{proof}

\section{Target fibres and exact tail census}

\begin{lemma}[decreasing-run inverse]\label{lem:fibre}
For $\tau\in\cI_n$ and $2\le k\le n$, the state $\rho_k(\tau)$ has first
descent at position $k-1$ if and only if $k\le r(\tau)+1$.
These valid states are distinct as $k$ varies.
\end{lemma}

\begin{proof}
The first $k$ entries of $\rho_k(\tau)$ are
\begin{equation}\label{eq:inverse-word}
             \tau_k,\tau_{k-1},\ldots,\tau_2,\tau_1.
\end{equation}
They increase through position $k-1$ precisely when
$\tau_2>\tau_3>\cdots>\tau_k$.  The next comparison is then the descent
$\tau_2>\tau_1$, which holds because $\tau\in\cI_n$.  This condition is
equivalent to $k\le r(\tau)+1$.  Different $k$ give different first entries
$\tau_k$, proving distinctness.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}(iv)]
Part (i) excludes predecessors for targets outside $\cI_n$.  Suppose
$\tau\in\cI_n$.  Every nonfixed predecessor $\pi$ has some first descent
$d$ and satisfies $\tau=\rho_{d+1}(\pi)$; since prefix reversal is an
involution, $\pi=\rho_{d+1}(\tau)$.  Lemma~\ref{lem:fibre} gives exactly the
values $2\le d+1\le r(\tau)+1$, and it also proves the converse for every
such value.  The only fixed state is $\iota$, which contributes one further
predecessor exactly when $\tau=\iota$.  For that target, $\rho_2(\iota)$ is
distinct from $\iota$, so \eqref{eq:fibre-size} follows.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}(iii)]
Part (ii) gives the depth-zero population $n!/3+1$ and shows that every image
state outside $\cR_n$ has depth one.  Such a state $\tau$ begins
$\tau_1<\tau_2<\tau_3$, so $r(\tau)=1$.  Lemma~\ref{lem:fibre} says that its
only predecessor is $\rho_2(\tau)$, which begins with a descent and lies
outside $\cI_n$.  The map
\begin{equation}\label{eq:depth-two-bijection}
 \cI_n\setminus\cR_n\longrightarrow\{\pi:\mu(\pi)=2\},
 \qquad \tau\longmapsto\rho_2(\tau)
\end{equation}
is therefore a bijection.  Indeed, every depth-two state must map to a
nonrecurrent image state, so the displayed construction also captures the
reverse inclusion.

Using parts (i)--(ii), the depth-two population is
\begin{equation}\label{eq:depth-two-count}
 |\cI_n|-|\cR_n|=\frac{n!}{2}-\frac{n!}{3}-1
                 =\frac{n!}{6}-1.
\end{equation}
All remaining states have depth one, giving
$n!-(n!/3+1)-(n!/6-1)=n!/2$.  This proves \eqref{eq:tail-census} and the
absence of longer tails.
\end{proof}

\section{Sharp fibres and the small boundaries}

\begin{proof}[Proof of Theorem~\ref{thm:main}(v)]
For a nonidentity image target, \eqref{eq:fibre-size} is at most $n-1$;
the identity has fibre two.  When $n\ge4$, equality can therefore occur only
when $r(\tau)=n-1$, or equivalently
$\tau_2>\tau_3>\cdots>\tau_n$.  Since also $\tau_1<\tau_2$, the value $n$
cannot occupy the first position.  It lies in the decreasing tail and must
be its first, largest entry: $\tau_2=n$.  Conversely, every target in
\eqref{eq:maximizers} has the full run and hence fibre $n-1$.  Choose
$\tau_1$ arbitrarily from $[n-1]$; the remaining entries are then forced in
decreasing order.  This gives exactly $n-1$ maximizers.

At $n=3$, the two full-run targets still have fibre two, and the identity's
extra fixed predecessor also gives fibre two.  This is the stated exception.
\end{proof}

\begin{proposition}[the $n=1,2,3$ atlases]\label{prop:small}
At $n=1$, the sole arrow is $1\mapsto1$.  Thus the image and recurrent set
are both $\{1\}$, the sole state has depth zero, and its fibre has size one.

At $n=2$, the arrows are $12\mapsto12$ and $21\mapsto12$.  Thus the image
and recurrent set are both $\{12\}$, the depth populations are $(1,1)$, and
$F_2^{-1}(12)=\{12,21\}$ is the unique maximum fibre.

At $n=3$, the full functional graph is
\begin{equation}\label{eq:n3-atlas}
 123\mapsto123,\qquad 132\leftrightarrow231,
 \qquad 213\mapsto123,\quad312\mapsto132,\quad321\mapsto231.
\end{equation}
The image and recurrent populations are both three, the depth populations
are $(3,3,0)$, and each image target has fibre two.  Thus the three
maximizers are $123,132,231$.
\end{proposition}

\begin{proof}
The $n=1$ claim is immediate.  At $n=2,3$, applying \eqref{eq:map} gives the
listed arrows; the identity then has its fixed predecessor in addition to
its two-prefix reversal.  These direct calculations isolate the $n=1,2$
boundaries not covered by the $n\ge3$ recurrent count and the only tie not
covered uniformly by the $n\ge4$ maximizer statement.
\end{proof}
