\node[source] (sdc) {%
  {\sffamily\scriptsize\bfseries [SD-C07: SYMBOLIC SOURCE]}\quad
  tensor atoms $F_p$, entropy lengths $\log p$, and
  $\mathcal L_s e_p=p^{-s}e_p$
};

\node[exact, below=8mm of sdc, xshift=-0.315\linewidth] (koszul) {%
  {\sffamily\scriptsize\bfseries [EXACT: KOSZUL / EXTERIOR]}\\[2pt]
  Atoms have canonical homological degree $1$ inside the Koszul functor.
  For $\operatorname{Re}s>1$,
  \[
    \begin{aligned}
      \mathrm{Ber}_{V_{\bar 1}}(I-\mathcal L_s)
        &=\zeta(s),\\[-1pt]
      \operatorname{Str}\Gamma(\mathcal L_s)
        &=\det(I-\mathcal L_s),\\[-1pt]
        &=\zeta(s)^{-1}.
    \end{aligned}
  \]
  \textbf{A2 orientation is exact.}
};

\node[boundary, below=8mm of sdc] (reversal) {%
  {\sffamily\scriptsize\bfseries [BOUNDARY: STABLE / UNSTABLE]}\\[2pt]
  Natural symbolic reversal produces
  \[
    \mathcal L_s\oplus\mathcal L_s .
  \]
  It preserves the entropy weight:
  \[
    \begin{gathered}
      s\longmapsto s,\\[-1pt]
      \textit{not}\quad s\longmapsto1-s .
    \end{gathered}
  \]
  No intrinsic Riemann duality follows.
};

\node[obstructed, below=8mm of sdc, xshift=0.315\linewidth] (group) {%
  {\sffamily\scriptsize\bfseries [OBSTRUCTION: GROUP COMPLETION]}\\[2pt]
  In $\mathbb Q_{+}^{\times}$, tensor inversion gives
  \[
    p^{-s}\longmapsto p^{s},
    \qquad s\longmapsto -s .
  \]
  Every monoidal $\mathbb Z/2$-parity is inversion-even.
  Centering at $1/2$ therefore requires extra structure.
};

\node[adversarial, below=8mm of reversal, xshift=0.1575\linewidth] (detthree) {%
  {\sffamily\scriptsize\bfseries [ADVERSARIAL $s\leftrightarrow1-s$ CHECK]}\\[2pt]
  Even granting the missing half-density pairing,
  \[
    D_3(s)=\det\nolimits_3(I-\mathcal L_s)\,
           \det\nolimits_3(I-\mathcal L_{1-s}),
    \qquad \frac13<\operatorname{Re}s<\frac23 .
  \]
  The paired determinant is symmetric but \textbf{zero-free} in this strip,
  while
  $\log D_3(s)=-\sum_{r\geq3}r^{-1}
  \sum_p\!\left(p^{-rs}+p^{-r(1-s)}\right)$:
  the decisive $r=1,2$ prime traces are deleted.
};
