\begin{abstract}
Replace every coordinate of a length-$n$ word over an $n$-letter alphabet by
the number of distinct letters in its strict prefix, and feed the entire
output word back into the same rule.  We prove a pointwise iterate formula:
after the first epoch, each further epoch inserts one forced identity
coordinate and shifts the original prefix-diversity path.  It follows that
the identity word is the unique recurrent point and that a word's exact
entrance time is determined by its longest all-distinct prefix.  For
$1\le t\le n-1$, the image consists of $2^{n-t-1}$ binary-rise paths with
an identity prefix, and the depth CDF is $(n)_{n-t}n^t$.  Every transient
target fibre is a local product of fresh-letter and old-letter choices, with
the identity and stabilized cases stated separately.  For $n\ge3$ the global height
is $n-1$, with exactly $n^{n-1}$ deepest words; the two-letter boundary is
explicit.  Restricted-growth encodings and prefix statistics receive no
contribution credit.  The retained iterative
conjunction is \textsc{owner\_amber}; circulation remains
\textsc{hold\_external}.
\end{abstract}

\section{Literal rule and contribution boundary}

Write $[n]_0=\{0,\ldots,n-1\}$ and $\W_n=[n]_0^n$ for $n\ge1$.  Define the
prefix-diversity self-map $P_n:\W_n\to\W_n$ by
\begin{equation}\label{eq:map}
 (P_nw)_i=|\{w_0,\ldots,w_{i-1}\}|,
 \qquad 0\le i<n,
\end{equation}
where the empty prefix has size zero.  Since $(P_nw)_i\le i<n$, no
totalisation is needed.

Restricted-growth functions and their set-partition encodings are classical
\cite{Wachs1994}.  More closely, Mansour and Vajnovszki define restricted
growth words through general prefix statistics and give efficient Gray-code
generation \cite{MansourVajnovszki2013}.  We therefore assign
restricted-growth vocabulary, first-occurrence patterns, prefix-statistic
evaluation, falling factorials, and generic finite-map bookkeeping no
contribution credit.  The scoped residual is only the autonomous iteration
of the literal vector \eqref{eq:map}, its pointwise delay law, and the joint
all-time clock/image/every-target inverse atlas.  A bounded search found no
source stating that conjunction.  This non-hit is neither novelty nor
priority evidence.

Internally, P132 thresholds binary prefix sums; P134 recomputes border arrays;
P138 and P139 use palindromic and Lyndon prefix structure; P164 passes a
cyclic equality mask into a linear cellular automaton; and P176 rotates by a
first-symbol frequency.  None yields \eqref{eq:map} or the delay identity
below, but all generic prefix and word-feedback language is subtracted.

For a word $w$, put $d=P_nw$.  Its increments record novelty:
\begin{equation}\label{eq:novelty}
 d_0=0,\qquad d_i-d_{i-1}=\mathbf1\{w_{i-1}\notin
 \{w_0,\ldots,w_{i-2}\}\}\quad(1\le i<n).
\end{equation}
In particular $d_1=1$ when $n\ge2$.

\section{The delay normal form}

\begin{theorem}[pointwise all-time iterate]\label{thm:iterate}
Let $w\in\W_n$, $d=P_nw$, and $t\ge1$.  With $r=t-1$,
\begin{equation}\label{eq:iterate}
 (P_n^tw)_i=
 \begin{cases}
 i,&i<r,\\
 r+d_{i-r},&i\ge r.
 \end{cases}
\end{equation}
For $r\ge n$, the first branch applies at every coordinate.
\end{theorem}

\begin{proof}
Every prefix-diversity word $d$ starts at zero and rises at each step by zero
or one.  Hence its prefix $d_0,\ldots,d_{i-1}$ contains every integer from
zero through $d_{i-1}$.  Applying \eqref{eq:map} to such a path gives
\begin{equation}\label{eq:shift}
 (P_nd)_0=0,\qquad (P_nd)_i=d_{i-1}+1\quad(i\ge1).
\end{equation}
The right-hand side is again a zero-one-rise path.  Iterating
\eqref{eq:shift} inserts one more forced identity coordinate on the left and
shifts the old path one place right.  Starting from $P_nw=d$ proves
\eqref{eq:iterate} by induction.
\end{proof}

Thus the entire future of $w$ is already coded by the novelty decisions in
its first image; no orbit search is needed.

\section{Images, recurrence, and pointwise clocks}

Let $e=(0,1,\ldots,n-1)$.  For $w\in\W_n$, let $\rho(w)$ be the length of
its longest all-distinct prefix.

\begin{theorem}[complete forward atlas]\label{thm:forward}
For $1\le t\le n-1$, the time-$t$ image is exactly the set of words $y$ with
\begin{equation}\label{eq:image}
 y_i=i\ (0\le i\le t),\qquad
 y_i-y_{i-1}\in\{0,1\}\ (t<i<n).
\end{equation}
Consequently,
\begin{equation}\label{eq:image-size}
                 |\im P_n^t|=2^{n-t-1}.
\end{equation}
The word $e$ is the unique fixed and recurrent state, and $P_n^{n-1}$ is
constant with value $e$.  The exact entrance time $\tau(w)$ is
\begin{equation}\label{eq:clock}
 \tau(w)=
 \begin{cases}
 0,&w=e,\\
 \max\{1,n-\rho(w)\},&w\ne e.
 \end{cases}
\end{equation}
For $n\ge3$, the sharp height is $n-1$; its complete deepest set is
$\{w:w_0=w_1\}$ and has size $n^{n-1}$.  At $n=2$, the height is one and
all three nonidentity words are deepest.  At $n=1$, the height is zero.
\end{theorem}

\begin{proof}
Formula \eqref{eq:iterate} forces \eqref{eq:image}.  Conversely, a target of
that form recovers the prefix
\begin{equation}\label{eq:recover-d}
 d_j=y_{j+t-1}-(t-1),\qquad 0\le j\le n-t.
\end{equation}
It begins $0,1$ and has zero-one rises, so choose a fresh source letter at
each rise and a previously seen letter at each flat.  There are always
enough letters, and the last $t$ source coordinates are arbitrary.  Thus
every target in \eqref{eq:image} occurs.  Its final $n-t-1$ increments are
free bits, proving \eqref{eq:image-size}.

For $n=1$ the claim is immediate.  For $n\ge2$, taking $t=n-1$ gives the
constant image $e$.  If $P_nw=w$, then $w_0=0,w_1=1$ and induction in
\eqref{eq:map} forces $w_i=i$; hence $e$ is the only fixed point and, because
every orbit reaches it, the only recurrent point.

For $t\ge1$, \eqref{eq:iterate} equals $e$ exactly when
$d_j=j$ through $j=n-t$, equivalently when the first $n-t$ letters of $w$
are distinct.  Taking the least such positive $t$ and separating the already
fixed word proves \eqref{eq:clock}.  For $n\ge3$, depth $n-1\ge2$ means
$\rho(w)=1$, which is exactly $w_0=w_1$; choose this repeated letter in $n$
ways and the remaining $n-2$ coordinates arbitrarily.  Directly, $\W_2$
consists of the fixed word $01$ and three depth-one words, while
$\W_1=\{0\}$.
\end{proof}

\begin{corollary}[exact depth CDF]\label{cor:depth}
For $1\le t\le n-1$,
\begin{equation}\label{eq:depth-cdf}
 |\{w:\tau(w)\le t\}|=(n)_{n-t}n^t,
\end{equation}
where $(n)_k=n(n-1)\cdots(n-k+1)$.  Thus every positive depth population is
the difference of consecutive values in \eqref{eq:depth-cdf}, with the
depth-zero value equal to one.
\end{corollary}

\begin{proof}
The proof of Theorem~\ref{thm:forward} says that $\tau(w)\le t$ precisely
when the first $n-t$ letters are distinct.  There are $(n)_{n-t}$ choices for
that prefix and $n^t$ choices for the invisible suffix.
\end{proof}

\section{All-time every-target fibres}

\begin{theorem}[local inverse product]\label{thm:fibre}
Let $1\le t\le n-1$ and let $y$ satisfy \eqref{eq:image}.  Recover
$d_0,\ldots,d_{n-t}$ by \eqref{eq:recover-d}.  Then
\begin{equation}\label{eq:fibre}
 |(P_n^t)^{-1}(y)|=
 n^{t+1}\prod_{q=1}^{n-t-1}
 \begin{cases}
 n-d_q,&d_{q+1}-d_q=1,\\
 d_q,&d_{q+1}-d_q=0.
 \end{cases}
\end{equation}
Targets outside \eqref{eq:image} have fibre zero.  In particular, the fibre
of $e$ is $(n)_{n-t}n^t$, exactly the depth CDF.
The product in \eqref{eq:fibre} is empty when $t=n-1$.  At time $t=0$,
$P_n^0$ is the identity, so every target has fibre one.  For every
$t\ge n-1$, the image is the singleton $\{e\}$, the fibre of $e$ has size
$n^n$, and every other fibre is zero.
\end{theorem}

\begin{proof}
The target exposes whether source position $q$ is a new letter for every
$1\le q\le n-t-1$.  Before that position, exactly $d_q$ letters have
appeared.  A rise therefore has $n-d_q$ choices and a flat has $d_q$
choices.  The first source letter has $n$ choices, while the last $t$ source
letters are invisible and contribute $n^t$.  These choices are independent
conditional on the displayed novelty path, proving \eqref{eq:fibre} and also
proving fibre-mass conservation without division by the image size.  For
$y=e$, every visible decision is a rise, yielding the falling factorial in
\eqref{eq:depth-cdf}.
\end{proof}

For orientation, at $n=4$ the word $0013$ follows
\begin{equation*}
 0013\longmapsto0112\longmapsto0122\longmapsto0123=e,
\end{equation*}
and its initial repetition witnesses the sharp depth three.  At $n=7$, the
exact depth histogram is
\begin{equation*}
 1,\ 35279,\ 88200,\ 164640,\ 216090,\ 201684,\ 117649.
\end{equation*}
