\section{Frozen affine source and path semantics}
\label{sec:source}

\subsection{The full positive semigroup}

Let
\begin{equation}
P=\Nzero\rtimes\N^{\times},\qquad
(b,a)(d,c)=(b+ad,ac).
\label{eq:affine-product}
\end{equation}
Set $u=(1,1)$ and $d_n=(0,n)$ for every $n\ge2$.  The labelled right Cayley
graph has edges
\[
x\xrightarrow{u}xu,\qquad x\xrightarrow{d_n}xd_n.
\]
The identity generator is excluded.  The alphabet contains every $n\ge2$,
not a prime subalphabet.

The full graph has countably infinite outdegree.  It is therefore used for a
formal path theorem, while the natural bounded whole-operator benchmark is
the following finite-generator slice.

\subsection{The bounded slice}

For $r\ge2$, define
\begin{equation}
P_r=\Nzero\rtimes\langle r\rangle
=\{(b,k):b,k\in\Nzero\},\qquad
(b,k)(d,\ell)=(b+r^kd,k+\ell).
\label{eq:slice-product}
\end{equation}
Writing $u=(1,0)$ and $v=(0,1)$ gives
\begin{equation}
vu=u^rv.
\label{eq:affine-relation}
\end{equation}
Every element has the unique normal form $u^bv^k$.  Right multiplication is
\begin{equation}
U(b,k)=(b+r^k,k),\qquad V(b,k)=(b,k+1).
\label{eq:right-actions}
\end{equation}

\begin{convention}[Clock and primitive class]
Every original edge has roof one and contributes one power of a free marker
$z$.  In a formally symmetrized graph, a reverse arc is a distinct edge and
also contributes one step.  A primitive orbit is a nonempty closed oriented
edge word modulo cyclic rotation that is not a positive temporal power.
Reflection is not identified.  Cyclic nonbacktracking forbids an oriented
edge followed by its formal reverse, including at the cyclic join.
\end{convention}

No first return, acceleration, quotient clock, accepted-support projector,
KMS support, prime symbol, or target spectral data enters this definition.

\subsection{Natural operators}

On $\Hh_r=\ell^2(P_r)$ let
\[
S\delta_x=\delta_{xu},\qquad T\delta_x=\delta_{xv}.
\]
For fixed $a,b>0$, set
\begin{equation}
\Aplus=aS+bT,\qquad
\Asym=a(S+S^*)+b(T+T^*).
\label{eq:vertex-operators}
\end{equation}
Let $\Er$ be the oriented edges of the symmetrized graph.  Give $U$-type
arcs weight $a$ and $V$-type arcs weight $b$.  The Hashimoto operator on
$\ell^2(\Er)$ is
\begin{equation}
B\delta_e=\sum_{\substack{o(f)=t(e)\\f\ne\bar e}}w(f)\delta_f.
\label{eq:hashimoto}
\end{equation}
These spaces and operators remain uninduced.  A determinant, if owned, would
be $\det(I-zA)$ with the same marker $z$.
