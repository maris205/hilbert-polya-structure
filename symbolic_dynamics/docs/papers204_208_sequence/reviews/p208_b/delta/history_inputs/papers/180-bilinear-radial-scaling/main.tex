\begin{abstract}
Let $B$ be a nondegenerate bilinear form on $V=\mathbb F_q^m$.  We determine
the finite polynomial map
$\Phi(u,v)=(B(u,v)u,B(u,v)v)$.  Its $t$th iterate is radial with exponent
$(3^t-1)/2$, while its bilinear value follows the scalar cube map.  This
gives every tail and eventual period from the decomposition of the initial
value's order into its $3$-part and prime-to-$3$ part.  We lift the scalar
description back to all pair targets: every-time
fibres are power-map root counts, except for a null cone collapsing to zero.
We also give every tail population, the recurrent census, image size, and
complete one-step fibre distribution; zero is the unique maximum-fibre
target.  Power maps, bilinear level counts, and generic polynomial functional
graphs receive zero contribution credit.  Status is
\texttt{OWNER\_AMBER / HOLD\_EXTERNAL}.
\end{abstract}

\section{Literal map and subtraction boundary}

Let $q$ be a prime power and $m\ge1$.  Fix a nondegenerate bilinear form
$B:V\times V\to\Fq$ and define
\begin{equation}\label{eq:map}
 \Phi(u,v)=\bigl(cu,cv\bigr),\qquad c=B(u,v).
\end{equation}
The form need not be symmetric.  A state means the ordered pair $(u,v)$;
write $0=(0,0)$.

Monomial dynamics over finite fields is established background
\cite{ColonReyesEtAl2006}, as are polynomial functional graphs
\cite{KonyaginEtAl2016} and, most directly, the structure of power-map
functional graphs over finite groups \cite{QureshiReis2023}.  We therefore
assign finite-field cyclicity, scalar power-map tails and roots, and static
level counts of $B$ zero contribution credit.  The retained conjunction is
the literal pair map \eqref{eq:map}, its radial iterate, the lift of every
scalar time/fibre statement to every pair target, the exceptional null cone,
and the complete population atlas.  A literal or conjugate owner triggers
withdrawal.  The bounded source non-hit used for internal drafting is not a
novelty certificate.

Internally, P102 and P103 contain scalar power-map reductions inside,
respectively, an involutive group-algebra norm map and double-adjugate matrix
dynamics; P125 is a formed-space pair map with a different shear update.
P171 is a Gram-type warning, but it lives over the Boolean semiring rather
than a finite-field bilinear radial map.  None supplies the present
source-to-pair lift, null-cone collapse, or bilinear population atlas, and all
shared scalar-power and formed-space vocabulary receives zero credit.

For later counts put
\begin{equation}\label{eq:QZ}
 Q=q^{m-1}(q^m-1),\qquad
 Z=q^{2m-1}+q^m-q^{m-1}.
\end{equation}

\begin{lemma}[bilinear levels]\label{lem:levels}
For each $d\in\Fq^*$, exactly $Q$ states satisfy $B(u,v)=d$, while exactly
$Z$ states lie on the null cone $B(u,v)=0$.
\end{lemma}

\begin{proof}
For each $u\ne0$, nondegeneracy makes $v\mapsto B(u,v)$ a nonzero linear
functional, so every value has $q^{m-1}$ preimages.  Thus a nonzero level has
$(q^m-1)q^{m-1}=Q$ points.  Subtracting the $q-1$ nonzero levels from
$q^{2m}$ gives $Z$.
\end{proof}

\section{Iterates, tails, and periods}

Set $a_t=(3^t-1)/2$, an ordinary integer exponent (also in characteristic
two).

\begin{theorem}[closed iterate and scalar factor]\label{thm:iterate}
For every $t\ge0$,
\begin{equation}\label{eq:iterate}
 \Phi^t(u,v)=\bigl(c^{a_t}u,c^{a_t}v\bigr),
 \qquad B(\Phi^t(u,v))=c^{3^t},
\end{equation}
with the evident exponent-zero convention at $t=0$.
\end{theorem}

\begin{proof}
If a pair is scaled by $\lambda$, bilinearity scales its $B$-value by
$\lambda^2$.  Hence one step sends $c$ to $c^3$.  If the accumulated radial
exponent is $a_t$, the next is $a_t+3^t=a_{t+1}$.  Induction proves both
identities, including $c=0$.
\end{proof}

For $c\ne0$, let $r=\ord_{\Fq^*}(c)$ and factor
$r=3^a s$ with $(s,3)=1$.  Write $\ord_{2s}(3)$ for the multiplicative
order of $3$ modulo the ordinary integer $2s$; in particular
$\ord_2(3)=1$.  Thus no division by two in $\Fq$ is intended.

\begin{theorem}[pointwise functional graph]\label{thm:orbit}
The zero state is fixed.  Every other state on the null cone has exact tail
one and eventual period one.  A state with $c\ne0$ has exact tail $a$ and
eventual period
\begin{equation}\label{eq:period}
 \ord_{2s}(3).
\end{equation}
\end{theorem}

\begin{proof}
A null state maps to zero, proving the first claims.  For $c\ne0$, the pair
itself has a nonzero coordinate, so two radial multiples are equal exactly
when their scalars are equal.  By \eqref{eq:iterate}, equality at epochs
$t$ and $t+\ell$ is equivalent to
\begin{equation}\label{eq:divisibility}
 3^a s\mid 3^t\frac{3^\ell-1}{2}.
\end{equation}
Because $3\nmid 3^\ell-1$, a cycle can first be reached exactly at $t=a$.
After cancelling $3^a$, the least positive $\ell$ satisfies
$2s\mid3^\ell-1$, which is \eqref{eq:period}.
\end{proof}

\section{Every-time fibres}

\begin{theorem}[all targets at every time]\label{thm:fibres}
At $t=0$, every target has exactly one predecessor under $\Phi^0$.  For
$t\ge1$, put $g_t=\gcd(3^t,q-1)$, and let $(x,y)$ be a target with
$d=B(x,y)$.  Then
\begin{equation}\label{eq:fibres}
 \#(\Phi^t)^{-1}(x,y)=
 \begin{cases}
 Z,&(x,y)=0,\\
 0,&(x,y)\ne0\text{ and }d=0,\\
 g_t,&d\ne0\text{ and }d^{(q-1)/g_t}=1,\\
 0,&d\ne0\text{ and }d^{(q-1)/g_t}\ne1.
 \end{cases}
\end{equation}
\end{theorem}

\begin{proof}
The time-zero statement is the identity-map boundary.  Suppose henceforth
that $t\ge1$.
The null cone is precisely the fibre of zero: its points collapse in one
step, while a nonzero bilinear value gives a nonzero radial scalar.  Now let
the target be nonzero.  A predecessor with initial value $c\ne0$ is forced by
\eqref{eq:iterate} to be
$c^{-a_t}(x,y)$.  Its bilinear value equals $c$ exactly when
$d=c^{2a_t+1}=c^{3^t}$.  Thus predecessors are in bijection with
$3^t$th roots of $d$ in the cyclic group $\Fq^*$.  The standard cyclic-group
root criterion gives the final two cases.
\end{proof}

This proof supplies an explicit predecessor, not just its number: each root
$c$ gives $c^{-a_t}(x,y)$.

\section{Population and image atlases}

Write $q-1=3^A h$ with $(h,3)=1$.

\begin{theorem}[complete tail census]\label{thm:census}
Among states with nonzero $B$-value, exactly $hQ$ are recurrent, and for
$1\le a\le A$ exactly
\begin{equation}\label{eq:tails}
 2\cdot3^{a-1}hQ
\end{equation}
have tail $a$.  Including the null cone, the total tail-zero population is
$1+hQ$; the tail-one population is
$Z-1+\1_{A\ge1}\,2hQ$; and for $2\le a\le A$ it is
\eqref{eq:tails}.  The sharp maximum tail is $\max\{1,A\}$.
\end{theorem}

\begin{proof}
The cyclic group $\Fq^*$ is the product of cyclic factors of orders $3^A$
and $h$.  Exactly $h$ elements have order prime to $3$.  For $a\ge1$,
exactly $\varphi(3^a)h=2\cdot3^{a-1}h$ elements have order with exact
$3$-primary part $3^a$.  Multiply these scalar counts by the common level
size $Q$ from Lemma~\ref{lem:levels}, then add zero and the $Z-1$ nonzero
null states.  Such null states ensure tail one exists for every $q,m$.
\end{proof}

\begin{theorem}[image and one-step fibres]\label{thm:image}
Let $g=\gcd(3,q-1)$.  Then
\begin{equation}\label{eq:image}
 |\operatorname{im}\Phi|=1+\frac{(q-1)Q}{g}.
\end{equation}
Every nonzero image target has exactly $g$ predecessors.  Zero has $Z$
predecessors and is the unique maximum-fibre target.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:fibres} at $t=1$.  The cube subgroup has
$(q-1)/g$ values, each supporting $Q$ pair targets, plus zero.  This proves
\eqref{eq:image} and uniformity.  Finally
$Z=q^{m-1}(q^m+q-1)>q-1\ge g$, so zero is uniquely maximal.
\end{proof}

All statements are deductive for arbitrary prime powers and nondegenerate
bilinear forms.  The accompanying prime-field enumeration is only finite
counterexample pressure; it is neither experimental nor novelty evidence.
