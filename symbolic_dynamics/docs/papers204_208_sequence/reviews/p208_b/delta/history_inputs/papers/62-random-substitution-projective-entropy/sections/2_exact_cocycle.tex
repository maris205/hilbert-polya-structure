\section{Disjoint decoding and the exact entropy cocycle}
\label{sec:exact}

For sets $S,T$ of words of a common length and integers $i,j\ge0$, write
$S^iT^j$ for the set of concatenations of $i$ members of $S$ followed by
$j$ members of $T$.  Fixed block length makes the concatenation map from the
Cartesian product injective.

\begin{lemma}[Persistent level separation]
\label{lem:separation}
For every $m\ge1$, the sets $\cU_m$ and $\cV_m$ are disjoint and consist of
words of length $\ell^m$.  Moreover,
\begin{equation}
 \cU_{m+1}=\cU_m^\ell\,\dot\cup\,\cV_m^\ell,
 \qquad
 \cV_{m+1}=\cU_m^p\cV_m^q,
 \label{eq:set-recursion}
\end{equation}
where the first union is disjoint.
\end{lemma}

\begin{proof}
At level one, $\cU_1=\{a^\ell,b^\ell\}$ and
$\cV_1=\{a^pb^q\}$.  Since $p,q\ge1$, these three words are distinct.
All have length $\ell$.

Assume separation at level $m$.  Every word in \cref{eq:set-recursion} has a
unique parsing into consecutive blocks of length $\ell^m$.  A word from
$\cU_m^\ell$ has block-type pattern $U^\ell$, one from $\cV_m^\ell$ has
pattern $V^\ell$, and one from $\cU_m^p\cV_m^q$ has pattern $U^pV^q$.
Because $\cU_m\cap\cV_m=\varnothing$ and $p,q\ge1$, the three block-type
patterns are pairwise different.  This proves both the stated disjointness
and injectivity of every product count.  The new common length is
$\ell\cdot\ell^m=\ell^{m+1}$.
\end{proof}

\begin{proposition}[Exact count and projective recurrences]
\label{prop:recurrence}
With $A_m=|\cU_m|$, $B_m=|\cV_m|$, and $r_m=B_m/A_m$, one has
\begin{align}
 A_{m+1}&=A_m^\ell+B_m^\ell,& A_1&=2,\label{eq:A-rec}\\
 B_{m+1}&=A_m^pB_m^q,& B_1&=1,\label{eq:B-rec}\\
 r_{m+1}&=\frac{r_m^q}{1+r_m^\ell},& r_1&=\frac12.\label{eq:r-rec}
\end{align}
In particular $0<r_{m+1}<r_m\le1/2$.
\end{proposition}

\begin{proof}
The cardinality identities follow from \cref{lem:separation}.  Division of
\cref{eq:B-rec} by \cref{eq:A-rec}, using $p+q=\ell$, gives
\cref{eq:r-rec}.  If $0<r<1$, then
$0<r^q/(1+r^\ell)<r$ for every $q\ge1$, so induction gives the final claim.
\end{proof}

Every realization in \cref{eq:substitution} is nonempty and has length
$\ell$.  Moreover, the substitution is primitive with exponent one in the
set-valued sense used by \cite{Mitchell2024}: for each source letter and each
target letter there is a realization of the source image containing the
target.  Indeed, the two realizations $a^\ell,b^\ell$ cover the two targets
for source $a$, while the sole realization $a^pb^q$ contains both targets for
source $b$ because $p,q\ge1$.  Thus the hypotheses of
\cite[Theorem~4.1]{Mitchell2024} apply, and its inflation-word normalization
is exactly $\ell^{-m}\log|\vartheta_{\ell,p}^m(a)|=\ell^{-m}\log A_m$.
We include the remaining algebra to fix every index.

\begin{theorem}[Exact entropy series and finite certificate]
\label{thm:entropy}
The topological entropy of the random substitution subshift generated by
$\vartheta_{\ell,p}$ is
\begin{equation}
 h_{\ell,p}=\frac{\log2}{\ell}
 +\sum_{m=1}^{\infty}\ell^{-(m+1)}\log(1+r_m^\ell),
 \label{eq:entropy}
\end{equation}
where $r_m$ is generated by \cref{eq:r-rec}.  For
$H_M=\ell^{-M}\log A_M$,
\begin{equation}
 H_M=\frac{\log2}{\ell}
 +\sum_{m=1}^{M-1}\ell^{-(m+1)}\log(1+r_m^\ell)
 \label{eq:HM}
\end{equation}
and
\begin{equation}
 0<h_{\ell,p}-H_M
 \le \frac{\ell^{-M}}{\ell-1}\log(1+r_M^\ell)
 \le\frac{\ell^{-M}}{\ell-1}r_M^\ell.
 \label{eq:tail-certificate}
\end{equation}
\end{theorem}

\begin{proof}
Taking logarithms in \cref{eq:A-rec} gives
\[
 \log A_{m+1}=\ell\log A_m+\log(1+r_m^\ell).
\]
After division by $\ell^{m+1}$ and telescoping from $m=1$ to $M-1$, this is
\cref{eq:HM}.  Mitchell's theorem identifies the limit of $H_M$ with the
topological entropy, yielding \cref{eq:entropy}.

By \cref{prop:recurrence}, $r_m$ decreases strictly.  Therefore every term
in the omitted series is positive and
\begin{align*}
 h_{\ell,p}-H_M
 &\le \log(1+r_M^\ell)
 \sum_{m=M}^{\infty}\ell^{-(m+1)}\\
 &=\frac{\ell^{-M}}{\ell-1}\log(1+r_M^\ell).
\end{align*}
The last inequality uses $\log(1+x)\le x$.
\end{proof}

\begin{remark}[Histories are not words]
The exact recursion is valid because \cref{lem:separation} proves absence of
collisions at every level.  Counting substitution choices without that lemma
would count histories and could overestimate inflation-word entropy.
\end{remark}
