\section{Positive acyclicity, backtracks, and relation cycles}
\label{sec:trichotomy}

\begin{theorem}[Full affine trichotomy]
\label{thm:full-trichotomy}
For the full right Cayley graph of \cref{eq:affine-product}:
\begin{enumerate}
\item $h(b,a)=b+a$ strictly increases along every positive edge, so there is
no nonempty directed closed path.
\item Adjoining a distinct reverse arc to every edge creates a primitive
length-two backtrack $e\bar e$ at every retained edge.
\item After immediate reversals are forbidden, each $n\ge2$ and base vertex
$x$ supports the simple primitive relation cycle
\[
C_{n,x}=d_nu\,\bar d_n\,\bar u^n,
\]
of length $n+3$.
\item The unweighted all-$n$ adjacency does not map any basis vector into
$\ell^2(P)$.
\end{enumerate}
\end{theorem}

\begin{proof}
For $x=(b,a)$,
\[
h(xu)-h(x)=a>0,
\qquad h(xd_n)-h(x)=a(n-1)>0.
\]
This proves acyclicity.  Formal reversal makes $e\bar e$ closed, and no
positive loop exists from which it could be a square.  Finally,
$d_nu=u^nd_n$.  The two positive paths have common endpoint; their internal
vertices are disjoint because the $d_nu$ branch has multiplier $an$ while
the $u^nd_n$ branch keeps multiplier $a$ until the final edge.  Traversing
one path and the reverse of the other is simple and cyclically reduced, hence
primitive.  The unweighted image of $\delta_x$ contains the orthonormal
family $\{\delta_{xd_n}:n\ge2\}$, so it has infinite norm.
\end{proof}

The same result has a bounded finite-generator form.

\begin{theorem}[Bounded affine benchmark]
\label{thm:bounded-benchmark}
For every $r\ge2$ and $a,b>0$:
\begin{enumerate}
\item the positive graph of $P_r$ is acyclic;
\item $\|\Aplus\|\le a+b$ and $\Aplus$ is noncompact;
\item $\Asym$ is bounded selfadjoint, noncompact, and contains a two-step
backtrack at every edge;
\item $B$ is bounded and noncompact, and it retains the primitive relation
cycle
\begin{equation}
C_{r,x}=vu\,\bar v\,\bar u^r
\label{eq:relation-cycle}
\end{equation}
of length $r+3$ and weight $a^{r+1}b^2$;
\item infinitely many translated copies of \cref{eq:relation-cycle} are
edge-disjoint, so its formal order-$(r+3)$ diagonal sum diverges.
\end{enumerate}
\end{theorem}

\begin{proof}
The frozen height $h_r(b,k)=b+r^k$ increases by $r^k$ along $U$ and by
$(r-1)r^k$ along $V$.  Boundedness and noncompactness of the three operators are proved in
\cref{prop:noncompactness} and \cref{app:proofs}.
\Cref{eq:affine-relation} closes the word in
\cref{eq:relation-cycle}.  It is cyclically reduced, and its single positive
$v$ occurrence excludes a proper temporal power.  The word uses $r+1$
$U$-type arcs and two $V$-type arcs, giving weight $a^{r+1}b^2$.
Base vertices $(0,2j)$ use disjoint pairs of height levels; a subsequence
therefore gives infinitely many edge-disjoint translated cycles.  Their
order-$(r+3)$ diagonal contributions are uniformly positive, so the formal
diagonal sum diverges.
\end{proof}

\begin{remark}
The relation cycle is presentation algebra, not arithmetic selection.  It is
present for prime, composite, and prime-power values of $r$.  The composite
baseline $r=4$ makes this control explicit.
\end{remark}
